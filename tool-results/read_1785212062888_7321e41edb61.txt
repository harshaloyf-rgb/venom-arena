  2175→// ---------------------------------------------------------------------------
  2176→// ReplayPlayer sub-component — renders last 30s of gameplay on a canvas

  2179→interface ReplayPlayerProps {
  2180→  frames: GameSnapshot[];
  2181→  myId: string;
  2182→  onClose: () => void;
  2183→}
  2184→
  2185→const REPLAY_SPEEDS = [0.25, 0.5, 1, 2] as const;
  2186→
  2187→function ReplayPlayer({ frames, myId, onClose }: ReplayPlayerProps) {
  2188→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  2189→  const rafRef = useRef<number | null>(null);
  2190→  const frameIdxRef = useRef(0);
  2191→  const playingRef = useRef(true);
  2192→  const speedRef = useRef(1);
  2193→  const zoomRef = useRef(0.8);
  2194→  const lastTimeRef = useRef(0);
  2195→  const [playing, setPlaying] = useState(true);
  2196→  const [speed, setSpeed] = useState(1);
  2197→  const [zoom, setZoom] = useState(0.8);
  2198→  const [frameIdx, setFrameIdx] = useState(0);
  2199→  const totalFrames = frames.length;
  2200→
  2201→  useEffect(() => {
  2202→    const canvas = canvasRef.current;
  2203→    if (!canvas || totalFrames === 0) return;
  2204→
  2205→    const ctx = canvas.getContext('2d');
  2206→    if (!ctx) return;
  2207→
  2208→    const w = canvas.clientWidth;
  2209→    const h = canvas.clientHeight;
  2210→    const dpr = window.devicePixelRatio || 1;
  2211→    canvas.width = w * dpr;
  2212→    canvas.height = h * dpr;
  2213→    ctx.scale(dpr, dpr);
  2214→
  2215→    const render = (now: number) => {
  2216→      if (!playingRef.current) {
  2217→        lastTimeRef.current = now;
  2218→        rafRef.current = requestAnimationFrame(render);
  2219→        return;
  2220→      }
  2221→
  2222→      const dt = now - lastTimeRef.current;
  2223→      const frameInterval = 50 / speedRef.current; // 50ms per frame at 1x
  2224→      if (dt >= frameInterval) {
  2225→        frameIdxRef.current = (frameIdxRef.current + 1) % totalFrames;
  2226→        lastTimeRef.current = now;
  2227→        setFrameIdx(frameIdxRef.current);
  2228→      }
  2229→
  2230→      const snap = frames[frameIdxRef.current];
  2231→      if (!snap) {
  2232→        rafRef.current = requestAnimationFrame(render);
  2233→        return;
  2234→      }
  2235→
