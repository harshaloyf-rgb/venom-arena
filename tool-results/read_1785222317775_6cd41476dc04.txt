   318→// ----------------------------------------------------------------------------
   319→// Dynamic Map Scaling (Online Mode)
   320→// ----------------------------------------------------------------------------
   321→export const MAP_MIN_RADIUS = 1500;  // radius when 1 player
   322→export const MAP_MAX_RADIUS = 8000;  // radius when 1000 players (increased for density)
   323→export const MAP_BREATH_AMPLITUDE = 40;  // breathing oscillation
   324→export const MAP_BREATH_CYCLE_MS = 10000;
   325→export const MAX_ARENA_PLAYERS = 1000;
   326→
   327→/** Compute dynamic map radius based on real player count. */
   328→export function getDynamicMapRadius(realPlayerCount: number, elapsedMs?: number): number {
   329→  const minP = 1;
   330→  const maxP = MAX_ARENA_PLAYERS;
   331→  const count = Math.max(minP, Math.min(maxP, realPlayerCount));
   332→  // sqrt scaling: 1 player -> 1500, ~31 players -> ~3000, 1000 players -> 5000
   333→  const baseRadius = MAP_MIN_RADIUS + (MAP_MAX_RADIUS - MAP_MIN_RADIUS) * Math.sqrt((count - 1) / (maxP - 1));
   334→  // Add breathing
   335→  if (elapsedMs !== undefined) {
   336→    const cycle = (elapsedMs % MAP_BREATH_CYCLE_MS) / MAP_BREATH_CYCLE_MS;
   337→    return baseRadius + Math.sin(cycle * Math.PI * 2) * MAP_BREATH_AMPLITUDE;
   338→  }
   339→  return baseRadius;
   340→}
