   176→// ----------------------------------------------------------------------------
   177→// Spatial hash grid — slimmed-down client port of the server's grid.
   178→// Items are bucketed into square cells; queries return a deduplicated Map.
   179→// ----------------------------------------------------------------------------
   180→
   181→class SpatialHashGrid {
   182→  private readonly cellSize: number;
   183→  private readonly cells: Map<string, Map<string, GridItem>> = new Map();
   184→
   185→  constructor(cellSize = 120) {
   186→    this.cellSize = cellSize;
   187→  }
   188→
   189→  private key(cx: number, cy: number): string {
   190→    return cx + ':' + cy;
   191→  }
   192→
   193→  clear(): void {
   194→    this.cells.clear();
   195→  }
   196→
   197→  insert(item: GridItem): void {
   198→    const minCx = Math.floor((item.x - item.radius) / this.cellSize);
   199→    const maxCx = Math.floor((item.x + item.radius) / this.cellSize);
   200→    const minCy = Math.floor((item.y - item.radius) / this.cellSize);
   201→    const maxCy = Math.floor((item.y + item.radius) / this.cellSize);
   202→    for (let cx = minCx; cx <= maxCx; cx++) {
   203→      for (let cy = minCy; cy <= maxCy; cy++) {
   204→        const k = this.key(cx, cy);
   205→        let bucket = this.cells.get(k);
   206→        if (!bucket) {
   207→          bucket = new Map();
   208→          this.cells.set(k, bucket);
   209→        }
   210→        bucket.set(item.id, item);
   211→      }
   212→    }
   213→  }
   214→
   215→  queryRadius(x: number, y: number, r: number): Map<string, GridItem> {
   216→    const out = new Map<string, GridItem>();
   217→    const minCx = Math.floor((x - r) / this.cellSize);
   218→    const maxCx = Math.floor((x + r) / this.cellSize);
   219→    const minCy = Math.floor((y - r) / this.cellSize);
   220→    const maxCy = Math.floor((y + r) / this.cellSize);
   221→    for (let cx = minCx; cx <= maxCx; cx++) {
   222→      for (let cy = minCy; cy <= maxCy; cy++) {
   223→        const bucket = this.cells.get(this.key(cx, cy));
   224→        if (!bucket) continue;
   225→        for (const [id, item] of bucket) {
   226→          if (!out.has(id)) out.set(id, item);
   227→        }
   228→      }
   229→    }
   230→    return out;
   231→  }
   232→}

   294→function angularDelta(current: number, desired: number): number {
   295→  let diff = desired - current;
   296→  while (diff > Math.PI) diff -= 2 * Math.PI;
   297→  while (diff < -Math.PI) diff += 2 * Math.PI;
   298→  return diff;
   299→}
   300→
   301→function turnToward(current: number, desired: number, maxStep: number): number {
   302→  const diff = angularDelta(current, desired);
   303→  if (Math.abs(diff) <= maxStep) return desired;
   304→  return current + Math.sign(diff) * maxStep;
   305→}

   307→function dist(ax: number, ay: number, bx: number, by: number): number {
   308→  return Math.hypot(ax - bx, ay - by);
   309→}
   310→
   311→/** Random point within a circle of maxR around center. */
   312→function randomPointInCircle(cx: number, cy: number, maxR: number): Vec2 {
   313→  const r = Math.sqrt(Math.random()) * maxR;
   314→  const theta = Math.random() * Math.PI * 2;
   315→  return {
   316→    x: cx + Math.cos(theta) * r,
   317→    y: cy + Math.sin(theta) * r,
   318→  };
   319→}
   320→
   321→function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   322→  const pts: Vec2[] = [];
   323→  for (let i = 0; i < length; i++) {
   324→    pts.push({
   325→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   326→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   327→    });
   328→  }
   329→  return pts;
   330→}

   336→/** Pick a weighted random FoodOrbConfig. */
   337→function weightedRandomOrb(): FoodOrbConfig {
   338→  let r = Math.random();
   339→  for (let i = 0; i < ALL_FOOD_ORBS.length; i++) {
   340→    r -= FOOD_ORB_WEIGHTS[i];
   341→    if (r <= 0) return ALL_FOOD_ORBS[i];
   342→  }
   343→  return ALL_FOOD_ORBS[0]; // fallback to small
   344→}
   345→
   346→/** Create a Food object at a random position around the player. */
   347→function createFoodOrb(idPrefix: string, idCounter: { value: number }, cx: number, cy: number, farSpawn: boolean): Food {
   348→  const orb = weightedRandomOrb();
   349→  const radius = farSpawn ? FOOD_SPAWN_RADIUS_FAR : FOOD_SPAWN_RADIUS_NEAR;
   350→  const pos = randomPointInCircle(cx, cy, radius);
   351→  return {
   352→    id: `${idPrefix}-food-${idCounter.value++}`,
   353→    x: pos.x,
   354→    y: pos.y,
   355→    size: orb.radius,
   356→    value: orb.value,
   357→    orbSize: orb.size,
   358→    color: orb.color,
   359→    glowColor: orb.glowColor,
   360→  };
   361→}

   385→
   386→function computeDeathFoodDrop(
   387→  totalScore: number,
   388→  deathX: number,
   389→  deathY: number,
   390→  idPrefix: string,
   391→  idCounter: { value: number },
   392→): Food[] {
   393→  const result: Food[] = [];
   394→  let remaining = Math.max(0, totalScore);
   395→
   396→  // Large orbs (value 5)
   397→  const largeOrb = ALL_FOOD_ORBS[2]; // FOOD_ORB_LARGE
   398→  const largeCount = Math.floor(remaining / largeOrb.value);
   399→  for (let i = 0; i < largeCount; i++) {
   400→    const offset = randomPointInCircle(deathX, deathY, 30 + largeCount * 0.5);
   401→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, largeOrb));
   402→  }
   403→  remaining -= largeCount * largeOrb.value;
   404→
   405→  // Medium orbs (value 3)
   406→  const mediumOrb = ALL_FOOD_ORBS[1]; // FOOD_ORB_MEDIUM
   407→  const mediumCount = Math.floor(remaining / mediumOrb.value);
   408→  for (let i = 0; i < mediumCount; i++) {
   409→    const offset = randomPointInCircle(deathX, deathY, 30 + mediumCount * 0.5);
   410→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, mediumOrb));
   411→  }
   412→  remaining -= mediumCount * mediumOrb.value;
   413→
   414→  // Small orbs (value 1) for the remainder
   415→  const smallOrb = ALL_FOOD_ORBS[0]; // FOOD_ORB_SMALL
   416→  for (let i = 0; i < remaining; i++) {
   417→    const offset = randomPointInCircle(deathX, deathY, 30 + remaining * 0.5);
   418→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, smallOrb));
   419→  }
   420→
   421→  return result;
   422→}

   594→  }

   790→  private onTouchStart(e: TouchEvent): void {
   791→    if (this.joystick) return;
   792→    const t = this.findJoystickTouch(e.touches);
   793→    if (!t) return;
   794→    e.preventDefault();
   795→    const rect = this.canvas.getBoundingClientRect();
   796→    this.joystick = {
   797→      active: true,
   798→      pointerId: t.identifier,
   799→      originX: t.clientX - rect.left,
   800→      originY: t.clientY - rect.top,
   801→      curX: t.clientX - rect.left,
   802→      curY: t.clientY - rect.top,
   803→    };
   804→  }
   805→
   806→  private onTouchMove(e: TouchEvent): void {
   807→    const js = this.joystick;
   808→    if (!js) return;
   809→    let t: Touch | null = null;
   810→    for (let i = 0; i < e.touches.length; i++) {
   811→      if (e.touches[i].identifier === js.pointerId) {
   812→        t = e.touches[i];
   813→        break;
   814→      }
   815→    }
   816→    if (!t) return;
   817→    e.preventDefault();
   818→    const rect = this.canvas.getBoundingClientRect();
   819→    js.curX = t.clientX - rect.left;
   820→    js.curY = t.clientY - rect.top;
   821→    const dx = js.curX - js.originX;
   822→    const dy = js.curY - js.originY;
   823→    const d = Math.hypot(dx, dy);
   824→    const mag = Math.min(1, d / JOYSTICK_MAX_RADIUS_PX);
   825→    if (mag > JOYSTICK_DEADZONE) {
   826→      this.touchAngle = Math.atan2(dy, dx);
   827→      this.touchBoost = mag > JOYSTICK_BOOST_MAGNITUDE;
   828→    } else {
   829→      this.touchAngle = null;
   830→      this.touchBoost = false;
   831→    }
   832→  }
   833→
   834→  private onTouchEnd(e: TouchEvent): void {
   835→    const js = this.joystick;
   836→    if (!js) return;
   837→    let stillActive = false;
   838→    for (let i = 0; i < e.touches.length; i++) {
   839→      if (e.touches[i].identifier === js.pointerId) {
   840→        stillActive = true;
   841→        break;
   842→      }
   843→    }
   844→    if (!stillActive) {
   845→      this.joystick = null;
   846→      this.touchAngle = null;
   847→      this.touchBoost = false;
   848→    }
   849→  }
   850→
   851→  private findJoystickTouch(touches: TouchList): Touch | null {
   852→    const rect = this.canvas.getBoundingClientRect();
   853→    for (let i = 0; i < touches.length; i++) {
   854→      const t = touches[i];
   855→      const x = t.clientX - rect.left;
   856→      const y = t.clientY - rect.top;
   857→      if (x < rect.width / 2 && y > rect.height / 2) {
   858→        return t;
   859→      }
   860→    }
   861→    return null;
   862→  }

   971→  // Main rAF loop
   972→  // --------------------------------------------------------------------------
   973→
   974→  private frame = (now: number): void => {
   975→    if (this.stopped) return;
   976→    this.rafId = requestAnimationFrame(this.frame);
   977→    const dt = now - this.lastFrameTime;
   978→    this.lastFrameTime = now;
   979→
   980→    this.updateFps(now);
   981→    this.updateParticles(dt);
   982→
   983→    // Physics ticks at ~TICK_MS (30 Hz) cadence.
   984→    if (this.state === 'playing' || this.isPostDeathRecording) {
   985→      this.accumulator += dt;
   986→      let safety = 4;
   987→      while (this.accumulator >= TICK_MS && safety > 0) {
   988→        this.tickPhysics(now);
   989→        this.accumulator -= TICK_MS;
   990→        safety--;
   991→      }
   992→      if (safety === 0) this.accumulator = 0;
   993→    }
   994→
   995→    // Post-death recording: count down ticks
   996→    if (this.isPostDeathRecording) {
   997→      this.postDeathTicksRemaining--;
   998→      if (this.postDeathTicksRemaining <= 0) {
   999→        this.finishPostDeathRecording();
  1000→      }
  1001→    }
  1002→
  1003→    this.render(now);
  1004→    this.updateHUD();
  1005→  };
  1006→
  1007→  private accumulator: number = 0;
  1008→
  1009→  // --------------------------------------------------------------------------
  1010→  // Physics tick (one server-tick equivalent)
  1011→  // --------------------------------------------------------------------------
  1012→
  1013→  private tickPhysics(now: number): void {
  1014→    this.tick++;
  1015→    const p = this.player;
  1016→    if (!p) return;
  1017→
  1018→    // During post-death recording, run simplified bot-only physics
  1019→    if (this.isPostDeathRecording) {
  1020→      this.tickPostDeathPhysics(now);
  1021→      this.captureReplaySnapshot();
  1022→      return;
  1023→    }
  1024→
  1025→    if (p.isDead) return;
  1026→
  1027→    // 1) Compute player desired angle + boost.
  1028→    const { angle, boost } = this.computePlayerInput();
  1029→    if (angle !== null) p.desiredAngle = angle;
  1030→    p.wantsBoost = boost;
  1031→    p.isBoosting = boost && p.points.length > BOOST_MIN_LENGTH + INITIAL_BODY_LENGTH - 20;
  1032→
  1033→    // Extraction progress (3-second hold).
  1034→    if (p.isExtracting) {
  1035→      p.extractionProgress += TICK_MS;
  1036→      if (p.extractionProgress >= EXTRACT_DURATION_MS) {
  1037→        this.finishExtract();
  1038→        return;
  1039→      }
  1040→    }
  1041→
  1042→    // 2) Move player.
  1043→    this.tickSnakeMovement(p, p.desiredAngle, p.wantsBoost);
  1044→
  1045→    // 3) Bots: think + move.
  1046→    for (const bot of this.bots.values()) {
  1047→      this.tickBot(bot, now);
  1048→    }
  1049→
  1050→    // Process boost food drops (from tail shedding)
  1051→    if (this.boostDropQueue.length > 0) {
  1052→      for (const drop of this.boostDropQueue) {
  1053→        this.foods.push({
  1054→          id: `food-${this.arena.id}-${this.idCounterObj.value++}`,
  1055→          x: drop.x,
  1056→          y: drop.y,
  1057→          size: 3,
  1058→          value: 1,
  1059→          isStarChip: false,
  1060→          color: '#34d399',
  1061→          glowColor: '#10b981',
  1062→          orbSize: 'small',
  1063→        });
  1064→      }
  1065→      this.boostDropQueue.length = 0;
  1066→    }
  1067→
  1068→    // 4) Build spatial grid for collision + food queries.
  1069→    this.grid.clear();
  1070→    this.insertSnakeIntoGrid(p);
  1071→    for (const bot of this.bots.values()) {
  1072→      if (!bot.isDead) this.insertSnakeIntoGrid(bot);
  1073→    }
  1074→    for (const food of this.foods) {
  1075→      this.grid.insert({
  1076→        id: food.id,
  1077→        kind: 'food',
  1078→        x: food.x,
  1079→        y: food.y,
  1080→        radius: food.size,
  1081→        value: food.value,
  1082→        foodRef: food,
  1083→      });
  1084→    }
  1085→
  1086→    // 5) Eat food (player + bots).
  1087→    this.eatFood();
  1088→
  1089→    // 6) Collision detection (body collision + head-on collision). NO wall death.
  1090→    const deaths = this.detectCollisions(now);
  1091→
  1092→    // 7) Apply deaths + drop food.
  1093→    let playerDied = false;
  1094→    const newDropFoods: Food[] = [];
  1095→    for (const d of deaths) {
  1096→      if (d.deadId === p.id) {
  1097→        playerDied = true;
  1098→        // Drop player's food
  1099→        const playerTotal = INITIAL_SPAWN_SCORE + p.score;
  1100→        newDropFoods.push(...computeDeathFoodDrop(playerTotal, p.points[0].x, p.points[0].y, this.arena.id, this.idCounterObj));
  1101→        continue;
  1102→      }
  1103→      const bot = this.bots.get(d.deadId);
  1104→      if (bot) {
  1105→        // Drop bot's food
  1106→        const botTotal = INITIAL_SPAWN_SCORE + bot.score;
  1107→        newDropFoods.push(...computeDeathFoodDrop(botTotal, bot.points[0].x, bot.points[0].y, this.arena.id, this.idCounterObj));
  1108→        bot.isDead = true;
  1109→        // Credit kill to the killer if it's the player.
  1110→        if (d.killerId === p.id) {
  1111→          p.kills++;
  1112→        }
  1113→      }
  1114→    }
  1115→
  1116→    // Add dropped food to the world.
  1117→    if (newDropFoods.length > 0) {
  1118→      this.foods.push(...newDropFoods);
  1119→    }
  1120→
  1121→    // 8) Remove dead bots and respawn to maintain count.
  1122→    if (deaths.length > 0) {
  1123→      const toRemove: string[] = [];
  1124→      for (const [id, bot] of this.bots) {
  1125→        if (bot.isDead) toRemove.push(id);
  1126→      }
  1127→      for (const id of toRemove) {
  1128→        this.bots.delete(id);
  1129→      }
  1130→    }
  1131→    // Respawn bots near the player to maintain count.
  1132→    const pHead = p.points[0];
  1133→    while (this.bots.size < this.arena.botsCount) {
  1134→      const bot = this.spawnBot(pHead.x, pHead.y);
  1135→      // Ensure bots spawn in a safe area (far from player AND other bots)
  1136→      let safe = false;
  1137→      for (let attempt = 0; attempt < 20; attempt++) {
  1138→        const spawnPt = bot.points[0];
  1139→        const distFromPlayer = dist(spawnPt.x, spawnPt.y, pHead.x, pHead.y);
  1140→        if (distFromPlayer < 300) {
  1141→          // Re-spawn further from player
  1142→          const far = randomPointInCircle(pHead.x, pHead.y, BOT_SPAWN_RADIUS);
  1143→          bot.points = initialBody(far.x, far.y, bot.angle, INITIAL_BODY_LENGTH);
  1144→          continue;
  1145→        }
  1146→        // Also check distance from other bots
  1147→        let tooClose = false;
  1148→        for (const other of this.bots.values()) {
  1149→          if (other.isDead || other.points.length === 0) continue;
  1150→          if (dist(spawnPt.x, spawnPt.y, other.points[0].x, other.points[0].y) < 300) {
  1151→            tooClose = true;
  1152→            break;
  1153→          }
  1154→        }
  1155→        if (tooClose) {
  1156→          const far = randomPointInCircle(pHead.x, pHead.y, BOT_SPAWN_RADIUS);
  1157→          bot.points = initialBody(far.x, far.y, bot.angle, INITIAL_BODY_LENGTH);
  1158→          continue;
  1159→        }
  1160→        safe = true;
  1161→        break;
  1162→      }
  1163→      this.bots.set(bot.id, bot);
  1164→    }
  1165→
  1166→    // 9) Player death → enter post-death recording (15s of continued simulation).
  1167→    if (playerDied && !this.isPostDeathRecording) {
  1168→      p.isDead = true;
  1169→      this.enterPostDeathRecording();
  1170→      // Don't return — continue to replenish food, expire chat, update camera
  1171→    }
  1172→
  1173→    // 10) Replenish food toward target (spawn around player).
  1174→    this.replenishFood();
  1175→
  1176→    // 11) Expire chat bubbles.
  1177→    this.expireChat(now);
  1178→
  1179→    // 12) Update camera target.
  1180→    this.updateCamera();
  1181→
  1182→    // 13) Capture replay snapshot.
  1183→    this.captureReplaySnapshot();
  1184→  }

  1188→  // --------------------------------------------------------------------------

  1190→  private tickSnakeMovement(snake: SnakeBase, desiredAngle: number, wantsBoost: boolean): void {
  1191→    if (snake.points.length === 0 || snake.isDead) return;
  1192→
  1193→    // Turn rate: max(0.045, 0.15 - totalScore*0.0006).
  1194→    const totalScore = INITIAL_SPAWN_SCORE + snake.score;
  1195→    const turnRate = Math.max(TURN_MIN, TURN_BASE - totalScore * TURN_SCORE_FACTOR);
  1196→    snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
  1197→
  1198→    // Speed.
  1199→    let speed = BASE_SPEED;
  1200→    if (snake.isExtracting) {
  1201→      speed = EXTRACT_GLIDE_SPEED;
  1202→    } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
  1203→      speed = BOOST_SPEED;
  1204→      snake.isBoosting = true;
  1205→      snake.boostFrameCounter++;
  1206→      if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
  1207→        snake.boostFrameCounter = 0;
  1208→        if (snake.points.length > BOOST_MIN_LENGTH) {
  1209→          // Record tail position BEFORE popping (for food drop)
  1210→          const tail = snake.points[snake.points.length - 1];
  1211→          this.boostDropQueue.push({
  1212→            x: tail.x,
  1213→            y: tail.y,
  1214→          });
  1215→          snake.points.pop();
  1216→          // Reduce food-score (score can go below 0 but body is clamped)
  1217→          snake.score = Math.max(0, snake.score - 1);
  1218→        }
  1219→      }
  1220→    } else {
  1221→      snake.isBoosting = false;
  1222→    }
  1223→
  1224→    // Move head.
  1225→    const head = snake.points[0];
  1226→    const nx = head.x + Math.cos(snake.angle) * speed;
  1227→    const ny = head.y + Math.sin(snake.angle) * speed;
  1228→    snake.points.unshift({ x: nx, y: ny });
  1229→
  1230→    // Grow / shrink body to target length based on SCORE.
  1231→    // targetLen = INITIAL_BODY_LENGTH + foodScore, clamped to [BOOST_MIN_LENGTH+1, MAX_BODY_LENGTH]
  1232→    const targetLen = Math.max(
  1233→      BOOST_MIN_LENGTH + 1,
  1234→      Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + snake.score),
  1235→    );
  1236→    while (snake.points.length > targetLen) snake.points.pop();
  1237→
  1238→    // Size: 8 + sqrt(totalScore) * 0.4.
  1239→    snake.size = SIZE_BASE + Math.sqrt(INITIAL_SPAWN_SCORE + snake.score) * SIZE_SCORE_FACTOR;
  1240→  }

  1246→  private tickBot(bot: BotSession, now: number): void {
  1247→    if (bot.points.length === 0 || bot.isDead) return;
  1248→
  1249→    if (now >= bot.nextThinkAt) {
  1250→      bot.nextThinkAt = now + BOT_THINK_INTERVAL_MS + Math.floor(Math.random() * BOT_THINK_JITTER_MS);
  1251→      const head = bot.points[0];
  1252→
  1253→      // --- Find nearest food ---
  1254→      let bestFood: GridItem | null = null;
  1255→      let bestFoodDist = Infinity;
  1256→      const foodQuery = this.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
  1257→      for (const item of foodQuery.values()) {
  1258→        if (item.kind !== 'food') continue;
  1259→        if ((item.value ?? 0) <= 0) continue;
  1260→        const d = dist(head.x, head.y, item.x, item.y);
  1261→        if (d < bestFoodDist) {
  1262→          bestFoodDist = d;
  1263→          bestFood = item;
  1264→        }
  1265→      }
  1266→
  1267→      // --- Find nearest threat (foreign body segment) ---
  1268→      let threatX = 0;
  1269→      let threatY = 0;
  1270→      let threatDist = Infinity;
  1271→      const threatQuery = this.grid.queryRadius(head.x, head.y, BOT_THREAT_SCAN_RADIUS);
  1272→      for (const item of threatQuery.values()) {
  1273→        if (item.kind !== 'segment') continue;
  1274→        if (item.snakeId === bot.id) continue;
  1275→        if (item.segIdx === 0) continue;
  1276→        const d = dist(head.x, head.y, item.x, item.y);
  1277→        if (d < threatDist) {
  1278→          threatDist = d;
  1279→          threatX = item.x;
  1280→          threatY = item.y;
  1281→        }
  1282→      }
  1283→
  1284→      let desired: number;
  1285→      const shouldFlee =
  1286→        threatDist < 140 &&
  1287→        (bot.personality === 'coward' ||
  1288→          bot.personality === 'extractor' ||
  1289→          (bot.personality === 'opportunist' && threatDist < 90));
  1290→
  1291→      // --- Predictive evasion against the human player ---
  1292→      const p = this.player;
  1293→      if (p && !p.isDead && p.points.length > 0 && now >= p.spawnProtectedUntil) {
  1294→        const playerHead = p.points[0];
  1295→        const playerDist = dist(head.x, head.y, playerHead.x, playerHead.y);
  1296→        if (playerDist < BOT_EVADE_RADIUS * 2) {
  1297→          // Predict where the player will be in N ticks.
  1298→          const predictedX = playerHead.x + Math.cos(p.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1299→          const predictedY = playerHead.y + Math.sin(p.angle) * BOT_PREDICT_AHEAD_TICKS;
  1300→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1301→
  1302→          // If the bot is on a collision course with the predicted position, evade.
  1303→          if (distToPredicted < (bot.size + p.size) * 3) {
  1304→            // Evade perpendicular to the player's heading.
  1305→            const evadeAngle = p.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1306→            desired = evadeAngle;
  1307→            bot.desiredAngle = desired;
  1308→            const wantsBoost = bot.personality === 'hunter' && bot.score > 10;
  1309→            this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1310→            return;
  1311→          }
  1312→        }
  1313→      }
  1314→
  1315→      if (shouldFlee) {
  1316→        desired = Math.atan2(head.y - threatY, head.x - threatX);
  1317→      } else if (bestFood) {
  1318→        desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
  1319→      } else {
  1320→        desired = bot.angle + (Math.random() - 0.5) * 0.4;
  1321→      }
  1322→
  1323→      // NO edge avoidance — infinite map, no boundaries.
  1324→
  1325→      bot.desiredAngle = desired;
  1326→    }
  1327→
  1328→    // Hunter personality occasionally boosts.
  1329→    const wantsBoost = bot.personality === 'hunter' && bot.score > 5 && Math.random() < 0.05;
  1330→    this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1331→  }

  1389→  private detectCollisions(now: number): { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] {
  1390→    const deaths: { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] = [];
  1391→    const seenDead = new Set<string>();
  1392→    const all: SnakeBase[] = [];
  1393→    if (this.player && !this.player.isDead) all.push(this.player);
  1394→    for (const bot of this.bots.values()) {
  1395→      if (!bot.isDead) all.push(bot);
  1396→    }
  1397→
  1398→    // --- Head-to-head collision (checked first, takes priority) ---
  1399→    const headOnChecked = new Set<string>();
  1400→    for (let i = 0; i < all.length; i++) {
  1401→      for (let j = i + 1; j < all.length; j++) {
  1402→        const a = all[i];
  1403→        const b = all[j];
  1404→        if (a.isDead || b.isDead) continue;
  1405→        if (now < a.spawnProtectedUntil || now < b.spawnProtectedUntil) continue;
  1406→        const pairKey = a.id < b.id ? `${a.id}:${b.id}` : `${b.id}:${a.id}`;
  1407→        if (headOnChecked.has(pairKey)) continue;
  1408→        headOnChecked.add(pairKey);
  1409→
  1410→        const ha = a.points[0];
  1411→        const hb = b.points[0];
  1412→        const d = dist(ha.x, ha.y, hb.x, hb.y);
  1413→        const hitDist = (a.size + b.size) * HEAD_ON_HIT_FACTOR;
  1414→        if (d >= hitDist) continue;
  1415→
  1416→        // Head-on collision! Determine winner based on rules.
  1417→        const aBoosting = a.isBoosting;
  1418→        const bBoosting = b.isBoosting;
  1419→        const aTotalScore = INITIAL_SPAWN_SCORE + a.score;
  1420→        const bTotalScore = INITIAL_SPAWN_SCORE + b.score;
  1421→
  1422→        let aDies = false;
  1423→        let bDies = false;
  1424→
  1425→        if (aTotalScore === bTotalScore) {
  1426→          // Tie — both die
  1427→          aDies = true;
  1428→          bDies = true;
  1429→        } else if (aBoosting && !bBoosting) {
  1430→          // A is boosting, B is not → A wins (boost advantage)
  1431→          bDies = true;
  1432→        } else if (bBoosting && !aBoosting) {
  1433→          // B is boosting, A is not → B wins (boost advantage)
  1434→          aDies = true;
  1435→        } else {
  1436→          // Both boosting or neither boosting → higher score wins
  1437→          if (aTotalScore > bTotalScore) {
  1438→            bDies = true;
  1439→          } else {
  1440→            aDies = true;
  1441→          }
  1442→        }
  1443→
  1444→        if (aDies && !seenDead.has(a.id)) {
  1445→          deaths.push({ deadId: a.id, killerId: b.id, cause: 'headon' });
  1446→          seenDead.add(a.id);
  1447→        }
  1448→        if (bDies && !seenDead.has(b.id)) {
  1449→          deaths.push({ deadId: b.id, killerId: a.id, cause: 'headon' });
  1450→          seenDead.add(b.id);
  1451→        }
  1452→      }
  1453→    }
  1454→
  1455→    // --- Body collision: head vs foreign non-head segment ---
  1456→    for (const snake of all) {
  1457→      if (snake.isDead) continue;
  1458→      if (snake.points.length === 0) continue;
  1459→      if (now < snake.spawnProtectedUntil) continue;
  1460→      if (seenDead.has(snake.id)) continue;
  1461→      const head = snake.points[0];
  1462→
  1463→      // NO wall collision — infinite map.
  1464→
  1465→      const queryR = snake.size + 30;
  1466→      const nearby = this.grid.queryRadius(head.x, head.y, queryR);
  1467→      for (const item of nearby.values()) {
  1468→        if (item.kind !== 'segment') continue;
  1469→        // Own-collision immunity — skip own segments.
  1470→        if (item.snakeId === snake.id) continue;
  1471→        // Skip head segments (handled by head-on collision above).
  1472→        if (item.segIdx === 0) continue;
  1473→        // Neck protection: skip first N segments behind the head (close-call safety)
  1474→        if (item.segIdx <= 5) continue;
  1475→        const d = dist(head.x, head.y, item.x, item.y);
  1476→        if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
  1477→          deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
  1478→          seenDead.add(snake.id);
  1479→          break;
  1480→        }
  1481→      }
  1482→    }
  1483→    return deaths;
  1484→  }
