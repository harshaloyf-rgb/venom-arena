     1→// ============================================================================
     2→// index.ts — Venom Arena Socket.IO game server (mini-service, port 3001).
     3→// ----------------------------------------------------------------------------
     4→// Server-authoritative multiplayer snake. The Next.js app (port 3000) talks
     5→// to this server via the Caddy gateway using io("/?XTransformPort=3001").
     6→//
     7→// Architectural fixes vs the OLD broken server (see worklog ANALYSIS-1):
     8→//  * Auth is mandatory — sockets without a valid JWT (verified via
     9→//    POST /api/match/verify) are disconnected at the middleware. No
    10→//    "auth optional, admit anyway" path.
    11→//  * Identity is the verify response only. We never trust any subsequent
    12→//    client-supplied userTag/name/color/skin.
    13→//  * One socket per userTag — a new connection kicks the prior socket.
    14→//  * Movement is server-authoritative — clients send only `angle`; the
    15→//    server computes position. Teleport is impossible.
    16→//  * Buy-in is deducted atomically by /api/match/join (Next.js + Prisma
    17→//    transaction). We never touch the DB directly.
    18→//  * Match results are reported exactly once via /api/match/result, guarded
    19→//    by a `matchSettling` flag to prevent double-credit on disconnect races.
    20→//  * Recursive setTimeout (not setInterval) for both tick and broadcast
    21→//    loops — a slow tick can't overlap the next.
    22→//  * Every tick is wrapped in try/catch; one bad snake never kills the loop.
    23→//  * Spatial hash grid (see spatial-grid.ts) keeps collision detection
    24→//    near-linear instead of O(n²).
    25→//  * Broadcasts happen at 20 Hz, not every tick, and snapshot points are
    26→//    downsampled to 60 to keep payloads small.
    27→//  * uncaughtException / unhandledRejection are logged, never fatal.
    28→//  * Graceful shutdown on SIGINT/SIGTERM broadcasts `server_shutdown` then exits.
    29→// ============================================================================

    92→type LogLevel = 'info' | 'warn' | 'error';
    93→function log(level: LogLevel, msg: string): void {
    94→  const now = new Date();
    95→  const hh = String(now.getHours()).padStart(2, '0');
    96→  const mm = String(now.getMinutes()).padStart(2, '0');
    97→  const ss = String(now.getSeconds()).padStart(2, '0');
    98→  const prefix = `[${hh}:${mm}:${ss}] [${level.toUpperCase()}]`;
    99→  if (level === 'error') {
   100→    process.stderr.write(`${prefix} ${msg}\n`);
   101→  } else {
   102→    process.stdout.write(`${prefix} ${msg}\n`);
   103→  }
   104→}

   131→/** Call POST /api/match/verify to validate the client's JWT and fetch its identity. */
   132→async function verifyToken(token: string): Promise<VerifyResponse> {
   133→  const controller = new AbortController();
   134→  const timer = setTimeout(() => controller.abort(), HTTP_TIMEOUT_MS);
   135→  try {
   136→    const res = await fetch(`${NEXT_APP_URL}/api/match/verify`, {
   137→      method: 'POST',
   138→      headers: { 'Content-Type': 'application/json', 'x-internal-secret': INTERNAL_SECRET },
   139→      body: JSON.stringify({ token }),
   140→      signal: controller.signal,
   141→    });
   142→    return (await res.json()) as VerifyResponse;
   143→  } finally {
   144→    clearTimeout(timer);
   145→  }
   146→}

   177→/** Call POST /api/match/result — credits/debits the player's account. */
   178→async function reportMatchResult(payload: ReportResultPayload): Promise<MatchResultResponse | null> {
   179→  const controller = new AbortController();
   180→  const timer = setTimeout(() => controller.abort(), HTTP_TIMEOUT_MS);
   181→  try {
   182→    const res = await fetch(`${NEXT_APP_URL}/api/match/result`, {
   183→      method: 'POST',
   184→      headers: { 'Content-Type': 'application/json', 'x-internal-secret': INTERNAL_SECRET },
   185→      body: JSON.stringify(payload),
   186→      signal: controller.signal,
   187→    });
   188→    if (!res.ok) {
   189→      log('error', `match/result HTTP ${res.status} for ${payload.userTag}`);
   190→      return null;
   191→    }
   192→    return (await res.json()) as MatchResultResponse;
   193→  } catch (err) {
   194→    log('error', `match/result fetch failed for ${payload.userTag}: ${(err as Error).message}`);
   195→    return null;
   196→  } finally {
   197→    clearTimeout(timer);
   198→  }
   199→}

   228→/** Max real players per arena shard before auto-creating a new instance. */
   229→const MAX_PLAYERS_PER_SHARD = 1000;
   230→
   231→/**
   232→ * Get-or-create an ArenaRoom for `arenaId`. If the primary shard is full
   233→ * (>= MAX_PLAYERS_PER_SHARD real players), find or create the next shard.
   234→ * Shards are keyed as `{arenaId}` (shard 0), `{arenaId}#2` (shard 1), etc.
   235→ */
   236→function getOrCreateRoom(arenaId: string): ArenaRoom | null {
   237→  const arena = getArenaById(arenaId);
   238→  if (!arena) return null;
   239→
   240→  // Try shard 0, then 1, 2, ... until we find one with capacity.
   241→  let shardIdx = 0;
   242→  while (true) {
   243→    const roomKey = shardIdx === 0 ? arenaId : `${arenaId}#${shardIdx + 1}`;
   244→    let room = rooms.get(roomKey);
   245→    if (room && room.players.size < MAX_PLAYERS_PER_SHARD) {
   246→      return room;
   247→    }
   248→    if (!room) {
   249→      // Create a new shard.
   250→      room = createArenaRoom(arena);
   251→      rooms.set(roomKey, room);
   252→      ensureBots(room);
   253→      log('info', `Created arena shard ${roomKey} with ${room.bots.size} bots (shard ${shardIdx})`);
   254→      return room;
   255→    }
   256→    shardIdx++;
   257→    // Safety: don't infinite-loop if somehow all shards are full (shouldn't happen).
   258→    if (shardIdx > 200) {
   259→      log('error', `Arena ${arenaId} has >200 shards, rejecting join`);
   260→      return null;
   261→    }
   262→  }
   263→}

   292→/**
   293→ * Settle a player's match: call /api/match/result, emit `match_result` to
   294→ * the player, and remove them from the room. Idempotent via the
   295→ * `matchSettling` flag — concurrent calls short-circuit.
   296→ */
   297→async function settleMatch(
   298→  room: ArenaRoom,
   299→  session: PlayerSession,
   300→  outcome: 'extract' | 'death',
   301→  killer?: SnakeBase,
   302→): Promise<void> {
   303→  if (session.matchSettling) return;
   304→  session.matchSettling = true;
   305→  session.isExtracting = false;
   306→  session.extractionProgress = 0;
   307→
   308→  const now = Date.now();
   309→  const durationSeconds = Math.max(0, Math.floor((now - session.joinedAt) / 1000));
   310→  const carriedChips = Math.max(0, Math.floor(session.carriedChips));
   311→  const kills = session.kills;
   312→  const killerTag = killer?.userTag;
   313→  const score = session.score;
   314→
   315→  // Dynamic commission: 0% if <=3 real players in arena, 35% if >=4.
   316→  const realPlayerCount = room.players.size;
   317→  const commissionRate = realPlayerCount <= 3 ? 0 : 0.35;
   318→  const commission = outcome === 'extract' ? Math.floor(carriedChips * commissionRate) : 0;
   319→  const bankedAmount = outcome === 'extract' ? (carriedChips - commission) : 0;
   320→
   321→  const result = await reportMatchResult({
   322→    userTag: session.identity.userTag,
   323→    arenaId: room.arena.id,
   324→    outcome,
   325→    carriedChips,
   326→    kills,
   327→    durationSeconds,
   328→    killerTag,
   329→    score,
   330→    bankedAmount,
   331→  });
   332→
   333→  const payload: MatchResultClientPayload = {
   334→    outcome,
   335→    arenaId: room.arena.id,
   336→    arenaName: room.arena.name,
   337→    chipsExtracted: outcome === 'extract' ? carriedChips : 0,
   338→    commission,
   339→    bankedAmount,
   340→    kills,
   341→    score,
   342→    xpGained: result?.xpGained ?? 0,
   343→    newLevel: result?.newLevel ?? session.identity.level,
   344→    newBankedChips: result?.newBankedChips ?? 0,
   345→    durationSeconds,
   346→  };
   347→
   348→  const socket = io.sockets.sockets.get(session.id);
   349→  if (socket) {
   350→    socket.emit('match_result', payload);
   351→    if (outcome === 'death') {
   352→      socket.emit('death', {
   353→        killerId: killer?.id,
   354→        killerName: killer?.name,
   355→        killerTag: killer?.userTag,
   356→        killerColor: killer?.color,
   357→      });
   358→    }
   359→  }
   360→
   361→  room.players.delete(session.id);
   362→  if (socket) {
   363→    const data = getSocketData(socket);
   364→    data.playerSession = undefined;
   365→  }
   366→  log('info', `${session.identity.userTag} ${outcome} in ${room.arena.id}: chips=${carriedChips} kills=${kills} dur=${durationSeconds}s`);
   367→}

   369→// ----------------------------------------------------------------------------
   370→// Tick loop (recursive setTimeout — a slow tick can't overlap the next)
   371→// ----------------------------------------------------------------------------
   372→
   373→function tickRoom(room: ArenaRoom, now: number): void {
   374→  room.tick++;
   375→
   376→  // 1) Rebuild spatial grid from scratch each tick.
   377→  room.grid.clear();
   378→
   379→  // 2) Insert all snake body segments (skip dead / empty).
   380→  for (const snake of collectAllSnakes(room)) {
   381→    if (snake.isDead) continue;
   382→    if (snake.points.length === 0) continue;
   383→    for (let i = 0; i < snake.points.length; i++) {
   384→      const p = snake.points[i];
   385→      room.grid.insert({
   386→        id: `${snake.id}:${i}`,
   387→        kind: 'segment',
   388→        x: p.x,
   389→        y: p.y,
   390→        radius: snake.size,
   391→        snakeId: snake.id,
   392→        segIdx: i,
   393→      });
   394→    }
   395→  }
   396→
   397→  // 3) Insert all food (skip already-eaten with value=0).
   398→  for (const food of room.foods) {
   399→    if (food.value <= 0) continue;
   400→    room.grid.insert({
   401→      id: food.id,
   402→      kind: 'food',
   403→      x: food.x,
   404→      y: food.y,
   405→      radius: food.size,
   406→      value: food.value,
   407→      isStarChip: food.isStarChip,
   408→      color: food.color,
   409→      foodRef: food, // <-- reference to the REAL food object so eatFood can zero it
   410→    });
   411→  }
   412→
   413→  // 4) Bot AI tick.
   414→  for (const bot of room.bots.values()) {
   415→    if (bot.isDead) continue;
   416→    try {
   417→      tickBot(bot, room, now);
   418→    } catch (err) {
   419→      log('error', `Bot tick error in ${room.arena.id}: ${(err as Error).message}`);
   420→    }
   421→  }
   422→
   423→  // 5) Player movement (server-authoritative — apply desired angle + speed).
   424→  for (const session of room.players.values()) {
   425→    if (session.isDead || session.matchSettling) continue;
   426→    try {
   427→      tickSnakeMovement(session, session.desiredAngle, session.wantsBoost);
   428→    } catch (err) {
   429→      log('error', `Player movement error: ${(err as Error).message}`);
   430→    }
   431→  }
   432→
   433→  // 6) Collision detection (body + head-on).
   434→  let deaths: PendingDeath[] = [];
   435→  try {
   436→    deaths = detectCollisions(room, now);
   437→    // Head-on collisions processed AFTER body collisions (already-dead snakes skipped).
   438→    const headOnDeaths = detectHeadOnCollisions(room, now);
   439→    deaths = deaths.concat(headOnDeaths);
   440→  } catch (err) {
   441→    log('error', `Collision detection error in ${room.arena.id}: ${(err as Error).message}`);
   442→  }
   443→
   444→  // 7) Apply deaths with CORRECT drop rules per spec:
   445→  //    Body/headOn collision: drop score orbs (all snakes) + 10 stars (real players only).
   446→  //    Wall death: drop 0 food (score destroyed) + 10 stars (real players only).
   447→  //    Bot in selfDestruct wall death: drop 0 food, 0 stars (vanish cleanly).
   448→  for (const death of deaths) {
   449→    const dead = room.players.get(death.deadId) || (room.bots.get(death.deadId) as SnakeBase | undefined);
   450→    if (!dead || dead.isDead) continue;
   451→    dead.isDead = true;
   452→
   453→    const isBotSelfDestruct = dead.isBot && (dead as BotSession).botState === 'selfDestruct';
   454→    const headX = dead.points[0]?.x ?? 0;
   455→    const headY = dead.points[0]?.y ?? 0;
   456→
   457→    if (death.cause === 'wall') {
   458→      // Wall death: NO food orbs (score destroyed).
   459→      // Stars: real players always get 10 stars; selfDestruct bots get 0.
   460→      if (dead.isPlayer && dead.carriedChips > 0) {
   461→        dropStarsAtDeath(room, headX, headY, dead.carriedChips);
   462→      }
   463→      // Bot wall death (selfDestruct): 0 food, 0 stars — vanish cleanly.
   464→    } else {
   465→      // Body or headOn collision: drop score orbs (sum = snake.score) + 10 stars (players only).
   466→      if (!isBotSelfDestruct) {
   467→        dropScoreOrbsAtBody(room, dead.points, dead.score, dead.color);
   468→        if (dead.isPlayer && dead.carriedChips > 0) {
   469→          dropStarsAtDeath(room, headX, headY, dead.carriedChips);
   470→        }
   471→      }
   472→    }
   473→
   474→    // Credit killer's kill counter (only if killer is alive). Bots don't track kills.
   475→    const killer = death.killerId && death.killerId !== 'wall'
   476→      ? (room.players.get(death.killerId) ?? room.bots.get(death.killerId))
   477→      : undefined;
   478→    if (killer && !killer.isDead && killer.isPlayer) {
   479→      (killer as PlayerSession).kills += 1;
   480→    }
   481→
   482→    if (dead.isPlayer) {
   483→      const session = dead as PlayerSession;
   484→      log('info', `${session.identity.userTag} died in ${room.arena.id} (cause=${death.cause}, killer=${death.killerId ?? 'none'})`);
   485→      // Settle asynchronously — never throw out of the tick loop.
   486→      settleMatch(room, session, 'death', death.cause === 'wall' ? undefined : killer).catch((err) => {
   487→        log('error', `settleMatch(death) failed: ${(err as Error).message}`);
   488→      });
   489→    } else {
   490→      // Bot dies: remove + respawn a fresh one to keep the bot count stable.
   491→      room.bots.delete(dead.id);
   492→      try {
   493→        const fresh = spawnNewBot(room);
   494→        if (fresh) room.bots.set(fresh.id, fresh);
   495→      } catch (err) {
   496→        log('error', `Bot respawn error: ${(err as Error).message}`);
   497→      }
   498→    }
   499→  }
   500→
   501→  // 8) Food collision (credits carriedChips, marks food as eaten via sentinel).
   502→  try {
   503→    eatFood(room);
   504→  } catch (err) {
   505→    log('error', `eatFood error in ${room.arena.id}: ${(err as Error).message}`);
   506→  }
   507→
   508→  // 9) Replenish food up to target.
   509→  try {
   510→    replenishFood(room);
   511→  } catch (err) {
   512→    log('error', `replenishFood error in ${room.arena.id}: ${(err as Error).message}`);
   513→  }
   514→
   515→  // 10) Extraction progress for players. NO zone check — extract anywhere.
   516→  for (const session of room.players.values()) {
   517→    if (session.isDead || session.matchSettling) continue;
   518→    if (!session.isExtracting) continue;
   519→
   520→    session.extractionProgress += TICK_MS;
   521→    const progress = Math.min(1, session.extractionProgress / EXTRACT_DURATION_MS);
   522→    const sock = io.sockets.sockets.get(session.id);
   523→    if (sock) sock.emit('extract_progress', { progress });
   524→
   525→    if (session.extractionProgress >= EXTRACT_DURATION_MS) {
   526→      // Extraction success — settle asynchronously.
   527→      settleMatch(room, session, 'extract').catch((err) => {
   528→        log('error', `settleMatch(extract) failed: ${(err as Error).message}`);
   529→      });
   530→    }
   531→  }
   532→
   533→  // 11) Recompute leader + expire chat bubbles.
   534→  try {
   535→    recomputeLeader(room);
   536→  } catch (err) {
   537→    log('error', `recomputeLeader error: ${(err as Error).message}`);
   538→  }
   539→  try {
   540→    expireChat(room, now);
   541→  } catch (err) {
   542→    log('error', `expireChat error: ${(err as Error).message}`);
   543→  }
   544→}
   545→
   546→function tickOnce(): void {
   547→  const now = Date.now();
   548→  try {
   549→    for (const room of rooms.values()) {
   550→      // Skip empty rooms (no players) for CPU savings — bots stay idle.
   551→      if (room.players.size === 0) continue;
   552→      try {
   553→        tickRoom(room, now);
   554→      } catch (err) {
   555→        log('error', `Tick error in arena ${room.arena.id}: ${(err as Error).message}`);
   556→      }
   557→    }
   558→  } catch (err) {
   559→    log('error', `Tick loop fatal: ${(err as Error).message}`);
   560→  } finally {
   561→    setTimeout(tickOnce, TICK_MS);
   562→  }
   563→}

   574→// ----------------------------------------------------------------------------
   575→// Broadcast loop (20 Hz — independent of tick rate)
   576→// ----------------------------------------------------------------------------
   577→
   578→function broadcastOnce(): void {
   579→  try {
   580→    for (const room of rooms.values()) {
   581→      if (room.players.size === 0) continue;
   582→      // Build one base snapshot per room, then customize yourRank per viewer.
   583→      for (const socketId of room.players.keys()) {
   584→        let snapshot: ReturnType<typeof buildSnapshot>;
   585→        try {
   586→          snapshot = buildSnapshot(room, socketId);
   587→        } catch (err) {
   588→          log('error', `Snapshot build error in ${room.arena.id}: ${(err as Error).message}`);
   589→          continue;
   590→        }
   591→        io.to(socketId).emit('snapshot', snapshot);
   592→      }
   593→    }
   594→  } catch (err) {
   595→    log('error', `Broadcast loop fatal: ${(err as Error).message}`);
   596→  } finally {
   597→    setTimeout(broadcastOnce, BROADCAST_MS);
   598→  }
   599→}

   601→// ----------------------------------------------------------------------------
   602→// Socket.IO auth middleware
   603→// ----------------------------------------------------------------------------
   604→
   605→io.use(async (socket, next) => {
   606→  const token = (socket.handshake.auth as { token?: unknown })?.token;
   607→  if (typeof token !== 'string' || token.length === 0) {
   608→    return next(new Error('No token provided'));
   609→  }
   610→  try {
   611→    const result = await verifyToken(token);
   612→    if (!result.ok || !result.player) {
   613→      return next(new Error(result.reason || 'invalid_token'));
   614→    }
   615→    getSocketData(socket).identity = result.player;
   616→    return next();
   617→  } catch (err) {
   618→    log('warn', `Verify fetch failed: ${(err as Error).message}`);
   619→    return next(new Error('verify_failed'));
   620→  }
   621→});

   623→// ----------------------------------------------------------------------------
   624→// Connection handler
   625→// ----------------------------------------------------------------------------
   626→
   627→io.on('connection', (socket: Socket) => {
   628→  const data = getSocketData(socket);
   629→  const identity = data.identity;
   630→  if (!identity) {
   631→    // Should never happen — middleware rejects unauthed sockets.
   632→    socket.disconnect(true);
   633→    return;
   634→  }
   635→
   636→  // One socket per userTag: kick any prior session.
   637→  const prior = userTagToSocket.get(identity.userTag);
   638→  if (prior && prior !== socket.id) {
   639→    const priorSock = io.sockets.sockets.get(prior);
   640→    if (priorSock) {
   641→      priorSock.emit('kicked', { reason: 'Another session opened for this account' });
   642→      priorSock.disconnect(true);
   643→    }
   644→  }
   645→  userTagToSocket.set(identity.userTag, socket.id);
   646→  log('info', `Socket connected: ${identity.userTag} (${socket.id})`);
   647→
   648→  // ----- join_arena -----
   649→  socket.on('join_arena', (payload: unknown) => {
   650→    void handleJoinArena(socket, payload);
   651→  });
   652→
   653→  // ----- input -----
   654→  socket.on('input', (payload: unknown) => {
   655→    handleInput(socket, payload);
   656→  });
   657→
   658→  // ----- extract -----
   659→  socket.on('extract', () => {
   660→    handleExtract(socket);
   661→  });
   662→
   663→  // ----- cancel_extract -----
   664→  socket.on('cancel_extract', () => {
   665→    handleCancelExtract(socket);
   666→  });
   667→
   668→  // ----- chat -----
   669→  socket.on('chat', (payload: unknown) => {
   670→    handleChat(socket, payload);
   671→  });
   672→
   673→  // ----- leave -----
   674→  socket.on('leave', () => {
   675→    handleLeave(socket, 'client requested');
   676→  });
   677→
   678→  // ----- disconnect -----
   679→  socket.on('disconnect', (reason: string) => {
   680→    handleLeave(socket, `disconnected (${reason})`);
   681→    // Clean up userTag → socket mapping (only if we still own it).
   682→    const ident = getSocketData(socket).identity;
   683→    if (ident && userTagToSocket.get(ident.userTag) === socket.id) {
   684→      userTagToSocket.delete(ident.userTag);
   685→    }
   686→    log('info', `Socket disconnected: ${ident?.userTag ?? 'unknown'} (${reason})`);
   687→  });
   688→
   689→  socket.on('error', (err: Error) => {
   690→    log('error', `Socket error (${socket.id}): ${err.message}`);
   691→  });
   692→});

   698→async function handleJoinArena(socket: Socket, payload: unknown): Promise<void> {
   699→  const data = getSocketData(socket);
   700→  const identity = data.identity;
   701→  if (!identity) return;
   702→
   703→  if (!payload || typeof payload !== 'object') {
   704→    socket.emit('join_error', { reason: 'invalid_arena' });
   705→    return;
   706→  }
   707→  const arenaId = String((payload as { arenaId?: unknown }).arenaId || '');
   708→  const arena = getArenaById(arenaId);
   709→  if (!arena) {
   710→    socket.emit('join_error', { reason: 'invalid_arena' });
   711→    return;
   712→  }
   713→
   714→  // Already in a match? Reject.
   715→  if (data.playerSession) {
   716→    socket.emit('join_error', { reason: 'already_in_match' });
   717→    return;
   718→  }
   719→
   720→  // Server-to-server: atomically deduct buyIn on the Next.js side.
   721→  let joinResult;
   722→  try {
   723→    joinResult = await joinMatch(identity.userTag, arenaId);
   724→  } catch (err) {
   725→    log('error', `joinMatch fetch failed: ${(err as Error).message}`);
   726→    socket.emit('join_error', { reason: 'invalid_arena' });
   727→    return;
   728→  }
   729→
   730→  if (!joinResult.ok) {
   731→    const reason = joinResult.reason;
   732→    if (reason === 'insufficient_chips') {
   733→      socket.emit('join_error', { reason: 'insufficient_chips' });
   734→      return;
   735→    }
   736→    if (reason === 'banned') {
   737→      socket.emit('join_error', { reason: 'banned' });
   738→      socket.emit('kicked', { reason: 'Account banned' });
   739→      socket.disconnect(true);
   740→      return;
   741→    }
   742→    socket.emit('join_error', { reason: 'invalid_arena' });
   743→    return;
   744→  }
   745→
   746→  // Spawn the player.
   747→  const room = getOrCreateRoom(arenaId);
   748→  if (!room) {
   749→    socket.emit('join_error', { reason: 'invalid_arena' });
   750→    return;
   751→  }
   752→
   753→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   754→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount));
   755→  const spawn = randomSpawnPoint(mapRadius - 200, room.mapCenterX, room.mapCenterY);
   756→  const angle = Math.random() * Math.PI * 2;
   757→  const session: PlayerSession = {
   758→    id: socket.id,
   759→    name: identity.name,
   760→    userTag: identity.userTag,
   761→    country: identity.country,
   762→    points: initialBody(spawn.x, spawn.y, angle, INITIAL_BODY_LENGTH),
   763→    angle,
   764→    size: SIZE_BASE + Math.sqrt(INITIAL_SPAWN_SCORE) * SIZE_SCORE_FACTOR,
   765→    color: identity.color,
   766→    secondaryColor: identity.secondaryColor,
   767→    isPlayer: true,
   768→    isBot: false,
   769→    carriedChips: 0,
   770→    score: INITIAL_SPAWN_SCORE,
   771→    boostFrameCounter: 0,
   772→    isExtracting: false,
   773→    extractionProgress: 0,
   774→    isDead: false,
   775→    spawnProtectedUntil: Date.now() + RESPAWN_INVULN_MS,
   776→    identity,
   777→    desiredAngle: angle,
   778→    wantsBoost: false,
   779→    kills: 0,
   780→    joinedAt: Date.now(),
   781→    lastInputAt: 0,
   782→    inputDropCount: 0,
   783→    lastChatAt: 0,
   784→    arenaId,
   785→    matchSettling: false,
   786→  };
   787→
   788→  room.players.set(socket.id, session);
   789→  data.playerSession = session;
   790→
   791→  socket.emit('joined', {
   792→    arenaId,
   793→    worldSize: WORLD_SIZE,
   794→    yourId: socket.id,
   795→  });
   796→
   797→  log('info', `${identity.userTag} joined ${arenaId} (buyIn=${arena.buyIn})`);
   798→}

   800→/** Validate + rate-limit a client input packet. Server applies position; client only sends angle. */
   801→function handleInput(socket: Socket, payload: unknown): void {
   802→  const session = getSocketData(socket).playerSession;
   803→  if (!session || session.isDead || session.matchSettling) return;
   804→
   805→  if (!payload || typeof payload !== 'object') return;
   806→  const obj = payload as { angle?: unknown; wantsBoost?: unknown };
   807→  const angleRaw = obj.angle;
   808→  const wantsBoostRaw = obj.wantsBoost;
   809→
   810→  if (typeof angleRaw !== 'number' || !Number.isFinite(angleRaw)) return;
   811→  if (typeof wantsBoostRaw !== 'boolean') return;
   812→
   813→  // Normalize angle into [0, 2π).
   814→  let angle = angleRaw;
   815→  while (angle < 0) angle += 2 * Math.PI;
   816→  while (angle >= 2 * Math.PI) angle -= 2 * Math.PI;
   817→
   818→  // Rate-limit: max MAX_SNAPSHOTS_PER_SECOND per second.
   819→  const now = Date.now();
   820→  const delta = now - session.lastInputAt;
   821→  if (delta < INPUT_MIN_INTERVAL_MS) {
   822→    session.inputDropCount++;
   823→    if (session.inputDropCount % 20 === 0) {
   824→      log('warn', `${session.identity.userTag} input flood: ${session.inputDropCount} packets dropped`);
   825→    }
   826→    return;
   827→  }
   828→  session.lastInputAt = now;
   829→  session.inputDropCount = 0;
   830→
   831→  session.desiredAngle = angle;
   832→  session.wantsBoost = wantsBoostRaw;
   833→}

   855→/** Validate + rate-limit a chat message; broadcast to the room. */
   856→function handleChat(socket: Socket, payload: unknown): void {
   857→  const session = getSocketData(socket).playerSession;
   858→  const identity = getSocketData(socket).identity;
   859→  if (!session || !identity) return;
   860→  if (!payload || typeof payload !== 'object') return;
   861→  const raw = (payload as { message?: unknown }).message;
   862→  if (typeof raw !== 'string') return;
   863→  const message = raw.trim().slice(0, CHAT_MAX_LEN);
   864→  if (message.length === 0) return;
   865→
   866→  const now = Date.now();
   867→  if (now - session.lastChatAt < CHAT_MIN_INTERVAL_MS) return;
   868→  session.lastChatAt = now;
   869→
   870→  session.chatMessage = message;
   871→  session.chatExpiry = now + 4000;
   872→
   873→  const room = session.arenaId ? rooms.get(session.arenaId) : null;
   874→  if (!room) return;
   875→  const chatPayload = {
   876→    senderId: session.id,
   877→    senderName: session.name,
   878→    senderTag: session.userTag,
   879→    message,
   880→  };
   881→  for (const socketId of room.players.keys()) {
   882→    io.to(socketId).emit('chat', chatPayload);
   883→  }
   884→}

   886→/** Remove a player from their arena; scatter carried chips as star-chips (no DB write). */
   887→function handleLeave(socket: Socket, reason: string): void {
   888→  const data = getSocketData(socket);
   889→  const session = data.playerSession;
   890→  if (!session) return;
   891→
   892→  const room = session.arenaId ? rooms.get(session.arenaId) : null;
   893→  if (room) {
   894→    // Forfeit carried chips as 10 star collectibles at death position.
   895→    if (!session.isDead && !session.matchSettling && session.carriedChips > 0) {
   896→      try {
   897→        dropStarsAtDeath(room, session.points[0]?.x ?? 0, session.points[0]?.y ?? 0, session.carriedChips);
   898→      } catch (err) {
   899→        log('error', `dropStarsAtDeath error on leave: ${(err as Error).message}`);
   900→      }
   901→    }
   902→    room.players.delete(socket.id);
   903→  }
   904→
   905→  data.playerSession = undefined;
   906→  log('info', `${session.identity.userTag} left arena ${session.arenaId ?? '?'} (${reason})`);
   907→}

   922→function gracefulShutdown(signal: string): void {
   923→  log('info', `${signal} received — broadcasting server_shutdown and exiting`);
   924→  try {
   925→    io.emit('server_shutdown', {});
   926→  } catch (err) {
   927→    log('error', `shutdown broadcast failed: ${(err as Error).message}`);
   928→  }
   929→  // Give sockets a moment to flush, then exit.
   930→  setTimeout(() => process.exit(0), 400);
   931→  // Hard exit fallback if flush hangs.
   932→  setTimeout(() => process.exit(1), 2000).unref();
   933→}
   934→
   935→process.on('SIGINT', () => gracefulShutdown('SIGINT'));
   936→process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
   937→// Log SIGHUP so we can diagnose if the sandbox is hanging up the process.
   938→process.on('SIGHUP', () => {
   939→  log('warn', 'SIGHUP received — ignoring (nohup-style)');
   940→});
   941→// Log SIGPIPE so we don't die silently if a socket write fails on a closed pipe.
   942→process.on('SIGPIPE', () => {
   943→  log('warn', 'SIGPIPE received — ignoring');
   944→});

   946→// Diagnostic: log when the event loop is about to drain. If bun decides to

   949→  log('warn', `beforeExit code=${code} — event loop empty, process will exit`);
