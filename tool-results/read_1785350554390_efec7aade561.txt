    16→//  * **Opacity layering** — larger snakes fade when a smaller snake is near.
    17→//  * **Score model** — INITIAL_SPAWN_SCORE + food eaten, body from score.

   197→// ----------------------------------------------------------------------------
   198→// Spatial hash grid — slimmed-down client port of the server's grid.
   199→// Items are bucketed into square cells; queries return a deduplicated Map.
   200→// ----------------------------------------------------------------------------
   201→
   202→class SpatialHashGrid {
   203→  private readonly cellSize: number;
   204→  private readonly cells: Map<string, Map<string, GridItem>> = new Map();
   205→
   206→  constructor(cellSize = 120) {
   207→    this.cellSize = cellSize;
   208→  }
   209→
   210→  private key(cx: number, cy: number): string {
   211→    return cx + ':' + cy;
   212→  }
   213→
   214→  clear(): void {
   215→    this.cells.clear();
   216→  }
   217→
   218→  insert(item: GridItem): void {
   219→    const minCx = Math.floor((item.x - item.radius) / this.cellSize);
   220→    const maxCx = Math.floor((item.x + item.radius) / this.cellSize);
   221→    const minCy = Math.floor((item.y - item.radius) / this.cellSize);
   222→    const maxCy = Math.floor((item.y + item.radius) / this.cellSize);
   223→    for (let cx = minCx; cx <= maxCx; cx++) {
   224→      for (let cy = minCy; cy <= maxCy; cy++) {
   225→        const k = this.key(cx, cy);
   226→        let bucket = this.cells.get(k);
   227→        if (!bucket) {
   228→          bucket = new Map();
   229→          this.cells.set(k, bucket);
   230→        }
   231→        bucket.set(item.id, item);
   232→      }
   233→    }
   234→  }
   235→
   236→  queryRadius(x: number, y: number, r: number): Map<string, GridItem> {
   237→    const out = new Map<string, GridItem>();
   238→    const minCx = Math.floor((x - r) / this.cellSize);
   239→    const maxCx = Math.floor((x + r) / this.cellSize);
   240→    const minCy = Math.floor((y - r) / this.cellSize);
   241→    const maxCy = Math.floor((y + r) / this.cellSize);
   242→    for (let cx = minCx; cx <= maxCx; cx++) {
   243→      for (let cy = minCy; cy <= maxCy; cy++) {
   244→        const bucket = this.cells.get(this.key(cx, cy));
   245→        if (!bucket) continue;
   246→        for (const [id, item] of bucket) {
   247→          if (!out.has(id)) out.set(id, item);
   248→        }
   249→      }
   250→    }
   251→    return out;
   252→  }
   253→}

   321→function angularDelta(current: number, desired: number): number {
   322→  let diff = desired - current;
   323→  while (diff > Math.PI) diff -= 2 * Math.PI;
   324→  while (diff < -Math.PI) diff += 2 * Math.PI;
   325→  return diff;
   326→}
   327→
   328→function turnToward(current: number, desired: number, maxStep: number): number {
   329→  const diff = angularDelta(current, desired);
   330→  if (Math.abs(diff) <= maxStep) return desired;
   331→  return current + Math.sign(diff) * maxStep;
   332→}

   334→function dist(ax: number, ay: number, bx: number, by: number): number {
   335→  return Math.hypot(ax - bx, ay - by);
   336→}
   337→
   338→/** Random point within a circle of maxR around center. */
   339→function randomPointInCircle(cx: number, cy: number, maxR: number): Vec2 {
   340→  const r = Math.sqrt(Math.random()) * maxR;
   341→  const theta = Math.random() * Math.PI * 2;
   342→  return {
   343→    x: cx + Math.cos(theta) * r,
   344→    y: cy + Math.sin(theta) * r,
   345→  };
   346→}

   348→function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   349→  const pts: Vec2[] = [];
   350→  for (let i = 0; i < length; i++) {
   351→    pts.push({
   352→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   353→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   354→    });
   355→  }
   356→  return pts;
   357→}

   363→/** Pick a weighted random FoodOrbConfig. */
   364→function weightedRandomOrb(): FoodOrbConfig {
   365→  let r = Math.random();
   366→  for (let i = 0; i < ALL_FOOD_ORBS.length; i++) {
   367→    r -= FOOD_ORB_WEIGHTS[i];
   368→    if (r <= 0) return ALL_FOOD_ORBS[i];
   369→  }
   370→  return ALL_FOOD_ORBS[0]; // fallback to small
   371→}

   375→  const orb = weightedRandomOrb();

   405→// Death food drop — greedily split score into Large(5)/Medium(3)/Small(1)
   406→// ----------------------------------------------------------------------------
   407→
   408→interface DeathDropResult {
   409→  foods: Food[];
   410→  /** IDs of dead snakes that should drop food. */
   411→}
   412→
   413→function computeDeathOrbs(totalScore: number): { small: number; medium: number; large: number } {
   414→  let remaining = totalScore;
   415→  const large = Math.floor(remaining / 5);
   416→  remaining -= large * 5;
   417→  const medium = Math.floor(remaining / 3);
   418→  remaining -= medium * 3;
   419→  const small = remaining;
   420→  return { small, medium, large };
   421→}
   422→
   423→function computeDeathFoodDrop(
   424→  totalScore: number,
   425→  bodyPoints: Vec2[],
   426→  idPrefix: string,
   427→  idCounter: { value: number },
   428→): Food[] {
   429→  const result: Food[] = [];
   430→  if (!bodyPoints || bodyPoints.length === 0 || totalScore <= 0) return result;
   431→
   432→  const { small, medium, large } = computeDeathOrbs(totalScore);
   433→  let orbIdx = 0;
   434→  const totalOrbs = small + medium + large;
   435→  if (totalOrbs === 0) return result;
   436→
   437→  // Build orb sequence
   438→  const orbSequence: Array<{ value: number; size: number; color: string; glowColor: string; orbSize: string }> = [];
   439→  for (let i = 0; i < large; i++) orbSequence.push({ value: 5, size: 8, color: '#f472b6', glowColor: '#ec4899', orbSize: 'large' });
   440→  for (let i = 0; i < medium; i++) orbSequence.push({ value: 3, size: 5, color: '#38bdf8', glowColor: '#0ea5e9', orbSize: 'medium' });
   441→  for (let i = 0; i < small; i++) orbSequence.push({ value: 1, size: 3, color: '#34d399', glowColor: '#10b981', orbSize: 'small' });
   442→
   443→  // Shuffle
   444→  for (let i = orbSequence.length - 1; i > 0; i--) {
   445→    const j = Math.floor(Math.random() * (i + 1));
   446→    [orbSequence[i], orbSequence[j]] = [orbSequence[j], orbSequence[i]];
   447→  }
   448→
   449→  // Distribute orbs evenly along the body
   450→  const scatter = 15;
   451→  for (const orb of orbSequence) {
   452→    const segIdx = Math.min(bodyPoints.length - 1, Math.floor((orbIdx / totalOrbs) * bodyPoints.length));
   453→    const pt = bodyPoints[segIdx];
   454→    result.push({
   455→      id: `${idPrefix}-death-${idCounter.value++}`,
   456→      x: pt.x + (Math.random() - 0.5) * scatter,
   457→      y: pt.y + (Math.random() - 0.5) * scatter,
   458→      size: orb.size,
   459→      value: orb.value,
   460→      orbSize: orb.orbSize as 'small' | 'medium' | 'large',
   461→      color: orb.color,
   462→      glowColor: orb.glowColor,
   463→    });
   464→    orbIdx++;
   465→  }
   466→
   467→  return result;
   468→}

   768→  /** Create 1000 virtual bot definitions with cheap positions spread around (cx, cy). */
   769→  private createVirtualBotPool(cx: number, cy: number): void {
   770→    this.virtualBots = [];
   771→    for (let i = 0; i < VIRTUAL_BOT_COUNT; i++) {
   772→      const baseName = BOT_NAMES[i % BOT_NAMES.length];
   773→      const name = i < BOT_NAMES.length ? baseName : `${baseName}-${Math.floor(i / BOT_NAMES.length) + 1}`;
   774→      const skin = BOT_SKINS[i % BOT_SKINS.length];
   775→      const personality = PERSONALITIES[i % PERSONALITIES.length];
   776→      const botId = `bot-${this.arena.id}-${i}`;
   777→      const initialScore = Math.floor(Math.random() * 80);
   778→
   779→      // Spread bots in a large ring. Use a mix of near and far positions.
   780→      const spawn = randomPointInCircle(cx, cy, VIRTUAL_WORLD_RADIUS);
   781→      const botAngle = Math.random() * Math.PI * 2;
   782→
   783→      this.virtualBots.push({
   784→        idx: i,
   785→        id: botId,
   786→        botId,
   787→        name,
   788→        personality,
   789→        color: skin.color,
   790→        secondaryColor: skin.secondaryColor,
   791→        initialScore,
   792→        x: spawn.x,
   793→        y: spawn.y,
   794→        angle: botAngle,
   795→        score: initialScore,
   796→        isActive: false,
   797→      });
   798→    }
   799→  }
   800→
   801→  /** Cheaply update all virtual bot positions (straight-line wander). Called every tick but is very lightweight. */
   802→  private updateVirtualBotPositions(): void {
   803→    for (let i = 0; i < this.virtualBots.length; i++) {
   804→      const vb = this.virtualBots[i];
   805→      if (vb.isActive) continue; // active bots are moved by full physics
   806→      // Random angle tweak occasionally (use simple condition instead of Math.random for speed)
   807→      if (((this.tick + i) * 7919) % 1000 < 20) { // deterministic pseudo-random
   808→        vb.angle += (((this.tick + i) * 3571) % 1000 - 500) * 0.0016;
   809→      }
   810→      vb.x += Math.cos(vb.angle) * VIRTUAL_BOT_SPEED;
   811→      vb.y += Math.sin(vb.angle) * VIRTUAL_BOT_SPEED;
   812→    }
   813→  }
   814→
   815→  /** Activate virtual bots near the player, deactivate those far away. */
   816→  private activateNearbyBots(): void {
   817→    const p = this.player;
   818→    if (!p || p.points.length === 0) return;
   819→    const px = p.points[0].x;
   820→    const py = p.points[0].y;
   821→
   822→    // 1) Deactivate active bots that moved too far from the player.
   823→    const toDeactivate: string[] = [];
   824→    for (const [id, bot] of this.bots) {
   825→      if (bot.points.length === 0) continue;
   826→      const head = bot.points[0];
   827→      const d = dist(head.x, head.y, px, py);
   828→      if (d > DEACTIVATION_RADIUS) {
   829→        // Save state back to virtual bot
   830→        const vb = this.virtualBots[bot.virtualIdx];
   831→        if (vb) {
   832→          vb.x = head.x;
   833→          vb.y = head.y;
   834→          vb.angle = bot.angle;
   835→          vb.score = bot.score;
   836→          vb.isActive = false;
   837→        }
   838→        toDeactivate.push(id);
   839→      }
   840→    }
   841→    for (const id of toDeactivate) {
   842→      this.bots.delete(id);
   843→    }
   844→
   845→    // 2) Activate virtual bots within range (up to MAX_ACTIVE_BOTS).
   846→    // Only scan virtual pool every 10 ticks to reduce overhead.
   847→    if (this.tick % 10 !== 0) return;
   848→    if (this.bots.size >= MAX_ACTIVE_BOTS) return;
   849→
   850→    // Quick activation: scan virtual bots, find inactive ones within range.
   851→    // No sorting needed — just activate up to the limit.
   852→    for (let i = 0; i < this.virtualBots.length && this.bots.size < MAX_ACTIVE_BOTS; i++) {
   853→      const vb = this.virtualBots[i];
   854→      if (vb.isActive) continue;
   855→      const d = dist(vb.x, vb.y, px, py);
   856→      if (d < ACTIVATION_RADIUS) {
   857→        const bot = this.createActiveBotFromVirtual(vb, px, py);
   858→        this.bots.set(bot.id, bot);
   859→        vb.isActive = true;
   860→      }
   861→    }
   862→  }

   952→  private onTouchStart(e: TouchEvent): void {
   953→    if (this.joystick) return;
   954→    const t = this.findJoystickTouch(e.touches);
   955→    if (!t) return;
   956→    e.preventDefault();
   957→    const rect = this.canvas.getBoundingClientRect();
   958→    this.joystick = {
   959→      active: true,
   960→      pointerId: t.identifier,
   961→      originX: t.clientX - rect.left,
   962→      originY: t.clientY - rect.top,
   963→      curX: t.clientX - rect.left,
   964→      curY: t.clientY - rect.top,
   965→    };
   966→  }
   967→
   968→  private onTouchMove(e: TouchEvent): void {
   969→    const js = this.joystick;
   970→    if (!js) return;
   971→    let t: Touch | null = null;
   972→    for (let i = 0; i < e.touches.length; i++) {
   973→      if (e.touches[i].identifier === js.pointerId) {
   974→        t = e.touches[i];
   975→        break;
   976→      }
   977→    }
   978→    if (!t) return;
   979→    e.preventDefault();
   980→    const rect = this.canvas.getBoundingClientRect();
   981→    js.curX = t.clientX - rect.left;
   982→    js.curY = t.clientY - rect.top;
   983→    const dx = js.curX - js.originX;
   984→    const dy = js.curY - js.originY;
   985→    const d = Math.hypot(dx, dy);
   986→    const mag = Math.min(1, d / JOYSTICK_MAX_RADIUS_PX);
   987→    if (mag > JOYSTICK_DEADZONE) {
   988→      this.touchAngle = Math.atan2(dy, dx);
   989→      this.touchBoost = mag > JOYSTICK_BOOST_MAGNITUDE;
   990→    } else {
   991→      this.touchAngle = null;
   992→      this.touchBoost = false;
   993→    }
   994→  }
   995→
   996→  private onTouchEnd(e: TouchEvent): void {
   997→    const js = this.joystick;
   998→    if (!js) return;
   999→    let stillActive = false;
  1000→    for (let i = 0; i < e.touches.length; i++) {
  1001→      if (e.touches[i].identifier === js.pointerId) {
  1002→        stillActive = true;
  1003→        break;
  1004→      }
  1005→    }
  1006→    if (!stillActive) {
  1007→      this.joystick = null;
  1008→      this.touchAngle = null;
  1009→      this.touchBoost = false;
  1010→    }
  1011→  }

  1013→  private findJoystickTouch(touches: TouchList): Touch | null {
  1014→    const rect = this.canvas.getBoundingClientRect();
  1015→    for (let i = 0; i < touches.length; i++) {
  1016→      const t = touches[i];
  1017→      const x = t.clientX - rect.left;
  1018→      const y = t.clientY - rect.top;
  1019→      if (x < rect.width / 2 && y > rect.height / 2) {
  1020→        return t;
  1021→      }
  1022→    }
  1023→    return null;
  1024→  }

  1026→  private onKeyDown(e: KeyboardEvent): void {
  1027→    const k = e.key.toLowerCase();
  1028→    if (
  1029→      k === 'arrowup' ||
  1030→      k === 'arrowdown' ||
  1031→      k === 'arrowleft' ||
  1032→      k === 'arrowright' ||
  1033→      k === ' ' ||
  1034→      k === 'spacebar'
  1035→    ) {
  1036→      e.preventDefault();
  1037→    }
  1038→    if (k === 'escape' && this.state !== 'playing') {
  1039→      this.handleExitToLobby();
  1040→      return;
  1041→    }
  1042→    if (k === 'e' && this.state === 'playing' && !this.extractHold) {
  1043→      this.extractHold = true;
  1044→      this.beginExtract();
  1045→    }
  1046→    if (this.state === 'playing' && ['1', '2', '3', '4', '5'].includes(k)) {
  1047→      const idx = parseInt(k, 10) - 1;
  1048→      if (idx >= 0 && idx < QUICK_EMOTES.length) {
  1049→        this.setPlayerChat(QUICK_EMOTES[idx]);
  1050→      }
  1051→    }
  1052→    const normalized = k === 'spacebar' ? ' ' : k;
  1053→    this.keys.add(normalized);
  1054→  }

  1059→      this.extractHold = false;
  1060→      this.cancelExtract();
  1061→    }
  1062→    const normalized = k === 'spacebar' ? ' ' : k;
  1063→    this.keys.delete(normalized);
  1064→  }

  1074→  }

  1091→    if (p.isDead) return;

  1094→    p.extractionProgress = 0;

  1136→  private frame = (now: number): void => {
  1137→    if (this.stopped) return;
  1138→    this.rafId = requestAnimationFrame(this.frame);
  1139→    const dt = now - this.lastFrameTime;
  1140→    this.lastFrameTime = now;
  1141→
  1142→    this.updateFps(now);
  1143→    this.updateParticles(dt);
  1144→
  1145→    // Physics ticks at ~TICK_MS (30 Hz) cadence.
  1146→    if (this.state === 'playing' || this.isPostDeathRecording) {
  1147→      this.accumulator += dt;
  1148→      let safety = 4;
  1149→      while (this.accumulator >= TICK_MS && safety > 0) {
  1150→        this.tickPhysics(now);
  1151→        this.accumulator -= TICK_MS;
  1152→        safety--;
  1153→      }
  1154→      if (safety === 0) this.accumulator = 0;
  1155→    }
  1156→
  1157→    // Post-death recording: count down ticks
  1158→    if (this.isPostDeathRecording) {
  1159→      this.postDeathTicksRemaining--;
  1160→      if (this.postDeathTicksRemaining <= 0) {
  1161→        this.finishPostDeathRecording();
  1162→      }
  1163→    }
  1164→
  1165→    this.render(now);
  1166→    this.updateHUD();
  1167→  };

  1174→
  1175→  private tickPhysics(now: number): void {
  1176→    this.tick++;
  1177→    const p = this.player;
  1178→    if (!p) return;
  1179→
  1180→    // During post-death recording, run simplified bot-only physics
  1181→    if (this.isPostDeathRecording) {
  1182→      this.tickPostDeathPhysics(now);
  1183→      this.captureReplaySnapshot();
  1184→      return;
  1185→    }
  1186→
  1187→    if (p.isDead) return;
  1188→
  1189→    // 1) Compute player desired angle + boost.
  1190→    const { angle, boost } = this.computePlayerInput();
  1191→    if (angle !== null) p.desiredAngle = angle;
  1192→    p.wantsBoost = boost;
  1193→    p.isBoosting = boost && p.points.length > BOOST_MIN_LENGTH + INITIAL_BODY_LENGTH - 20;
  1194→
  1195→    // Extraction progress (3-second hold).
  1196→    if (p.isExtracting) {
  1197→      p.extractionProgress += TICK_MS;
  1198→      if (p.extractionProgress >= EXTRACT_DURATION_MS) {
  1199→        this.finishExtract();
  1200→        return;
  1201→      }
  1202→    }
  1203→
  1204→    // 2) Move player.
  1205→    this.tickSnakeMovement(p, p.desiredAngle, p.wantsBoost);
  1206→
  1207→    // 3) Update virtual bot positions cheaply + manage activation/deactivation.
  1208→    this.updateVirtualBotPositions();
  1209→    this.activateNearbyBots();
  1210→
  1211→    // 4) Bots: think + move (only active bots).
  1212→    for (const bot of this.bots.values()) {
  1213→      this.tickBot(bot, now);
  1214→    }
  1215→
  1216→    // Process boost food drops (from tail shedding)
  1217→    if (this.boostDropQueue.length > 0) {
  1218→      for (const drop of this.boostDropQueue) {
  1219→        this.foods.push({
  1220→          id: `food-${this.arena.id}-${this.idCounterObj.value++}`,
  1221→          x: drop.x,
  1222→          y: drop.y,
  1223→          size: 3,
  1224→          value: 1,
  1225→          isStarChip: false,
  1226→          color: '#34d399',
  1227→          glowColor: '#10b981',
  1228→          orbSize: 'small',
  1229→        });
  1230→      }
  1231→      this.boostDropQueue.length = 0;
  1232→    }
  1233→
  1234→    // 4) Build spatial grid for collision + food queries.
  1235→    this.grid.clear();
  1236→    this.insertSnakeIntoGrid(p);
  1237→    for (const bot of this.bots.values()) {
  1238→      if (!bot.isDead) this.insertSnakeIntoGrid(bot);
  1239→    }
  1240→    for (const food of this.foods) {
  1241→      this.grid.insert({
  1242→        id: food.id,
  1243→        kind: 'food',
  1244→        x: food.x,
  1245→        y: food.y,
  1246→        radius: food.size,
  1247→        value: food.value,
  1248→        foodRef: food,
  1249→      });
  1250→    }
  1251→
  1252→    // 5) Eat food (player + bots).
  1253→    this.eatFood();
  1254→
  1255→    // 6) Collision detection (body collision + head-on collision). NO wall death.
  1256→    const deaths = this.detectCollisions(now);
  1257→
  1258→    // 7) Apply deaths + drop food.
  1259→    let playerDied = false;
  1260→    const newDropFoods: Food[] = [];
  1261→    for (const d of deaths) {
  1262→      if (d.deadId === p.id) {
  1263→        playerDied = true;
  1264→        // Drop player's food
  1265→        const playerTotal = INITIAL_SPAWN_SCORE + p.score;
  1266→        newDropFoods.push(...computeDeathFoodDrop(playerTotal, p.points, this.arena.id, this.idCounterObj));
  1267→        continue;
  1268→      }
  1269→      const bot = this.bots.get(d.deadId);
  1270→      if (bot) {
  1271→        // Drop bot's food
  1272→        const botTotal = INITIAL_SPAWN_SCORE + bot.score;
  1273→        newDropFoods.push(...computeDeathFoodDrop(botTotal, bot.points, this.arena.id, this.idCounterObj));
  1274→        bot.isDead = true;
  1275→        // Credit kill to the killer if it's the player.
  1276→        if (d.killerId === p.id) {
  1277→          p.kills++;
  1278→        }
  1279→      }
  1280→    }
  1281→
  1282→    // Add dropped food to the world.
  1283→    if (newDropFoods.length > 0) {
  1284→      this.foods.push(...newDropFoods);
  1285→    }
  1286→
  1287→    // 8) Remove dead bots and recycle their virtual bots.
  1288→    if (deaths.length > 0) {
  1289→      const pHead = p.points[0];
  1290→      const toRemove: string[] = [];
  1291→      for (const [id, bot] of this.bots) {
  1292→        if (bot.isDead) {
  1293→          // Recycle the virtual bot to a new random position
  1294→          const vb = this.virtualBots[bot.virtualIdx];
  1295→          if (vb) {
  1296→            this.respawnVirtualBot(vb, pHead.x, pHead.y);
  1297→          }
  1298→          toRemove.push(id);
  1299→        }
  1300→      }
  1301→      for (const id of toRemove) {
  1302→        this.bots.delete(id);
  1303→      }
  1304→    }
  1305→
  1306→    // 9) Player death → enter post-death recording (15s of continued simulation).
  1307→    if (playerDied && !this.isPostDeathRecording) {
  1308→      p.isDead = true;
  1309→      this.enterPostDeathRecording();
  1310→      // Don't return — continue to replenish food, expire chat, update camera
  1311→    }
  1312→
  1313→    // 10) Replenish food toward target (spawn around player).
  1314→    this.replenishFood();
  1315→
  1316→    // 11) Expire chat bubbles.
  1317→    this.expireChat(now);
  1318→
  1319→    // 12) Update camera target.
  1320→    this.updateCamera();
  1321→
  1322→    // 13) Capture replay snapshot.
  1323→    this.captureReplaySnapshot();
  1324→  }

  1326→  // --------------------------------------------------------------------------
  1327→  // Snake movement (server-authoritative formula)
  1328→  // --------------------------------------------------------------------------
  1329→
  1330→  private tickSnakeMovement(snake: SnakeBase, desiredAngle: number, wantsBoost: boolean): void {
  1331→    if (snake.points.length === 0 || snake.isDead) return;
  1332→
  1333→    // Turn rate: max(TURN_MIN, TURN_BASE - totalScore*TURN_SCORE_FACTOR).
  1334→    const totalScore = INITIAL_SPAWN_SCORE + snake.score;
  1335→    const turnRate = Math.max(TURN_MIN, TURN_BASE - totalScore * TURN_SCORE_FACTOR);
  1336→    snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
  1337→
  1338→    // Speed.
  1339→    let speed = BASE_SPEED;
  1340→    if (snake.isExtracting) {
  1341→      speed = EXTRACT_GLIDE_SPEED;
  1342→    } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
  1343→      speed = BOOST_SPEED;
  1344→      snake.isBoosting = true;
  1345→      snake.boostFrameCounter++;
  1346→      if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
  1347→        snake.boostFrameCounter = 0;
  1348→        if (snake.points.length > BOOST_MIN_LENGTH) {
  1349→          // Record tail position BEFORE popping (for food drop)
  1350→          const tail = snake.points[snake.points.length - 1];
  1351→          this.boostDropQueue.push({
  1352→            x: tail.x,
  1353→            y: tail.y,
  1354→          });
  1355→          snake.points.pop();
  1356→          // Reduce food-score (score can go below 0 but body is clamped)
  1357→          snake.score = Math.max(0, snake.score - 1);
  1358→        }
  1359→      }
  1360→    } else {
  1361→      snake.isBoosting = false;
  1362→    }
  1363→
  1364→    // Move head.
  1365→    const head = snake.points[0];
  1366→    const nx = head.x + Math.cos(snake.angle) * speed;
  1367→    const ny = head.y + Math.sin(snake.angle) * speed;
  1368→    snake.points.unshift({ x: nx, y: ny });
  1369→
  1370→    // Grow / shrink body to target length based on SCORE.
  1371→    // targetLen = INITIAL_BODY_LENGTH + foodScore, clamped to [BOOST_MIN_LENGTH+1, MAX_BODY_LENGTH]
  1372→    const targetLen = Math.max(
  1373→      BOOST_MIN_LENGTH + 1,
  1374→      Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + snake.score),
  1375→    );
  1376→    while (snake.points.length > targetLen) snake.points.pop();
  1377→
  1378→    // Size: 8 + sqrt(totalScore) * 0.4.
  1379→    snake.size = SIZE_BASE + Math.sqrt(INITIAL_SPAWN_SCORE + snake.score) * SIZE_SCORE_FACTOR;
  1380→  }

  1383→  // Bot AI — no self-destruct, seek food, predictive evasion
  1384→  // --------------------------------------------------------------------------
  1385→
  1386→  private tickBot(bot: BotSession, now: number): void {
  1387→    if (bot.points.length === 0 || bot.isDead) return;
  1388→
  1389→    if (now >= bot.nextThinkAt) {
  1390→      bot.nextThinkAt = now + BOT_THINK_INTERVAL_MS + Math.floor(Math.random() * BOT_THINK_JITTER_MS);
  1391→      const head = bot.points[0];
  1392→
  1393→      // --- Find nearest food ---
  1394→      let bestFood: GridItem | null = null;
  1395→      let bestFoodDist = Infinity;
  1396→      const foodQuery = this.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
  1397→      for (const item of foodQuery.values()) {
  1398→        if (item.kind !== 'food') continue;
  1399→        if ((item.value ?? 0) <= 0) continue;
  1400→        const d = dist(head.x, head.y, item.x, item.y);
  1401→        if (d < bestFoodDist) {
  1402→          bestFoodDist = d;
  1403→          bestFood = item;
  1404→        }
  1405→      }
  1406→
  1407→      // --- Find nearest threat (foreign body segment) ---
  1408→      let threatX = 0;
  1409→      let threatY = 0;
  1410→      let threatDist = Infinity;
  1411→      const threatQuery = this.grid.queryRadius(head.x, head.y, BOT_THREAT_SCAN_RADIUS);
  1412→      for (const item of threatQuery.values()) {
  1413→        if (item.kind !== 'segment') continue;
  1414→        if (item.snakeId === bot.id) continue;
  1415→        if (item.segIdx === 0) continue;
  1416→        const d = dist(head.x, head.y, item.x, item.y);
  1417→        if (d < threatDist) {
  1418→          threatDist = d;
  1419→          threatX = item.x;
  1420→          threatY = item.y;
  1421→        }
  1422→      }
  1423→
  1424→      let desired: number;
  1425→      const shouldFlee = threatDist < 150; // ALL bots dodge body segments when close
  1426→
  1427→      // --- Predictive evasion against the human player ---
  1428→      const p = this.player;
  1429→      if (p && !p.isDead && p.points.length > 0 && now >= p.spawnProtectedUntil) {
  1430→        const playerHead = p.points[0];
  1431→        const playerDist = dist(head.x, head.y, playerHead.x, playerHead.y);
  1432→        if (playerDist < BOT_EVADE_RADIUS * 2) {
  1433→          // Predict where the player will be in N ticks.
  1434→          const predictedX = playerHead.x + Math.cos(p.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1435→          const predictedY = playerHead.y + Math.sin(p.angle) * BOT_PREDICT_AHEAD_TICKS;
  1436→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1437→
  1438→          // If the bot is on a collision course with the predicted position, evade.
  1439→          if (distToPredicted < (bot.size + p.size) * 3) {
  1440→            // Evade perpendicular to the player's heading.
  1441→            const evadeAngle = p.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1442→            desired = evadeAngle;
  1443→            bot.desiredAngle = desired;
  1444→            const wantsBoost = bot.personality === 'hunter' && bot.score > 10;
  1445→            this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1446→            return;
  1447→          }
  1448→        }
  1449→      }
  1450→
  1451→      // Also check nearby bot heads for predictive evasion
  1452→      for (const other of this.bots.values()) {
  1453→        if (other.id === bot.id || other.isDead || other.points.length === 0) continue;
  1454→        const otherHead = other.points[0];
  1455→        const otherDist = dist(head.x, head.y, otherHead.x, otherHead.y);
  1456→        if (otherDist < 200) {
  1457→          const predictedX = otherHead.x + Math.cos(other.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1458→          const predictedY = otherHead.y + Math.sin(other.angle) * BOT_PREDICT_AHEAD_TICKS;
  1459→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1460→          if (distToPredicted < (bot.size + other.size) * 3) {
  1461→            const evadeAngle = other.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1462→            desired = evadeAngle;
  1463→            bot.desiredAngle = desired;
  1464→            this.tickSnakeMovement(bot, bot.desiredAngle, false);
  1465→            return;
  1466→          }
  1467→        }
  1468→      }
  1469→
  1470→      if (shouldFlee) {
  1471→        desired = Math.atan2(head.y - threatY, head.x - threatX);
  1472→      } else if (bestFood) {
  1473→        desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
  1474→      } else {
  1475→        desired = bot.angle + (Math.random() - 0.5) * 0.4;
  1476→      }
  1477→
  1478→      // NO edge avoidance — infinite map, no boundaries.
  1479→
  1480→      bot.desiredAngle = desired;
  1481→    }
  1482→
  1483→    // Hunter personality occasionally boosts.
  1484→    const wantsBoost = bot.personality === 'hunter' && bot.score > 5 && Math.random() < 0.05;
  1485→    this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1486→  }

  1540→  // --------------------------------------------------------------------------
  1541→  // Collision detection (body + head-on) — NO wall death
  1542→  // --------------------------------------------------------------------------
  1543→
  1544→  private detectCollisions(now: number): { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] {
  1545→    const deaths: { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] = [];
  1546→    const seenDead = new Set<string>();
  1547→    const all: SnakeBase[] = [];
  1548→    if (this.player && !this.player.isDead) all.push(this.player);
  1549→    for (const bot of this.bots.values()) {
  1550→      if (!bot.isDead) all.push(bot);
  1551→    }
  1552→
  1553→    // --- Head-to-head collision (checked first, takes priority) ---
  1554→    const headOnChecked = new Set<string>();
  1555→    for (let i = 0; i < all.length; i++) {
  1556→      for (let j = i + 1; j < all.length; j++) {
  1557→        const a = all[i];
  1558→        const b = all[j];
  1559→        if (a.isDead || b.isDead) continue;
  1560→        if (now < a.spawnProtectedUntil || now < b.spawnProtectedUntil) continue;
  1561→        const pairKey = a.id < b.id ? `${a.id}:${b.id}` : `${b.id}:${a.id}`;
  1562→        if (headOnChecked.has(pairKey)) continue;
  1563→        headOnChecked.add(pairKey);
  1564→
  1565→        const ha = a.points[0];
  1566→        const hb = b.points[0];
  1567→        const d = dist(ha.x, ha.y, hb.x, hb.y);
  1568→        const hitDist = (a.size + b.size) * HEAD_ON_HIT_FACTOR;
  1569→        if (d >= hitDist) continue;
  1570→
  1571→        // Head-on collision! Determine winner based on rules.
  1572→        const aBoosting = a.isBoosting;
  1573→        const bBoosting = b.isBoosting;
  1574→        const aTotalScore = INITIAL_SPAWN_SCORE + a.score;
  1575→        const bTotalScore = INITIAL_SPAWN_SCORE + b.score;
  1576→
  1577→        let aDies = false;
  1578→        let bDies = false;
  1579→
  1580→        if (aTotalScore === bTotalScore) {
  1581→          // Tie — both die
  1582→          aDies = true;
  1583→          bDies = true;
  1584→        } else if (aBoosting && !bBoosting) {
  1585→          // A is boosting, B is not → A wins (boost advantage)
  1586→          bDies = true;
  1587→        } else if (bBoosting && !aBoosting) {
  1588→          // B is boosting, A is not → B wins (boost advantage)
  1589→          aDies = true;
  1590→        } else {
  1591→          // Both boosting or neither boosting → higher score wins
  1592→          if (aTotalScore > bTotalScore) {
  1593→            bDies = true;
  1594→          } else {
  1595→            aDies = true;
  1596→          }
  1597→        }
  1598→
  1599→        if (aDies && !seenDead.has(a.id)) {
  1600→          deaths.push({ deadId: a.id, killerId: b.id, cause: 'headon' });
  1601→          seenDead.add(a.id);
  1602→        }
  1603→        if (bDies && !seenDead.has(b.id)) {
  1604→          deaths.push({ deadId: b.id, killerId: a.id, cause: 'headon' });
  1605→          seenDead.add(b.id);
  1606→        }
  1607→      }
  1608→    }
  1609→
  1610→    // --- Body collision: head vs foreign non-head segment ---
  1611→    for (const snake of all) {
  1612→      if (snake.isDead) continue;
  1613→      if (snake.points.length === 0) continue;
  1614→      if (now < snake.spawnProtectedUntil) continue;
  1615→      if (seenDead.has(snake.id)) continue;
  1616→      const head = snake.points[0];
  1617→
  1618→      // NO wall collision — infinite map.
  1619→
  1620→      const queryR = snake.size + 30;
  1621→      const nearby = this.grid.queryRadius(head.x, head.y, queryR);
  1622→      for (const item of nearby.values()) {
  1623→        if (item.kind !== 'segment') continue;
  1624→        // Own-collision immunity — skip own segments.
  1625→        if (item.snakeId === snake.id) continue;
  1626→        // Skip head segments (handled by head-on collision above).
  1627→        if (item.segIdx === 0) continue;
  1628→        // Neck protection: skip first N segments behind the head (close-call safety)
  1629→        if (item.segIdx <= 5) continue;
  1630→        const d = dist(head.x, head.y, item.x, item.y);
  1631→        if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
  1632→          deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
  1633→          seenDead.add(snake.id);
  1634→          break;
  1635→        }
