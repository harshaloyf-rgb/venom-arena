   206→   206→function angularDelta(current: number, desired: number): number {
   207→   207→  let diff = desired - current;
   208→   208→  while (diff > Math.PI) diff -= 2 * Math.PI;
   209→   209→  while (diff < -Math.PI) diff += 2 * Math.PI;
   210→   210→  return diff;
   211→   211→}
   212→   212→
   213→   213→export function turnToward(current: number, desired: number, maxStep: number): number {
   214→   214→  const diff = angularDelta(current, desired);
   215→   215→  if (Math.abs(diff) <= maxStep) return desired;
   216→   216→  return current + Math.sign(diff) * maxStep;
   217→   217→}
   218→   218→
   219→   219→function dist(ax: number, ay: number, bx: number, by: number): number {
   220→   220→  return Math.hypot(ax - bx, ay - by);
   221→   221→}

   223→   223→// ----------------------------------------------------------------------------
   224→   224→// Weighted random picker (for food orb spawning)
   225→   225→// ----------------------------------------------------------------------------
   226→   226→
   227→   227→function weightedRandomIndex(weights: number[]): number {
   228→   228→  const r = Math.random();
   229→   229→  let cumulative = 0;
   230→   230→  for (let i = 0; i < weights.length; i++) {
   231→   231→    cumulative += weights[i];
   232→   232→    if (r <= cumulative) return i;
   233→   233→  }
   234→   234→  return weights.length - 1;
   235→   235→}

   241→   241→/** Random point inside a disc of radius `maxR` centered at (cx, cy). */
   242→   242→export function randomSpawnPoint(maxR: number, cx: number = 0, cy: number = 0): Vec2 {
   243→   243→  const r = Math.sqrt(Math.random()) * maxR * 0.85;
   244→   244→  const theta = Math.random() * Math.PI * 2;
   245→   245→  return { x: cx + Math.cos(theta) * r, y: cy + Math.sin(theta) * r };
   246→   246→}

   248→   248→/**
   249→   249→ * Build an initial body: `length` points strung out behind the head at
   250→   250→ * SEGMENT_SPACING px apart, all along the angle's reverse direction.
   251→   251→ */
   252→   252→export function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   253→   253→  const pts: Vec2[] = [];
   254→   254→  for (let i = 0; i < length; i++) {
   255→   255→    pts.push({
   256→   256→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   257→   257→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   258→   258→    });
   259→   259→  }
   260→   260→  return pts;
   261→   261→}

   284→   284→export function spawnBot(room: ArenaRoom): BotSession {
   285→   285→  const idx = room.botIdCounter++;
   286→   286→  const name = BOT_NAMES[idx % BOT_NAMES.length] + (idx >= BOT_NAMES.length ? '-' + Math.floor(idx / BOT_NAMES.length) : '');
   287→   287→  const skin = BOT_SKINS[idx % BOT_SKINS.length];
   288→   288→  const personalities: BotPersonality[] = ['scavenger', 'opportunist', 'hunter', 'extractor', 'coward'];
   289→   289→  const personality = personalities[idx % personalities.length];
   290→   290→
   291→   291→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   292→   292→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount));
   293→   293→  const spawn = randomSpawnPoint(mapRadius - 200, room.mapCenterX, room.mapCenterY);
   294→   294→  const angle = Math.random() * Math.PI * 2;
   295→   295→  const botId = `bot-${room.arena.id}-${idx}`;
   296→   296→
   297→   297→  return {
   298→   298→    id: botId,
   299→   299→    botId,
   300→   300→    name,
   301→   301→    points: initialBody(spawn.x, spawn.y, angle, INITIAL_BODY_LENGTH),
   302→   302→    angle,
   303→   303→    size: SIZE_BASE + Math.sqrt(INITIAL_SPAWN_SCORE) * SIZE_SCORE_FACTOR,
   304→   304→    color: skin.color,

   306→   306→    isPlayer: false,
   307→   307→    isBot: true,
   308→   308→    carriedChips: 0,
   309→   309→    score: INITIAL_SPAWN_SCORE,
   310→   310→    boostFrameCounter: 0,
   311→   311→    isExtracting: false,
   312→   312→    extractionProgress: 0,
   313→   313→    isDead: false,
   314→   314→    spawnProtectedUntil: Date.now() + RESPAWN_INVULN_MS,
   315→   315→    personality,
   316→   316→    nextThinkAt: 0,

   354→   354→// ----------------------------------------------------------------------------
   355→   355→// Death drop helpers
   356→   356→// ----------------------------------------------------------------------------
   357→   357→
   358→   358→/**
   359→   359→ * Compute the mix of Small(1), Medium(3), Large(5) orbs that sum to exactly totalScore.
   360→   360→ * Greedily picks Large first, then Medium, then Small.
   361→   361→ */
   362→   362→export function computeDeathOrbs(totalScore: number): { small: number; medium: number; large: number } {
   363→   363→  let remaining = totalScore;
   364→   364→  const large = Math.floor(remaining / 5);
   365→   365→  remaining -= large * 5;
   366→   366→  const medium = Math.floor(remaining / 3);
   367→   367→  remaining -= medium * 3;
   368→   368→  const small = remaining;
   369→   369→  return { small, medium, large };
   370→   370→}
   371→   371→
   372→   372→/**
   373→   373→ * Drop score orbs (S/M/L) that sum to exactly totalScore along a body path.
   374→   374→ * Used for body-collision deaths (NOT wall deaths).
   375→   375→ */
   376→   376→export function dropScoreOrbsAtBody(
   377→   377→  room: ArenaRoom,
   378→   378→  bodyPoints: Vec2[],
   379→   379→  totalScore: number,
   380→   380→  snakeColor: string,
   381→   381→): void {
   382→   382→  if (!bodyPoints || bodyPoints.length === 0 || totalScore <= 0) return;
   383→   383→
   384→   384→  const { small, medium, large } = computeDeathOrbs(totalScore);
   385→   385→  let orbIdx = 0;
   386→   386→  const totalOrbs = small + medium + large;
   387→   387→  if (totalOrbs === 0) return;
   388→   388→
   389→   389→  // Interleave S/M/L along the body for visual spread
   390→   390→  const orbSequence: Array<{ value: number; size: number; color: string; glowColor: string; orbSize: 'small' | 'medium' | 'large' }> = [];
   391→   391→  for (let i = 0; i < large; i++) orbSequence.push({ value: 5, size: 8, color: '#f472b6', glowColor: '#ec4899', orbSize: 'large' });
   392→   392→  for (let i = 0; i < medium; i++) orbSequence.push({ value: 3, size: 5, color: '#38bdf8', glowColor: '#0ea5e9', orbSize: 'medium' });
   393→   393→  for (let i = 0; i < small; i++) orbSequence.push({ value: 1, size: 3, color: '#34d399', glowColor: '#10b981', orbSize: 'small' });
   394→   394→
   395→   395→  // Shuffle for visual variety
   396→   396→  for (let i = orbSequence.length - 1; i > 0; i--) {
   397→   397→    const j = Math.floor(Math.random() * (i + 1));
   398→   398→    [orbSequence[i], orbSequence[j]] = [orbSequence[j], orbSequence[i]];
   399→   399→  }
   400→   400→
   401→   401→  for (const orb of orbSequence) {
   402→   402→    const segIdx = Math.min(bodyPoints.length - 1, Math.floor((orbIdx / totalOrbs) * bodyPoints.length));
   403→   403→    const pt = bodyPoints[segIdx];
   404→   404→    const scatter = 20;
   405→   405→    const id = `food-${room.arena.id}-${room.foodIdCounter++}`;
   406→   406→    room.foods.push({
   407→   407→      id,
   408→   408→      x: pt.x + (Math.random() - 0.5) * scatter,
   409→   409→      y: pt.y + (Math.random() - 0.5) * scatter,
   410→   410→      size: orb.size,
   411→   411→      value: orb.value,
   412→   412→      isStarChip: false,
   413→   413→      color: orb.color,
   414→   414→      glowColor: orb.glowColor,
   415→   415→      orbSize: orb.orbSize,
   416→   416→    });
   417→   417→    orbIdx++;
   418→   418→  }
   419→   419→}
   420→   420→
   421→   421→/**
   422→   422→ * Drop exactly 10 Star collectibles at a death position.
   423→   423→ * Each star = floor(chips / 10), remainder goes to the last star.
   424→   424→ * ONLY for real players. Bots never drop stars.
   425→   425→ */
   426→   426→export function dropStarsAtDeath(
   427→   427→  room: ArenaRoom,
   428→   428→  x: number,
   429→   429→  y: number,
   430→   430→  chips: number,
   431→   431→): void {
   432→   432→  if (chips <= 0) return;
   433→   433→
   434→   434→  const valuePerStar = Math.floor(chips / STAR_DROP_COUNT);
   435→   435→  const remainder = chips - valuePerStar * STAR_DROP_COUNT;
   436→   436→  const scatter = 40;
   437→   437→
   438→   438→  for (let i = 0; i < STAR_DROP_COUNT; i++) {
   439→   439→    const value = valuePerStar + (i === STAR_DROP_COUNT - 1 ? remainder : 0);
   440→   440→    if (value <= 0) continue;
   441→   441→    const id = `food-${room.arena.id}-${room.foodIdCounter++}`;
   442→   442→    const angle = (i / STAR_DROP_COUNT) * Math.PI * 2;
   443→   443→    const dist = scatter * 0.5 + Math.random() * scatter * 0.5;
   444→   444→    room.foods.push({
   445→   445→      id,
   446→   446→      x: x + Math.cos(angle) * dist,
   447→   447→      y: y + Math.sin(angle) * dist,
   448→   448→      size: 12,
   449→   449→      value,
   450→   450→      isStarChip: true,
   451→   451→      color: '#fbbf24',
   452→   452→      glowColor: '#f59e0b',
   453→   453→    });
   454→   454→  }
   455→   455→}

   457→   457→/**
   458→   458→ * Force one bot into self-destruct state to make room for a new human player.
   459→   459→ * If no bots available, does nothing.
   460→   460→ */
   461→   461→export function displaceBotForPlayer(room: ArenaRoom): void {
   462→   462→  if (room.bots.size === 0) return;
   463→   463→  // Pick a random harvesting bot to self-destruct
   464→   464→  const candidates = [...room.bots.values()].filter(b => !b.isDead && b.botState === 'harvesting');
   465→   465→  if (candidates.length === 0) return;
   466→   466→  const bot = candidates[Math.floor(Math.random() * candidates.length)];
   467→   467→  bot.botState = 'selfDestruct';
   468→   468→  bot.wantsBoost = true; // boost toward wall faster
   469→   469→}

   479→   479→/**
   480→   480→ * Move a snake one tick. Server-authoritative: only desiredAngle from client.
   481→   481→ */
   482→   482→export function tickSnakeMovement(
   483→   483→  snake: SnakeBase,
   484→   484→  desiredAngle: number,
   485→   485→  wantsBoost: boolean,
   486→   486→): void {
   487→   487→  if (snake.points.length === 0) return;
   488→   488→  if (snake.isDead) return;
   489→   489→
   490→   490→  snake.wantsBoost = wantsBoost;
   491→   491→
   492→   492→  // 1) Turn — rate-limited, bigger snakes turn slower.
   493→   493→  const turnRate = Math.max(TURN_MIN, TURN_BASE - snake.score * TURN_SCORE_FACTOR);
   494→   494→  snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
   495→   495→
   496→   496→  // 2) Speed: extracting < normal < boost.
   497→   497→  let speed = BASE_SPEED;
   498→   498→  if (snake.isExtracting) {
   499→   499→    speed = EXTRACT_GLIDE_SPEED;
   500→   500→  } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
   501→   501→    speed = BOOST_SPEED;
   502→   502→    snake.boostFrameCounter++;
   503→   503→    if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
   504→   504→      snake.boostFrameCounter = 0;
   505→   505→      if (snake.points.length > BOOST_MIN_LENGTH) {
   506→   506→        snake.points.pop();
   507→   507→        snake.score = Math.max(INITIAL_SPAWN_SCORE, snake.score - 1);
   508→   508→      }
   509→   509→    }
   510→   510→  } else {
   511→   511→    snake.wantsBoost = false;
   512→   512→    snake.boostFrameCounter = 0;
   513→   513→  }
   514→   514→
   515→   515→  // 3) Move head.
   516→   516→  const head = snake.points[0];
   517→   517→  const nx = head.x + Math.cos(snake.angle) * speed;
   518→   518→  const ny = head.y + Math.sin(snake.angle) * speed;
   519→   519→
   520→   520→  // 4) Unshift new head.
   521→   521→  snake.points.unshift({ x: nx, y: ny });
   522→   522→
   523→   523→  // 5) Body length = INITIAL_BODY_LENGTH + (score - INITIAL_SPAWN_SCORE)
   524→   524→  const targetLen = Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + Math.max(0, snake.score - INITIAL_SPAWN_SCORE));
   525→   525→  while (snake.points.length > targetLen) snake.points.pop();
   526→   526→
   527→   527→  // 6) Size formula.
   528→   528→  snake.size = SIZE_BASE + Math.sqrt(snake.score) * SIZE_SCORE_FACTOR;
   529→   529→}

   535→   535→/**
   536→   536→ * Per-bot AI tick. All bots seek food and evade human players.
   537→   537→ * Self-destruct bots navigate toward the wall.
   538→   538→ */
   539→   539→export function tickBot(bot: BotSession, room: ArenaRoom, now: number): void {
   540→   540→  if (bot.points.length === 0 || bot.isDead) return;
   541→   541→
   542→   542→  const head = bot.points[0];
   543→   543→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   544→   544→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount), now);
   545→   545→
   546→   546→  // --- Self-destruct state: navigate to nearest wall ---
   547→   547→  if (bot.botState === 'selfDestruct') {
   548→   548→    // Find direction to nearest wall boundary
   549→   549→    const toWall = Math.atan2(room.mapCenterY - head.y, room.mapCenterX - head.x);
   550→   550→    // Navigate AWAY from center (toward wall)
   551→   551→    const awayFromCenter = Math.atan2(head.y - room.mapCenterY, head.x - room.mapCenterX);
   552→   552→    // Steer toward nearest wall point
   553→   553→    bot.desiredAngle = awayFromCenter;
   554→   554→    bot.wantsBoost = true;
   555→   555→
   556→   556→    // Still eat food on the way (prioritize survival)
   557→   557→    if (now >= bot.nextThinkAt) {
   558→   558→      bot.nextThinkAt = now + 80;
   559→   559→      const foodQuery = room.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
   560→   560→      let bestFood: GridItem | null = null;
   561→   561→      let bestFoodDist = Infinity;
   562→   562→      for (const item of foodQuery.values()) {
   563→   563→        if (item.kind !== 'food') continue;
   564→   564→        if ((item.value ?? 0) <= 0) continue;
   565→   565→        if (item.isStarChip) continue; // bots ignore stars
   566→   566→        const d = dist(head.x, head.y, item.x, item.y);
   567→   567→        if (d < bestFoodDist) {
   568→   568→          bestFoodDist = d;
   569→   569→          bestFood = item;
   570→   570→        }
   571→   571→      }
   572→   572→      // Blend food-seeking with wall-seeking (70% wall, 30% food if close)
   573→   573→      if (bestFood && bestFoodDist < 150) {
   574→   574→        const foodAngle = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
   575→   575→        bot.desiredAngle = turnToward(awayFromCenter, foodAngle, MAX_TURN_PER_TICK * 0.6);
   576→   576→      }
   577→   577→    }
   578→   578→
   579→   579→    tickSnakeMovement(bot, bot.desiredAngle, bot.wantsBoost);
   580→   580→    return;
   581→   581→  }
   582→   582→
   583→   583→  // --- Harvesting state ---
   584→   584→  if (now >= bot.nextThinkAt) {
   585→   585→    bot.nextThinkAt = now + 120 + Math.floor(Math.random() * 80);
   586→   586→
   587→   587→    // Find nearest food
   588→   588→    let bestFood: GridItem | null = null;
   589→   589→    let bestFoodDist = Infinity;
   590→   590→    const foodQuery = room.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
   591→   591→    for (const item of foodQuery.values()) {
   592→   592→      if (item.kind !== 'food') continue;
   593→   593→      if ((item.value ?? 0) <= 0) continue;
   594→   594→      if (item.isStarChip) continue; // bots ignore stars
   595→   595→      const d = dist(head.x, head.y, item.x, item.y);
   596→   596→      if (d < bestFoodDist) {
   597→   597→        bestFoodDist = d;
   598→   598→        bestFood = item;
   599→   599→      }
   600→   600→    }
   601→   601→
   602→   602→    // Find nearest human player (for evasion)
   603→   603→    let nearPlayerDist = Infinity;
   604→   604→    let nearPlayerAngle = 0;
   605→   605→    let nearPlayerVx = 0;
   606→   606→    let nearPlayerVy = 0;
   607→   607→    for (const p of room.players.values()) {
   608→   608→      if (p.isDead || p.matchSettling) continue;
   609→   609→      if (p.points.length === 0) continue;
   610→   610→      const ph = p.points[0];
   611→   611→      const d = dist(head.x, head.y, ph.x, ph.y);
   612→   612→      if (d < nearPlayerDist) {
   613→   613→        nearPlayerDist = d;
   614→   614→        nearPlayerAngle = Math.atan2(ph.y - head.y, ph.x - head.x);
   615→   615→        // Estimate player velocity from head angle
   616→   616→        nearPlayerVx = Math.cos(p.angle) * BASE_SPEED;
   617→   617→        nearPlayerVy = Math.sin(p.angle) * BASE_SPEED;
   618→   618→      }
   619→   619→    }
   620→   620→
   621→   621→    // Also check nearby body segments (for collision avoidance)
   622→   622→    let threatX = 0;
   623→   623→    let threatY = 0;
   624→   624→    let threatDist = Infinity;
   625→   625→    const threatQuery = room.grid.queryRadius(head.x, head.y, 150);
   626→   626→    for (const item of threatQuery.values()) {
   627→   627→      if (item.kind !== 'segment') continue;
   628→   628→      if (item.snakeId === bot.id) continue;
   629→   629→      if (item.segIdx === 0) continue;
   630→   630→      const d = dist(head.x, head.y, item.x, item.y);
   631→   631→      if (d < threatDist) {
   632→   632→        threatDist = d;
   633→   633→        threatX = item.x;
   634→   634→        threatY = item.y;
   635→   635→      }
   636→   636→    }
   637→   637→
   638→   638→    let desired: number;
   639→   639→
   640→   640→    // Priority 1: Evasion of body segments (immediate danger)
   641→   641→    if (threatDist < 140) {
   642→   642→      desired = Math.atan2(head.y - threatY, head.x - threatX);
   643→   643→    }
   644→   644→    // Priority 2: Predictive evasion of human players
   645→   645→    else if (nearPlayerDist < BOT_EVADE_RADIUS) {
   646→   646→      // Project player position 8 ticks ahead
   647→   647→      const futurePx = (room.players.values().toArray().find(p => !p.isDead)?.points[0]?.x ?? 0) + nearPlayerVx * 8;
   648→   648→      const futurePy = (room.players.values().toArray().find(p => !p.isDead)?.points[0]?.y ?? 0) + nearPlayerVy * 8;
   649→   649→      const futureD = dist(head.x, head.y, futurePx, futurePy);
   650→   650→      if (futureD < BOT_EVADE_RADIUS) {
   651→   651→        // Steer perpendicular to the player's approach vector
   652→   652→        const perpAngle = nearPlayerAngle + Math.PI / 2 * (Math.random() > 0.5 ? 1 : -1);
   653→   653→        desired = perpAngle;
   654→   654→      } else {
   655→   655→        // Just steer away
   656→   656→        desired = Math.atan2(head.y - (room.players.values().toArray().find(p => !p.isDead)?.points[0]?.y ?? 0), head.x - (room.players.values().toArray().find(p => !p.isDead)?.points[0]?.x ?? 0));
   657→   657→      }
   658→   658→    }
   659→   659→    // Priority 3: Seek food
   660→   660→    else if (bestFood) {
   661→   661→      desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
   662→   662→    }
   663→   663→    // Priority 4: Wander
   664→   664→    else {
   665→   665→      desired = bot.angle + (Math.random() - 0.5) * 0.4;
   666→   666→    }
   667→   667→
   668→   668→    // Edge avoidance: if near map boundary, turn back toward center
   669→   669→    const distFromCenter = Math.hypot(head.x - room.mapCenterX, head.y - room.mapCenterY);
   670→   670→    if (distFromCenter > mapRadius - 300) {
   671→   671→      const toCenter = Math.atan2(room.mapCenterY - head.y, room.mapCenterX - head.x);
   672→   672→      desired = turnToward(desired, toCenter, MAX_TURN_PER_TICK * 2);
   673→   673→    }
   674→   674→
   675→   675→    bot.desiredAngle = desired;
   676→   676→  }
   677→   677→
   678→   678→  // Check self-destruct threshold
   679→   679→  if (bot.score >= BOT_SELF_DESTRUCT_THRESHOLD && bot.botState === 'harvesting') {
   680→   680→    bot.botState = 'selfDestruct';
   681→   681→  }
   682→   682→
   683→   683→  // Bots don't boost in harvesting mode
   684→   684→  tickSnakeMovement(bot, bot.desiredAngle, false);
   685→   685→}

   691→   691→/** A pending death detected during collision iteration. */
   692→   692→export interface PendingDeath {
   693→   693→  deadId: string;
   694→   694→  killerId?: string;
   695→   695→  cause: 'body' | 'wall' | 'headOn';

   698→   698→/**
   699→   699→ * Detect head-to-body collisions via spatial grid.
   700→   700→ * Head hits any foreign body segment → that head's owner dies.
   701→   701→ */
   702→   702→export function detectCollisions(room: ArenaRoom, now: number): PendingDeath[] {
   703→   703→  const deaths: PendingDeath[] = [];
   704→   704→  const seenDead = new Set<string>();
   705→   705→
   706→   706→  const allSnakes: SnakeBase[] = collectAllSnakes(room);
   707→   707→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   708→   708→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount), now);
   709→   709→
   710→   710→  for (const snake of allSnakes) {
   711→   711→    if (snake.isDead) continue;
   712→   712→    if (snake.points.length === 0) continue;
   713→   713→    if (now < snake.spawnProtectedUntil) continue;
   714→   714→    if (seenDead.has(snake.id)) continue;
   715→   715→
   716→   716→    const head = snake.points[0];
   717→   717→
   718→   718→    // Wall collision
   719→   719→    const distFromCenter = Math.hypot(head.x - room.mapCenterX, head.y - room.mapCenterY);
   720→   720→    if (distFromCenter > mapRadius) {
   721→   721→      deaths.push({ deadId: snake.id, killerId: 'wall', cause: 'wall' });
   722→   722→      seenDead.add(snake.id);
   723→   723→      continue;
   724→   724→    }
   725→   725→
   726→   726→    // Head-to-body collision
   727→   727→    const queryR = snake.size + 30;
   728→   728→    const nearby = room.grid.queryRadius(head.x, head.y, queryR);
   729→   729→
   730→   730→    for (const item of nearby.values()) {
   731→   731→      if (item.kind !== 'segment') continue;
   732→   732→      if (item.snakeId === snake.id) continue;
   733→   733→      if (item.segIdx === 0) continue; // head-to-head handled separately
   734→   734→      const d = dist(head.x, head.y, item.x, item.y);
   735→   735→      if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
   736→   736→        deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
   737→   737→        seenDead.add(snake.id);
   738→   738→        break;
   739→   739→      }
   740→   740→    }
   741→   741→  }
   742→   742→
   743→   743→  return deaths;
   744→   744→}
   745→   745→
   746→   746→/**
   747→   747→ * Detect head-on (head-to-head) collisions.
   748→   748→ * Rules:
   749→   749→ *   A) No boost / both boosting → higher score survives
   750→   750→ *   B) Smaller boosting, larger steady → smaller survives
   751→   751→ *   C) Both boosting → higher score survives (same as A)
   752→   752→ *   Tie → both die
   753→   753→ */
   754→   754→export function detectHeadOnCollisions(room: ArenaRoom, now: number): PendingDeath[] {
   755→   755→  const deaths: PendingDeath[] = [];
   756→   756→  const seenDead = new Set<string>();
   757→   757→
   758→   758→  const allSnakes: SnakeBase[] = collectAllSnakes(room);
   759→   759→  const headMap = new Map<string, SnakeBase>();
   760→   760→  for (const s of allSnakes) {
   761→   761→    if (s.isDead || s.points.length === 0 || now < s.spawnProtectedUntil) continue;
   762→   762→    headMap.set(s.id, s);
   763→   763→  }
   764→   764→
   765→   765→  const processed = new Set<string>();
   766→   766→  for (const snakeA of allSnakes) {
   767→   767→    if (snakeA.isDead || snakeA.points.length === 0 || now < snakeA.spawnProtectedUntil) continue;
   768→   768→    if (seenDead.has(snakeA.id)) continue;
   769→   769→    if (processed.has(snakeA.id)) continue;
   770→   770→
   771→   771→    const headA = snakeA.points[0];
   772→   772→    const queryR = snakeA.size + 20;
   773→   773→    const nearby = room.grid.queryRadius(headA.x, headA.y, queryR);
   774→   774→
   775→   775→    for (const item of nearby.values()) {
   776→   776→      if (item.kind !== 'segment') continue;
   777→   777→      if (item.segIdx !== 0) continue; // only heads
   778→   778→      const snakeB = headMap.get(item.snakeId ?? '');
   779→   779→      if (!snakeB || snakeB.isDead || snakeB.id === snakeA.id) continue;
   780→   780→      if (seenDead.has(snakeB.id)) continue;
   781→   781→      if (processed.has(snakeB.id)) continue;
   782→   782→
   783→   783→      const headB = snakeB.points[0];
   784→   784→      const d = dist(headA.x, headA.y, headB.x, headB.y);
   785→   785→      if (d < (snakeA.size + snakeB.size) * HEAD_ON_HIT_FACTOR) {
   786→   786→        processed.add(snakeA.id);
   787→   787→        processed.add(snakeB.id);
   788→   788→
   789→   789→        const aBoosting = snakeA.wantsBoost;
   790→   790→        const bBoosting = snakeB.wantsBoost;
   791→   791→        const aScore = snakeA.score;
   792→   792→        const bScore = snakeB.score;
   793→   793→
   794→   794→        let loserId: string | null = null;
   795→   795→
   796→   796→        if (aScore === bScore) {
   797→   797→          // Tie: both die
   798→   798→          deaths.push({ deadId: snakeA.id, killerId: snakeB.id, cause: 'headOn' });
   799→   799→          deaths.push({ deadId: snakeB.id, killerId: snakeA.id, cause: 'headOn' });
   800→   800→          seenDead.add(snakeA.id);
   801→   801→          seenDead.add(snakeB.id);
   802→   802→        } else if (aScore > bScore) {
   803→   803→          // A is bigger
   804→   804→          if (bBoosting && !aBoosting) {
   805→   805→            // Rule B: smaller boosting vs larger steady → smaller survives
   806→   806→            loserId = snakeA.id;
   807→   807→          } else {
   808→   808→            // Rule A/C: bigger survives
   809→   809→            loserId = snakeB.id;
   810→   810→          }
   811→   811→        } else {
   812→   812→          // B is bigger
   813→   813→          if (aBoosting && !bBoosting) {
   814→   814→            // Rule B: smaller boosting vs larger steady → smaller survives
   815→   815→            loserId = snakeB.id;
   816→   816→          } else {
   817→   817→            // Rule A/C: bigger survives
   818→   818→            loserId = snakeA.id;
   819→   819→          }
   820→   820→        }
   821→   821→
   822→   822→        if (loserId && aScore !== bScore) {
   823→   823→          const winnerId = loserId === snakeA.id ? snakeB.id : snakeA.id;
   824→   824→          deaths.push({ deadId: loserId, killerId: winnerId, cause: 'headOn' });
   825→   825→          seenDead.add(loserId);
   826→   826→        }
   827→   827→        break;
   828→   828→      }
   829→   829→    }
   830→   830→  }
   831→   831→
   832→   832→  return deaths;
   833→   833→}

   835→   835→// ----------------------------------------------------------------------------
   836→   836→// Food eating
   837→   837→// ----------------------------------------------------------------------------
   838→   838→
   839→   839→/**
   840→   840→ * Eat food via spatial grid.
   841→   841→ * Regular food: snake.score += value (1, 3, or 5). ALL snakes eat.
   842→   842→ * Star chips: carriedChips += value, score += STAR_CHIP_GROW. ONLY real players.
   843→   843→ * Bots NEVER collect star chips.
   844→   844→ */
   845→   845→export function eatFood(room: ArenaRoom): void {
   846→   846→  const allSnakes: SnakeBase[] = collectAllSnakes(room);
   847→   847→  for (const snake of allSnakes) {
   848→   848→    if (snake.isDead) continue;
   849→   849→    if (snake.points.length === 0) continue;
   850→   850→    const head = snake.points[0];
   851→   851→    const nearby = room.grid.queryRadius(head.x, head.y, snake.size + 20);
   852→   852→    for (const item of nearby.values()) {
   853→   853→      if (item.kind !== 'food') continue;
   854→   854→      if ((item.value ?? 0) <= 0) continue;
   855→   855→      const d = dist(head.x, head.y, item.x, item.y);
   856→   856→      if (d < snake.size + (item.radius || 6) + 6) {
   857→   857→        if (item.isStarChip) {
   858→   858→          // Star chip: ONLY real players collect (bots skip).
   859→   859→          if (!snake.isPlayer) continue;
   860→   860→          snake.carriedChips += item.value ?? 0;
   861→   861→          snake.score += STAR_CHIP_GROW;
   862→   862→        } else {
   863→   863→          // Regular food orb: ALL snakes eat, score += value.
   864→   864→          snake.score += item.value ?? 1;
   865→   865→        }
   866→   866→        // Zero BOTH the grid item and the real food object.
   867→   867→        item.value = 0;
   868→   868→        if (item.foodRef) item.foodRef.value = 0;
   869→   869→      }
   870→   870→    }
   871→   871→  }
   872→   872→}

   878→   878→/** Iterator: yield every snake (player + bot) in the room. */
   879→   879→export function collectAllSnakes(room: ArenaRoom): SnakeBase[] {
   880→   880→  const out: SnakeBase[] = [];
   881→   881→  for (const p of room.players.values()) out.push(p);
   882→   882→  for (const b of room.bots.values()) out.push(b);
   883→   883→  return out;
   884→   884→}

   886→   886→/**
   887→   887→ * Build the broadcast GameSnapshot for an arena.
   888→   888→ */
   889→   889→export function buildSnapshot(room: ArenaRoom, viewerId?: string): GameSnapshot {
   890→   890→  const snakes: SnakeSnapshot[] = [];
   891→   891→  const now = Date.now();
   892→   892→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   893→   893→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount), now);
   894→   894→
   895→   895→  for (const snake of collectAllSnakes(room)) {
   896→   896→    let points = snake.points;
