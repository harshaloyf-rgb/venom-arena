    86→/** Returns the world-space rectangle visible in the viewport, plus margin. */
    87→export function computeVisibleRect(rc: FrameRenderCtx, marginPx = 100): VisibleRect {
    88→  const halfW = rc.w / 2 / rc.zoom;
    89→  const halfH = rc.h / 2 / rc.zoom;
    90→  const m = marginPx / rc.zoom;
    91→  return {
    92→    left: rc.camX - halfW - m,
    93→    right: rc.camX + halfW + m,
    94→    top: rc.camY - halfH - m,
    95→    bottom: rc.camY + halfH + m,
    96→  };
    97→}

    99→/** Quick AABB-vs-rect test. */
   100→function rectContainsPoint(rect: VisibleRect, x: number, y: number): boolean {
   101→  return x >= rect.left && x <= rect.right && y >= rect.top && y <= rect.bottom;
   102→}
   103→
   104→/** Returns true if any point in the list is inside the visible rect. */
   105→function snakeIsVisible(pts: { x: number; y: number }[], rect: VisibleRect): boolean {
   106→  // Cheap check: head + tail. If both off-screen on the SAME side we still
   107→  // need to scan all points (snake might span the screen). For most cases
   108→  // the head check suffices.
   109→  if (pts.length === 0) return false;
   110→  if (rectContainsPoint(rect, pts[0].x, pts[0].y)) return true;
   111→  const last = pts[pts.length - 1];
   112→  if (rectContainsPoint(rect, last.x, last.y)) return true;
   113→  // Fallback: walk the downsampled polyline.
   114→  const stride = pts.length > 60 ? 4 : 2;
   115→  for (let i = 0; i < pts.length; i += stride) {
   116→    if (rectContainsPoint(rect, pts[i].x, pts[i].y)) return true;
   117→  }
   118→  return false;
   119→}

   121→/**
   122→ * Cache a metallic radial gradient. Bucketed by 2-pixel radius so we don't
   123→ * cache a gradient for every floating-point size value.
   124→ */
   125→function getMetallicGradient(
   126→  rc: FrameRenderCtx,
   127→  radius: number,
   128→  color: string,
   129→  secondary: string | undefined,
   130→): CanvasGradient | string {
   131→  const bucket = Math.max(4, Math.round(radius));
   132→  const key = `${color}:${secondary ?? ''}:${bucket}`;
   133→  const cached = rc.metallicCache.get(key);
   134→  if (cached) return cached;
   135→  const { ctx } = rc;
   136→  // Gradient is built in device pixels — we always rebuild on resize because
   137→  // the cache Map is reset by the resize handler.
   138→  const g = ctx.createRadialGradient(-bucket * 0.35, -bucket * 0.35, bucket * 0.1, 0, 0, bucket);
   139→  g.addColorStop(0, '#f8fafc');
   140→  g.addColorStop(0.35, secondary || color);
   141→  g.addColorStop(1, color);
   142→  rc.metallicCache.set(key, g);
   143→  return g;
   144→}

   146→/**
   147→ * Breathing arena radius at a given time. Matches the server's
   148→ * `getMapRadius(elapsedMs)` formula exactly:
   149→ *   baseRadius + sin(cycleProgress * 2π) * amplitude
   150→ */
   151→export function getArenaRadius(now: number): number {
   152→  const cycleTime = (now % MAP_BREATH_CYCLE_MS) / MAP_BREATH_CYCLE_MS;
   153→  const sinVal = Math.sin(cycleTime * Math.PI * 2);
   154→  return MAP_BASE_RADIUS + sinVal * MAP_BREATH_AMPLITUDE;
   155→}

   161→/**
   162→ * Draws the breathing circular arena: background fill (deep slate `#020617`)
   163→ * clipped to the circle, a subtle grid (`#1e293b`, gridSize 60) inside the
   164→ * circle, and the neon-rose boundary (`#f43f5e`, lineWidth 10, shadowBlur 16).
   165→ */
   166→export function drawGrid(rc: FrameRenderCtx): void {
   167→  const { ctx, worldSize, camX, camY, zoom, w, h } = rc;
   168→  const cx = worldSize / 2;
   169→  const cy = worldSize / 2;
   170→  const radius = getArenaRadius(rc.now);
   171→
   172→  // --- Background fill (entire arena disc) ---
   173→  ctx.save();
   174→  // Clip to arena circle so nothing draws outside.
   175→  ctx.beginPath();
   176→  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   177→  ctx.fillStyle = '#020617'; // Deep Slate (AUDIT-A line 2572)
   178→  ctx.fill();
   179→  ctx.clip();
   180→
   181→  // --- Grid lines (subtle slate-800, gridSize 60) ---
   182→  const grid = 60; // AUDIT-A line 2578
   183→  const vis = computeVisibleRect(rc);
   184→  const startX = Math.max(0, Math.floor(vis.left / grid) * grid);
   185→  const endX = Math.min(worldSize, Math.ceil(vis.right / grid) * grid);
   186→  const startY = Math.max(0, Math.floor(vis.top / grid) * grid);
   187→  const endY = Math.min(worldSize, Math.ceil(vis.bottom / grid) * grid);
   188→
   189→  ctx.strokeStyle = '#1e293b'; // AUDIT-A line 2576
   190→  ctx.lineWidth = 1 / zoom;
   191→  ctx.beginPath();
   192→  for (let x = startX; x <= endX; x += grid) {
   193→    ctx.moveTo(x, startY);
   194→    ctx.lineTo(x, endY);
   195→  }
   196→  for (let y = startY; y <= endY; y += grid) {
   197→    ctx.moveTo(startX, y);
   198→    ctx.lineTo(endX, y);
   199→  }
   200→  ctx.stroke();
   201→  ctx.restore(); // remove clip
   202→
   203→  // --- Outer boundary (neon rose, breathing) ---
   204→  // AUDIT-A lines 2595-2603: strokeStyle #f43f5e, lineWidth 10, shadowBlur 16.
   205→  ctx.save();
   206→  ctx.lineWidth = 10;
   207→  ctx.strokeStyle = '#f43f5e';
   208→  if (!rc.lowQuality) {
   209→    ctx.shadowColor = '#f43f5e';
   210→    ctx.shadowBlur = 16;
   211→  }
   212→  ctx.beginPath();
   213→  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   214→  ctx.stroke();
   215→  ctx.shadowBlur = 0;
   216→  ctx.restore();
   217→
   218→  // Silence unused-param lint for camX/camY/w/h (they are part of the public
   219→  // type even if grid math doesn't need them directly).
   220→  void camX;
   221→  void camY;
   222→  void w;
   223→  void h;
   224→}

   230→/**
   231→ * Draws all food in two passes:
   232→ *  1. Regular food — batched by color into a single Path2D + fill per color
   233→ *     (filled circle in its own color).
   234→ *  2. Star chips — drawn individually as 5-pointed gold stars with a soft
   235→ *     `shadowColor #eab308`, `shadowBlur 6` glow.
   236→ */
   237→export function drawFood(rc: FrameRenderCtx, foods: FoodSnapshot[]): void {
   238→  const { ctx } = rc;
   239→  const vis = computeVisibleRect(rc);
   240→
   241→  // --- Regular food: batched by color ---
   242→  // Group foods into a per-color Path2D. The browser keeps one path open
   243→  // per color, then we issue a single fill() per color. This turns 800
   244→  // individual fills into ~6 fills (one per palette color).
   245→  const pathsByColor = new Map<string, Path2D>();
   246→  const starChips: FoodSnapshot[] = [];
   247→
   248→  for (let i = 0; i < foods.length; i++) {
   249→    const f = foods[i];
   250→    if (
   251→      f.x < vis.left ||
   252→      f.x > vis.right ||
   253→      f.y < vis.top ||
   254→      f.y > vis.bottom
   255→    ) {
   256→      continue; // culled
   257→    }
   258→    if (f.isStarChip) {
   259→      starChips.push(f);
   260→      continue;
   261→    }
   262→    let p = pathsByColor.get(f.color);
   263→    if (!p) {
   264→      p = new Path2D();
   265→      pathsByColor.set(f.color, p);
   266→    }
   267→    p.moveTo(f.x + f.size, f.y);
   268→    p.arc(f.x, f.y, f.size, 0, Math.PI * 2);
   269→  }
   270→
   271→  for (const [color, path] of pathsByColor) {
   272→    ctx.fillStyle = color;
   273→    ctx.fill(path);
   274→  }
   275→
   276→  // --- Star chips: 5-pointed gold star with shadowBlur 6 ---
   277→  if (starChips.length === 0) return;
   278→  ctx.save();
   279→  for (let i = 0; i < starChips.length; i++) {
   280→    const f = starChips[i];
   281→    const outer = Math.max(2, f.size + 1);
   282→    const inner = Math.max(1, f.size / 2);
   283→    drawStar(ctx, f.x, f.y, 5, outer, inner, f.color || '#fbbf24', rc.lowQuality);
   284→  }
   285→  ctx.restore();
   286→}

   288→/** Draws a 5-pointed star centered at (x,y) with `outer` outer radius and `inner` inner radius. */
   289→function drawStar(
   290→  ctx: CanvasRenderingContext2D,
   291→  x: number,
   292→  y: number,
   293→  points: number,
   294→  outer: number,
   295→  inner: number,
   296→  color: string,
   297→  lowQuality: boolean,
   298→): void {
   299→  ctx.save();
   300→  if (!lowQuality) {
   301→    ctx.shadowColor = '#eab308';
   302→    ctx.shadowBlur = 6;
   303→  }
   304→  ctx.beginPath();
   305→  for (let i = 0; i < points * 2; i++) {
   306→    const r = i % 2 === 0 ? outer : inner;
   307→    const angle = (Math.PI / points) * i - Math.PI / 2;
   308→    const px = x + Math.cos(angle) * r;
   309→    const py = y + Math.sin(angle) * r;
   310→    if (i === 0) ctx.moveTo(px, py);
   311→    else ctx.lineTo(px, py);
   312→  }
   313→  ctx.closePath();
   314→  ctx.fillStyle = color;
   315→  ctx.fill();
   316→  if (!lowQuality) ctx.shadowBlur = 0;
   317→  ctx.restore();
   318→}

   324→/**
   325→ * Draws a single snake. The body is rendered as ONE thick stroked polyline
   326→ * (with an outline underlay) — not N individual circles — which eliminates
   327→ * per-segment gradient, per-segment shadowBlur, per-segment save/restore,
   328→ * and per-frame object allocation.
   329→ *
   330→ * The head is drawn separately with eyes and (for the player's own snake,
   331→ * high-quality only) a glow halo.
   332→ *
   333→ * Player skin patterns (rainbow / neon / metallic / camo) are applied when
   334→ * `rc.playerSkin.pattern` is set and the snake is the local player.
   335→ */
   336→export function drawSnake(rc: FrameRenderCtx, snake: SnakeSnapshot): void {
   337→  const pts = snake.points;
   338→  if (!pts || pts.length === 0) return; // C13 guard
   339→
   340→  const vis = computeVisibleRect(rc);
   341→  if (!snakeIsVisible(pts, vis)) return; // P16 cull
   342→
   343→  const { ctx, zoom, lowQuality } = rc;
   344→  const isMe = snake.id === rc.myId;
   345→
   346→  // Downsample long snakes (defensive — server already caps at ~120).
   347→  const stride = pts.length > 60 ? 2 : 1;
   348→
   349→  // Body width in world units. `size` is the radius; visual width = 2*size.
   350→  const radius = Math.max(2, snake.size);
   351→  const width = radius * 2;
   352→
   353→  // --- Outline (underlay) ---
   354→  if (snake.secondaryColor) {
   355→    ctx.save();
   356→    ctx.lineCap = 'round';
   357→    ctx.lineJoin = 'round';
   358→    ctx.lineWidth = width + 4 / zoom;
   359→    ctx.strokeStyle = snake.secondaryColor;
   360→    ctx.globalAlpha = 0.55;
   361→    ctx.beginPath();
   362→    ctx.moveTo(pts[0].x, pts[0].y);
   363→    for (let i = stride; i < pts.length; i += stride) {
   364→      ctx.lineTo(pts[i].x, pts[i].y);
   365→    }
   366→    ctx.stroke();
   367→    ctx.restore();
   368→  }
   369→
   370→  // --- Body ---
   371→  ctx.save();
   372→  ctx.lineCap = 'round';
   373→  ctx.lineJoin = 'round';
   374→  ctx.lineWidth = width;
   375→
   376→  const pattern = isMe ? rc.playerSkin?.pattern : undefined;
   377→  if (pattern === 'metallic' && snake.secondaryColor) {
   378→    // Use cached gradient — keyed by size bucket so we don't re-build per
   379→    // snake. Stroking still happens per-snake (correct transform).
   380→    const g = getMetallicGradient(rc, radius, snake.color, snake.secondaryColor);
   381→    ctx.strokeStyle = g;
   382→    ctx.beginPath();
   383→    ctx.moveTo(pts[0].x, pts[0].y);
   384→    for (let i = stride; i < pts.length; i += stride) {
   385→      ctx.lineTo(pts[i].x, pts[i].y);
   386→    }
   387→    ctx.stroke();
   388→  } else if (pattern === 'rainbow' && !lowQuality) {
   389→    // Rainbow: stroke in hue-cycling chunks. Cheap — only a handful of
   390→    // stroke() calls per snake.
   391→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   392→    ctx.lineWidth = width;
   393→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   394→      const hue = (rc.now * 0.05 + i * 14) % 360;
   395→      ctx.strokeStyle = `hsl(${hue}, 90%, 55%)`;
   396→      ctx.beginPath();
   397→      ctx.moveTo(pts[i].x, pts[i].y);
   398→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   399→      for (let j = i + stride; j <= end; j += stride) {
   400→        ctx.lineTo(pts[j].x, pts[j].y);
   401→      }
   402→      ctx.stroke();
   403→    }
   404→  } else if (pattern === 'neon' && !lowQuality) {
   405→    // Neon: alternate cyan / purple along the body (AUDIT-A line 2762-2765).
   406→    const chunkSize = Math.max(4, Math.floor(pts.length / 10));
   407→    ctx.lineWidth = width;
   408→    let chunkIndex = 0;
   409→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   410→      const ratio = (Math.sin(rc.now * 0.009 - chunkIndex * 0.28) + 1) / 2;
   411→      ctx.strokeStyle = ratio > 0.5 ? '#06b6d4' : '#a855f7';
   412→      ctx.beginPath();
   413→      ctx.moveTo(pts[i].x, pts[i].y);
   414→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   415→      for (let j = i + stride; j <= end; j += stride) {
   416→        ctx.lineTo(pts[j].x, pts[j].y);
   417→      }
   418→      ctx.stroke();
   419→      chunkIndex += 1;
   420→    }
   421→  } else if (pattern === 'camo') {
   422→    // Camo: alternating earth-tone segments (AUDIT-A line 2768-2770).
   423→    const camoColors = ['#15803d', '#854d0e', '#78350f', '#166534'];
   424→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   425→    let chunkIndex = 0;
   426→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   427→      ctx.strokeStyle = camoColors[chunkIndex % camoColors.length];
   428→      ctx.beginPath();
   429→      ctx.moveTo(pts[i].x, pts[i].y);
   430→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   431→      for (let j = i + stride; j <= end; j += stride) {
   432→        ctx.lineTo(pts[j].x, pts[j].y);
   433→      }
   434→      ctx.stroke();
   435→      chunkIndex += 1;
   436→    }
   437→  } else {
   438→    // Default: solid snake.color (alternating 2-color handled by outline).
   439→    ctx.strokeStyle = snake.color;
   440→    ctx.beginPath();
   441→    ctx.moveTo(pts[0].x, pts[0].y);
   442→    for (let i = stride; i < pts.length; i += stride) {
   443→      ctx.lineTo(pts[i].x, pts[i].y);
   444→    }
   445→    ctx.stroke();
   446→  }
   447→  ctx.restore();
   448→
   449→  // --- Head ---
   450→  const head = pts[0];
   451→  ctx.save();
   452→  // Player-only glow halo, high-quality only (FIXES P1).
   453→  if (isMe && !lowQuality) {
   454→    ctx.shadowColor = snake.color;
   455→    ctx.shadowBlur = 14;
   456→  }
   457→  ctx.beginPath();
   458→  ctx.arc(head.x, head.y, radius, 0, Math.PI * 2);
   459→  ctx.fillStyle = snake.color;
   460→  ctx.fill();
   461→  if (isMe && !lowQuality) ctx.shadowBlur = 0;
   462→
   463→  // Eyes — direction-based positioning.
   464→  const eyeOffset = radius * 0.45;
   465→  const eyeR = Math.max(1.5, radius * 0.32);
   466→  const pupilR = Math.max(0.8, radius * 0.18);
   467→  const angle = snake.angle;
   468→  const perp = angle + Math.PI / 2;
   469→  const ex1 = head.x + Math.cos(angle) * eyeOffset * 0.4 + Math.cos(perp) * eyeOffset;
   470→  const ey1 = head.y + Math.sin(angle) * eyeOffset * 0.4 + Math.sin(perp) * eyeOffset;
   471→  const ex2 = head.x + Math.cos(angle) * eyeOffset * 0.4 - Math.cos(perp) * eyeOffset;
   472→  const ey2 = head.y + Math.sin(angle) * eyeOffset * 0.4 - Math.sin(perp) * eyeOffset;
   473→
   474→  ctx.fillStyle = '#ffffff';
   475→  ctx.beginPath();
   476→  ctx.arc(ex1, ey1, eyeR, 0, Math.PI * 2);
   477→  ctx.arc(ex2, ey2, eyeR, 0, Math.PI * 2);
   478→  ctx.fill();
   479→
   480→  ctx.fillStyle = '#0a0a0a';
   481→  ctx.beginPath();
   482→  ctx.arc(ex1 + Math.cos(angle) * pupilR, ey1 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   483→  ctx.arc(ex2 + Math.cos(angle) * pupilR, ey2 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   484→  ctx.fill();
   485→
   486→  // Spawn-protection ring.
   487→  if (snake.spawnProtected) {
   488→    ctx.strokeStyle = 'rgba(255,255,255,0.6)';
   489→    ctx.lineWidth = 2 / zoom;
   490→    ctx.beginPath();
   491→    ctx.arc(head.x, head.y, radius + 4 / zoom, 0, Math.PI * 2);
   492→    ctx.stroke();
   493→  }
   494→
   495→  // Extraction ring.
   496→  if (snake.isExtracting) {
   497→    ctx.strokeStyle = 'rgba(16, 185, 129, 0.95)';
   498→    ctx.lineWidth = 3 / zoom;
   499→    ctx.setLineDash([6 / zoom, 4 / zoom]);
   500→    ctx.beginPath();
   501→    ctx.arc(head.x, head.y, radius + 8 / zoom, 0, Math.PI * 2);
   502→    ctx.stroke();
   503→    ctx.setLineDash([]); // RESET — fixes R11
   504→  }
   505→  ctx.restore();
   506→
   507→  // --- Name label ---
   508→  if (snake.name) {
   509→    ctx.save();
   510→    ctx.font = `${Math.max(10, 12 / zoom)}px monospace`;
   511→    ctx.textAlign = 'center';
   512→    ctx.textBaseline = 'bottom';
   513→    const labelY = head.y - radius - 6 / zoom;
   514→    ctx.fillStyle = isMe ? '#22c55e' : 'rgba(226, 232, 240, 0.85)';
   515→    ctx.fillText(snake.name, head.x, labelY);
   516→    if (snake.userTag) {
   517→      ctx.fillStyle = 'rgba(148, 163, 184, 0.7)';
   518→      ctx.font = `${Math.max(8, 9 / zoom)}px monospace`;
   519→      ctx.fillText(snake.userTag, head.x, labelY - 12 / zoom);
   520→    }
   521→    ctx.restore();
   522→  }
   523→
   524→  // --- Chat bubble ---
   525→  if (snake.chatMessage) {
   526→    drawChatBubble(rc, head.x, head.y - radius - 24 / zoom, snake.chatMessage);
   527→  }
   528→}

   530→/** Draws a small chat bubble above a snake head. */
   531→function drawChatBubble(
   532→  rc: FrameRenderCtx,
   533→  x: number,
   534→  y: number,
   535→  msg: string,
   536→): void {
   537→  const { ctx, zoom } = rc;
   538→  const text = msg.length > 80 ? msg.slice(0, 78) + '…' : msg;
   539→  ctx.save();
   540→  ctx.font = `${Math.max(10, 11 / zoom)}px monospace`;
   541→  const metrics = ctx.measureText(text);
   542→  const padX = 6 / zoom;
   543→  const padY = 3 / zoom;
   544→  const tw = metrics.width + padX * 2;
   545→  const th = 14 / zoom + padY * 2;
   546→  ctx.fillStyle = 'rgba(15, 23, 42, 0.92)';
   547→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.6)'; // indigo (AUDIT-A line 3072)
   548→  ctx.lineWidth = 1.5 / zoom;
   549→  // Rounded rect (manual since older canvas may not have roundRect).
   550→  const rx = x - tw / 2;
   551→  const ry = y - th;
   552→  const rr = 4 / zoom;
   553→  ctx.beginPath();
   554→  ctx.moveTo(rx + rr, ry);
   555→  ctx.lineTo(rx + tw - rr, ry);
   556→  ctx.arcTo(rx + tw, ry, rx + tw, ry + rr, rr);
   557→  ctx.lineTo(rx + tw, ry + th - rr);
   558→  ctx.arcTo(rx + tw, ry + th, rx + tw - rr, ry + th, rr);
   559→  ctx.lineTo(rx + rr, ry + th);
   560→  ctx.arcTo(rx, ry + th, rx, ry + th - rr, rr);
   561→  ctx.lineTo(rx, ry + rr);
   562→  ctx.arcTo(rx, ry, rx + rr, ry, rr);
   563→  ctx.closePath();
   564→  ctx.fill();
   565→  ctx.stroke();
   566→  ctx.fillStyle = '#f8fafc';
   567→  ctx.textAlign = 'center';
   568→  ctx.textBaseline = 'middle';
   569→  ctx.fillText(text, x, y - th / 2);
   570→  ctx.restore();
   571→}

   577→/** Draws the particle array (caller caps at MAX_PARTICLES = 200). */
   578→export function drawParticles(rc: FrameRenderCtx, particles: Particle[]): void {
   579→  if (particles.length === 0) return;
   580→  const { ctx, zoom } = rc;
   581→  ctx.save();
   582→  ctx.globalCompositeOperation = 'lighter';
   583→  for (let i = 0; i < particles.length; i++) {
   584→    const p = particles[i];
   585→    if (p.life <= 0) continue;
   586→    const alpha = Math.max(0, Math.min(1, p.life / p.maxLife));
   587→    ctx.globalAlpha = alpha;
   588→    ctx.fillStyle = p.color;
   589→    ctx.beginPath();
   590→    ctx.arc(p.x, p.y, p.size / zoom, 0, Math.PI * 2);
   591→    ctx.fill();
   592→  }
   593→  ctx.globalAlpha = 1;
   594→  ctx.globalCompositeOperation = 'source-over';
   595→  ctx.restore();
   596→}

   598→// ---------------------------------------------------------------------------
   599→// Minimap (drawn in screen-space, no camera transform)
   600→// ---------------------------------------------------------------------------
   601→
   602→export interface MinimapArgs {
   603→  ctx: CanvasRenderingContext2D;
   604→  /** Top-left of minimap in CSS pixels. */
   605→  x: number;
   606→  y: number;
   607→  /** Minimap size (square). */
   608→  size: number;
   609→  worldSize: number;
   610→  /** Current breathing arena radius (world units). */
   611→  arenaRadius: number;
   612→  snakes: SnakeSnapshot[];
   613→  myId: string;
   614→  /**
   615→   * World-space radius around the player the radar should cover. Snakes
   616→   * outside this range are NOT rendered on the small corner minimap
   617→   * (BUILD-13: small minimap = "nearby snakes within ~1800px"). Defaults
   618→   * to WORLD_SIZE/2 (legacy full-radar behaviour) when omitted.
   619→   */
   620→  range?: number;
   621→}
   622→
   623→/**
   624→ * Draws the circular radar minimap. Player is rendered as an indigo dot at
   625→ * the centre (the radar is centered on the player's head). The arena
   626→ * boundary is a dashed rose circle. Bots render as rose dots, other real
   627→ * players as emerald dots — matches AUDIT-A radar visual language.
   628→ *
   629→ * BUILD-13: `range` controls how far the radar sees. The small corner
   630→ * minimap passes ~1800 so only nearby snakes are shown; the full-map
   631→ * overlay passes WORLD_SIZE/2 (or simply uses drawFullMap).
   632→ */
   633→export function drawMinimap(args: MinimapArgs): void {
   634→  const { ctx, x, y, size, worldSize, arenaRadius, snakes, myId } = args;
   635→  const cx = x + size / 2;
   636→  const cy = y + size / 2;
   637→  const r = size / 2;
   638→  // Radar range — default to WORLD_SIZE/2 (legacy). Small corner minimap
   639→  // passes 1800 to show only nearby snakes (BUILD-13 spec).
   640→  const radarRange = args.range ?? WORLD_SIZE / 2;
   641→  const scale = r / radarRange;
   642→
   643→  ctx.save();
   644→  // Background circle (radar disc)
   645→  ctx.beginPath();
   646→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   647→  ctx.fillStyle = 'rgba(2, 6, 23, 0.85)';
   648→  ctx.fill();
   649→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.4)';
   650→  ctx.lineWidth = 1.5;
   651→  ctx.stroke();
   652→
   653→  // Clip to minimap circle.
   654→  ctx.beginPath();
   655→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   656→  ctx.clip();
   657→
   658→  // Concentric rings (AUDIT-A lines 3520-3522).
   659→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
   660→  ctx.lineWidth = 1;
   661→  for (const inset of [2, 5, 8]) {
   662→    ctx.beginPath();
   663→    ctx.arc(cx, cy, r - inset, 0, Math.PI * 2);
   664→    ctx.stroke();
   665→  }
   666→
   667→  // Crosshairs.
   668→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.4)';
   669→  ctx.beginPath();
   670→  ctx.moveTo(cx, y);
   671→  ctx.lineTo(cx, y + size);
   672→  ctx.moveTo(x, cy);
   673→  ctx.lineTo(x + size, cy);
   674→  ctx.stroke();
   675→
   676→  // Arena boundary (dashed rose) — centered on world center.
   677→  // World center relative to player's radar position.
   678→  const mySnake = snakes.find((s) => s.id === myId);
   679→  const myHead = mySnake?.points?.[0];
   680→  const px = myHead ? myHead.x : worldSize / 2;
   681→  const py = myHead ? myHead.y : worldSize / 2;
   682→  const worldCenterOffsetX = (worldSize / 2 - px) * scale;
   683→  const worldCenterOffsetY = (worldSize / 2 - py) * scale;
   684→  const arenaR = arenaRadius * scale;
   685→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.6)';
   686→  ctx.lineWidth = 1.2;
   687→  ctx.setLineDash([4, 4]);
   688→  ctx.beginPath();
   689→  ctx.arc(cx + worldCenterOffsetX, cy + worldCenterOffsetY, arenaR, 0, Math.PI * 2);
   690→  ctx.stroke();
   691→  ctx.setLineDash([]);
   692→
   693→  // Snakes (player centered → drawn at centre; others offset).
   694→  for (let i = 0; i < snakes.length; i++) {
   695→    const s = snakes[i];
   696→    if (!s.points || s.points.length === 0) continue;
   697→    const head = s.points[0];
   698→    const dx = (head.x - px) * scale;
   699→    const dy = (head.y - py) * scale;
   700→    const dist = Math.hypot(dx, dy);
   701→    if (dist > r) continue; // off-radar
   702→    const mx = cx + dx;
   703→    const my = cy + dy;
   704→    ctx.beginPath();
   705→    if (s.id === myId) {
   706→      ctx.arc(mx, my, 3, 0, Math.PI * 2);
   707→      ctx.fillStyle = '#818cf8'; // indigo-400 (AUDIT-A line 3525)
   708→    } else if (s.isBot) {
   709→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
   710→      ctx.fillStyle = '#f43f5e'; // rose-500 (AUDIT-A line 3577)
   711→    } else {
   712→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
   713→      ctx.fillStyle = '#34d399'; // emerald-400 (AUDIT-A line 3577)
   714→    }
   715→    ctx.fill();
   716→  }
   717→  ctx.restore();
   718→}
   719→
   720→// ---------------------------------------------------------------------------
   721→// Full Map overlay (BUILD-13) — drawn in screen-space, no camera transform.
   722→// Press `M` to toggle. Shows the ENTIRE arena with all snakes as dots.
   723→// ---------------------------------------------------------------------------
   724→
   725→export interface FullMapArgs {
   726→  ctx: CanvasRenderingContext2D;
   727→  /** Canvas CSS-pixel width. */
   728→  w: number;
   729→  /** Canvas CSS-pixel height. */
   730→  h: number;
   731→  worldSize: number;
   732→  /** Current breathing arena radius (world units). */
   733→  arenaRadius: number;
   734→  snakes: SnakeSnapshot[];
   735→  myId: string;
   736→}
   737→
   738→/**
   739→ * Draws a full-screen overlay of the entire arena. The arena circle is
   740→ * centered on screen and scaled to fit the smaller of (w, h) with padding.
   741→ * All snakes are rendered as dots — player = indigo (larger), bots = rose,
   742→ * other real humans = emerald. A "Press M to close" hint is drawn at the
   743→ * bottom. The player's own dot pulses with a ring.
   744→ */
   745→export function drawFullMap(args: FullMapArgs): void {
   746→  const { ctx, w, h, worldSize, arenaRadius, snakes, myId } = args;
   747→  // --- Background fill (dim slate) ---
   748→  ctx.save();
   749→  ctx.setTransform(1, 0, 0, 1, 0, 0);
   750→  ctx.fillStyle = 'rgba(2, 6, 23, 0.94)'; // deep slate, near-opaque
   751→  ctx.fillRect(0, 0, w, h);
   752→
   753→  // --- Layout: arena centred on screen ---
   754→  const cx = w / 2;
   755→  const cy = h / 2;
   756→  // Fit the arena diameter (2 * arenaRadius) plus a small margin into the
   757→  // smaller of width / height. Margin = 80px so labels fit.
   758→  const margin = 80;
   759→  const fitDim = Math.min(w, h) - margin * 2;
   760→  const arenaDiameter = arenaRadius * 2;
   761→  const scale = fitDim / arenaDiameter;
   762→  const screenR = arenaRadius * scale;
   763→
   764→  // World-center → screen mapping. World center = (worldSize/2, worldSize/2).
   765→  // All snake positions are in world coords; subtract world center, scale,
   766→  // then add screen center.
   767→  const wcx = worldSize / 2;
   768→  const wcy = worldSize / 2;
   769→  const toScreenX = (wx: number) => cx + (wx - wcx) * scale;
   770→  const toScreenY = (wy: number) => cy + (wy - wcy) * scale;
   771→
   772→  // --- Concentric range rings (faint) ---
   773→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
   774→  ctx.lineWidth = 1;
   775→  for (const frac of [0.25, 0.5, 0.75]) {
   776→    ctx.beginPath();
   777→    ctx.arc(cx, cy, screenR * frac, 0, Math.PI * 2);
   778→    ctx.stroke();
   779→  }
   780→  // Crosshairs.
   781→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.5)';
   782→  ctx.beginPath();
   783→  ctx.moveTo(cx, cy - screenR);
   784→  ctx.lineTo(cx, cy + screenR);
   785→  ctx.moveTo(cx - screenR, cy);
   786→  ctx.lineTo(cx + screenR, cy);
   787→  ctx.stroke();
   788→
   789→  // --- Arena boundary (dashed rose) ---
   790→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.8)';
   791→  ctx.lineWidth = 2;
   792→  ctx.setLineDash([6, 6]);
   793→  ctx.beginPath();
   794→  ctx.arc(cx, cy, screenR, 0, Math.PI * 2);
   795→  ctx.stroke();
   796→  ctx.setLineDash([]);
   797→
   798→  // --- Snakes ---
   799→  for (let i = 0; i < snakes.length; i++) {
   800→    const s = snakes[i];
   801→    if (!s.points || s.points.length === 0) continue;
   802→    const head = s.points[0];
   803→    const sx = toScreenX(head.x);
   804→    const sy = toScreenY(head.y);
   805→    ctx.beginPath();
   806→    if (s.id === myId) {
   807→      // Player: bigger indigo dot + pulsing ring.
   808→      ctx.arc(sx, sy, 5, 0, Math.PI * 2);
   809→      ctx.fillStyle = '#818cf8';
   810→      ctx.fill();
   811→      ctx.strokeStyle = 'rgba(129, 140, 248, 0.7)';
   812→      ctx.lineWidth = 2;
   813→      ctx.beginPath();
   814→      ctx.arc(sx, sy, 9, 0, Math.PI * 2);
   815→      ctx.stroke();
   816→    } else if (s.isBot) {
   817→      ctx.arc(sx, sy, 2.5, 0, Math.PI * 2);
   818→      ctx.fillStyle = '#f43f5e'; // rose
   819→      ctx.fill();
   820→    } else {
   821→      ctx.arc(sx, sy, 3, 0, Math.PI * 2);
   822→      ctx.fillStyle = '#34d399'; // emerald — other humans
   823→      ctx.fill();
   824→    }
   825→  }
   826→
   827→  // --- Title + close hint ---
   828→  ctx.fillStyle = 'rgba(226, 232, 240, 0.95)';
   829→  ctx.font = 'bold 14px monospace';
   830→  ctx.textAlign = 'center';
   831→  ctx.textBaseline = 'top';
   832→  ctx.fillText('ARENA OVERVIEW — ALL SNAKES', cx, 16);
   833→  ctx.fillStyle = 'rgba(148, 163, 184, 0.85)';
   834→  ctx.font = '11px monospace';
   835→  ctx.fillText('Press M to close', cx, h - 24);
   836→
   837→  // Legend (top-left).
   838→  ctx.textAlign = 'left';
   839→  ctx.font = '11px monospace';
   840→  const legendX = 20;
   841→  let legendY = 20;
   842→  const drawLegend = (color: string, label: string, dotR: number) => {
   843→    ctx.beginPath();
   844→    ctx.arc(legendX + 6, legendY + 6, dotR, 0, Math.PI * 2);
   845→    ctx.fillStyle = color;
   846→    ctx.fill();
   847→    ctx.fillStyle = 'rgba(226, 232, 240, 0.85)';
   848→    ctx.fillText(label, legendX + 18, legendY);
   849→    legendY += 18;
   850→  };
   851→  drawLegend('#818cf8', 'You', 5);
   852→  drawLegend('#34d399', 'Real Players', 3);
   853→  drawLegend('#f43f5e', 'Bots', 2.5);
   854→
   855→  ctx.restore();
   856→}
