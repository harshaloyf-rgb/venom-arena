    29→
    30→type ToastType = 'success' | 'error' | 'info';
    31→type ToastFn = (msg: string, type?: ToastType) => void;
    32→
    33→function notify(msg: string, type: ToastType, onToast?: ToastFn) {
    34→  if (onToast) onToast(msg, type);
    35→  else if (type === 'error') toast.error(msg);
    36→  else if (type === 'info') toast.info(msg);
    37→  else toast.success(msg);
    38→}
    39→
    40→interface PlayerInspectorModalProps {
    41→  player?: InspectedPlayer | null;
    42→  onClose: () => void;
    43→  onToast?: ToastFn;
    44→}

    46→type Tab = 'overview' | 'stats' | 'logs' | 'loadout';

    48→interface MatchLog {
    49→  arena: string;
    50→  outcome: 'Extracted' | 'Eliminated';
    51→  chips: number;
    52→  time: string;

    56→function buildMatchHistory(p: InspectedPlayer): MatchLog[] {
    57→  const bigChip = p.bankedChips >= 10_000_000 ? 10_000_000 : 2_500_000;
    58→  return [
    59→    { arena: 'Tier-05 Crore High Roller', outcome: 'Extracted', chips: bigChip, time: '10 mins ago', kills: 14 },
    60→    { arena: 'Tier-04 Platinum Arena', outcome: 'Extracted', chips: 1_500_000, time: '2 hours ago', kills: 8 },
    61→    { arena: 'Tier-03 Viper Boundary', outcome: 'Extracted', chips: 500_000, time: '1 day ago', kills: 5 },
    62→    { arena: 'Tier-05 Crore High Roller', outcome: 'Eliminated', chips: -200_000, time: '2 days ago', kills: 3 },
    63→  ];
    64→}
