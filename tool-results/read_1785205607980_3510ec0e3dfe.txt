     3→// ============================================================================
     4→// offline-engine.ts — Pure client-side snake game engine for "Venom Arena".
     5→// ----------------------------------------------------------------------------
     6→// This module runs a complete slither.io-style game ENTIRELY in the browser.
     7→// No Socket.IO connection, no server round-trips, no chips, no minimap. It is
     8→// used by `game-canvas.tsx` whenever the active arena's id starts with
     9→// `practice-` (the three free offline practice tiers).
    10→//
    11→// Design tenets (from the locked BUILD-14 spec):
    12→//  * **No server connection** — runs purely on requestAnimationFrame.
    13→//  * **No chips** — chip-less mode. No star chips, no carried chips, no
    14→//    buy-in, no commission. Food only grows body (+1 score).
    15→//  * **No map** — mapless. No minimap, no full-map overlay.
    16→//  * **Ranking by score** (body length), not chips.
    17→//  * **Leaderboard**: top 10 by score (you + bots), collapsible.
    18→//  * **Extract any time** — hold 3 s → ends run, shows final score, ZERO XP.
    19→//  * **Death**: body collision = die + restart. Wall hit = die + restart.
    20→//    No drops (no food drop, no star drop — chips don't exist).
    21→//  * **Boost**: costs body length (every 40 frames drops 1 tail segment).
    22→//
    23→// Performance:
    24→//  * Spatial hash grid for collision + food queries (O(n)-ish, not O(n²)).
    25→//  * Snake bodies capped at MAX_BODY_LENGTH (120); rendered via downsampled
    26→//    stride-2 polylines (max 60 stroke points).
    27→//  * Viewport culling (skip off-screen snakes/food).
    28→//  * Adaptive quality: FPS < 40 for 2 s → low-quality mode (no glow / shadow).
    29→//  * Particles capped at 200.
    30→//
    31→// The engine owns its own DOM HUD overlays (appended to `canvas.parentElement`)
    32→// and tears them all down on `stop()`. The host React component just has to
    33→// instantiate, start, and forward the `onExit` / `onStateChange` callbacks.
    34→// ============================================================================

   157→// ----------------------------------------------------------------------------
   158→// Spatial hash grid — slimmed-down client port of the server's grid.
   159→// Items are bucketed into square cells; queries return a deduplicated Map.
   160→// ----------------------------------------------------------------------------
   161→
   162→class SpatialHashGrid {
   163→  private readonly cellSize: number;
   164→  private readonly cells: Map<string, Map<string, GridItem>> = new Map();
   165→
   166→  constructor(cellSize = 120) {
   167→    this.cellSize = cellSize;
   168→  }
   169→
   170→  private key(cx: number, cy: number): string {
   171→    return cx + ':' + cy;
   172→  }
   173→
   174→  clear(): void {
   175→    this.cells.clear();
   176→  }
   177→
   178→  insert(item: GridItem): void {
   179→    const minCx = Math.floor((item.x - item.radius) / this.cellSize);
   180→    const maxCx = Math.floor((item.x + item.radius) / this.cellSize);
   181→    const minCy = Math.floor((item.y - item.radius) / this.cellSize);
   182→    const maxCy = Math.floor((item.y + item.radius) / this.cellSize);
   183→    for (let cx = minCx; cx <= maxCx; cx++) {
   184→      for (let cy = minCy; cy <= maxCy; cy++) {
   185→        const k = this.key(cx, cy);
   186→        let bucket = this.cells.get(k);
   187→        if (!bucket) {
   188→          bucket = new Map();
   189→          this.cells.set(k, bucket);
   190→        }
   191→        bucket.set(item.id, item);
   192→      }
   193→    }
   194→  }
   195→
   196→  queryRadius(x: number, y: number, r: number): Map<string, GridItem> {
   197→    const out = new Map<string, GridItem>();
   198→    const minCx = Math.floor((x - r) / this.cellSize);
   199→    const maxCx = Math.floor((x + r) / this.cellSize);
   200→    const minCy = Math.floor((y - r) / this.cellSize);
   201→    const maxCy = Math.floor((y + r) / this.cellSize);
   202→    for (let cx = minCx; cx <= maxCx; cx++) {
   203→      for (let cy = minCy; cy <= maxCy; cy++) {
   204→        const bucket = this.cells.get(this.key(cx, cy));
   205→        if (!bucket) continue;
   206→        for (const [id, item] of bucket) {
   207→          if (!out.has(id)) out.set(id, item);
   208→        }
   209→      }
   210→    }
   211→    return out;
   212→  }
   213→}

   262→// ----------------------------------------------------------------------------
   263→// Math helpers
   264→// ----------------------------------------------------------------------------
   265→
   266→function angularDelta(current: number, desired: number): number {
   267→  let diff = desired - current;
   268→  while (diff > Math.PI) diff -= 2 * Math.PI;
   269→  while (diff < -Math.PI) diff += 2 * Math.PI;
   270→  return diff;
   271→}
   272→
   273→function turnToward(current: number, desired: number, maxStep: number): number {
   274→  const diff = angularDelta(current, desired);
   275→  if (Math.abs(diff) <= maxStep) return desired;
   276→  return current + Math.sign(diff) * maxStep;
   277→}
   278→
   279→function dist(ax: number, ay: number, bx: number, by: number): number {
   280→  return Math.hypot(ax - bx, ay - by);
   281→}
   282→
   283→function randomSpawnPoint(maxR: number): Vec2 {
   284→  const r = Math.sqrt(Math.random()) * maxR;
   285→  const theta = Math.random() * Math.PI * 2;
   286→  return {
   287→    x: WORLD_RADIUS + Math.cos(theta) * r,
   288→    y: WORLD_RADIUS + Math.sin(theta) * r,
   289→  };
   290→}
   291→
   292→function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   293→  const pts: Vec2[] = [];
   294→  for (let i = 0; i < length; i++) {
   295→    pts.push({
   296→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   297→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   298→    });
   299→  }
   300→  return pts;
   301→}

   577→  private spawnBot(): BotSession {

   583→    const personality = PERSONALITIES[idx % PERSONALITIES.length];

   585→    const angle = Math.random() * Math.PI * 2;
   586→    const botId = `bot-${this.arena.id}-${idx}`;
   587→    return {
   588→      id: botId,
   589→      botId,
   590→      name,
   591→      points: initialBody(spawn.x, spawn.y, angle, INITIAL_BODY_LENGTH),
   592→      angle,

   662→  private onTouchStart(e: TouchEvent): void {
   663→    if (this.joystick) return;
   664→    const t = this.findJoystickTouch(e.touches);
   665→    if (!t) return;
   666→    e.preventDefault();
   667→    const rect = this.canvas.getBoundingClientRect();
   668→    this.joystick = {
   669→      active: true,
   670→      pointerId: t.identifier,
   671→      originX: t.clientX - rect.left,
   672→      originY: t.clientY - rect.top,
   673→      curX: t.clientX - rect.left,
   674→      curY: t.clientY - rect.top,
   675→    };
   676→  }
   677→
   678→  private onTouchMove(e: TouchEvent): void {
   679→    const js = this.joystick;
   680→    if (!js) return;
   681→    let t: Touch | null = null;
   682→    for (let i = 0; i < e.touches.length; i++) {
   683→      if (e.touches[i].identifier === js.pointerId) {
   684→        t = e.touches[i];
   685→        break;
   686→      }
   687→    }
   688→    if (!t) return;
   689→    e.preventDefault();
   690→    const rect = this.canvas.getBoundingClientRect();
   691→    js.curX = t.clientX - rect.left;
   692→    js.curY = t.clientY - rect.top;
   693→    const dx = js.curX - js.originX;
   694→    const dy = js.curY - js.originY;
   695→    const d = Math.hypot(dx, dy);
   696→    const mag = Math.min(1, d / JOYSTICK_MAX_RADIUS_PX);
   697→    if (mag > JOYSTICK_DEADZONE) {
   698→      this.touchAngle = Math.atan2(dy, dx);
   699→      this.touchBoost = mag > JOYSTICK_BOOST_MAGNITUDE;
   700→    } else {
   701→      this.touchAngle = null;
   702→      this.touchBoost = false;
   703→    }
   704→  }
   705→
   706→  private onTouchEnd(e: TouchEvent): void {
   707→    const js = this.joystick;
   708→    if (!js) return;
   709→    let stillActive = false;
   710→    for (let i = 0; i < e.touches.length; i++) {
   711→      if (e.touches[i].identifier === js.pointerId) {
   712→        stillActive = true;
   713→        break;
   714→      }
   715→    }
   716→    if (!stillActive) {
   717→      this.joystick = null;
   718→      this.touchAngle = null;
   719→      this.touchBoost = false;
   720→    }
   721→  }

   723→  private findJoystickTouch(touches: TouchList): Touch | null {
   724→    const rect = this.canvas.getBoundingClientRect();
   725→    for (let i = 0; i < touches.length; i++) {
   726→      const t = touches[i];
   727→      const x = t.clientX - rect.left;
   728→      const y = t.clientY - rect.top;
   729→      if (x < rect.width / 2 && y > rect.height / 2) {
   730→        return t;
   731→      }
   732→    }
   733→    return null;
   734→  }

   837→  private frame = (now: number): void => {
   838→    if (this.stopped) return;
   839→    this.rafId = requestAnimationFrame(this.frame);
   840→    const dt = now - this.lastFrameTime;
   841→    this.lastFrameTime = now;
   842→
   843→    this.updateFps(now);
   844→    this.updateParticles(dt);
   845→
   846→    // Physics ticks at ~TICK_MS (30 Hz) cadence — we run the sim in fixed
   847→    // steps but render every rAF. Accumulator pattern prevents spiral-of-death.
   848→    if (this.state === 'playing') {
   849→      this.accumulator += dt;
   850→      let safety = 4;
   851→      while (this.accumulator >= TICK_MS && safety > 0) {
   852→        this.tickPhysics(now);
   853→        this.accumulator -= TICK_MS;
   854→        safety--;
   855→      }
   856→      if (safety === 0) this.accumulator = 0;
   857→    }
   858→
   859→    this.render(now);
   860→    this.updateHUD();
   861→  };

   865→  // --------------------------------------------------------------------------
   866→  // Physics tick (one server-tick equivalent)
   867→  // --------------------------------------------------------------------------
   868→
   869→  private tickPhysics(now: number): void {
   870→    this.tick++;
   871→    const p = this.player;
   872→    if (!p || p.isDead) return;
   873→
   874→    // 1) Compute player desired angle + boost.
   875→    const { angle, boost } = this.computePlayerInput();
   876→    if (angle !== null) p.desiredAngle = angle;
   877→    p.wantsBoost = boost;
   878→
   879→    // Extraction progress (3-second hold).
   880→    if (p.isExtracting) {
   881→      p.extractionProgress += TICK_MS;
   882→      if (p.extractionProgress >= EXTRACT_DURATION_MS) {
   883→        this.finishExtract();
   884→        return;
   885→      }
   886→    }
   887→
   888→    // 2) Move player.
   889→    this.tickSnakeMovement(p, p.desiredAngle, p.wantsBoost);
   890→
   891→    // 3) Bots: think + move.
   892→    for (const bot of this.bots.values()) {
   893→      this.tickBot(bot, now);
   894→    }
   895→
   896→    // 4) Build spatial grid for collision + food queries.
   897→    this.grid.clear();
   898→    this.insertSnakeIntoGrid(p);
   899→    for (const bot of this.bots.values()) {
   900→      if (!bot.isDead) this.insertSnakeIntoGrid(bot);
   901→    }
   902→    for (const food of this.foods) {
   903→      if (food.value > 0) {
   904→        this.grid.insert({
   905→          id: food.id,
   906→          kind: 'food',
   907→          x: food.x,
   908→          y: food.y,
   909→          radius: food.size,
   910→          value: food.value,
   911→          foodRef: food,
   912→        });
   913→      }
   914→    }
   915→
   916→    // 5) Eat food (player + bots). No star chips — only regular food (+1 score).
   917→    this.eatFood();
   918→
   919→    // 6) Collision detection (head-to-body = death; wall = death).
   920→    const deaths = this.detectCollisions(now);
   921→
   922→    // 7) Apply deaths.
   923→    let playerDied = false;
   924→    for (const d of deaths) {
   925→      if (d.deadId === p.id) {
   926→        playerDied = true;
   927→        continue;
   928→      }
   929→      const bot = this.bots.get(d.deadId);
   930→      if (bot) {
   931→        // No drops (chip-less mode) — just mark dead + respawn.
   932→        bot.isDead = true;
   933→        // Credit kill to the killer if it's the player.
   934→        if (d.killerId === p.id) {
   935→          p.kills++;
   936→        }
   937→      }
   938→    }
   939→
   940→    // 8) Remove dead bots and respawn to maintain count.
   941→    if (deaths.length > 0) {
   942→      const toRemove: string[] = [];
   943→      for (const [id, bot] of this.bots) {
   944→        if (bot.isDead) toRemove.push(id);
   945→      }
   946→      for (const id of toRemove) {
   947→        this.bots.delete(id);
   948→      }
   949→    }
   950→    // Respawn bots to maintain count.
   951→    while (this.bots.size < this.arena.botsCount) {
   952→      const bot = this.spawnBot();
   953→      this.bots.set(bot.id, bot);
   954→    }
   955→
   956→    // 9) Player death → death screen.
   957→    if (playerDied) {
   958→      p.isDead = true;
   959→      this.handlePlayerDeath();
   960→      return;
   961→    }
   962→
   963→    // 10) Replenish food toward target.
   964→    this.replenishFood();
   965→
   966→    // 11) Expire chat bubbles.
   967→    this.expireChat(now);
   968→
   969→    // 12) Update camera target.
   970→    this.updateCamera();
   971→  }

   973→  // --------------------------------------------------------------------------
   974→  // Snake movement (server-authoritative formula)
   975→  // --------------------------------------------------------------------------
   976→
   977→  private tickSnakeMovement(snake: SnakeBase, desiredAngle: number, wantsBoost: boolean): void {
   978→    if (snake.points.length === 0 || snake.isDead) return;
   979→
   980→    // Turn rate: max(0.045, 0.15 - score*0.0006).
   981→    const turnRate = Math.max(TURN_MIN, TURN_BASE - snake.score * TURN_SCORE_FACTOR);
   982→    snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
   983→
   984→    // Speed.
   985→    let speed = BASE_SPEED;
   986→    if (snake.isExtracting) {
   987→      speed = EXTRACT_GLIDE_SPEED;
   988→    } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
   989→      speed = BOOST_SPEED;
   990→      snake.boostFrameCounter++;
   991→      if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
   992→        snake.boostFrameCounter = 0;
   993→        if (snake.points.length > BOOST_MIN_LENGTH) {
   994→          snake.points.pop();
   995→          snake.score = Math.max(0, snake.score - 1);
   996→        }
   997→      }
   998→    }
   999→
  1000→    // Move head.
  1001→    const head = snake.points[0];
  1002→    const nx = head.x + Math.cos(snake.angle) * speed;
  1003→    const ny = head.y + Math.sin(snake.angle) * speed;
  1004→    snake.points.unshift({ x: nx, y: ny });
  1005→
  1006→    // Grow / shrink body to target length based on SCORE.
  1007→    const targetLen = Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + snake.score);
  1008→    while (snake.points.length > targetLen) snake.points.pop();
  1009→
  1010→    // Size: 8 + sqrt(score) * 0.4.
  1011→    snake.size = SIZE_BASE + Math.sqrt(snake.score) * SIZE_SCORE_FACTOR;
  1012→  }
  1013→
  1014→  // --------------------------------------------------------------------------
  1015→  // Bot AI (scavenger / opportunist / hunter / extractor / coward)
  1016→  // --------------------------------------------------------------------------
  1017→
  1018→  private tickBot(bot: BotSession, now: number): void {
  1019→    if (bot.points.length === 0 || bot.isDead) return;
  1020→
  1021→    if (now >= bot.nextThinkAt) {
  1022→      bot.nextThinkAt = now + BOT_THINK_INTERVAL_MS + Math.floor(Math.random() * BOT_THINK_JITTER_MS);
  1023→      const head = bot.points[0];
  1024→
  1025→      // Find nearest food.
  1026→      let bestFood: GridItem | null = null;
  1027→      let bestFoodDist = Infinity;
  1028→      const foodQuery = this.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
  1029→      for (const item of foodQuery.values()) {
  1030→        if (item.kind !== 'food') continue;
  1031→        if ((item.value ?? 0) <= 0) continue;
  1032→        const d = dist(head.x, head.y, item.x, item.y);
  1033→        if (d < bestFoodDist) {
  1034→          bestFoodDist = d;
  1035→          bestFood = item;
  1036→        }
  1037→      }
  1038→
  1039→      // Find nearest threat (foreign body segment within threat scan radius).
  1040→      let threatX = 0;
  1041→      let threatY = 0;
  1042→      let threatDist = Infinity;
  1043→      const threatQuery = this.grid.queryRadius(head.x, head.y, BOT_THREAT_SCAN_RADIUS);
  1044→      for (const item of threatQuery.values()) {
  1045→        if (item.kind !== 'segment') continue;
  1046→        if (item.snakeId === bot.id) continue;
  1047→        if (item.segIdx === 0) continue;
  1048→        const d = dist(head.x, head.y, item.x, item.y);
  1049→        if (d < threatDist) {
  1050→          threatDist = d;
  1051→          threatX = item.x;
  1052→          threatY = item.y;
  1053→        }
  1054→      }
  1055→
  1056→      let desired: number;
  1057→      const shouldFlee =
  1058→        threatDist < 140 &&
  1059→        (bot.personality === 'coward' ||
  1060→          bot.personality === 'extractor' ||
  1061→          (bot.personality === 'opportunist' && threatDist < 90));
  1062→
  1063→      if (shouldFlee) {
  1064→        desired = Math.atan2(head.y - threatY, head.x - threatX);
  1065→      } else if (bestFood) {
  1066→        desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
  1067→      } else {
  1068→        desired = bot.angle + (Math.random() - 0.5) * 0.4;
  1069→      }
  1070→
  1071→      // Edge avoidance: turn back toward center if near the wall.
  1072→      const headR = Math.hypot(head.x - WORLD_RADIUS, head.y - WORLD_RADIUS);
  1073→      if (headR > MAP_BASE_RADIUS - BOT_EDGE_AVOID_RADIUS) {
  1074→        const toCenter = Math.atan2(WORLD_RADIUS - head.y, WORLD_RADIUS - head.x);
  1075→        desired = turnToward(desired, toCenter, BOT_MAX_TURN_PER_TICK * 2);
  1076→      }
  1077→
  1078→      bot.desiredAngle = desired;
  1079→    }
  1080→
  1081→    // Hunter personality occasionally boosts (no chips required, just body length).
  1082→    const wantsBoost = bot.personality === 'hunter' && bot.score > 5 && Math.random() < 0.05;
  1083→    this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1084→  }

  1090→  private insertSnakeIntoGrid(snake: SnakeBase): void {
  1091→    const pts = snake.points;

  1095→        id: `${snake.id}:${i}`,
  1096→        kind: 'segment',
  1097→        x: pts[i].x,
  1098→        y: pts[i].y,
  1099→        radius: snake.size,
  1100→        snakeId: snake.id,
  1101→        segIdx: i,
  1102→      });
  1103→    }
  1104→  }

  1110→  private eatFood(): void {
  1111→    const p = this.player;
  1112→    if (p && !p.isDead) this.eatFoodForSnake(p);
  1113→    for (const bot of this.bots.values()) {
  1114→      if (!bot.isDead) this.eatFoodForSnake(bot);
  1115→    }
  1116→  }
  1117→
  1118→  private eatFoodForSnake(snake: SnakeBase): void {
  1119→    if (snake.points.length === 0) return;
  1120→    const head = snake.points[0];
  1121→    const nearby = this.grid.queryRadius(head.x, head.y, snake.size + 20);
  1122→    for (const item of nearby.values()) {
  1123→      if (item.kind !== 'food') continue;
  1124→      if ((item.value ?? 0) <= 0) continue;
  1125→      const d = dist(head.x, head.y, item.x, item.y);
  1126→      if (d < snake.size + (item.radius || 6) + 6) {
  1127→        // Regular food only (+1 score, no chips). No star chips in offline mode.
  1128→        snake.score += REGULAR_FOOD_GROW;
  1129→        item.value = 0;
  1130→        if (item.foodRef) item.foodRef.value = 0;
  1131→        // Spawn small particle burst for player only.
  1132→        if (snake.isPlayer) {
  1133→          this.spawnEatParticles(item.x, item.y, item.radius || 6);
  1134→        }
  1135→      }
  1136→    }
  1137→  }

  1141→  // --------------------------------------------------------------------------

  1143→  private detectCollisions(now: number): { deadId: string; killerId?: string; cause: 'body' | 'wall' }[] {
  1144→    const deaths: { deadId: string; killerId?: string; cause: 'body' | 'wall' }[] = [];
  1145→    const seenDead = new Set<string>();
  1146→    const all: SnakeBase[] = [];
  1147→    if (this.player && !this.player.isDead) all.push(this.player);
  1148→    for (const bot of this.bots.values()) {
  1149→      if (!bot.isDead) all.push(bot);
  1150→    }
  1151→
  1152→    const mapRadius = getArenaRadius(now);
  1153→
  1154→    for (const snake of all) {
  1155→      if (snake.isDead) continue;
  1156→      if (snake.points.length === 0) continue;
  1157→      if (now < snake.spawnProtectedUntil) continue;
  1158→      if (seenDead.has(snake.id)) continue;
  1159→      const head = snake.points[0];
  1160→
  1161→      // Wall collision: outside breathing map radius.
  1162→      const distFromCenter = Math.hypot(head.x - WORLD_RADIUS, head.y - WORLD_RADIUS);
  1163→      if (distFromCenter > mapRadius) {
  1164→        deaths.push({ deadId: snake.id, killerId: 'wall', cause: 'wall' });
  1165→        seenDead.add(snake.id);
  1166→        continue;
  1167→      }
  1168→
  1169→      // Body collision: head vs foreign non-head segment.
  1170→      const queryR = snake.size + 30;
  1171→      const nearby = this.grid.queryRadius(head.x, head.y, queryR);
  1172→      for (const item of nearby.values()) {
  1173→        if (item.kind !== 'segment') continue;
  1174→        if (item.snakeId === snake.id) continue;
  1175→        if (item.segIdx === 0) continue; // head-to-head ignored (no self-collision; head-to-head doesn't kill either)
  1176→        const d = dist(head.x, head.y, item.x, item.y);
  1177→        if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
  1178→          deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
  1179→          seenDead.add(snake.id);
  1180→          break;
  1181→        }
  1182→      }
  1183→    }
  1184→    return deaths;
  1185→  }

  1262→  private computePlayerInput(): { angle: number | null; boost: boolean } {

  1288→        return { angle: Math.atan2(dy, dx), boost: spaceHeld };

  1484→  private updateFps(now: number): void {
  1485→    const acc = this.fpsAccum;
  1486→    acc.frames += 1;
  1487→    if (acc.lastSecond === 0) acc.lastSecond = now;
  1488→    const dt = now - acc.lastSecond;
  1489→    if (dt >= 1000) {
  1490→      const measured = (acc.frames * 1000) / dt;
  1491→      this.fps = Math.round(measured);
  1492→      if (measured < FPS_LOW_THRESHOLD) {
  1493→        if (acc.lowSince === 0) acc.lowSince = now;
  1494→        if (acc.highSince !== 0) acc.highSince = 0;
  1495→        if (now - acc.lowSince >= FPS_LOW_DURATION_MS && !this.lowQuality) {
  1496→          this.lowQuality = true;
  1497→        }
  1498→      } else if (measured > FPS_HIGH_THRESHOLD) {
  1499→        if (acc.highSince === 0) acc.highSince = now;
  1500→        if (acc.lowSince !== 0) acc.lowSince = 0;
  1501→        if (now - acc.highSince >= FPS_HIGH_DURATION_MS && this.lowQuality) {
  1502→          this.lowQuality = false;
  1503→        }
  1504→      } else {
  1505→        acc.lowSince = 0;
  1506→        acc.highSince = 0;
  1507→      }
  1508→      acc.frames = 0;
  1509→      acc.lastSecond = now;
  1510→    }
  1511→  }
