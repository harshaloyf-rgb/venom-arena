     1→// ============================================================================
     2→// index.ts — Venom Arena Socket.IO game server (mini-service, port 3001).
     3→// ----------------------------------------------------------------------------
     4→// Server-authoritative multiplayer snake. The Next.js app (port 3000) talks
     5→// to this server via the Caddy gateway using io("/?XTransformPort=3001").
     6→//
     7→// Architectural fixes vs the OLD broken server (see worklog ANALYSIS-1):
     8→//  * Auth is mandatory — sockets without a valid JWT (verified via
     9→//    POST /api/match/verify) are disconnected at the middleware. No
    10→//    "auth optional, admit anyway" path.
    11→//  * Identity is the verify response only. We never trust any subsequent
    12→//    client-supplied userTag/name/color/skin.
    13→//  * One socket per userTag — a new connection kicks the prior socket.
    14→//  * Movement is server-authoritative — clients send only `angle`; the
    15→//    server computes position. Teleport is impossible.
    16→//  * Buy-in is deducted atomically by /api/match/join (Next.js + Prisma
    17→//    transaction). We never touch the DB directly.
    18→//  * Match results are reported exactly once via /api/match/result, guarded
    19→//    by a `matchSettling` flag to prevent double-credit on disconnect races.
    20→//  * Recursive setTimeout (not setInterval) for both tick and broadcast
    21→//    loops — a slow tick can't overlap the next.
    22→//  * Every tick is wrapped in try/catch; one bad snake never kills the loop.
    23→//  * Spatial hash grid (see spatial-grid.ts) keeps collision detection
    24→//    near-linear instead of O(n²).
    25→//  * Broadcasts happen at 20 Hz, not every tick, and snapshot points are
    26→//    downsampled to 60 to keep payloads small.
    27→//  * uncaughtException / unhandledRejection are logged, never fatal.
    28→//  * Graceful shutdown on SIGINT/SIGTERM broadcasts `server_shutdown` then exits.
    29→// ============================================================================

    87→type LogLevel = 'info' | 'warn' | 'error';
    88→function log(level: LogLevel, msg: string): void {
    89→  const now = new Date();
    90→  const hh = String(now.getHours()).padStart(2, '0');
    91→  const mm = String(now.getMinutes()).padStart(2, '0');
    92→  const ss = String(now.getSeconds()).padStart(2, '0');
    93→  const prefix = `[${hh}:${mm}:${ss}] [${level.toUpperCase()}]`;
    94→  if (level === 'error') {
    95→    process.stderr.write(`${prefix} ${msg}\n`);
    96→  } else {
    97→    process.stdout.write(`${prefix} ${msg}\n`);
    98→  }
    99→}

   126→/** Call POST /api/match/verify to validate the client's JWT and fetch its identity. */
   127→async function verifyToken(token: string): Promise<VerifyResponse> {
   128→  const controller = new AbortController();
   129→  const timer = setTimeout(() => controller.abort(), HTTP_TIMEOUT_MS);
   130→  try {
   131→    const res = await fetch(`${NEXT_APP_URL}/api/match/verify`, {
   132→      method: 'POST',
   133→      headers: { 'Content-Type': 'application/json', 'x-internal-secret': INTERNAL_SECRET },
   134→      body: JSON.stringify({ token }),
   135→      signal: controller.signal,
   136→    });
   137→    return (await res.json()) as VerifyResponse;
   138→  } finally {
   139→    clearTimeout(timer);
   140→  }
   141→}

   172→/** Call POST /api/match/result — credits/debits the player's account. */
   173→async function reportMatchResult(payload: ReportResultPayload): Promise<MatchResultResponse | null> {
   174→  const controller = new AbortController();
   175→  const timer = setTimeout(() => controller.abort(), HTTP_TIMEOUT_MS);
   176→  try {
   177→    const res = await fetch(`${NEXT_APP_URL}/api/match/result`, {
   178→      method: 'POST',
   179→      headers: { 'Content-Type': 'application/json', 'x-internal-secret': INTERNAL_SECRET },
   180→      body: JSON.stringify(payload),
   181→      signal: controller.signal,
   182→    });
   183→    if (!res.ok) {
   184→      log('error', `match/result HTTP ${res.status} for ${payload.userTag}`);
   185→      return null;
   186→    }
   187→    return (await res.json()) as MatchResultResponse;
   188→  } catch (err) {
   189→    log('error', `match/result fetch failed for ${payload.userTag}: ${(err as Error).message}`);
   190→    return null;
   191→  } finally {
   192→    clearTimeout(timer);
   193→  }
   194→}

   223→/** Max real players per arena shard before auto-creating a new instance. */
   224→const MAX_PLAYERS_PER_SHARD = 1000;
   225→
   226→/**
   227→ * Get-or-create an ArenaRoom for `arenaId`. If the primary shard is full
   228→ * (>= MAX_PLAYERS_PER_SHARD real players), find or create the next shard.
   229→ * Shards are keyed as `{arenaId}` (shard 0), `{arenaId}#2` (shard 1), etc.
   230→ */
   231→function getOrCreateRoom(arenaId: string): ArenaRoom | null {
   232→  const arena = getArenaById(arenaId);
   233→  if (!arena) return null;
   234→
   235→  // Try shard 0, then 1, 2, ... until we find one with capacity.
   236→  let shardIdx = 0;
   237→  while (true) {
   238→    const roomKey = shardIdx === 0 ? arenaId : `${arenaId}#${shardIdx + 1}`;
   239→    let room = rooms.get(roomKey);
   240→    if (room && room.players.size < MAX_PLAYERS_PER_SHARD) {
   241→      return room;
   242→    }
   243→    if (!room) {
   244→      // Create a new shard.
   245→      room = createArenaRoom(arena);
   246→      rooms.set(roomKey, room);
   247→      ensureBots(room);
   248→      log('info', `Created arena shard ${roomKey} with ${room.bots.size} bots (shard ${shardIdx})`);
   249→      return room;
   250→    }
   251→    shardIdx++;
   252→    // Safety: don't infinite-loop if somehow all shards are full (shouldn't happen).
   253→    if (shardIdx > 200) {
   254→      log('error', `Arena ${arenaId} has >200 shards, rejecting join`);
   255→      return null;
   256→    }
   257→  }
   258→}

   287→/**
   288→ * Settle a player's match: call /api/match/result, emit `match_result` to
   289→ * the player, and remove them from the room. Idempotent via the
   290→ * `matchSettling` flag — concurrent calls short-circuit.
   291→ */
   292→async function settleMatch(
   293→  room: ArenaRoom,
   294→  session: PlayerSession,
   295→  outcome: 'extract' | 'death',
   296→  killer?: SnakeBase,
   297→): Promise<void> {
   298→  if (session.matchSettling) return;
   299→  session.matchSettling = true;
   300→  session.isExtracting = false;
   301→  session.extractionProgress = 0;
   302→
   303→  const now = Date.now();
   304→  const durationSeconds = Math.max(0, Math.floor((now - session.joinedAt) / 1000));
   305→  const carriedChips = Math.max(0, Math.floor(session.carriedChips));
   306→  const kills = session.kills;
   307→  const killerTag = killer?.userTag;
   308→  const score = session.score;
   309→
   310→  // Dynamic commission: 0% if <=3 real players in arena, 35% if >=4.
   311→  const realPlayerCount = room.players.size;
   312→  const commissionRate = realPlayerCount <= 3 ? 0 : 0.35;
   313→  const commission = outcome === 'extract' ? Math.floor(carriedChips * commissionRate) : 0;
   314→  const bankedAmount = outcome === 'extract' ? (carriedChips - commission) : 0;
   315→
   316→  const result = await reportMatchResult({
   317→    userTag: session.identity.userTag,
   318→    arenaId: room.arena.id,
   319→    outcome,
   320→    carriedChips,
   321→    kills,
   322→    durationSeconds,
   323→    killerTag,
   324→    score,
   325→    bankedAmount,
   326→  });
   327→
   328→  const payload: MatchResultClientPayload = {
   329→    outcome,
   330→    arenaId: room.arena.id,
   331→    arenaName: room.arena.name,
   332→    chipsExtracted: outcome === 'extract' ? carriedChips : 0,
   333→    commission,
   334→    bankedAmount,
   335→    kills,
   336→    score,
   337→    xpGained: result?.xpGained ?? 0,
   338→    newLevel: result?.newLevel ?? session.identity.level,
   339→    newBankedChips: result?.newBankedChips ?? 0,
   340→    durationSeconds,
   341→  };
   342→
   343→  const socket = io.sockets.sockets.get(session.id);
   344→  if (socket) {
   345→    socket.emit('match_result', payload);
   346→    if (outcome === 'death') {
   347→      socket.emit('death', {
   348→        killerId: killer?.id,
   349→        killerName: killer?.name,
   350→        killerTag: killer?.userTag,
   351→        killerColor: killer?.color,
   352→      });
   353→    }
   354→  }
   355→
   356→  room.players.delete(session.id);
   357→  if (socket) {
   358→    const data = getSocketData(socket);
   359→    data.playerSession = undefined;
   360→  }
   361→  log('info', `${session.identity.userTag} ${outcome} in ${room.arena.id}: chips=${carriedChips} kills=${kills} dur=${durationSeconds}s`);
   362→}

   364→// ----------------------------------------------------------------------------
   365→// Tick loop (recursive setTimeout — a slow tick can't overlap the next)
   366→// ----------------------------------------------------------------------------
   367→
   368→function tickRoom(room: ArenaRoom, now: number): void {
   369→  room.tick++;
   370→
   371→  // 1) Rebuild spatial grid from scratch each tick.
   372→  room.grid.clear();
   373→
   374→  // 2) Insert all snake body segments (skip dead / empty).
   375→  for (const snake of collectAllSnakes(room)) {
   376→    if (snake.isDead) continue;
   377→    if (snake.points.length === 0) continue;
   378→    for (let i = 0; i < snake.points.length; i++) {
   379→      const p = snake.points[i];
   380→      room.grid.insert({
   381→        id: `${snake.id}:${i}`,
   382→        kind: 'segment',
   383→        x: p.x,
   384→        y: p.y,
   385→        radius: snake.size,
   386→        snakeId: snake.id,
   387→        segIdx: i,
   388→      });
   389→    }
   390→  }
   391→
   392→  // 3) Insert all food (skip already-eaten with value=0).
   393→  for (const food of room.foods) {
   394→    if (food.value <= 0) continue;
   395→    room.grid.insert({
   396→      id: food.id,
   397→      kind: 'food',
   398→      x: food.x,
   399→      y: food.y,
   400→      radius: food.size,
   401→      value: food.value,
   402→      isStarChip: food.isStarChip,
   403→      color: food.color,
   404→      foodRef: food, // <-- reference to the REAL food object so eatFood can zero it
   405→    });
   406→  }
   407→
   408→  // 4) Bot AI tick.
   409→  for (const bot of room.bots.values()) {
   410→    if (bot.isDead) continue;
   411→    try {
   412→      tickBot(bot, room, now);
   413→    } catch (err) {
   414→      log('error', `Bot tick error in ${room.arena.id}: ${(err as Error).message}`);
   415→    }
   416→  }
   417→
   418→  // 5) Player movement (server-authoritative — apply desired angle + speed).
   419→  for (const session of room.players.values()) {
   420→    if (session.isDead || session.matchSettling) continue;
   421→    try {
   422→      tickSnakeMovement(session, session.desiredAngle, session.wantsBoost);
   423→    } catch (err) {
   424→      log('error', `Player movement error: ${(err as Error).message}`);
   425→    }
   426→  }
   427→
   428→  // 6) Collision detection (collect deaths, apply after iteration).
   429→  let deaths: Awaited<ReturnType<typeof detectCollisions>> = [];
   430→  try {
   431→    deaths = detectCollisions(room, now);
   432→  } catch (err) {
   433→    log('error', `Collision detection error in ${room.arena.id}: ${(err as Error).message}`);
   434→  }
   435→
   436→  // 7) Apply deaths with CORRECT drop rules:
   437→  //    Body collision death: drop food (all) + star chips (real players only) at body positions.
   438→  //    Wall death: drop star chips only (real players only). Bots drop nothing on wall death.
   439→  for (const death of deaths) {
   440→    const dead = room.players.get(death.deadId) || (room.bots.get(death.deadId) as SnakeBase | undefined);
   441→    if (!dead || dead.isDead) continue;
   442→    dead.isDead = true;
   443→
   444→    if (death.cause === 'body') {
   445→      // Body collision: drop food at body (ALL snakes) + star chips (real players only).
   446→      dropFoodAtBody(room, dead.points, dead.color);
   447→      if (dead.isPlayer && dead.carriedChips > 0) {
   448→        dropStarChipsAtBody(room, dead.points, dead.carriedChips);
   449→      }
   450→    } else {
   451→      // Wall death: drop star chips ONLY (real players only). Bots drop nothing.
   452→      if (dead.isPlayer && dead.carriedChips > 0) {
   453→        dropStarChipsAtBody(room, dead.points, dead.carriedChips);
   454→      }
   455→    }
   456→
   457→    // Credit killer's kill counter (only if killer is alive). Bots don't track kills.
   458→    const killer = death.killerId && death.killerId !== 'wall'
   459→      ? (room.players.get(death.killerId) ?? room.bots.get(death.killerId))
   460→      : undefined;
   461→    if (killer && !killer.isDead && killer.isPlayer) {
   462→      (killer as PlayerSession).kills += 1;
   463→    }
   464→
   465→    if (dead.isPlayer) {
   466→      const session = dead as PlayerSession;
   467→      log('info', `${session.identity.userTag} died in ${room.arena.id} (cause=${death.cause}, killer=${death.killerId ?? 'none'})`);
   468→      // Settle asynchronously — never throw out of the tick loop.
   469→      settleMatch(room, session, 'death', death.cause === 'wall' ? undefined : killer).catch((err) => {
   470→        log('error', `settleMatch(death) failed: ${(err as Error).message}`);
   471→      });
   472→    } else {
   473→      // Bot dies: remove + respawn a fresh one to keep the bot count stable.
   474→      room.bots.delete(dead.id);
   475→      try {
   476→        const fresh = spawnNewBot(room);
   477→        if (fresh) room.bots.set(fresh.id, fresh);
   478→      } catch (err) {
   479→        log('error', `Bot respawn error: ${(err as Error).message}`);
   480→      }
   481→    }
   482→  }
   483→
   484→  // 8) Food collision (credits carriedChips, marks food as eaten via sentinel).
   485→  try {
   486→    eatFood(room);
   487→  } catch (err) {
   488→    log('error', `eatFood error in ${room.arena.id}: ${(err as Error).message}`);
   489→  }
   490→
   491→  // 9) Replenish food up to target.
   492→  try {
   493→    replenishFood(room);
   494→  } catch (err) {
   495→    log('error', `replenishFood error in ${room.arena.id}: ${(err as Error).message}`);
   496→  }
   497→
   498→  // 10) Extraction progress for players. NO zone check — extract anywhere.
   499→  for (const session of room.players.values()) {
   500→    if (session.isDead || session.matchSettling) continue;
   501→    if (!session.isExtracting) continue;
   502→
   503→    session.extractionProgress += TICK_MS;
   504→    const progress = Math.min(1, session.extractionProgress / EXTRACT_DURATION_MS);
   505→    const sock = io.sockets.sockets.get(session.id);
   506→    if (sock) sock.emit('extract_progress', { progress });
   507→
   508→    if (session.extractionProgress >= EXTRACT_DURATION_MS) {
   509→      // Extraction success — settle asynchronously.
   510→      settleMatch(room, session, 'extract').catch((err) => {
   511→        log('error', `settleMatch(extract) failed: ${(err as Error).message}`);
   512→      });
   513→    }
   514→  }
   515→
   516→  // 11) Recompute leader + expire chat bubbles.
   517→  try {
   518→    recomputeLeader(room);
   519→  } catch (err) {
   520→    log('error', `recomputeLeader error: ${(err as Error).message}`);
   521→  }
   522→  try {
   523→    expireChat(room, now);
   524→  } catch (err) {
   525→    log('error', `expireChat error: ${(err as Error).message}`);
   526→  }
   527→}

   529→function tickOnce(): void {
   530→  const now = Date.now();
   531→  try {
   532→    for (const room of rooms.values()) {
   533→      // Skip empty rooms (no players) for CPU savings — bots stay idle.
   534→      if (room.players.size === 0) continue;
   535→      try {
   536→        tickRoom(room, now);
   537→      } catch (err) {
   538→        log('error', `Tick error in arena ${room.arena.id}: ${(err as Error).message}`);
   539→      }
   540→    }
   541→  } catch (err) {
   542→    log('error', `Tick loop fatal: ${(err as Error).message}`);
   543→  } finally {
   544→    setTimeout(tickOnce, TICK_MS);
   545→  }
   546→}

   557→// ----------------------------------------------------------------------------
   558→// Broadcast loop (20 Hz — independent of tick rate)
   559→// ----------------------------------------------------------------------------
   560→
   561→function broadcastOnce(): void {
   562→  try {
   563→    for (const room of rooms.values()) {
   564→      if (room.players.size === 0) continue;
   565→      // Build one base snapshot per room, then customize yourRank per viewer.
   566→      for (const socketId of room.players.keys()) {
   567→        let snapshot: ReturnType<typeof buildSnapshot>;
   568→        try {
   569→          snapshot = buildSnapshot(room, socketId);
   570→        } catch (err) {
   571→          log('error', `Snapshot build error in ${room.arena.id}: ${(err as Error).message}`);
   572→          continue;
   573→        }
   574→        io.to(socketId).emit('snapshot', snapshot);
   575→      }
   576→    }
   577→  } catch (err) {
   578→    log('error', `Broadcast loop fatal: ${(err as Error).message}`);
   579→  } finally {
   580→    setTimeout(broadcastOnce, BROADCAST_MS);
   581→  }
   582→}

   585→// Socket.IO auth middleware

   588→io.use(async (socket, next) => {
   589→  const token = (socket.handshake.auth as { token?: unknown })?.token;
   590→  if (typeof token !== 'string' || token.length === 0) {
   591→    return next(new Error('No token provided'));
   592→  }
   593→  try {
   594→    const result = await verifyToken(token);
   595→    if (!result.ok || !result.player) {
   596→      return next(new Error(result.reason || 'invalid_token'));
   597→    }
   598→    getSocketData(socket).identity = result.player;
   599→    return next();
   600→  } catch (err) {
   601→    log('warn', `Verify fetch failed: ${(err as Error).message}`);
   602→    return next(new Error('verify_failed'));
   603→  }
   604→});

   606→// ----------------------------------------------------------------------------
   607→// Connection handler
   608→// ----------------------------------------------------------------------------
   609→
   610→io.on('connection', (socket: Socket) => {
   611→  const data = getSocketData(socket);
   612→  const identity = data.identity;
   613→  if (!identity) {
   614→    // Should never happen — middleware rejects unauthed sockets.
   615→    socket.disconnect(true);
   616→    return;
   617→  }
   618→
   619→  // One socket per userTag: kick any prior session.
   620→  const prior = userTagToSocket.get(identity.userTag);
   621→  if (prior && prior !== socket.id) {
   622→    const priorSock = io.sockets.sockets.get(prior);
   623→    if (priorSock) {
   624→      priorSock.emit('kicked', { reason: 'Another session opened for this account' });
   625→      priorSock.disconnect(true);
   626→    }
   627→  }
   628→  userTagToSocket.set(identity.userTag, socket.id);
   629→  log('info', `Socket connected: ${identity.userTag} (${socket.id})`);
   630→
   631→  // ----- join_arena -----
   632→  socket.on('join_arena', (payload: unknown) => {
   633→    void handleJoinArena(socket, payload);
   634→  });
   635→
   636→  // ----- input -----
   637→  socket.on('input', (payload: unknown) => {
   638→    handleInput(socket, payload);
   639→  });
   640→
   641→  // ----- extract -----
   642→  socket.on('extract', () => {
   643→    handleExtract(socket);
   644→  });
   645→
   646→  // ----- cancel_extract -----
   647→  socket.on('cancel_extract', () => {
   648→    handleCancelExtract(socket);
   649→  });
   650→
   651→  // ----- chat -----
   652→  socket.on('chat', (payload: unknown) => {
   653→    handleChat(socket, payload);
   654→  });
   655→
   656→  // ----- leave -----
   657→  socket.on('leave', () => {
   658→    handleLeave(socket, 'client requested');
   659→  });
   660→
   661→  // ----- disconnect -----
   662→  socket.on('disconnect', (reason: string) => {
   663→    handleLeave(socket, `disconnected (${reason})`);
   664→    // Clean up userTag → socket mapping (only if we still own it).
   665→    const ident = getSocketData(socket).identity;
   666→    if (ident && userTagToSocket.get(ident.userTag) === socket.id) {
   667→      userTagToSocket.delete(ident.userTag);
   668→    }
   669→    log('info', `Socket disconnected: ${ident?.userTag ?? 'unknown'} (${reason})`);
   670→  });
   671→
   672→  socket.on('error', (err: Error) => {
   673→    log('error', `Socket error (${socket.id}): ${err.message}`);
   674→  });
   675→});

   681→async function handleJoinArena(socket: Socket, payload: unknown): Promise<void> {
   682→  const data = getSocketData(socket);
   683→  const identity = data.identity;
   684→  if (!identity) return;
   685→
   686→  if (!payload || typeof payload !== 'object') {
   687→    socket.emit('join_error', { reason: 'invalid_arena' });
   688→    return;
   689→  }
   690→  const arenaId = String((payload as { arenaId?: unknown }).arenaId || '');
   691→  const arena = getArenaById(arenaId);
   692→  if (!arena) {
   693→    socket.emit('join_error', { reason: 'invalid_arena' });
   694→    return;
   695→  }
   696→
   697→  // Already in a match? Reject.
   698→  if (data.playerSession) {
   699→    socket.emit('join_error', { reason: 'already_in_match' });
   700→    return;
   701→  }
   702→
   703→  // Server-to-server: atomically deduct buyIn on the Next.js side.
   704→  let joinResult;
   705→  try {
   706→    joinResult = await joinMatch(identity.userTag, arenaId);
   707→  } catch (err) {
   708→    log('error', `joinMatch fetch failed: ${(err as Error).message}`);
   709→    socket.emit('join_error', { reason: 'invalid_arena' });
   710→    return;
   711→  }
   712→
   713→  if (!joinResult.ok) {
   714→    const reason = joinResult.reason;
   715→    if (reason === 'insufficient_chips') {
   716→      socket.emit('join_error', { reason: 'insufficient_chips' });
   717→      return;
   718→    }
   719→    if (reason === 'banned') {
   720→      socket.emit('join_error', { reason: 'banned' });
   721→      socket.emit('kicked', { reason: 'Account banned' });
   722→      socket.disconnect(true);
   723→      return;
   724→    }
   725→    socket.emit('join_error', { reason: 'invalid_arena' });
   726→    return;
   727→  }
   728→
   729→  // Spawn the player.
   730→  const room = getOrCreateRoom(arenaId);
   731→  if (!room) {
   732→    socket.emit('join_error', { reason: 'invalid_arena' });
   733→    return;
   734→  }
   735→
   736→  const spawn = randomSpawnPoint(MAP_BASE_RADIUS - 200);
   737→  const angle = Math.random() * Math.PI * 2;
   738→  const session: PlayerSession = {
   739→    id: socket.id,
   740→    name: identity.name,
   741→    userTag: identity.userTag,
   742→    country: identity.country,
   743→    points: initialBody(spawn.x, spawn.y, angle, INITIAL_BODY_LENGTH),
   744→    angle,
   745→    size: SIZE_BASE,
   746→    color: identity.color,
   747→    secondaryColor: identity.secondaryColor,
   748→    isPlayer: true,
   749→    isBot: false,
   750→    carriedChips: 0,
   751→    score: 0,
   752→    boostFrameCounter: 0,
   753→    isExtracting: false,
   754→    extractionProgress: 0,
   755→    isDead: false,
   756→    spawnProtectedUntil: Date.now() + RESPAWN_INVULN_MS,
   757→    identity,
   758→    desiredAngle: angle,
   759→    wantsBoost: false,
   760→    kills: 0,
   761→    joinedAt: Date.now(),
   762→    lastInputAt: 0,
   763→    inputDropCount: 0,
   764→    lastChatAt: 0,
   765→    arenaId,
   766→    matchSettling: false,
   767→  };
   768→
   769→  room.players.set(socket.id, session);
   770→  data.playerSession = session;
   771→
   772→  socket.emit('joined', {
   773→    arenaId,
   774→    worldSize: WORLD_SIZE,
   775→    yourId: socket.id,
   776→  });
   777→
   778→  log('info', `${identity.userTag} joined ${arenaId} (buyIn=${arena.buyIn})`);
   779→}

   781→/** Validate + rate-limit a client input packet. Server applies position; client only sends angle. */
   782→function handleInput(socket: Socket, payload: unknown): void {
   783→  const session = getSocketData(socket).playerSession;
   784→  if (!session || session.isDead || session.matchSettling) return;
   785→
   786→  if (!payload || typeof payload !== 'object') return;
   787→  const obj = payload as { angle?: unknown; wantsBoost?: unknown };
   788→  const angleRaw = obj.angle;
   789→  const wantsBoostRaw = obj.wantsBoost;
   790→
   791→  if (typeof angleRaw !== 'number' || !Number.isFinite(angleRaw)) return;
   792→  if (typeof wantsBoostRaw !== 'boolean') return;
   793→
   794→  // Normalize angle into [0, 2π).
   795→  let angle = angleRaw;
   796→  while (angle < 0) angle += 2 * Math.PI;
   797→  while (angle >= 2 * Math.PI) angle -= 2 * Math.PI;
   798→
   799→  // Rate-limit: max MAX_SNAPSHOTS_PER_SECOND per second.
   800→  const now = Date.now();
   801→  const delta = now - session.lastInputAt;
   802→  if (delta < INPUT_MIN_INTERVAL_MS) {
   803→    session.inputDropCount++;
   804→    if (session.inputDropCount % 20 === 0) {
   805→      log('warn', `${session.identity.userTag} input flood: ${session.inputDropCount} packets dropped`);
   806→    }
   807→    return;
   808→  }
   809→  session.lastInputAt = now;
   810→  session.inputDropCount = 0;
   811→
   812→  session.desiredAngle = angle;
   813→  session.wantsBoost = wantsBoostRaw;
   814→}

   816→/** Start extraction. Validates the player is within the extract zone. */
   817→function handleExtract(socket: Socket): void {
   818→  const session = getSocketData(socket).playerSession;
   819→  if (!session || session.isDead || session.matchSettling) return;
   820→  if (session.isExtracting) return;
   821→
   822→  // NO zone check — extract anywhere (matches original design).
   823→  session.isExtracting = true;
   824→  session.extractionProgress = 0;
   825→  socket.emit('extract_start', { durationMs: EXTRACT_DURATION_MS });
   826→}

   829→function handleCancelExtract(socket: Socket): void {
   830→  const session = getSocketData(socket).playerSession;
   831→  if (!session) return;
   832→  session.isExtracting = false;
   833→  session.extractionProgress = 0;
   834→}

   836→/** Validate + rate-limit a chat message; broadcast to the room. */
   837→function handleChat(socket: Socket, payload: unknown): void {
   838→  const session = getSocketData(socket).playerSession;
   839→  const identity = getSocketData(socket).identity;
   840→  if (!session || !identity) return;
   841→  if (!payload || typeof payload !== 'object') return;
   842→  const raw = (payload as { message?: unknown }).message;
   843→  if (typeof raw !== 'string') return;
   844→  const message = raw.trim().slice(0, CHAT_MAX_LEN);
   845→  if (message.length === 0) return;
   846→
   847→  const now = Date.now();
   848→  if (now - session.lastChatAt < CHAT_MIN_INTERVAL_MS) return;
   849→  session.lastChatAt = now;
   850→
   851→  session.chatMessage = message;
   852→  session.chatExpiry = now + 4000;
   853→
   854→  const room = session.arenaId ? rooms.get(session.arenaId) : null;
   855→  if (!room) return;
   856→  const chatPayload = {
   857→    senderId: session.id,
   858→    senderName: session.name,
   859→    senderTag: session.userTag,
   860→    message,
   861→  };
   862→  for (const socketId of room.players.keys()) {
   863→    io.to(socketId).emit('chat', chatPayload);
   864→  }
   865→}

   867→/** Remove a player from their arena; scatter carried chips as star-chips (no DB write). */
   868→function handleLeave(socket: Socket, reason: string): void {
   869→  const data = getSocketData(socket);
   870→  const session = data.playerSession;
   871→  if (!session) return;
   872→
   873→  const room = session.arenaId ? rooms.get(session.arenaId) : null;
   874→  if (room) {
   875→    // Forfeit carried chips as star-chips at body positions (no scatter).
   876→    if (!session.isDead && !session.matchSettling && session.carriedChips > 0) {
   877→      try {
   878→        dropStarChipsAtBody(room, session.points, session.carriedChips);
   879→      } catch (err) {
   880→        log('error', `dropStarChipsAtBody error on leave: ${(err as Error).message}`);
   881→      }
   882→    }
   883→    room.players.delete(socket.id);
   884→  }
   885→
   886→  data.playerSession = undefined;
   887→  log('info', `${session.identity.userTag} left arena ${session.arenaId ?? '?'} (${reason})`);
   888→}

   903→function gracefulShutdown(signal: string): void {
   904→  log('info', `${signal} received — broadcasting server_shutdown and exiting`);
   905→  try {
   906→    io.emit('server_shutdown', {});
   907→  } catch (err) {
   908→    log('error', `shutdown broadcast failed: ${(err as Error).message}`);
   909→  }
   910→  // Give sockets a moment to flush, then exit.
   911→  setTimeout(() => process.exit(0), 400);
   912→  // Hard exit fallback if flush hangs.
   913→  setTimeout(() => process.exit(1), 2000).unref();
   914→}
   915→
   916→process.on('SIGINT', () => gracefulShutdown('SIGINT'));
   917→process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
   918→// Log SIGHUP so we can diagnose if the sandbox is hanging up the process.
   919→process.on('SIGHUP', () => {
   920→  log('warn', 'SIGHUP received — ignoring (nohup-style)');
   921→});
   922→// Log SIGPIPE so we don't die silently if a socket write fails on a closed pipe.
   923→process.on('SIGPIPE', () => {
   924→  log('warn', 'SIGPIPE received — ignoring');
   925→});
   926→
   927→// Diagnostic: log when the event loop is about to drain. If bun decides to
   928→// exit cleanly (no more refs), this tells us why.
   929→process.on('beforeExit', (code) => {
   930→  log('warn', `beforeExit code=${code} — event loop empty, process will exit`);
   931→});
   932→process.on('exit', (code) => {
   933→  log('warn', `exit code=${code}`);
   934→});

   943→  for (const room of rooms.values()) n += room.players.size;
