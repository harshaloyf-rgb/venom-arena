    86→    86→/** Returns the world-space rectangle visible in the viewport, plus margin. */
    87→    87→export function computeVisibleRect(rc: FrameRenderCtx, marginPx = 100): VisibleRect {
    88→    88→  const halfW = rc.w / 2 / rc.zoom;
    89→    89→  const halfH = rc.h / 2 / rc.zoom;
    90→    90→  const m = marginPx / rc.zoom;
    91→    91→  return {
    92→    92→    left: rc.camX - halfW - m,
    93→    93→    right: rc.camX + halfW + m,
    94→    94→    top: rc.camY - halfH - m,
    95→    95→    bottom: rc.camY + halfH + m,
    96→    96→  };
    97→    97→}

    99→    99→/** Quick AABB-vs-rect test. */
   100→   100→function rectContainsPoint(rect: VisibleRect, x: number, y: number): boolean {
   101→   101→  return x >= rect.left && x <= rect.right && y >= rect.top && y <= rect.bottom;
   102→   102→}

   104→   104→/** Returns true if any point in the list is inside the visible rect. */
   105→   105→function snakeIsVisible(pts: { x: number; y: number }[], rect: VisibleRect): boolean {
   106→   106→  // Cheap check: head + tail. If both off-screen on the SAME side we still
   107→   107→  // need to scan all points (snake might span the screen). For most cases
   108→   108→  // the head check suffices.
   109→   109→  if (pts.length === 0) return false;
   110→   110→  if (rectContainsPoint(rect, pts[0].x, pts[0].y)) return true;
   111→   111→  const last = pts[pts.length - 1];
   112→   112→  if (rectContainsPoint(rect, last.x, last.y)) return true;
   113→   113→  // Fallback: walk the downsampled polyline.
   114→   114→  const stride = pts.length > 60 ? 4 : 2;
   115→   115→  for (let i = 0; i < pts.length; i += stride) {
   116→   116→    if (rectContainsPoint(rect, pts[i].x, pts[i].y)) return true;
   117→   117→  }
   118→   118→  return false;
   119→   119→}

   121→   121→/**
   122→   122→ * Cache a metallic radial gradient. Bucketed by 2-pixel radius so we don't
   123→   123→ * cache a gradient for every floating-point size value.
   124→   124→ */
   125→   125→function getMetallicGradient(
   126→   126→  rc: FrameRenderCtx,
   127→   127→  radius: number,
   128→   128→  color: string,
   129→   129→  secondary: string | undefined,
   130→   130→): CanvasGradient | string {
   131→   131→  const bucket = Math.max(4, Math.round(radius));
   132→   132→  const key = `${color}:${secondary ?? ''}:${bucket}`;
   133→   133→  const cached = rc.metallicCache.get(key);
   134→   134→  if (cached) return cached;
   135→   135→  const { ctx } = rc;
   136→   136→  // Gradient is built in device pixels — we always rebuild on resize because
   137→   137→  // the cache Map is reset by the resize handler.
   138→   138→  const g = ctx.createRadialGradient(-bucket * 0.35, -bucket * 0.35, bucket * 0.1, 0, 0, bucket);
   139→   139→  g.addColorStop(0, '#f8fafc');
   140→   140→  g.addColorStop(0.35, secondary || color);
   141→   141→  g.addColorStop(1, color);
   142→   142→  rc.metallicCache.set(key, g);
   143→   143→  return g;
   144→   144→}

   146→   146→/**
   147→   147→ * Breathing arena radius at a given time. Matches the server's
   148→   148→ * `getMapRadius(elapsedMs)` formula exactly:
   149→   149→ *   baseRadius + sin(cycleProgress * 2π) * amplitude
   150→   150→ */
   151→   151→export function getArenaRadius(now: number): number {
   152→   152→  const cycleTime = (now % MAP_BREATH_CYCLE_MS) / MAP_BREATH_CYCLE_MS;
   153→   153→  const sinVal = Math.sin(cycleTime * Math.PI * 2);
   154→   154→  return MAP_BASE_RADIUS + sinVal * MAP_BREATH_AMPLITUDE;
   155→   155→}

   161→   161→/**
   162→   162→ * Draws the breathing circular arena: background fill (deep slate `#020617`)
   163→   163→ * clipped to the circle, a subtle grid (`#1e293b`, gridSize 60) inside the
   164→   164→ * circle, and the neon-rose boundary (`#f43f5e`, lineWidth 10, shadowBlur 16).
   165→   165→ */
   166→   166→export function drawGrid(rc: FrameRenderCtx): void {
   167→   167→  const { ctx, worldSize, camX, camY, zoom, w, h } = rc;
   168→   168→  const cx = worldSize / 2;
   169→   169→  const cy = worldSize / 2;
   170→   170→  const radius = getArenaRadius(rc.now);
   171→   171→
   172→   172→  // --- Background fill (entire arena disc) ---
   173→   173→  ctx.save();
   174→   174→  // Clip to arena circle so nothing draws outside.
   175→   175→  ctx.beginPath();
   176→   176→  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   177→   177→  ctx.fillStyle = '#020617'; // Deep Slate (AUDIT-A line 2572)
   178→   178→  ctx.fill();
   179→   179→  ctx.clip();
   180→   180→
   181→   181→  // --- Grid lines (subtle slate-800, gridSize 60) ---
   182→   182→  const grid = 60; // AUDIT-A line 2578
   183→   183→  const vis = computeVisibleRect(rc);
   184→   184→  const startX = Math.max(0, Math.floor(vis.left / grid) * grid);
   185→   185→  const endX = Math.min(worldSize, Math.ceil(vis.right / grid) * grid);
   186→   186→  const startY = Math.max(0, Math.floor(vis.top / grid) * grid);
   187→   187→  const endY = Math.min(worldSize, Math.ceil(vis.bottom / grid) * grid);
   188→   188→
   189→   189→  ctx.strokeStyle = '#1e293b'; // AUDIT-A line 2576
   190→   190→  ctx.lineWidth = 1 / zoom;
   191→   191→  ctx.beginPath();
   192→   192→  for (let x = startX; x <= endX; x += grid) {
   193→   193→    ctx.moveTo(x, startY);
   194→   194→    ctx.lineTo(x, endY);
   195→   195→  }
   196→   196→  for (let y = startY; y <= endY; y += grid) {
   197→   197→    ctx.moveTo(startX, y);
   198→   198→    ctx.lineTo(endX, y);
   199→   199→  }
   200→   200→  ctx.stroke();
   201→   201→  ctx.restore(); // remove clip
   202→   202→
   203→   203→  // --- Outer boundary (neon rose, breathing) ---
   204→   204→  // AUDIT-A lines 2595-2603: strokeStyle #f43f5e, lineWidth 10, shadowBlur 16.
   205→   205→  ctx.save();
   206→   206→  ctx.lineWidth = 10;
   207→   207→  ctx.strokeStyle = '#f43f5e';
   208→   208→  if (!rc.lowQuality) {
   209→   209→    ctx.shadowColor = '#f43f5e';
   210→   210→    ctx.shadowBlur = 16;
   211→   211→  }
   212→   212→  ctx.beginPath();
   213→   213→  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   214→   214→  ctx.stroke();
   215→   215→  ctx.shadowBlur = 0;
   216→   216→  ctx.restore();
   217→   217→
   218→   218→  // Silence unused-param lint for camX/camY/w/h (they are part of the public
   219→   219→  // type even if grid math doesn't need them directly).
   220→   220→  void camX;
   221→   221→  void camY;
   222→   222→  void w;
   223→   223→  void h;
   224→   224→}

   230→   230→/**
   231→   231→ * Draws all food in two passes:
   232→   232→ *  1. Regular food — batched by color into a single Path2D + fill per color
   233→   233→ *     (filled circle in its own color).
   234→   234→ *  2. Star chips — drawn individually as 5-pointed gold stars with a soft
   235→   235→ *     `shadowColor #eab308`, `shadowBlur 6` glow.
   236→   236→ */
   237→   237→export function drawFood(rc: FrameRenderCtx, foods: FoodSnapshot[]): void {
   238→   238→  const { ctx } = rc;
   239→   239→  const vis = computeVisibleRect(rc);
   240→   240→
   241→   241→  // --- Regular food: batched by color ---
   242→   242→  // Group foods into a per-color Path2D. The browser keeps one path open
   243→   243→  // per color, then we issue a single fill() per color. This turns 800
   244→   244→  // individual fills into ~6 fills (one per palette color).
   245→   245→  const pathsByColor = new Map<string, Path2D>();
   246→   246→  const starChips: FoodSnapshot[] = [];
   247→   247→
   248→   248→  for (let i = 0; i < foods.length; i++) {
   249→   249→    const f = foods[i];
   250→   250→    if (
   251→   251→      f.x < vis.left ||
   252→   252→      f.x > vis.right ||
   253→   253→      f.y < vis.top ||
   254→   254→      f.y > vis.bottom
   255→   255→    ) {
   256→   256→      continue; // culled
   257→   257→    }
   258→   258→    if (f.isStarChip) {
   259→   259→      starChips.push(f);
   260→   260→      continue;
   261→   261→    }
   262→   262→    let p = pathsByColor.get(f.color);
   263→   263→    if (!p) {
   264→   264→      p = new Path2D();
   265→   265→      pathsByColor.set(f.color, p);
   266→   266→    }
   267→   267→    p.moveTo(f.x + f.size, f.y);
   268→   268→    p.arc(f.x, f.y, f.size, 0, Math.PI * 2);
   269→   269→  }
   270→   270→
   271→   271→  for (const [color, path] of pathsByColor) {
   272→   272→    ctx.fillStyle = color;
   273→   273→    ctx.fill(path);
   274→   274→  }
   275→   275→
   276→   276→  // --- Star chips: 5-pointed gold star with shadowBlur 6 ---
   277→   277→  if (starChips.length === 0) return;
   278→   278→  ctx.save();
   279→   279→  for (let i = 0; i < starChips.length; i++) {
   280→   280→    const f = starChips[i];
   281→   281→    const outer = Math.max(2, f.size + 1);
   282→   282→    const inner = Math.max(1, f.size / 2);
   283→   283→    drawStar(ctx, f.x, f.y, 5, outer, inner, f.color || '#fbbf24', rc.lowQuality);
   284→   284→  }
   285→   285→  ctx.restore();
   286→   286→}
   287→   287→
   288→   288→/** Draws a 5-pointed star centered at (x,y) with `outer` outer radius and `inner` inner radius. */
   289→   289→function drawStar(
   290→   290→  ctx: CanvasRenderingContext2D,
   291→   291→  x: number,
   292→   292→  y: number,
   293→   293→  points: number,
   294→   294→  outer: number,
   295→   295→  inner: number,
   296→   296→  color: string,
   297→   297→  lowQuality: boolean,
   298→   298→): void {
   299→   299→  ctx.save();
   300→   300→  if (!lowQuality) {
   301→   301→    ctx.shadowColor = '#eab308';
   302→   302→    ctx.shadowBlur = 6;
   303→   303→  }
   304→   304→  ctx.beginPath();
   305→   305→  for (let i = 0; i < points * 2; i++) {
   306→   306→    const r = i % 2 === 0 ? outer : inner;
   307→   307→    const angle = (Math.PI / points) * i - Math.PI / 2;
   308→   308→    const px = x + Math.cos(angle) * r;
   309→   309→    const py = y + Math.sin(angle) * r;
   310→   310→    if (i === 0) ctx.moveTo(px, py);
   311→   311→    else ctx.lineTo(px, py);
   312→   312→  }
   313→   313→  ctx.closePath();
   314→   314→  ctx.fillStyle = color;
   315→   315→  ctx.fill();
   316→   316→  if (!lowQuality) ctx.shadowBlur = 0;
   317→   317→  ctx.restore();
   318→   318→}

   321→   321→// Snakes

   324→   324→/**
   325→   325→ * Draws a single snake. The body is rendered as ONE thick stroked polyline
   326→   326→ * (with an outline underlay) — not N individual circles — which eliminates
   327→   327→ * per-segment gradient, per-segment shadowBlur, per-segment save/restore,
   328→   328→ * and per-frame object allocation.
   329→   329→ *
   330→   330→ * The head is drawn separately with eyes and (for the player's own snake,
   331→   331→ * high-quality only) a glow halo.
   332→   332→ *
   333→   333→ * Player skin patterns (rainbow / neon / metallic / camo) are applied when
   334→   334→ * `rc.playerSkin.pattern` is set and the snake is the local player.
   335→   335→ */
   336→   336→export function drawSnake(rc: FrameRenderCtx, snake: SnakeSnapshot): void {
   337→   337→  const pts = snake.points;
   338→   338→  if (!pts || pts.length === 0) return; // C13 guard
   339→   339→
   340→   340→  const vis = computeVisibleRect(rc);
   341→   341→  if (!snakeIsVisible(pts, vis)) return; // P16 cull
   342→   342→
   343→   343→  const { ctx, zoom, lowQuality } = rc;
   344→   344→  const isMe = snake.id === rc.myId;
   345→   345→
   346→   346→  // Downsample long snakes (defensive — server already caps at ~120).
   347→   347→  const stride = pts.length > 60 ? 2 : 1;
   348→   348→
   349→   349→  // Body width in world units. `size` is the radius; visual width = 2*size.
   350→   350→  const radius = Math.max(2, snake.size);
   351→   351→  const width = radius * 2;
   352→   352→
   353→   353→  // --- Outline (underlay) ---
   354→   354→  if (snake.secondaryColor) {
   355→   355→    ctx.save();
   356→   356→    ctx.lineCap = 'round';
   357→   357→    ctx.lineJoin = 'round';
   358→   358→    ctx.lineWidth = width + 4 / zoom;
   359→   359→    ctx.strokeStyle = snake.secondaryColor;
   360→   360→    ctx.globalAlpha = 0.55;
   361→   361→    ctx.beginPath();
   362→   362→    ctx.moveTo(pts[0].x, pts[0].y);
   363→   363→    for (let i = stride; i < pts.length; i += stride) {
   364→   364→      ctx.lineTo(pts[i].x, pts[i].y);
   365→   365→    }
   366→   366→    ctx.stroke();
   367→   367→    ctx.restore();
   368→   368→  }
   369→   369→
   370→   370→  // --- Body ---
   371→   371→  ctx.save();
   372→   372→  ctx.lineCap = 'round';
   373→   373→  ctx.lineJoin = 'round';
   374→   374→  ctx.lineWidth = width;
   375→   375→
   376→   376→  const pattern = isMe ? rc.playerSkin?.pattern : undefined;
   377→   377→  if (pattern === 'metallic' && snake.secondaryColor) {
   378→   378→    // Use cached gradient — keyed by size bucket so we don't re-build per
   379→   379→    // snake. Stroking still happens per-snake (correct transform).
   380→   380→    const g = getMetallicGradient(rc, radius, snake.color, snake.secondaryColor);
   381→   381→    ctx.strokeStyle = g;
   382→   382→    ctx.beginPath();
   383→   383→    ctx.moveTo(pts[0].x, pts[0].y);
   384→   384→    for (let i = stride; i < pts.length; i += stride) {
   385→   385→      ctx.lineTo(pts[i].x, pts[i].y);
   386→   386→    }
   387→   387→    ctx.stroke();
   388→   388→  } else if (pattern === 'rainbow' && !lowQuality) {
   389→   389→    // Rainbow: stroke in hue-cycling chunks. Cheap — only a handful of
   390→   390→    // stroke() calls per snake.
   391→   391→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   392→   392→    ctx.lineWidth = width;
   393→   393→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   394→   394→      const hue = (rc.now * 0.05 + i * 14) % 360;
   395→   395→      ctx.strokeStyle = `hsl(${hue}, 90%, 55%)`;
   396→   396→      ctx.beginPath();
   397→   397→      ctx.moveTo(pts[i].x, pts[i].y);
   398→   398→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   399→   399→      for (let j = i + stride; j <= end; j += stride) {
   400→   400→        ctx.lineTo(pts[j].x, pts[j].y);
   401→   401→      }
   402→   402→      ctx.stroke();
   403→   403→    }
   404→   404→  } else if (pattern === 'neon' && !lowQuality) {
   405→   405→    // Neon: alternate cyan / purple along the body (AUDIT-A line 2762-2765).
   406→   406→    const chunkSize = Math.max(4, Math.floor(pts.length / 10));
   407→   407→    ctx.lineWidth = width;
   408→   408→    let chunkIndex = 0;
   409→   409→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   410→   410→      const ratio = (Math.sin(rc.now * 0.009 - chunkIndex * 0.28) + 1) / 2;
   411→   411→      ctx.strokeStyle = ratio > 0.5 ? '#06b6d4' : '#a855f7';
   412→   412→      ctx.beginPath();
   413→   413→      ctx.moveTo(pts[i].x, pts[i].y);
   414→   414→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   415→   415→      for (let j = i + stride; j <= end; j += stride) {
   416→   416→        ctx.lineTo(pts[j].x, pts[j].y);
   417→   417→      }
   418→   418→      ctx.stroke();
   419→   419→      chunkIndex += 1;
   420→   420→    }
   421→   421→  } else if (pattern === 'camo') {
   422→   422→    // Camo: alternating earth-tone segments (AUDIT-A line 2768-2770).
   423→   423→    const camoColors = ['#15803d', '#854d0e', '#78350f', '#166534'];
   424→   424→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   425→   425→    let chunkIndex = 0;
   426→   426→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   427→   427→      ctx.strokeStyle = camoColors[chunkIndex % camoColors.length];
   428→   428→      ctx.beginPath();
   429→   429→      ctx.moveTo(pts[i].x, pts[i].y);
   430→   430→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   431→   431→      for (let j = i + stride; j <= end; j += stride) {
   432→   432→        ctx.lineTo(pts[j].x, pts[j].y);
   433→   433→      }
   434→   434→      ctx.stroke();
   435→   435→      chunkIndex += 1;
   436→   436→    }
   437→   437→  } else {
   438→   438→    // Default: solid snake.color (alternating 2-color handled by outline).
   439→   439→    ctx.strokeStyle = snake.color;
   440→   440→    ctx.beginPath();
   441→   441→    ctx.moveTo(pts[0].x, pts[0].y);
   442→   442→    for (let i = stride; i < pts.length; i += stride) {
   443→   443→      ctx.lineTo(pts[i].x, pts[i].y);
   444→   444→    }
   445→   445→    ctx.stroke();
   446→   446→  }
   447→   447→  ctx.restore();

   449→   449→  // --- Head ---
   450→   450→  const head = pts[0];
   451→   451→  ctx.save();

   453→   453→  if (isMe && !lowQuality) {
   454→   454→    ctx.shadowColor = snake.color;
   455→   455→    ctx.shadowBlur = 14;
   456→   456→  }
   457→   457→  ctx.beginPath();
   458→   458→  ctx.arc(head.x, head.y, radius, 0, Math.PI * 2);
   459→   459→  ctx.fillStyle = snake.color;
   460→   460→  ctx.fill();
   461→   461→  if (isMe && !lowQuality) ctx.shadowBlur = 0;
   462→   462→
   463→   463→  // Eyes — direction-based positioning.
   464→   464→  const eyeOffset = radius * 0.45;
   465→   465→  const eyeR = Math.max(1.5, radius * 0.32);
   466→   466→  const pupilR = Math.max(0.8, radius * 0.18);
   467→   467→  const angle = snake.angle;
   468→   468→  const perp = angle + Math.PI / 2;
   469→   469→  const ex1 = head.x + Math.cos(angle) * eyeOffset * 0.4 + Math.cos(perp) * eyeOffset;
   470→   470→  const ey1 = head.y + Math.sin(angle) * eyeOffset * 0.4 + Math.sin(perp) * eyeOffset;
   471→   471→  const ex2 = head.x + Math.cos(angle) * eyeOffset * 0.4 - Math.cos(perp) * eyeOffset;
   472→   472→  const ey2 = head.y + Math.sin(angle) * eyeOffset * 0.4 - Math.sin(perp) * eyeOffset;
   473→   473→
   474→   474→  ctx.fillStyle = '#ffffff';
   475→   475→  ctx.beginPath();
   476→   476→  ctx.arc(ex1, ey1, eyeR, 0, Math.PI * 2);
   477→   477→  ctx.arc(ex2, ey2, eyeR, 0, Math.PI * 2);
   478→   478→  ctx.fill();
   479→   479→
   480→   480→  ctx.fillStyle = '#0a0a0a';
   481→   481→  ctx.beginPath();
   482→   482→  ctx.arc(ex1 + Math.cos(angle) * pupilR, ey1 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   483→   483→  ctx.arc(ex2 + Math.cos(angle) * pupilR, ey2 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   484→   484→  ctx.fill();
   485→   485→
   486→   486→  // Spawn-protection ring.
   487→   487→  if (snake.spawnProtected) {
   488→   488→    ctx.strokeStyle = 'rgba(255,255,255,0.6)';
   489→   489→    ctx.lineWidth = 2 / zoom;
   490→   490→    ctx.beginPath();
   491→   491→    ctx.arc(head.x, head.y, radius + 4 / zoom, 0, Math.PI * 2);
   492→   492→    ctx.stroke();
   493→   493→  }
   494→   494→
   495→   495→  // Extraction ring.
   496→   496→  if (snake.isExtracting) {
   497→   497→    ctx.strokeStyle = 'rgba(16, 185, 129, 0.95)';
   498→   498→    ctx.lineWidth = 3 / zoom;
   499→   499→    ctx.setLineDash([6 / zoom, 4 / zoom]);
   500→   500→    ctx.beginPath();
   501→   501→    ctx.arc(head.x, head.y, radius + 8 / zoom, 0, Math.PI * 2);
   502→   502→    ctx.stroke();
   503→   503→    ctx.setLineDash([]); // RESET — fixes R11
   504→   504→  }
   505→   505→  ctx.restore();
   506→   506→
   507→   507→  // --- Name label ---
   508→   508→  if (snake.name) {
   509→   509→    ctx.save();
   510→   510→    ctx.font = `${Math.max(10, 12 / zoom)}px monospace`;
   511→   511→    ctx.textAlign = 'center';
   512→   512→    ctx.textBaseline = 'bottom';
   513→   513→    const labelY = head.y - radius - 6 / zoom;
   514→   514→    ctx.fillStyle = isMe ? '#22c55e' : 'rgba(226, 232, 240, 0.85)';
   515→   515→    ctx.fillText(snake.name, head.x, labelY);
   516→   516→    if (snake.userTag) {
   517→   517→      ctx.fillStyle = 'rgba(148, 163, 184, 0.7)';
   518→   518→      ctx.font = `${Math.max(8, 9 / zoom)}px monospace`;
   519→   519→      ctx.fillText(snake.userTag, head.x, labelY - 12 / zoom);
   520→   520→    }
   521→   521→    ctx.restore();
   522→   522→  }
   523→   523→
   524→   524→  // --- Chat bubble ---
   525→   525→  if (snake.chatMessage) {
   526→   526→    drawChatBubble(rc, head.x, head.y - radius - 24 / zoom, snake.chatMessage);
   527→   527→  }
   528→   528→}

   530→   530→/** Draws a small chat bubble above a snake head. */
   531→   531→function drawChatBubble(
   532→   532→  rc: FrameRenderCtx,
   533→   533→  x: number,
   534→   534→  y: number,
   535→   535→  msg: string,
   536→   536→): void {
   537→   537→  const { ctx, zoom } = rc;
   538→   538→  const text = msg.length > 80 ? msg.slice(0, 78) + '…' : msg;
   539→   539→  ctx.save();
   540→   540→  ctx.font = `${Math.max(10, 11 / zoom)}px monospace`;
   541→   541→  const metrics = ctx.measureText(text);
   542→   542→  const padX = 6 / zoom;
   543→   543→  const padY = 3 / zoom;
   544→   544→  const tw = metrics.width + padX * 2;
   545→   545→  const th = 14 / zoom + padY * 2;
   546→   546→  ctx.fillStyle = 'rgba(15, 23, 42, 0.92)';
   547→   547→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.6)'; // indigo (AUDIT-A line 3072)
   548→   548→  ctx.lineWidth = 1.5 / zoom;
   549→   549→  // Rounded rect (manual since older canvas may not have roundRect).
   550→   550→  const rx = x - tw / 2;
   551→   551→  const ry = y - th;
   552→   552→  const rr = 4 / zoom;
   553→   553→  ctx.beginPath();
   554→   554→  ctx.moveTo(rx + rr, ry);
   555→   555→  ctx.lineTo(rx + tw - rr, ry);
   556→   556→  ctx.arcTo(rx + tw, ry, rx + tw, ry + rr, rr);
   557→   557→  ctx.lineTo(rx + tw, ry + th - rr);
   558→   558→  ctx.arcTo(rx + tw, ry + th, rx + tw - rr, ry + th, rr);
   559→   559→  ctx.lineTo(rx + rr, ry + th);
   560→   560→  ctx.arcTo(rx, ry + th, rx, ry + th - rr, rr);
   561→   561→  ctx.lineTo(rx, ry + rr);
   562→   562→  ctx.arcTo(rx, ry, rx + rr, ry, rr);
   563→   563→  ctx.closePath();
   564→   564→  ctx.fill();
   565→   565→  ctx.stroke();
   566→   566→  ctx.fillStyle = '#f8fafc';
   567→   567→  ctx.textAlign = 'center';
   568→   568→  ctx.textBaseline = 'middle';
   569→   569→  ctx.fillText(text, x, y - th / 2);
   570→   570→  ctx.restore();
   571→   571→}

   578→   578→export function drawParticles(rc: FrameRenderCtx, particles: Particle[]): void {
   579→   579→  if (particles.length === 0) return;

   581→   581→  ctx.save();

   590→   590→    ctx.arc(p.x, p.y, p.size / zoom, 0, Math.PI * 2);
   591→   591→    ctx.fill();
   592→   592→  }
   593→   593→  ctx.globalAlpha = 1;
   594→   594→  ctx.globalCompositeOperation = 'source-over';
   595→   595→  ctx.restore();

   602→   602→export interface MinimapArgs {
   603→   603→  ctx: CanvasRenderingContext2D;
   604→   604→  /** Top-left of minimap in CSS pixels. */
   605→   605→  x: number;
   606→   606→  y: number;
   607→   607→  /** Minimap size (square). */
   608→   608→  size: number;
   609→   609→  worldSize: number;
   610→   610→  /** Current breathing arena radius (world units). */
   611→   611→  arenaRadius: number;
   612→   612→  snakes: SnakeSnapshot[];
   613→   613→  myId: string;
   614→   614→  /**
   615→   615→   * World-space radius around the player the radar should cover. Snakes
   616→   616→   * outside this range are NOT rendered on the small corner minimap
   617→   617→   * (BUILD-13: small minimap = "nearby snakes within ~1800px"). Defaults
   618→   618→   * to WORLD_SIZE/2 (legacy full-radar behaviour) when omitted.
   619→   619→   */
   620→   620→  range?: number;
   621→   621→}
   622→   622→
   623→   623→/**
   624→   624→ * Draws the circular radar minimap. Player is rendered as an indigo dot at
   625→   625→ * the centre (the radar is centered on the player's head). The arena
   626→   626→ * boundary is a dashed rose circle. Bots render as rose dots, other real
   627→   627→ * players as emerald dots — matches AUDIT-A radar visual language.
   628→   628→ *
   629→   629→ * BUILD-13: `range` controls how far the radar sees. The small corner
   630→   630→ * minimap passes ~1800 so only nearby snakes are shown; the full-map
   631→   631→ * overlay passes WORLD_SIZE/2 (or simply uses drawFullMap).
   632→   632→ */
   633→   633→export function drawMinimap(args: MinimapArgs): void {
   634→   634→  const { ctx, x, y, size, worldSize, arenaRadius, snakes, myId } = args;
   635→   635→  const cx = x + size / 2;
   636→   636→  const cy = y + size / 2;
   637→   637→  const r = size / 2;
   638→   638→  // Radar range — default to WORLD_SIZE/2 (legacy). Small corner minimap
   639→   639→  // passes 1800 to show only nearby snakes (BUILD-13 spec).
   640→   640→  const radarRange = args.range ?? WORLD_SIZE / 2;
   641→   641→  const scale = r / radarRange;
   642→   642→
   643→   643→  ctx.save();
   644→   644→  // Background circle (radar disc)
   645→   645→  ctx.beginPath();
   646→   646→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   647→   647→  ctx.fillStyle = 'rgba(2, 6, 23, 0.85)';
   648→   648→  ctx.fill();
   649→   649→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.4)';
   650→   650→  ctx.lineWidth = 1.5;
   651→   651→  ctx.stroke();
   652→   652→
   653→   653→  // Clip to minimap circle.
   654→   654→  ctx.beginPath();
   655→   655→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   656→   656→  ctx.clip();
   657→   657→
   658→   658→  // Concentric rings (AUDIT-A lines 3520-3522).
   659→   659→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
   660→   660→  ctx.lineWidth = 1;
   661→   661→  for (const inset of [2, 5, 8]) {
   662→   662→    ctx.beginPath();
   663→   663→    ctx.arc(cx, cy, r - inset, 0, Math.PI * 2);
   664→   664→    ctx.stroke();
   665→   665→  }
   666→   666→
   667→   667→  // Crosshairs.
   668→   668→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.4)';
   669→   669→  ctx.beginPath();
   670→   670→  ctx.moveTo(cx, y);
   671→   671→  ctx.lineTo(cx, y + size);
   672→   672→  ctx.moveTo(x, cy);
   673→   673→  ctx.lineTo(x + size, cy);
   674→   674→  ctx.stroke();
   675→   675→
   676→   676→  // Arena boundary (dashed rose) — centered on world center.
   677→   677→  // World center relative to player's radar position.
   678→   678→  const mySnake = snakes.find((s) => s.id === myId);
   679→   679→  const myHead = mySnake?.points?.[0];
   680→   680→  const px = myHead ? myHead.x : worldSize / 2;
   681→   681→  const py = myHead ? myHead.y : worldSize / 2;
   682→   682→  const worldCenterOffsetX = (worldSize / 2 - px) * scale;
   683→   683→  const worldCenterOffsetY = (worldSize / 2 - py) * scale;
   684→   684→  const arenaR = arenaRadius * scale;
   685→   685→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.6)';
   686→   686→  ctx.lineWidth = 1.2;
   687→   687→  ctx.setLineDash([4, 4]);
   688→   688→  ctx.beginPath();
   689→   689→  ctx.arc(cx + worldCenterOffsetX, cy + worldCenterOffsetY, arenaR, 0, Math.PI * 2);
   690→   690→  ctx.stroke();
   691→   691→  ctx.setLineDash([]);
   692→   692→
   693→   693→  // Snakes (player centered → drawn at centre; others offset).
   694→   694→  for (let i = 0; i < snakes.length; i++) {
   695→   695→    const s = snakes[i];
   696→   696→    if (!s.points || s.points.length === 0) continue;
   697→   697→    const head = s.points[0];
   698→   698→    const dx = (head.x - px) * scale;
   699→   699→    const dy = (head.y - py) * scale;
   700→   700→    const dist = Math.hypot(dx, dy);
   701→   701→    if (dist > r) continue; // off-radar
   702→   702→    const mx = cx + dx;
   703→   703→    const my = cy + dy;
   704→   704→    ctx.beginPath();
   705→   705→    if (s.id === myId) {
   706→   706→      ctx.arc(mx, my, 3, 0, Math.PI * 2);
   707→   707→      ctx.fillStyle = '#818cf8'; // indigo-400 (AUDIT-A line 3525)
   708→   708→    } else if (s.isBot) {
   709→   709→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
   710→   710→      ctx.fillStyle = '#f43f5e'; // rose-500 (AUDIT-A line 3577)
   711→   711→    } else {
   712→   712→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
   713→   713→      ctx.fillStyle = '#34d399'; // emerald-400 (AUDIT-A line 3577)
   714→   714→    }
   715→   715→    ctx.fill();
   716→   716→  }
   717→   717→  ctx.restore();
   718→   718→}

   761→   761→  const scale = fitDim / arenaDiameter;

   817→   817→      ctx.arc(sx, sy, 2.5, 0, Math.PI * 2);
   818→   818→      ctx.fillStyle = '#f43f5e'; // rose
   819→   819→      ctx.fill();

   828→   828→  ctx.fillStyle = 'rgba(226, 232, 240, 0.95)';

   843→   843→    ctx.beginPath();
   844→   844→    ctx.arc(legendX + 6, legendY + 6, dotR, 0, Math.PI * 2);
