     3→     3→// ============================================================================
     4→     4→// offline-engine.ts — Pure client-side snake game engine for "Venom Arena".
     5→     5→// ----------------------------------------------------------------------------
     6→     6→// This module runs a complete slither.io-style game ENTIRELY in the browser.
     7→     7→// No Socket.IO connection, no server round-trips, no chips, no minimap. It is
     8→     8→// used by `game-canvas.tsx` whenever the active arena's id starts with
     9→     9→// `practice-` (the three free offline practice tiers).
    10→    10→//
    11→    11→// Design tenets (from the locked BUILD-14 spec):
    12→    12→//  * **No server connection** — runs purely on requestAnimationFrame.
    13→    13→//  * **No chips** — chip-less mode. No star chips, no carried chips, no
    14→    14→//    buy-in, no commission. Food only grows body (+1 score).
    15→    15→//  * **No map** — mapless. No minimap, no full-map overlay.
    16→    16→//  * **Ranking by score** (body length), not chips.
    17→    17→//  * **Leaderboard**: top 10 by score (you + bots), collapsible.
    18→    18→//  * **Extract any time** — hold 3 s → ends run, shows final score, ZERO XP.
    19→    19→//  * **Death**: body collision = die + restart. Wall hit = die + restart.
    20→    20→//    No drops (no food drop, no star drop — chips don't exist).
    21→    21→//  * **Boost**: costs body length (every 40 frames drops 1 tail segment).
    22→    22→//
    23→    23→// Performance:
    24→    24→//  * Spatial hash grid for collision + food queries (O(n)-ish, not O(n²)).
    25→    25→//  * Snake bodies capped at MAX_BODY_LENGTH (120); rendered via downsampled
    26→    26→//    stride-2 polylines (max 60 stroke points).
    27→    27→//  * Viewport culling (skip off-screen snakes/food).
    28→    28→//  * Adaptive quality: FPS < 40 for 2 s → low-quality mode (no glow / shadow).
    29→    29→//  * Particles capped at 200.
    30→    30→//
    31→    31→// The engine owns its own DOM HUD overlays (appended to `canvas.parentElement`)
    32→    32→// and tears them all down on `stop()`. The host React component just has to
    33→    33→// instantiate, start, and forward the `onExit` / `onStateChange` callbacks.
    34→    34→// ============================================================================

   157→   157→// ----------------------------------------------------------------------------
   158→   158→// Spatial hash grid — slimmed-down client port of the server's grid.
   159→   159→// Items are bucketed into square cells; queries return a deduplicated Map.
   160→   160→// ----------------------------------------------------------------------------
   161→   161→
   162→   162→class SpatialHashGrid {
   163→   163→  private readonly cellSize: number;
   164→   164→  private readonly cells: Map<string, Map<string, GridItem>> = new Map();
   165→   165→
   166→   166→  constructor(cellSize = 120) {
   167→   167→    this.cellSize = cellSize;
   168→   168→  }
   169→   169→
   170→   170→  private key(cx: number, cy: number): string {
   171→   171→    return cx + ':' + cy;
   172→   172→  }
   173→   173→
   174→   174→  clear(): void {
   175→   175→    this.cells.clear();
   176→   176→  }
   177→   177→
   178→   178→  insert(item: GridItem): void {
   179→   179→    const minCx = Math.floor((item.x - item.radius) / this.cellSize);
   180→   180→    const maxCx = Math.floor((item.x + item.radius) / this.cellSize);
   181→   181→    const minCy = Math.floor((item.y - item.radius) / this.cellSize);
   182→   182→    const maxCy = Math.floor((item.y + item.radius) / this.cellSize);
   183→   183→    for (let cx = minCx; cx <= maxCx; cx++) {
   184→   184→      for (let cy = minCy; cy <= maxCy; cy++) {
   185→   185→        const k = this.key(cx, cy);
   186→   186→        let bucket = this.cells.get(k);
   187→   187→        if (!bucket) {
   188→   188→          bucket = new Map();
   189→   189→          this.cells.set(k, bucket);
   190→   190→        }
   191→   191→        bucket.set(item.id, item);
   192→   192→      }
   193→   193→    }
   194→   194→  }
   195→   195→
   196→   196→  queryRadius(x: number, y: number, r: number): Map<string, GridItem> {
   197→   197→    const out = new Map<string, GridItem>();
   198→   198→    const minCx = Math.floor((x - r) / this.cellSize);
   199→   199→    const maxCx = Math.floor((x + r) / this.cellSize);
   200→   200→    const minCy = Math.floor((y - r) / this.cellSize);
   201→   201→    const maxCy = Math.floor((y + r) / this.cellSize);
   202→   202→    for (let cx = minCx; cx <= maxCx; cx++) {
   203→   203→      for (let cy = minCy; cy <= maxCy; cy++) {
   204→   204→        const bucket = this.cells.get(this.key(cx, cy));
   205→   205→        if (!bucket) continue;
   206→   206→        for (const [id, item] of bucket) {
   207→   207→          if (!out.has(id)) out.set(id, item);
   208→   208→        }
   209→   209→      }
   210→   210→    }
   211→   211→    return out;
   212→   212→  }
   213→   213→}

   265→   265→
   266→   266→function angularDelta(current: number, desired: number): number {
   267→   267→  let diff = desired - current;
   268→   268→  while (diff > Math.PI) diff -= 2 * Math.PI;
   269→   269→  while (diff < -Math.PI) diff += 2 * Math.PI;
   270→   270→  return diff;
   271→   271→}
   272→   272→
   273→   273→function turnToward(current: number, desired: number, maxStep: number): number {
   274→   274→  const diff = angularDelta(current, desired);
   275→   275→  if (Math.abs(diff) <= maxStep) return desired;
   276→   276→  return current + Math.sign(diff) * maxStep;
   277→   277→}
   278→   278→
   279→   279→function dist(ax: number, ay: number, bx: number, by: number): number {
   280→   280→  return Math.hypot(ax - bx, ay - by);
   281→   281→}
   282→   282→
   283→   283→function randomSpawnPoint(maxR: number): Vec2 {
   284→   284→  const r = Math.sqrt(Math.random()) * maxR;
   285→   285→  const theta = Math.random() * Math.PI * 2;
   286→   286→  return {
   287→   287→    x: WORLD_RADIUS + Math.cos(theta) * r,
   288→   288→    y: WORLD_RADIUS + Math.sin(theta) * r,
   289→   289→  };
   290→   290→}
   291→   291→
   292→   292→function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   293→   293→  const pts: Vec2[] = [];
   294→   294→  for (let i = 0; i < length; i++) {
   295→   295→    pts.push({
   296→   296→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   297→   297→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   298→   298→    });
   299→   299→  }
   300→   300→  return pts;
   301→   301→}

   662→   662→  private onTouchStart(e: TouchEvent): void {
   663→   663→    if (this.joystick) return;
   664→   664→    const t = this.findJoystickTouch(e.touches);
   665→   665→    if (!t) return;
   666→   666→    e.preventDefault();
   667→   667→    const rect = this.canvas.getBoundingClientRect();
   668→   668→    this.joystick = {
   669→   669→      active: true,
   670→   670→      pointerId: t.identifier,
   671→   671→      originX: t.clientX - rect.left,
   672→   672→      originY: t.clientY - rect.top,
   673→   673→      curX: t.clientX - rect.left,
   674→   674→      curY: t.clientY - rect.top,
   675→   675→    };
   676→   676→  }
   677→   677→
   678→   678→  private onTouchMove(e: TouchEvent): void {
   679→   679→    const js = this.joystick;
   680→   680→    if (!js) return;
   681→   681→    let t: Touch | null = null;
   682→   682→    for (let i = 0; i < e.touches.length; i++) {
   683→   683→      if (e.touches[i].identifier === js.pointerId) {
   684→   684→        t = e.touches[i];
   685→   685→        break;
   686→   686→      }
   687→   687→    }
   688→   688→    if (!t) return;
   689→   689→    e.preventDefault();
   690→   690→    const rect = this.canvas.getBoundingClientRect();
   691→   691→    js.curX = t.clientX - rect.left;
   692→   692→    js.curY = t.clientY - rect.top;
   693→   693→    const dx = js.curX - js.originX;
   694→   694→    const dy = js.curY - js.originY;
   695→   695→    const d = Math.hypot(dx, dy);
   696→   696→    const mag = Math.min(1, d / JOYSTICK_MAX_RADIUS_PX);
   697→   697→    if (mag > JOYSTICK_DEADZONE) {
   698→   698→      this.touchAngle = Math.atan2(dy, dx);
   699→   699→      this.touchBoost = mag > JOYSTICK_BOOST_MAGNITUDE;
   700→   700→    } else {
   701→   701→      this.touchAngle = null;
   702→   702→      this.touchBoost = false;
   703→   703→    }
   704→   704→  }
   705→   705→
   706→   706→  private onTouchEnd(e: TouchEvent): void {
   707→   707→    const js = this.joystick;
   708→   708→    if (!js) return;
   709→   709→    let stillActive = false;
   710→   710→    for (let i = 0; i < e.touches.length; i++) {
   711→   711→      if (e.touches[i].identifier === js.pointerId) {
   712→   712→        stillActive = true;
   713→   713→        break;
   714→   714→      }
   715→   715→    }
   716→   716→    if (!stillActive) {
   717→   717→      this.joystick = null;
   718→   718→      this.touchAngle = null;
   719→   719→      this.touchBoost = false;
   720→   720→    }
   721→   721→  }
   722→   722→
   723→   723→  private findJoystickTouch(touches: TouchList): Touch | null {
   724→   724→    const rect = this.canvas.getBoundingClientRect();
   725→   725→    for (let i = 0; i < touches.length; i++) {
   726→   726→      const t = touches[i];
   727→   727→      const x = t.clientX - rect.left;
   728→   728→      const y = t.clientY - rect.top;
   729→   729→      if (x < rect.width / 2 && y > rect.height / 2) {
   730→   730→        return t;
   731→   731→      }
   732→   732→    }
   733→   733→    return null;
   734→   734→  }

   736→   736→  private onKeyDown(e: KeyboardEvent): void {
   737→   737→    const k = e.key.toLowerCase();
   738→   738→    if (
   739→   739→      k === 'arrowup' ||
   740→   740→      k === 'arrowdown' ||
   741→   741→      k === 'arrowleft' ||
   742→   742→      k === 'arrowright' ||
   743→   743→      k === ' ' ||
   744→   744→      k === 'spacebar'
   745→   745→    ) {
   746→   746→      e.preventDefault();

   749→   749→      this.handleExitToLobby();
   750→   750→      return;
   751→   751→    }
   752→   752→    if (k === 'e' && this.state === 'playing' && !this.extractHold) {
   753→   753→      this.extractHold = true;
   754→   754→      this.beginExtract();
   755→   755→    }
   756→   756→    if (this.state === 'playing' && ['1', '2', '3', '4', '5'].includes(k)) {
   757→   757→      const idx = parseInt(k, 10) - 1;
   758→   758→      if (idx >= 0 && idx < QUICK_EMOTES.length) {
   759→   759→        this.setPlayerChat(QUICK_EMOTES[idx]);
   760→   760→      }
   761→   761→    }
   762→   762→    const normalized = k === 'spacebar' ? ' ' : k;
   763→   763→    this.keys.add(normalized);
   764→   764→  }

   834→   834→  // Main rAF loop

   836→   836→
   837→   837→  private frame = (now: number): void => {
   838→   838→    if (this.stopped) return;
   839→   839→    this.rafId = requestAnimationFrame(this.frame);
   840→   840→    const dt = now - this.lastFrameTime;
   841→   841→    this.lastFrameTime = now;
   842→   842→
   843→   843→    this.updateFps(now);
   844→   844→    this.updateParticles(dt);
   845→   845→
   846→   846→    // Physics ticks at ~TICK_MS (30 Hz) cadence — we run the sim in fixed
   847→   847→    // steps but render every rAF. Accumulator pattern prevents spiral-of-death.
   848→   848→    if (this.state === 'playing') {
   849→   849→      this.accumulator += dt;
   850→   850→      let safety = 4;
   851→   851→      while (this.accumulator >= TICK_MS && safety > 0) {
   852→   852→        this.tickPhysics(now);
   853→   853→        this.accumulator -= TICK_MS;
   854→   854→        safety--;
   855→   855→      }
   856→   856→      if (safety === 0) this.accumulator = 0;
   857→   857→    }
   858→   858→
   859→   859→    this.render(now);
   860→   860→    this.updateHUD();
   861→   861→  };
   862→   862→
   863→   863→  private accumulator: number = 0;
   864→   864→
   865→   865→  // --------------------------------------------------------------------------
   866→   866→  // Physics tick (one server-tick equivalent)
   867→   867→  // --------------------------------------------------------------------------
   868→   868→
   869→   869→  private tickPhysics(now: number): void {
   870→   870→    this.tick++;
   871→   871→    const p = this.player;
   872→   872→    if (!p || p.isDead) return;
   873→   873→
   874→   874→    // 1) Compute player desired angle + boost.
   875→   875→    const { angle, boost } = this.computePlayerInput();
   876→   876→    if (angle !== null) p.desiredAngle = angle;
   877→   877→    p.wantsBoost = boost;
   878→   878→
   879→   879→    // Extraction progress (3-second hold).
   880→   880→    if (p.isExtracting) {
   881→   881→      p.extractionProgress += TICK_MS;
   882→   882→      if (p.extractionProgress >= EXTRACT_DURATION_MS) {
   883→   883→        this.finishExtract();
   884→   884→        return;
   885→   885→      }
   886→   886→    }
   887→   887→
   888→   888→    // 2) Move player.
   889→   889→    this.tickSnakeMovement(p, p.desiredAngle, p.wantsBoost);
   890→   890→
   891→   891→    // 3) Bots: think + move.
   892→   892→    for (const bot of this.bots.values()) {
   893→   893→      this.tickBot(bot, now);
   894→   894→    }
   895→   895→
   896→   896→    // 4) Build spatial grid for collision + food queries.
   897→   897→    this.grid.clear();
   898→   898→    this.insertSnakeIntoGrid(p);
   899→   899→    for (const bot of this.bots.values()) {
   900→   900→      if (!bot.isDead) this.insertSnakeIntoGrid(bot);
   901→   901→    }
   902→   902→    for (const food of this.foods) {
   903→   903→      if (food.value > 0) {
   904→   904→        this.grid.insert({
   905→   905→          id: food.id,
   906→   906→          kind: 'food',
   907→   907→          x: food.x,
   908→   908→          y: food.y,
   909→   909→          radius: food.size,
   910→   910→          value: food.value,
   911→   911→          foodRef: food,
   912→   912→        });
   913→   913→      }
   914→   914→    }
   915→   915→
   916→   916→    // 5) Eat food (player + bots). No star chips — only regular food (+1 score).
   917→   917→    this.eatFood();
   918→   918→
   919→   919→    // 6) Collision detection (head-to-body = death; wall = death).
   920→   920→    const deaths = this.detectCollisions(now);
   921→   921→
   922→   922→    // 7) Apply deaths.
   923→   923→    let playerDied = false;
   924→   924→    for (const d of deaths) {
   925→   925→      if (d.deadId === p.id) {
   926→   926→        playerDied = true;
   927→   927→        continue;
   928→   928→      }
   929→   929→      const bot = this.bots.get(d.deadId);
   930→   930→      if (bot) {
   931→   931→        // No drops (chip-less mode) — just mark dead + respawn.
   932→   932→        bot.isDead = true;
   933→   933→        // Credit kill to the killer if it's the player.
   934→   934→        if (d.killerId === p.id) {
   935→   935→          p.kills++;
   936→   936→        }
   937→   937→      }
   938→   938→    }
   939→   939→
   940→   940→    // 8) Remove dead bots and respawn to maintain count.
   941→   941→    if (deaths.length > 0) {
   942→   942→      const toRemove: string[] = [];
   943→   943→      for (const [id, bot] of this.bots) {
   944→   944→        if (bot.isDead) toRemove.push(id);
   945→   945→      }
   946→   946→      for (const id of toRemove) {
   947→   947→        this.bots.delete(id);
   948→   948→      }
   949→   949→    }
   950→   950→    // Respawn bots to maintain count.
   951→   951→    while (this.bots.size < this.arena.botsCount) {
   952→   952→      const bot = this.spawnBot();
   953→   953→      this.bots.set(bot.id, bot);
   954→   954→    }
   955→   955→
   956→   956→    // 9) Player death → death screen.
   957→   957→    if (playerDied) {
   958→   958→      p.isDead = true;
   959→   959→      this.handlePlayerDeath();
   960→   960→      return;
   961→   961→    }
   962→   962→
   963→   963→    // 10) Replenish food toward target.
   964→   964→    this.replenishFood();
   965→   965→
   966→   966→    // 11) Expire chat bubbles.
   967→   967→    this.expireChat(now);
   968→   968→
   969→   969→    // 12) Update camera target.
   970→   970→    this.updateCamera();
   971→   971→  }
   972→   972→
   973→   973→  // --------------------------------------------------------------------------
   974→   974→  // Snake movement (server-authoritative formula)
   975→   975→  // --------------------------------------------------------------------------
   976→   976→
   977→   977→  private tickSnakeMovement(snake: SnakeBase, desiredAngle: number, wantsBoost: boolean): void {
   978→   978→    if (snake.points.length === 0 || snake.isDead) return;
   979→   979→
   980→   980→    // Turn rate: max(0.045, 0.15 - score*0.0006).
   981→   981→    const turnRate = Math.max(TURN_MIN, TURN_BASE - snake.score * TURN_SCORE_FACTOR);
   982→   982→    snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
   983→   983→
   984→   984→    // Speed.
   985→   985→    let speed = BASE_SPEED;
   986→   986→    if (snake.isExtracting) {
   987→   987→      speed = EXTRACT_GLIDE_SPEED;
   988→   988→    } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
   989→   989→      speed = BOOST_SPEED;
   990→   990→      snake.boostFrameCounter++;
   991→   991→      if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
   992→   992→        snake.boostFrameCounter = 0;
   993→   993→        if (snake.points.length > BOOST_MIN_LENGTH) {
   994→   994→          snake.points.pop();
   995→   995→          snake.score = Math.max(0, snake.score - 1);
   996→   996→        }
   997→   997→      }
   998→   998→    }
   999→   999→
  1000→  1000→    // Move head.
  1001→  1001→    const head = snake.points[0];
  1002→  1002→    const nx = head.x + Math.cos(snake.angle) * speed;
  1003→  1003→    const ny = head.y + Math.sin(snake.angle) * speed;
  1004→  1004→    snake.points.unshift({ x: nx, y: ny });
  1005→  1005→
  1006→  1006→    // Grow / shrink body to target length based on SCORE.
  1007→  1007→    const targetLen = Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + snake.score);
  1008→  1008→    while (snake.points.length > targetLen) snake.points.pop();
  1009→  1009→
  1010→  1010→    // Size: 8 + sqrt(score) * 0.4.
  1011→  1011→    snake.size = SIZE_BASE + Math.sqrt(snake.score) * SIZE_SCORE_FACTOR;
  1012→  1012→  }
  1013→  1013→
  1014→  1014→  // --------------------------------------------------------------------------
  1015→  1015→  // Bot AI (scavenger / opportunist / hunter / extractor / coward)
  1016→  1016→  // --------------------------------------------------------------------------
  1017→  1017→
  1018→  1018→  private tickBot(bot: BotSession, now: number): void {
  1019→  1019→    if (bot.points.length === 0 || bot.isDead) return;
  1020→  1020→
  1021→  1021→    if (now >= bot.nextThinkAt) {
  1022→  1022→      bot.nextThinkAt = now + BOT_THINK_INTERVAL_MS + Math.floor(Math.random() * BOT_THINK_JITTER_MS);
  1023→  1023→      const head = bot.points[0];
  1024→  1024→
  1025→  1025→      // Find nearest food.
  1026→  1026→      let bestFood: GridItem | null = null;
  1027→  1027→      let bestFoodDist = Infinity;
  1028→  1028→      const foodQuery = this.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
  1029→  1029→      for (const item of foodQuery.values()) {
  1030→  1030→        if (item.kind !== 'food') continue;
  1031→  1031→        if ((item.value ?? 0) <= 0) continue;
  1032→  1032→        const d = dist(head.x, head.y, item.x, item.y);
  1033→  1033→        if (d < bestFoodDist) {
  1034→  1034→          bestFoodDist = d;
  1035→  1035→          bestFood = item;
  1036→  1036→        }
  1037→  1037→      }
  1038→  1038→
  1039→  1039→      // Find nearest threat (foreign body segment within threat scan radius).
  1040→  1040→      let threatX = 0;
  1041→  1041→      let threatY = 0;
  1042→  1042→      let threatDist = Infinity;
  1043→  1043→      const threatQuery = this.grid.queryRadius(head.x, head.y, BOT_THREAT_SCAN_RADIUS);
  1044→  1044→      for (const item of threatQuery.values()) {
  1045→  1045→        if (item.kind !== 'segment') continue;
  1046→  1046→        if (item.snakeId === bot.id) continue;
  1047→  1047→        if (item.segIdx === 0) continue;
  1048→  1048→        const d = dist(head.x, head.y, item.x, item.y);
  1049→  1049→        if (d < threatDist) {
  1050→  1050→          threatDist = d;
  1051→  1051→          threatX = item.x;
  1052→  1052→          threatY = item.y;
  1053→  1053→        }
  1054→  1054→      }
  1055→  1055→
  1056→  1056→      let desired: number;
  1057→  1057→      const shouldFlee =
  1058→  1058→        threatDist < 140 &&
  1059→  1059→        (bot.personality === 'coward' ||
  1060→  1060→          bot.personality === 'extractor' ||
  1061→  1061→          (bot.personality === 'opportunist' && threatDist < 90));
  1062→  1062→
  1063→  1063→      if (shouldFlee) {
  1064→  1064→        desired = Math.atan2(head.y - threatY, head.x - threatX);
  1065→  1065→      } else if (bestFood) {
  1066→  1066→        desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
  1067→  1067→      } else {
  1068→  1068→        desired = bot.angle + (Math.random() - 0.5) * 0.4;
  1069→  1069→      }
  1070→  1070→
  1071→  1071→      // Edge avoidance: turn back toward center if near the wall.
  1072→  1072→      const headR = Math.hypot(head.x - WORLD_RADIUS, head.y - WORLD_RADIUS);
  1073→  1073→      if (headR > MAP_BASE_RADIUS - BOT_EDGE_AVOID_RADIUS) {
  1074→  1074→        const toCenter = Math.atan2(WORLD_RADIUS - head.y, WORLD_RADIUS - head.x);
  1075→  1075→        desired = turnToward(desired, toCenter, BOT_MAX_TURN_PER_TICK * 2);
  1076→  1076→      }
  1077→  1077→
  1078→  1078→      bot.desiredAngle = desired;
  1079→  1079→    }
  1080→  1080→
  1081→  1081→    // Hunter personality occasionally boosts (no chips required, just body length).
  1082→  1082→    const wantsBoost = bot.personality === 'hunter' && bot.score > 5 && Math.random() < 0.05;
  1083→  1083→    this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1084→  1084→  }

  1110→  1110→  private eatFood(): void {

  1118→  1118→  private eatFoodForSnake(snake: SnakeBase): void {
  1119→  1119→    if (snake.points.length === 0) return;
  1120→  1120→    const head = snake.points[0];
  1121→  1121→    const nearby = this.grid.queryRadius(head.x, head.y, snake.size + 20);
  1122→  1122→    for (const item of nearby.values()) {
  1123→  1123→      if (item.kind !== 'food') continue;
  1124→  1124→      if ((item.value ?? 0) <= 0) continue;
  1125→  1125→      const d = dist(head.x, head.y, item.x, item.y);
  1126→  1126→      if (d < snake.size + (item.radius || 6) + 6) {
  1127→  1127→        // Regular food only (+1 score, no chips). No star chips in offline mode.
  1128→  1128→        snake.score += REGULAR_FOOD_GROW;
  1129→  1129→        item.value = 0;
  1130→  1130→        if (item.foodRef) item.foodRef.value = 0;
  1131→  1131→        // Spawn small particle burst for player only.
  1132→  1132→        if (snake.isPlayer) {
  1133→  1133→          this.spawnEatParticles(item.x, item.y, item.radius || 6);
  1134→  1134→        }
  1135→  1135→      }
  1136→  1136→    }
  1137→  1137→  }

  1142→  1142→
  1143→  1143→  private detectCollisions(now: number): { deadId: string; killerId?: string; cause: 'body' | 'wall' }[] {
  1144→  1144→    const deaths: { deadId: string; killerId?: string; cause: 'body' | 'wall' }[] = [];
  1145→  1145→    const seenDead = new Set<string>();
  1146→  1146→    const all: SnakeBase[] = [];
  1147→  1147→    if (this.player && !this.player.isDead) all.push(this.player);
  1148→  1148→    for (const bot of this.bots.values()) {
  1149→  1149→      if (!bot.isDead) all.push(bot);
  1150→  1150→    }
  1151→  1151→
  1152→  1152→    const mapRadius = getArenaRadius(now);
  1153→  1153→
  1154→  1154→    for (const snake of all) {
  1155→  1155→      if (snake.isDead) continue;
  1156→  1156→      if (snake.points.length === 0) continue;
  1157→  1157→      if (now < snake.spawnProtectedUntil) continue;
  1158→  1158→      if (seenDead.has(snake.id)) continue;
  1159→  1159→      const head = snake.points[0];
  1160→  1160→
  1161→  1161→      // Wall collision: outside breathing map radius.
  1162→  1162→      const distFromCenter = Math.hypot(head.x - WORLD_RADIUS, head.y - WORLD_RADIUS);
  1163→  1163→      if (distFromCenter > mapRadius) {
  1164→  1164→        deaths.push({ deadId: snake.id, killerId: 'wall', cause: 'wall' });
  1165→  1165→        seenDead.add(snake.id);
  1166→  1166→        continue;
  1167→  1167→      }
  1168→  1168→
  1169→  1169→      // Body collision: head vs foreign non-head segment.
  1170→  1170→      const queryR = snake.size + 30;
  1171→  1171→      const nearby = this.grid.queryRadius(head.x, head.y, queryR);
  1172→  1172→      for (const item of nearby.values()) {
  1173→  1173→        if (item.kind !== 'segment') continue;
  1174→  1174→        if (item.snakeId === snake.id) continue;
  1175→  1175→        if (item.segIdx === 0) continue; // head-to-head ignored (no self-collision; head-to-head doesn't kill either)
  1176→  1176→        const d = dist(head.x, head.y, item.x, item.y);
  1177→  1177→        if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
  1178→  1178→          deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
  1179→  1179→          seenDead.add(snake.id);
  1180→  1180→          break;
  1181→  1181→        }
  1182→  1182→      }
  1183→  1183→    }
  1184→  1184→    return deaths;
  1185→  1185→  }

  1262→  1262→  private computePlayerInput(): { angle: number | null; boost: boolean } {
  1263→  1263→    const spaceHeld = this.keys.has(' ') || this.keys.has('space') || this.boostHold;
  1264→  1264→    // Touch joystick
  1265→  1265→    if (this.touchAngle !== null) {
  1266→  1266→      return { angle: this.touchAngle, boost: this.touchBoost || spaceHeld };
  1267→  1267→    }
  1268→  1268→    // Keyboard
  1269→  1269→    const k = this.keys;
  1270→  1270→    let kx = 0;
  1271→  1271→    let ky = 0;
  1272→  1272→    if (k.has('w') || k.has('arrowup')) ky -= 1;
  1273→  1273→    if (k.has('s') || k.has('arrowdown')) ky += 1;
  1274→  1274→    if (k.has('a') || k.has('arrowleft')) kx -= 1;
  1275→  1275→    if (k.has('d') || k.has('arrowright')) kx += 1;
  1276→  1276→    if (kx !== 0 || ky !== 0) {
  1277→  1277→      return { angle: Math.atan2(ky, kx), boost: spaceHeld };
  1278→  1278→    }

  1280→  1280→    if (this.mouseActive) {
  1281→  1281→      const rect = this.canvas.getBoundingClientRect();
  1282→  1282→      const cx = rect.width / 2;
  1283→  1283→      const cy = rect.height / 2;
  1284→  1284→      const dx = this.mousePos.x - cx;
  1285→  1285→      const dy = this.mousePos.y - cy;
  1286→  1286→      const d = Math.hypot(dx, dy);
  1287→  1287→      if (d > MOUSE_DEADZONE_PX) {
  1288→  1288→        return { angle: Math.atan2(dy, dx), boost: spaceHeld };
  1289→  1289→      }
  1290→  1290→    }
  1291→  1291→    return { angle: null, boost: false };
  1292→  1292→  }

  1362→  1362→  private drawSnakeSnapshot(rc: FrameRenderCtx, snake: SnakeBase): void {
  1363→  1363→    // Downsample long snakes for the snapshot (cap rendered points at 60).
  1364→  1364→    let pts = snake.points;
  1365→  1365→    if (pts.length > MAX_SNAPSHOT_POINTS) {
  1366→  1366→      const step = pts.length / MAX_SNAPSHOT_POINTS;
  1367→  1367→      const down: Vec2[] = [];
  1368→  1368→      for (let i = 0; i < MAX_SNAPSHOT_POINTS; i++) {
  1369→  1369→        down.push(pts[Math.floor(i * step)]);
  1370→  1370→      }
  1371→  1371→      pts = down;
  1372→  1372→    }
  1373→  1373→    const now = rc.now;
  1374→  1374→    drawSnake(rc, {
  1375→  1375→      id: snake.id,
  1376→  1376→      name: snake.name,
  1377→  1377→      userTag: snake.userTag,
  1378→  1378→      points: pts,

  1396→  1396→  private drawJoystick(ctx: CanvasRenderingContext2D): void {
  1397→  1397→    const js = this.joystick;

  1400→  1400→    const dy = js.curY - js.originY;

  1422→  1422→  private updateParticles(dtMs: number): void {

  1424→  1424→    if (arr.length === 0) return;
  1425→  1425→    const dt = dtMs / 1000;
  1426→  1426→    for (let i = arr.length - 1; i >= 0; i--) {
  1427→  1427→      const p = arr[i];
  1428→  1428→      p.life -= dtMs;
  1429→  1429→      if (p.life <= 0) {
  1430→  1430→        arr.splice(i, 1);
  1431→  1431→        continue;
  1432→  1432→      }
  1433→  1433→      p.x += p.vx * dt;
  1434→  1434→      p.y += p.vy * dt;
  1435→  1435→      p.vx *= 0.96;
  1436→  1436→      p.vy *= 0.96;
  1437→  1437→    }
  1438→  1438→    if (arr.length > MAX_PARTICLES) {
  1439→  1439→      arr.splice(0, arr.length - MAX_PARTICLES);
  1440→  1440→    }

  1484→  1484→  private updateFps(now: number): void {
  1485→  1485→    const acc = this.fpsAccum;
  1486→  1486→    acc.frames += 1;
  1487→  1487→    if (acc.lastSecond === 0) acc.lastSecond = now;
  1488→  1488→    const dt = now - acc.lastSecond;
  1489→  1489→    if (dt >= 1000) {
  1490→  1490→      const measured = (acc.frames * 1000) / dt;
  1491→  1491→      this.fps = Math.round(measured);
  1492→  1492→      if (measured < FPS_LOW_THRESHOLD) {
  1493→  1493→        if (acc.lowSince === 0) acc.lowSince = now;
  1494→  1494→        if (acc.highSince !== 0) acc.highSince = 0;
  1495→  1495→        if (now - acc.lowSince >= FPS_LOW_DURATION_MS && !this.lowQuality) {
  1496→  1496→          this.lowQuality = true;
  1497→  1497→        }
  1498→  1498→      } else if (measured > FPS_HIGH_THRESHOLD) {
  1499→  1499→        if (acc.highSince === 0) acc.highSince = now;
  1500→  1500→        if (acc.lowSince !== 0) acc.lowSince = 0;
  1501→  1501→        if (now - acc.highSince >= FPS_HIGH_DURATION_MS && this.lowQuality) {
  1502→  1502→          this.lowQuality = false;
  1503→  1503→        }
  1504→  1504→      } else {
  1505→  1505→        acc.lowSince = 0;
  1506→  1506→        acc.highSince = 0;
  1507→  1507→      }
  1508→  1508→      acc.frames = 0;
  1509→  1509→      acc.lastSecond = now;
  1510→  1510→    }
  1511→  1511→  }
