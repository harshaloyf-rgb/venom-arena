   209→function angularDelta(current: number, desired: number): number {
   210→  let diff = desired - current;
   211→  while (diff > Math.PI) diff -= 2 * Math.PI;
   212→  while (diff < -Math.PI) diff += 2 * Math.PI;
   213→  return diff;
   214→}
   215→
   216→export function turnToward(current: number, desired: number, maxStep: number): number {
   217→  const diff = angularDelta(current, desired);
   218→  if (Math.abs(diff) <= maxStep) return desired;
   219→  return current + Math.sign(diff) * maxStep;
   220→}
   221→
   222→function dist(ax: number, ay: number, bx: number, by: number): number {
   223→  return Math.hypot(ax - bx, ay - by);
   224→}

   226→// ----------------------------------------------------------------------------
   227→// Weighted random picker (for food orb spawning)
   228→// ----------------------------------------------------------------------------
   229→
   230→function weightedRandomIndex(weights: number[]): number {
   231→  const r = Math.random();
   232→  let cumulative = 0;
   233→  for (let i = 0; i < weights.length; i++) {
   234→    cumulative += weights[i];
   235→    if (r <= cumulative) return i;
   236→  }
   237→  return weights.length - 1;
   238→}

   244→/** Random point inside a disc of radius `maxR` centered at (cx, cy). */
   245→export function randomSpawnPoint(maxR: number, cx: number = 0, cy: number = 0): Vec2 {
   246→  const r = Math.sqrt(Math.random()) * maxR * 0.85;
   247→  const theta = Math.random() * Math.PI * 2;
   248→  return { x: cx + Math.cos(theta) * r, y: cy + Math.sin(theta) * r };
   249→}

   251→/**
   252→ * Find a safe spawn point that is far from all existing snakes (players + bots).
   253→ * Tries SAFE_SPAWN_ATTEMPTS times, falls back to random if no safe spot found.
   254→ */
   255→export function findSafeSpawnPoint(room: ArenaRoom, maxR: number, cx: number = 0, cy: number = 0): Vec2 {
   256→  const allHeads: Vec2[] = [];
   257→  for (const s of collectAllSnakes(room)) {
   258→    if (s.isDead || s.points.length === 0) continue;
   259→    allHeads.push(s.points[0]);
   260→  }
   261→
   262→  for (let attempt = 0; attempt < SAFE_SPAWN_ATTEMPTS; attempt++) {
   263→    const pt = randomSpawnPoint(maxR, cx, cy);
   264→    let safe = true;
   265→
   266→    // Must be at least 500px inside the map boundary to avoid instant map death
   267→    const distFromCenter = dist(pt.x, pt.y, cx, cy);
   268→    if (distFromCenter > maxR - 500) {
   269→      safe = false;
   270→    }
   271→
   272→    // Must be far from all existing snake heads
   273→    if (safe) {
   274→      for (const head of allHeads) {
   275→        if (dist(pt.x, pt.y, head.x, head.y) < SAFE_SPAWN_MIN_DIST) {
   276→          safe = false;
   277→          break;
   278→        }
   279→      }
   280→    }
   281→    if (safe) return pt;
   282→  }
   283→  // Fallback: return a random point (better than no spawn at all)
   284→  return randomSpawnPoint(maxR, cx, cy);
   285→}

   287→/**
   288→ * Build an initial body: `length` points strung out behind the head at
   289→ * SEGMENT_SPACING px apart, all along the angle's reverse direction.
   290→ */
   291→export function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   292→  const pts: Vec2[] = [];
   293→  for (let i = 0; i < length; i++) {
   294→    pts.push({
   295→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   296→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   297→    });
   298→  }
   299→  return pts;
   300→}

   397→/**
   398→ * Compute the mix of Small(1), Medium(3), Large(5) orbs that sum to exactly totalScore.
   399→ * Greedily picks Large first, then Medium, then Small.
   400→ */
   401→export function computeDeathOrbs(totalScore: number): { small: number; medium: number; large: number } {
   402→  let remaining = totalScore;
   403→  const large = Math.floor(remaining / 5);
   404→  remaining -= large * 5;
   405→  const medium = Math.floor(remaining / 3);
   406→  remaining -= medium * 3;
   407→  const small = remaining;
   408→  return { small, medium, large };
   409→}

   411→/**
   412→ * Drop score orbs (S/M/L) that sum to exactly totalScore along a body path.
   413→ * Used for body-collision deaths (NOT wall deaths).
   414→ */
   415→export function dropScoreOrbsAtBody(
   416→  room: ArenaRoom,
   417→  bodyPoints: Vec2[],
   418→  totalScore: number,
   419→  snakeColor: string,
   420→): void {
   421→  if (!bodyPoints || bodyPoints.length === 0 || totalScore <= 0) return;
   422→
   423→  const { small, medium, large } = computeDeathOrbs(totalScore);
   424→  let orbIdx = 0;
   425→  const totalOrbs = small + medium + large;
   426→  if (totalOrbs === 0) return;
   427→
   428→  // Interleave S/M/L along the body for visual spread
   429→  const orbSequence: Array<{ value: number; size: number; color: string; glowColor: string; orbSize: 'small' | 'medium' | 'large' }> = [];
   430→  for (let i = 0; i < large; i++) orbSequence.push({ value: 5, size: 8, color: '#f472b6', glowColor: '#ec4899', orbSize: 'large' });
   431→  for (let i = 0; i < medium; i++) orbSequence.push({ value: 3, size: 5, color: '#38bdf8', glowColor: '#0ea5e9', orbSize: 'medium' });
   432→  for (let i = 0; i < small; i++) orbSequence.push({ value: 1, size: 3, color: '#34d399', glowColor: '#10b981', orbSize: 'small' });
   433→
   434→  // Shuffle for visual variety
   435→  for (let i = orbSequence.length - 1; i > 0; i--) {
   436→    const j = Math.floor(Math.random() * (i + 1));
   437→    [orbSequence[i], orbSequence[j]] = [orbSequence[j], orbSequence[i]];
   438→  }
   439→
   440→  for (const orb of orbSequence) {
   441→    const segIdx = Math.min(bodyPoints.length - 1, Math.floor((orbIdx / totalOrbs) * bodyPoints.length));
   442→    const pt = bodyPoints[segIdx];
   443→    const scatter = 20;
   444→    const id = `food-${room.arena.id}-${room.foodIdCounter++}`;
   445→    room.foods.push({
   446→      id,
   447→      x: pt.x + (Math.random() - 0.5) * scatter,
   448→      y: pt.y + (Math.random() - 0.5) * scatter,
   449→      size: orb.size,
   450→      value: orb.value,
   451→      isStarChip: false,
   452→      color: orb.color,
   453→      glowColor: orb.glowColor,
   454→      orbSize: orb.orbSize,
   455→    });
   456→    orbIdx++;
   457→  }
   458→}

   460→/**
   461→ * Drop exactly 10 Star collectibles at a death position.
   462→ * Each star = floor(chips / 10), remainder goes to the last star.
   463→ * ONLY for real players. Bots never drop stars.
   464→ */
   465→export function dropStarsAtDeath(
   466→  room: ArenaRoom,
   467→  x: number,
   468→  y: number,
   469→  chips: number,
   470→): void {
   471→  if (chips <= 0) return;
   472→
   473→  const valuePerStar = Math.floor(chips / STAR_DROP_COUNT);
   474→  const remainder = chips - valuePerStar * STAR_DROP_COUNT;
   475→  const scatter = 40;
   476→
   477→  // Always spawn exactly 10 stars. When chips < 10, some stars get value=1
   478→  // (remainder distributed to last stars) so all 10 are visible.
   479→  for (let i = 0; i < STAR_DROP_COUNT; i++) {
   480→    const value = Math.max(1, valuePerStar + (i === STAR_DROP_COUNT - 1 ? remainder : 0));
   481→    const id = `food-${room.arena.id}-${room.foodIdCounter++}`;
   482→    const angle = (i / STAR_DROP_COUNT) * Math.PI * 2;
   483→    const dist = scatter * 0.5 + Math.random() * scatter * 0.5;
   484→    room.foods.push({
   485→      id,
   486→      x: x + Math.cos(angle) * dist,
   487→      y: y + Math.sin(angle) * dist,
   488→      size: 12,
   489→      value,
   490→      isStarChip: true,
   491→      color: '#fbbf24',
   492→      glowColor: '#f59e0b',
   493→    });
   494→  }
   495→}

   497→/**
   498→ * Force one bot into self-destruct state to make room for a new human player.
   499→ * If no bots available, does nothing.
   500→ */
   501→export function displaceBotForPlayer(room: ArenaRoom): void {
   502→  if (room.bots.size === 0) return;
   503→  // Pick a random harvesting bot to self-destruct
   504→  const candidates = [...room.bots.values()].filter(b => !b.isDead && b.botState === 'harvesting');
   505→  if (candidates.length === 0) return;
   506→  const bot = candidates[Math.floor(Math.random() * candidates.length)];
   507→  bot.botState = 'selfDestruct';
   508→  bot.wantsBoost = false; // NEVER boost during self-destruct — go slowly
   509→}

   519→/**
   520→ * Move a snake one tick. Server-authoritative: only desiredAngle from client.
   521→ * Returns an array of food orbs to drop (from boost tail shedding).
   522→ */
   523→export function tickSnakeMovement(
   524→  snake: SnakeBase,
   525→  desiredAngle: number,
   526→  wantsBoost: boolean,
   527→): Vec2[] {
   528→  if (snake.points.length === 0) return [];
   529→  if (snake.isDead) return [];
   530→
   531→  const droppedFood: Vec2[] = [];
   532→  snake.wantsBoost = wantsBoost;
   533→
   534→  // 1) Turn — rate-limited, bigger snakes turn slower.
   535→  const turnRate = Math.max(TURN_MIN, TURN_BASE - snake.score * TURN_SCORE_FACTOR);
   536→  snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
   537→
   538→  // 2) Speed: extracting < normal < boost.
   539→  let speed = BASE_SPEED;
   540→  if (snake.isExtracting) {
   541→    speed = EXTRACT_GLIDE_SPEED;
   542→  } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
   543→    speed = BOOST_SPEED;
   544→    snake.boostFrameCounter++;
   545→    if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
   546→      snake.boostFrameCounter = 0;
   547→      if (snake.points.length > BOOST_MIN_LENGTH) {
   548→        // Record tail position BEFORE popping (for food drop)
   549→        const tail = snake.points[snake.points.length - 1];
   550→        droppedFood.push({ x: tail.x, y: tail.y });
   551→        snake.points.pop();
   552→        snake.score = Math.max(INITIAL_SPAWN_SCORE, snake.score - 1);
   553→      }
   554→    }
   555→  } else {
   556→    snake.wantsBoost = false;
   557→    snake.boostFrameCounter = 0;
   558→  }
   559→
   560→  // 3) Move head.
   561→  const head = snake.points[0];
   562→  const nx = head.x + Math.cos(snake.angle) * speed;
   563→  const ny = head.y + Math.sin(snake.angle) * speed;
   564→
   565→  // 4) Unshift new head.
   566→  snake.points.unshift({ x: nx, y: ny });
   567→
   568→  // 5) Body length = INITIAL_BODY_LENGTH + (score - INITIAL_SPAWN_SCORE)
   569→  const targetLen = Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + Math.max(0, snake.score - INITIAL_SPAWN_SCORE));
   570→  while (snake.points.length > targetLen) snake.points.pop();
   571→
   572→  // 6) Size formula.
   573→  snake.size = SIZE_BASE + Math.sqrt(snake.score) * SIZE_SCORE_FACTOR;
   574→
   575→  return droppedFood;
   576→}

   582→/**
   583→ * Per-bot AI tick. All bots seek food and evade human players.
   584→ * Self-destruct bots navigate toward the wall.
   585→ */
   586→export function tickBot(bot: BotSession, room: ArenaRoom, now: number): void {
   587→  if (bot.points.length === 0 || bot.isDead) return;
   588→
   589→  const head = bot.points[0];
   590→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   591→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount), now);
   592→
   593→  // --- Self-destruct state: navigate to nearest wall ---
   594→  if (bot.botState === 'selfDestruct') {
   595→    // Navigate AWAY from center (toward wall) — NEVER boost, go slowly.
   596→    const awayFromCenter = Math.atan2(head.y - room.mapCenterY, head.x - room.mapCenterX);
   597→    bot.desiredAngle = awayFromCenter;
   598→    bot.wantsBoost = false; // CRITICAL: never boost during self-destruct
   599→
   600→    // Still eat food on the way (prioritize nearby food, but keep moving toward wall)
   601→    if (now >= bot.nextThinkAt) {
   602→      bot.nextThinkAt = now + 120; // think less frequently (move slowly)
   603→      const foodQuery = room.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
   604→      let bestFood: GridItem | null = null;
   605→      let bestFoodDist = Infinity;
   606→      for (const item of foodQuery.values()) {
   607→        if (item.kind !== 'food') continue;
   608→        if ((item.value ?? 0) <= 0) continue;
   609→        if (item.isStarChip) continue; // bots ignore stars
   610→        const d = dist(head.x, head.y, item.x, item.y);
   611→        if (d < bestFoodDist) {
   612→          bestFoodDist = d;
   613→          bestFood = item;
   614→        }
   615→      }
   616→      // Blend food-seeking with wall-seeking (80% wall, 20% food if close)
   617→      if (bestFood && bestFoodDist < 120) {
   618→        const foodAngle = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
   619→        bot.desiredAngle = turnToward(awayFromCenter, foodAngle, 0.03);
   620→      }
   621→    }
   622→
   623→    tickSnakeMovement(bot, bot.desiredAngle, false); // NEVER boost
   624→    return;
   625→  }
   626→
   627→  // --- Harvesting state ---
   628→  if (now >= bot.nextThinkAt) {
   629→    bot.nextThinkAt = now + 120 + Math.floor(Math.random() * 80);
   630→
   631→    // Find nearest food
   632→    let bestFood: GridItem | null = null;
   633→    let bestFoodDist = Infinity;
   634→    const foodQuery = room.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
   635→    for (const item of foodQuery.values()) {
   636→      if (item.kind !== 'food') continue;
   637→      if ((item.value ?? 0) <= 0) continue;
   638→      if (item.isStarChip) continue; // bots ignore stars
   639→      const d = dist(head.x, head.y, item.x, item.y);
   640→      if (d < bestFoodDist) {
   641→        bestFoodDist = d;
   642→        bestFood = item;
   643→      }
   644→    }
   645→
   646→    // Find nearest human player (for evasion)
   647→    let nearPlayerDist = Infinity;
   648→    let nearPlayerAngle = 0;
   649→    let nearPlayerVx = 0;
   650→    let nearPlayerVy = 0;
   651→    let nearPlayerHx = 0;
   652→    let nearPlayerHy = 0;
   653→    for (const p of room.players.values()) {
   654→      if (p.isDead || p.matchSettling) continue;
   655→      if (p.points.length === 0) continue;
   656→      const ph = p.points[0];
   657→      const d = dist(head.x, head.y, ph.x, ph.y);
   658→      if (d < nearPlayerDist) {
   659→        nearPlayerDist = d;
   660→        nearPlayerAngle = Math.atan2(ph.y - head.y, ph.x - head.x);
   661→        nearPlayerHx = ph.x;
   662→        nearPlayerHy = ph.y;
   663→        // Estimate player velocity from head angle
   664→        nearPlayerVx = Math.cos(p.angle) * BASE_SPEED;
   665→        nearPlayerVy = Math.sin(p.angle) * BASE_SPEED;
   666→      }
   667→    }
   668→
   669→    // Also check nearby body segments (for collision avoidance)
   670→    let threatX = 0;
   671→    let threatY = 0;
   672→    let threatDist = Infinity;
   673→    const threatQuery = room.grid.queryRadius(head.x, head.y, 150);
   674→    for (const item of threatQuery.values()) {
   675→      if (item.kind !== 'segment') continue;
   676→      if (item.snakeId === bot.id) continue;
   677→      if (item.segIdx === 0) continue;
   678→      const d = dist(head.x, head.y, item.x, item.y);
   679→      if (d < threatDist) {
   680→        threatDist = d;
   681→        threatX = item.x;
   682→        threatY = item.y;
   683→      }
   684→    }
   685→
   686→    let desired: number;
   687→
   688→    // Priority 1: Evasion of body segments (immediate danger)
   689→    if (threatDist < 140) {
   690→      desired = Math.atan2(head.y - threatY, head.x - threatX);
   691→    }
   692→    // Priority 2: Predictive evasion of human players
   693→    else if (nearPlayerDist < BOT_EVADE_RADIUS) {
   694→      // Project nearest player position 8 ticks ahead
   695→      const futurePx = nearPlayerHx + nearPlayerVx * 8;
   696→      const futurePy = nearPlayerHy + nearPlayerVy * 8;
   697→      const futureD = dist(head.x, head.y, futurePx, futurePy);
   698→      if (futureD < BOT_EVADE_RADIUS) {
   699→        // Steer perpendicular to the player's approach vector
   700→        const perpAngle = nearPlayerAngle + Math.PI / 2 * (Math.random() > 0.5 ? 1 : -1);
   701→        desired = perpAngle;
   702→      } else {
   703→        // Just steer away from nearest player
   704→        desired = Math.atan2(head.y - nearPlayerHy, head.x - nearPlayerHx);
   705→      }
   706→    }
   707→    // Priority 3: Seek food
   708→    else if (bestFood) {
   709→      desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
   710→    }
   711→    // Priority 4: Wander
   712→    else {
   713→      desired = bot.angle + (Math.random() - 0.5) * 0.4;
   714→    }
   715→
   716→    // Edge avoidance: if near map boundary, turn back toward center
   717→    const distFromCenter = Math.hypot(head.x - room.mapCenterX, head.y - room.mapCenterY);
   718→    if (distFromCenter > mapRadius - 300) {
   719→      const toCenter = Math.atan2(room.mapCenterY - head.y, room.mapCenterX - head.x);
   720→      desired = turnToward(desired, toCenter, MAX_TURN_PER_TICK * 2);
   721→    }
   722→
   723→    bot.desiredAngle = desired;
   724→  }
   725→
   726→  // Check self-destruct threshold
   727→  if (bot.score >= BOT_SELF_DESTRUCT_THRESHOLD && bot.botState === 'harvesting') {
   728→    bot.botState = 'selfDestruct';
   729→  }
   730→
   731→  // Bots don't boost in harvesting mode
   732→  tickSnakeMovement(bot, bot.desiredAngle, false);
   733→}

   739→/** A pending death detected during collision iteration. */
   740→export interface PendingDeath {
   741→  deadId: string;
   742→  killerId?: string;

   746→/**
   747→ * Detect head-to-body collisions via spatial grid.
   748→ * Head hits any foreign body segment → that head's owner dies.
   749→ */
   750→export function detectCollisions(room: ArenaRoom, now: number): PendingDeath[] {
   751→  const deaths: PendingDeath[] = [];
   752→  const seenDead = new Set<string>();
   753→
   754→  const allSnakes: SnakeBase[] = collectAllSnakes(room);
   755→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   756→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount), now);
   757→
   758→  for (const snake of allSnakes) {
   759→    if (snake.isDead) continue;
   760→    if (snake.points.length === 0) continue;
   761→    if (now < snake.spawnProtectedUntil) continue;
   762→    if (seenDead.has(snake.id)) continue;
   763→
   764→    const head = snake.points[0];
   765→
   766→    // Wall collision
   767→    const distFromCenter = Math.hypot(head.x - room.mapCenterX, head.y - room.mapCenterY);
   768→    if (distFromCenter > mapRadius) {
   769→      deaths.push({ deadId: snake.id, killerId: 'wall', cause: 'wall' });
   770→      seenDead.add(snake.id);
   771→      continue;
   772→    }
   773→
   774→    // Head-to-body collision (with neck protection)
   775→    const queryR = snake.size + 30;
   776→    const nearby = room.grid.queryRadius(head.x, head.y, queryR);
   777→
   778→    for (const item of nearby.values()) {
   779→      if (item.kind !== 'segment') continue;
   780→      if (item.snakeId === snake.id) continue;
   781→      if (item.segIdx === 0) continue; // head-to-head handled separately
   782→      // Neck protection: skip first N segments behind the head (close-call safety)
   783→      if (item.segIdx <= NECK_PROTECTION_SEGS) continue;
   784→      const d = dist(head.x, head.y, item.x, item.y);
   785→      if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
   786→        deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
   787→        seenDead.add(snake.id);
   788→        break;
   789→      }
   790→    }
   791→  }
   792→
   793→  return deaths;
   794→}
   795→
   796→/**
   797→ * Detect head-on (head-to-head) collisions.
   798→ * Rules:
   799→ *   A) No boost / both boosting → higher score survives
   800→ *   B) Smaller boosting, larger steady → smaller survives
   801→ *   C) Both boosting → higher score survives (same as A)
   802→ *   Tie → both die
   803→ */
   804→export function detectHeadOnCollisions(room: ArenaRoom, now: number): PendingDeath[] {
   805→  const deaths: PendingDeath[] = [];
   806→  const seenDead = new Set<string>();
   807→
   808→  const allSnakes: SnakeBase[] = collectAllSnakes(room);
   809→  const headMap = new Map<string, SnakeBase>();
   810→  for (const s of allSnakes) {
   811→    if (s.isDead || s.points.length === 0 || now < s.spawnProtectedUntil) continue;
   812→    headMap.set(s.id, s);
   813→  }
   814→
   815→  const processed = new Set<string>();
   816→  for (const snakeA of allSnakes) {
   817→    if (snakeA.isDead || snakeA.points.length === 0 || now < snakeA.spawnProtectedUntil) continue;
   818→    if (seenDead.has(snakeA.id)) continue;
   819→    if (processed.has(snakeA.id)) continue;
   820→
   821→    const headA = snakeA.points[0];
   822→    const queryR = snakeA.size + 20;
   823→    const nearby = room.grid.queryRadius(headA.x, headA.y, queryR);
   824→
   825→    for (const item of nearby.values()) {
   826→      if (item.kind !== 'segment') continue;
   827→      if (item.segIdx !== 0) continue; // only heads
   828→      const snakeB = headMap.get(item.snakeId ?? '');
   829→      if (!snakeB || snakeB.isDead || snakeB.id === snakeA.id) continue;
   830→      if (seenDead.has(snakeB.id)) continue;
   831→      if (processed.has(snakeB.id)) continue;
   832→
   833→      const headB = snakeB.points[0];
   834→      const d = dist(headA.x, headA.y, headB.x, headB.y);
   835→      if (d < (snakeA.size + snakeB.size) * HEAD_ON_HIT_FACTOR) {
   836→        processed.add(snakeA.id);
   837→        processed.add(snakeB.id);
   838→
   839→        const aBoosting = snakeA.wantsBoost;
   840→        const bBoosting = snakeB.wantsBoost;
   841→        const aScore = snakeA.score;
   842→        const bScore = snakeB.score;
   843→
   844→        let loserId: string | null = null;
   845→
   846→        if (aScore === bScore) {
   847→          // Tie: both die
   848→          deaths.push({ deadId: snakeA.id, killerId: snakeB.id, cause: 'headOn' });
   849→          deaths.push({ deadId: snakeB.id, killerId: snakeA.id, cause: 'headOn' });
   850→          seenDead.add(snakeA.id);
   851→          seenDead.add(snakeB.id);
   852→        } else if (aScore > bScore) {
   853→          // A is bigger
   854→          if (bBoosting && !aBoosting) {
   855→            // Rule B: smaller boosting vs larger steady → smaller survives
   856→            loserId = snakeA.id;
   857→          } else {
   858→            // Rule A/C: bigger survives
   859→            loserId = snakeB.id;
   860→          }
   861→        } else {
   862→          // B is bigger
   863→          if (aBoosting && !bBoosting) {
   864→            // Rule B: smaller boosting vs larger steady → smaller survives
   865→            loserId = snakeB.id;
   866→          } else {
   867→            // Rule A/C: bigger survives
   868→            loserId = snakeA.id;
   869→          }
   870→        }
   871→
   872→        if (loserId && aScore !== bScore) {
   873→          const winnerId = loserId === snakeA.id ? snakeB.id : snakeA.id;
   874→          deaths.push({ deadId: loserId, killerId: winnerId, cause: 'headOn' });
   875→          seenDead.add(loserId);
   876→        }
   877→        break;
   878→      }
   879→    }
   880→  }
   881→
   882→  return deaths;
   883→}

   887→// ----------------------------------------------------------------------------
   888→
   889→/**
   890→ * Eat food via spatial grid.
   891→ * Regular food: snake.score += value (1, 3, or 5). ALL snakes eat.
   892→ * Star chips: carriedChips += value, score += STAR_CHIP_GROW. ONLY real players.
   893→ * Bots NEVER collect star chips.
   894→ */
   895→export function eatFood(room: ArenaRoom): void {
   896→  const allSnakes: SnakeBase[] = collectAllSnakes(room);
   897→  for (const snake of allSnakes) {
   898→    if (snake.isDead) continue;
   899→    if (snake.points.length === 0) continue;
   900→    const head = snake.points[0];
   901→    const nearby = room.grid.queryRadius(head.x, head.y, snake.size + 20);
   902→    for (const item of nearby.values()) {
   903→      if (item.kind !== 'food') continue;
   904→      if ((item.value ?? 0) <= 0) continue;
   905→      const d = dist(head.x, head.y, item.x, item.y);
   906→      if (d < snake.size + (item.radius || 6) + 6) {
   907→        if (item.isStarChip) {
   908→          // Star chip: ONLY real players collect (bots skip).
   909→          if (!snake.isPlayer) continue;
   910→          snake.carriedChips += item.value ?? 0;
   911→          snake.score += STAR_CHIP_GROW;
   912→        } else {
   913→          // Regular food orb: ALL snakes eat, score += value.
   914→          snake.score += item.value ?? 1;
   915→        }
   916→        // Zero BOTH the grid item and the real food object.
   917→        item.value = 0;
   918→        if (item.foodRef) item.foodRef.value = 0;
   919→      }
   920→    }
   921→  }
   922→}

   928→/** Iterator: yield every snake (player + bot) in the room. */
   929→export function collectAllSnakes(room: ArenaRoom): SnakeBase[] {
   930→  const out: SnakeBase[] = [];
   931→  for (const p of room.players.values()) out.push(p);
   932→  for (const b of room.bots.values()) out.push(b);
   933→  return out;

   936→/**
   937→ * Build the broadcast GameSnapshot for an arena.
   938→ */
   939→export function buildSnapshot(room: ArenaRoom, viewerId?: string): GameSnapshot {
   940→  const snakes: SnakeSnapshot[] = [];
   941→  const now = Date.now();
   942→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   943→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount), now);
   944→
   945→  for (const snake of collectAllSnakes(room)) {
   946→    let points = snake.points;
   947→    if (points.length > MAX_SNAPSHOT_POINTS) {
   948→      const step = points.length / MAX_SNAPSHOT_POINTS;
   949→      const downsampled: Vec2[] = [];
   950→      for (let i = 0; i < MAX_SNAPSHOT_POINTS; i++) {
   951→        downsampled.push(points[Math.floor(i * step)]);
   952→      }
   953→      points = downsampled;
   954→    }
   955→    snakes.push({
   956→      id: snake.id,
   957→      name: snake.name,
   958→      userTag: snake.userTag,
   959→      points,
   960→      angle: snake.angle,
   961→      size: snake.size,
   962→      color: snake.color,
   963→      secondaryColor: snake.secondaryColor,
   964→      isPlayer: snake.isPlayer,
   965→      isBot: snake.isBot,
   966→      carriedChips: Math.floor(snake.carriedChips),
   967→      score: snake.score,
   968→      isExtracting: snake.isExtracting,
   969→      extractionProgress: Math.min(1, snake.extractionProgress / EXTRACT_DURATION_MS),
   970→      isDead: snake.isDead,
   971→      spawnProtected: now < snake.spawnProtectedUntil,
   972→      chatMessage: snake.chatMessage,
   973→      country: snake.country,
   974→      isBoosting: snake.wantsBoost,
   975→      botState: snake.botState,
   976→    });
   977→  }
   978→
   979→  const foods: FoodSnapshot[] = room.foods.map((f) => ({
   980→    id: f.id,
   981→    x: f.x,
   982→    y: f.y,
   983→    size: f.size,
   984→    value: f.value,
   985→    isStarChip: f.isStarChip,
   986→    color: f.color,
   987→    glowColor: f.glowColor,
   988→    orbSize: f.orbSize,
   989→  }));
   990→
   991→  // Arena leaderboard: real players by carriedChips desc
   992→  const realPlayers = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling);
   993→  const sorted = realPlayers.sort((a, b) => b.carriedChips - a.carriedChips);
   994→  const arenaLeaderboard = sorted.slice(0, 10).map(p => ({
   995→    id: p.id,
   996→    name: p.name,
   997→    userTag: p.userTag,
   998→    carriedChips: Math.floor(p.carriedChips),
   999→    score: p.score,
  1000→    kills: p.kills,
  1001→    isPlayer: p.id === viewerId,
  1002→    country: p.country,
  1003→  }));
  1004→
  1005→  const yourRank = viewerId ? sorted.findIndex(p => p.id === viewerId) + 1 : 0;
  1006→  const commissionRate = realPlayerCount <= 3 ? 0 : 0.35;
  1007→
  1008→  return {
  1009→    arenaId: room.arena.id,
  1010→    tick: room.tick,
  1011→    snakes,
  1012→    foods,
  1013→    worldSize: WORLD_SIZE,
  1014→    mapRadius,
  1015→    mapCenterX: room.mapCenterX,
  1016→    mapCenterY: room.mapCenterY,
  1017→    leaderId: room.leaderId,
  1018→    leaderChips: room.leaderChips,
  1019→    realPlayerCount,
  1020→    yourRank: yourRank > 0 ? yourRank : 0,
  1021→    arenaLeaderboard,
  1022→    commissionRate,
  1023→  };
  1024→}
