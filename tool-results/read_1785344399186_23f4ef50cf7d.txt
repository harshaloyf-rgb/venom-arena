     1→'use client';
     2→
     3→import { useMemo, useState } from 'react';
     4→import { useAuth } from '@/components/providers/auth-provider';
     5→import {
     6→  INITIAL_FRIENDS,
     7→  INITIAL_RIVALS,
     8→  GLOBAL_COMMUNITY_PLAYERS,
     9→  SOCIAL_COUNTRY_FILTER,
    10→  PUBLIC_CLANS,
    11→  PRESET_EMBLEMS,
    12→  BOT_REPLIES,
    13→  countryFlag,
    14→  countryName,
    15→  ARENA_TIERS,
    16→  type MockFriend,
    17→  type MockRival,
    18→  type GlobalPlayer,
    19→} from '@/lib/game-config';
    20→import {
    21→  GlowBlob,
    22→  MicroLabel,
    23→  NotSignedIn,
    24→  notify,
    25→  type ToastFn,
    26→} from './_panel-primitives';
    27→import {
    28→  Users,
    29→  Shield,
    30→  Swords,
    31→  Globe,
    32→  UserPlus,
    33→  Gift,
    34→  Eye,
    35→  Send,
    36→  X,
    37→  Check,
    38→  Coins,
    39→  MessageSquare,
    40→  Search,
    41→  Plus,
    42→  LogOut,
    43→  Award,
    44→} from 'lucide-react';
    45→
    46→interface SocialPanelProps {
    47→  onToast?: ToastFn;
    48→  onSpectateFriend?: (name: string, color: string) => void;
    49→  onJoinArena?: (arenaId: string) => void;
    50→}
    51→
    52→type TopTab = 'friends' | 'syndicate';
    53→type FriendsSubTab = 'friends' | 'rivals' | 'search';
    54→
    55→const FRIEND_CHIPS: Record<string, number> = {
    56→  ApexViper: 1200,
    57→  ShadowSlinker: 150,
    58→  CoinGobbler: 40,
    59→  VenomKing: 250000,
    60→};
    61→
    62→function friendChips(name: string, level: number): number {
    63→  if (FRIEND_CHIPS[name] !== undefined) return FRIEND_CHIPS[name];
    64→  return level * 45;
    65→}
    66→
    67→const STATUS_LABELS: Record<MockFriend['status'] | MockRival['status'], string> = {
    68→  online: 'Lobby',
    69→  idle: 'Idle',
    70→  'in-match': 'In-Arena',
    71→  offline: 'Offline',
    72→};
    73→
    74→const STATUS_BADGES: Record<MockRival['status'], string> = {
    75→  'in-match': '⚔️ Playing Arena',
    76→  online: '🟢 Online',
    77→  idle: '🟢 Online',
    78→  offline: '⚪ Offline',
    79→};
    80→
    81→export function SocialPanel({ onToast, onSpectateFriend, onJoinArena }: SocialPanelProps) {
    82→  const { player, refresh } = useAuth();
    83→  const [topTab, setTopTab] = useState<TopTab>('friends');
    84→  const [friendsSub, setFriendsSub] = useState<FriendsSubTab>('friends');
    85→  const [friends, setFriends] = useState<MockFriend[]>(INITIAL_FRIENDS);
    86→  const [rivals, setRivals] = useState<MockRival[]>(INITIAL_RIVALS);
    87→  const [search, setSearch] = useState('');
    88→  const [countryFilter, setCountryFilter] = useState('ALL');
    89→  const [addFriendInput, setAddFriendInput] = useState('');
    90→  const [joinedClanId, setJoinedClanId] = useState<string | null>(null);
    91→  const [showCreateClan, setShowCreateClan] = useState(false);
    92→  const [clanForm, setClanForm] = useState({ name: '', tag: '', emblem: PRESET_EMBLEMS[0], description: '' });
    93→  const [clanChat, setClanChat] = useState<{ author: string; text: string; ts: string }[]>([]);
    94→  const [chatInput, setChatInput] = useState('');
    95→  const [depositAmount, setDepositAmount] = useState('');
    96→  const [coOpFriend, setCoOpFriend] = useState<MockFriend | null>(null);
    97→
    98→  const filteredGlobalPlayers = useMemo(() => {
    99→    return GLOBAL_COMMUNITY_PLAYERS.filter((p) => {
   100→      if (countryFilter !== 'ALL' && p.country !== countryFilter) return false;
   101→      if (search.trim()) {
   102→        const q = search.toLowerCase();
   103→        return p.name.toLowerCase().includes(q) || p.userTag.toLowerCase().includes(q);
   104→      }
   105→      return true;
   106→    });
   107→  }, [search, countryFilter]);
   108→
   109→  if (!player) return <NotSignedIn />;
   110→
   111→  const joinedClan = joinedClanId ? PUBLIC_CLANS.find((c) => c.id === joinedClanId) : null;
   112→
   113→  function handleAddFriend() {
   114→    const name = addFriendInput.trim();
   115→    if (!name) {
   116→      notify('Please enter a player tag or name.', 'error', onToast);
   117→      return;
   118→    }
   119→    const tagNum = Math.floor(1000 + Math.random() * 9000);
   120→    const finalTag = name.substring(0, 4).toUpperCase() + '-' + tagNum;
   121→    const newFriend: MockFriend = {
   122→      id: `f-${Date.now()}`,
   123→      name,
   124→      userTag: finalTag,
   125→      status: 'offline',
   126→      level: 5 + Math.floor(Math.random() * 30),
   127→      skinColor: '#22d3ee',
   128→      giftSent: false,
   129→      giftReceived: false,
   130→    };
   131→    setFriends((prev) => [...prev, newFriend]);
   132→    setAddFriendInput('');
   133→    notify(`Friend request sent to ${name} (${finalTag})! 🤝`, 'success', onToast);
   134→  }
   135→
   136→  function handleSendGift(f: MockFriend) {
   137→    setFriends((prev) => prev.map((x) => (x.id === f.id ? { ...x, giftSent: true } : x)));
   138→    notify(`Sent 25 Daily Chips Gift to ${f.name}! 🎁`, 'success', onToast);
   139→  }
   140→
   141→  function handleClaimGift(f: MockFriend) {
   142→    setFriends((prev) => prev.map((x) => (x.id === f.id ? { ...x, giftReceived: false } : x)));
   143→    notify(`Claimed 25 chips gift from ${f.name}! 🪙`, 'success', onToast);
   144→    void refresh();
   145→  }
   146→
   147→  function handleSpectate(f: MockFriend) {
   148→    if (f.status !== 'in-match') return;
   149→    if (onSpectateFriend) onSpectateFriend(f.name, f.skinColor);
   150→    else notify(`Joining spectating server for ${f.name}... 👁️`, 'info', onToast);
   151→  }
   152→
   153→  function handleInviteFriend(f: MockFriend) {
   154→    if (f.status === 'offline') return;
   155→    setCoOpFriend(f);
   156→  }
   157→
   158→  function handleRemoveFriend(f: MockFriend) {
   159→    setFriends((prev) => prev.filter((x) => x.id !== f.id));
   160→    notify(`Removed ${f.name} from friends list.`, 'info', onToast);
   161→  }
   162→
   163→  function handleHuntRival(r: MockRival) {
   164→    notify(`⚔️ HUNT INITIATED: Entering ${r.currentArenaName} to take down ${r.name}!`, 'info', onToast);
   165→    if (onJoinArena) onJoinArena('tier-2');
   166→  }
   167→
   168→  function handleConvertRival(r: MockRival) {
   169→    setFriends((prev) => [...prev, {
   170→      id: `f-${Date.now()}`,
   171→      name: r.name,
   172→      userTag: r.userTag,
   173→      status: r.status,
   174→      currentArenaName: r.currentArenaName,
   175→      level: r.level,
   176→      skinColor: '#22d3ee',
   177→      giftSent: false,
   178→      giftReceived: false,
   179→    }]);
   180→    setRivals((prev) => prev.filter((x) => x.id !== r.id));
   181→    notify(`${r.name} converted from rival to friend!`, 'success', onToast);
   182→  }
   183→
   184→  function handleRemoveRival(r: MockRival) {
   185→    setRivals((prev) => prev.filter((x) => x.id !== r.id));
   186→    notify(`Removed ${r.name} from rivals list.`, 'info', onToast);
   187→  }
   188→
   189→  function handleJoinClan(clanId: string) {
   190→    const clan = PUBLIC_CLANS.find((c) => c.id === clanId);
   191→    if (!clan) return;
   192→    if (joinedClanId) {
   193→      notify('You are already in a clan! Leave your current clan first.', 'error', onToast);
   194→      return;
   195→    }
   196→    setJoinedClanId(clanId);
   197→    setClanChat([
   198→      { author: clan.members[0]?.name || 'Leader', text: BOT_REPLIES[Math.floor(Math.random() * BOT_REPLIES.length)], ts: 'just now' },
   199→    ]);
   200→    notify(`Welcome to ${clan.name} [${clan.tag}]! 🛡️`, 'success', onToast);
   201→  }
   202→
   203→  function handleLeaveClan() {
   204→    if (!joinedClan) return;
   205→    notify(`Left ${joinedClan.name} [${joinedClan.tag}].`, 'info', onToast);
   206→    setJoinedClanId(null);
   207→    setClanChat([]);
   208→  }
   209→
   210→  function handleCreateClan() {
   211→    if (clanForm.name.length < 4) {
   212→      notify('Syndicate Name must be at least 4 characters.', 'error', onToast);
   213→      return;
   214→    }
   215→    if (clanForm.tag.length < 3 || clanForm.tag.length > 4) {
   216→      notify('Clan Tag must be 3-4 characters.', 'error', onToast);
   217→      return;
   218→    }
   219→    if (player.bankedChips < 500) {
   220→      notify('You need at least 500 chips to register a Syndicate.', 'error', onToast);
   221→      return;
   222→    }
   223→    notify(`Syndicate "${clanForm.name}" [${clanForm.tag.toUpperCase()}] established!`, 'success', onToast);
   224→    setShowCreateClan(false);
   225→    setClanForm({ name: '', tag: '', emblem: PRESET_EMBLEMS[0], description: '' });
   226→  }
   227→
   228→  function handleDeposit() {
   229→    const amt = parseInt(depositAmount, 10);
   230→    if (!amt || amt <= 0) {
   231→      notify('Enter a valid deposit amount.', 'error', onToast);
   232→      return;
   233→    }
   234→    if (amt > player.bankedChips) {
   235→      notify('Insufficient chips to deposit.', 'error', onToast);
   236→      return;
   237→    }
   238→    notify(`Deposited ${amt}c to syndicate vault (+${Math.floor(amt * 0.1)} Clan XP)!`, 'success', onToast);
   239→    setDepositAmount('');
   240→    void refresh();
   241→  }
   242→
   243→  function handleSendChat() {
   244→    if (!chatInput.trim() || !joinedClan) return;
   245→    const msg = { author: player.name, text: chatInput.trim(), ts: 'just now' };
   246→    setClanChat((prev) => [...prev, msg]);
   247→    setChatInput('');
   248→    // Bot reply
   249→    setTimeout(() => {
   250→      const botMsg = BOT_REPLIES[Math.floor(Math.random() * BOT_REPLIES.length)];
   251→      const botAuthor = joinedClan.members[Math.floor(Math.random() * joinedClan.members.length)]?.name || 'Member';
   252→      setClanChat((prev) => [...prev, { author: botAuthor, text: botMsg, ts: 'just now' }]);
   253→    }, 1500);
   254→  }
   255→
   256→  return (
   257→    <div className="relative rounded-2xl border border-slate-800/80 bg-slate-900/60 shadow-md p-5 sm:p-6 overflow-hidden">
   258→      <GlowBlob color="bg-violet-500/10" className="-top-12 -right-12 w-56 h-56" />
   259→
   260→      {/* TOP TABS */}
   261→      <div className="relative flex flex-wrap items-center gap-1.5 bg-slate-950 p-1 rounded-xl border border-slate-800/60 mb-5">
   262→        <button
   263→          type="button"
   264→          onClick={() => setTopTab('friends')}
   265→          className={`px-3 py-1.5 rounded-lg text-xs font-bold flex items-center gap-1.5 transition border ${topTab === 'friends' ? 'bg-violet-500/20 border-violet-500/40 text-violet-300' : 'text-slate-500 hover:text-slate-300 border-transparent'}`}
   266→        >
   267→          <Users className="w-3.5 h-3.5" /> Friends &amp; Global Search ({friends.length})
   268→        </button>
   269→        <button
   270→          type="button"
   271→          onClick={() => setTopTab('syndicate')}
   272→          className={`px-3 py-1.5 rounded-lg text-xs font-bold flex items-center gap-1.5 transition border ${topTab === 'syndicate' ? 'bg-violet-500/20 border-violet-500/40 text-violet-300' : 'text-slate-500 hover:text-slate-300 border-transparent'}`}
   273→        >
   274→          <Shield className="w-3.5 h-3.5" /> Competitive Syndicate {joinedClan ? `[${joinedClan.tag}]` : ''}
   275→        </button>
   276→      </div>
   277→
   278→      {/* FRIENDS TAB */}
   279→      {topTab === 'friends' && (
   280→        <div className="space-y-4">
   281→          {/* Sub-tabs */}
   282→          <div className="flex flex-wrap items-center gap-1.5 bg-slate-950 p-1 rounded-xl border border-slate-800/60">
   283→            <SubTabBtn active={friendsSub === 'friends'} onClick={() => setFriendsSub('friends')} icon={Users} label={`My Friends (${friends.length})`} />
   284→            <SubTabBtn active={friendsSub === 'rivals'} onClick={() => setFriendsSub('rivals')} icon={Swords} label={`My Rivals (${rivals.length})`} />
   285→            <SubTabBtn active={friendsSub === 'search'} onClick={() => setFriendsSub('search')} icon={Globe} label="Search Global Players" />
   286→          </div>
   287→
   288→          {/* Add friend bar */}
   289→          <div className="flex items-center gap-2">
   290→            <input
   291→              type="text"
   292→              value={addFriendInput}
   293→              onChange={(e) => setAddFriendInput(e.target.value)}
   294→              placeholder="Enter Player Tag or Name (e.g. Cobra-4231)..."
   295→              className="flex-1 bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-xs text-white placeholder:text-slate-600 focus:outline-none focus:border-violet-500/50"
   296→            />
   297→            <button
   298→              type="button"
   299→              onClick={handleAddFriend}
   300→              className="px-4 py-2 rounded-xl bg-violet-600 hover:bg-violet-500 text-white font-bold text-xs uppercase tracking-wider transition flex items-center gap-1.5"
   301→            >
   302→              <UserPlus className="w-3.5 h-3.5" /> Add Friend
   303→            </button>
   304→          </div>
   305→
   306→          {/* My Friends */}
   307→          {friendsSub === 'friends' && (
   308→            <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   309→              {friends.length === 0 ? (
   310→                <div className="md:col-span-2 p-6 rounded-xl border border-slate-800 bg-slate-950/60 text-center">
   311→                  <Users className="w-8 h-8 text-slate-600 mx-auto mb-2" />
   312→                  <h4 className="text-sm font-bold text-white">Your Friends List is Empty</h4>
   313→                  <p className="text-xs text-slate-400 mt-1">
   314→                    Use &quot;Search Global Players&quot; above or enter rival tags to connect,
   315→                    gift daily free chips, and play!
   316→                  </p>
   317→                </div>
   318→              ) : (
   319→                friends.map((f) => (
   320→                  <FriendCard
   321→                    key={f.id}
   322→                    f={f}
   323→                    onSendGift={() => handleSendGift(f)}
   324→                    onClaimGift={() => handleClaimGift(f)}
   325→                    onSpectate={() => handleSpectate(f)}
   326→                    onInvite={() => handleInviteFriend(f)}
   327→                    onRemove={() => handleRemoveFriend(f)}
   328→                  />
   329→                ))
   330→              )}
   331→            </div>
   332→          )}
   333→
   334→          {/* My Rivals */}
   335→          {friendsSub === 'rivals' && (
   336→            <div className="space-y-3">
   337→              <div className="flex items-center justify-between flex-wrap gap-2">
   338→                <div>
   339→                  <h3 className="text-sm font-bold text-white uppercase tracking-wider">RIVALRY &amp; REVENGE TRACKER</h3>
   340→                  <p className="text-xs text-slate-400 mt-1 max-w-2xl">
   341→                    Players who eliminated you or collided with you in arena matches.
   342→                    Track their online status and join their exact arena to seek revenge!
   343→                  </p>
   344→                </div>
   345→                <span className="text-[10px] font-mono bg-rose-500/10 border border-rose-500/30 text-rose-300 px-2 py-0.5 rounded-full">
   346→                  {rivals.length} Active Rivals
   347→                </span>
   348→              </div>
   349→              {rivals.length === 0 ? (
   350→                <div className="p-6 rounded-xl border border-slate-800 bg-slate-950/60 text-center">
   351→                  <Swords className="w-8 h-8 text-slate-600 mx-auto mb-2" />
   352→                  <h4 className="text-sm font-bold text-white">No Rivals in Your List</h4>
   353→                  <p className="text-xs text-slate-400 mt-1">
   354→                    When you get eliminated or collide with players in matches,
   355→                    click &quot;ADD RIVAL&quot; on the game-over screen to track them here!
   356→                  </p>
   357→                </div>
   358→              ) : (
   359→                <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
   360→                  {rivals.map((r) => (
   361→                    <RivalCard
   362→                      key={r.id}
   363→                      r={r}
   364→                      onHunt={() => handleHuntRival(r)}
   365→                      onConvert={() => handleConvertRival(r)}
   366→                      onRemove={() => handleRemoveRival(r)}
   367→                    />
   368→                  ))}
   369→                </div>
   370→              )}
   371→            </div>
   372→          )}
   373→
   374→          {/* Global Search */}
   375→          {friendsSub === 'search' && (
   376→            <div className="space-y-3">
   377→              <div className="flex flex-wrap items-center gap-2">
   378→                <div className="relative flex-1 min-w-[200px]">
   379→                  <Search className="w-3.5 h-3.5 text-slate-500 absolute left-3 top-1/2 -translate-y-1/2" />
   380→                  <input
   381→                    type="text"
   382→                    value={search}
   383→                    onChange={(e) => setSearch(e.target.value)}
   384→                    placeholder="Search players globally by Name or Tag (e.g. Cobra, #IND-8821)..."
   385→                    className="w-full bg-slate-950 border border-slate-800 rounded-xl pl-9 pr-3 py-2 text-xs text-white placeholder:text-slate-600 focus:outline-none focus:border-violet-500/50"
   386→                  />
   387→                </div>
   388→                <select
   389→                  value={countryFilter}
   390→                  onChange={(e) => setCountryFilter(e.target.value)}
   391→                  className="bg-slate-950 border border-slate-800 rounded-xl px-3 py-2 text-xs text-white font-mono focus:outline-none focus:border-violet-500/50"
   392→                >
   393→                  {SOCIAL_COUNTRY_FILTER.map((c) => (
   394→                    <option key={c.code} value={c.code}>
   395→                      {c.flag} {c.name}
   396→                    </option>
   397→                  ))}
   398→                </select>
   399→              </div>
   400→
   401→              <div className="rounded-2xl border border-slate-800/60 bg-slate-950/80 overflow-hidden">
   402→                <ol className="divide-y divide-slate-900 max-h-[55vh] overflow-y-auto va-scroll">
   403→                  {filteredGlobalPlayers.length === 0 ? (
   404→                    <li className="p-6 text-center text-xs text-slate-500">No players match your search.</li>
   405→                  ) : (
   406→                    filteredGlobalPlayers.map((p) => {
   407→                      const connected = friends.some((f) => f.userTag === p.userTag);
   408→                      return (
   409→                        <li key={p.userTag} className="px-4 py-3 text-sm flex items-center justify-between gap-3 hover:bg-slate-900/40 transition-colors">
   410→                          <div className="flex items-center gap-3 min-w-0">
   411→                            <div className="w-8 h-8 rounded-lg flex items-center justify-center text-base shrink-0" style={{ background: `${p.skinColor}20`, border: `1px solid ${p.skinColor}40` }} aria-hidden>
   412→                              {countryFlag(p.country)}
   413→                            </div>
   414→                            <div className="min-w-0">
   415→                              <div className="font-bold text-white truncate flex items-center gap-1.5">
   416→                                {p.name}
   417→                                <span className="text-[10px] font-mono text-slate-500">#{p.userTag}</span>
   418→                              </div>
   419→                              <div className="text-[10px] font-mono text-slate-400">
   420→                                🪙 {(p.chips / 1000).toFixed(0)}k · Lvl {p.level} · {STATUS_LABELS[p.status]}
   421→                              </div>
   422→                            </div>
   423→                          </div>
   424→                          {connected ? (
   425→                            <span className="inline-flex items-center gap-1 text-[10px] font-bold text-emerald-400 bg-emerald-500/10 border border-emerald-500/30 px-2 py-1 rounded-full">
   426→                              <Check className="w-3 h-3" /> Connected
   427→                            </span>
   428→                          ) : (
   429→                            <button
   430→                              type="button"
   431→                              onClick={() => {
   432→                                setFriends((prev) => [...prev, {
   433→                                  id: `f-${Date.now()}`,
   434→                                  name: p.name,
   435→                                  userTag: p.userTag,
   436→                                  status: p.status,
   437→                                  level: p.level,
   438→                                  skinColor: p.skinColor,
   439→                                  giftSent: false,
   440→                                  giftReceived: false,
   441→                                }]);
   442→                                notify(`Connected with ${p.name}! 🤝`, 'success', onToast);
   443→                              }}
   444→                              className="px-3 py-1.5 rounded-lg text-[10px] font-bold bg-violet-600/20 border border-violet-500/40 text-violet-300 hover:bg-violet-600 hover:text-white transition flex items-center gap-1"
   445→                            >
   446→                              <UserPlus className="w-3 h-3" /> Connect
   447→                            </button>
   448→                          )}
   449→                        </li>
   450→                      );
   451→                    })
   452→                  )}
   453→                  {/* Custom tag not in presets */}
   454→                  {search.trim() && filteredGlobalPlayers.length === 0 && (
   455→                    <li className="p-4 text-center text-xs text-slate-400">
   456→                      <p className="font-bold text-white">Custom Player Tag: &quot;{search}&quot;</p>
   457→                      <p className="mt-1">Found on Global Server. Connect &amp; send invite request.</p>
   458→                      <button
   459→                        type="button"
   460→                        onClick={() => {
   461→                          setFriends((prev) => [...prev, {
   462→                            id: `f-${Date.now()}`,
   463→                            name: search.trim(),
   464→                            userTag: search.trim().toUpperCase(),
   465→                            status: 'offline',
   466→                            level: 1,
   467→                            skinColor: '#22d3ee',
   468→                            giftSent: false,
   469→                            giftReceived: false,
   470→                          }]);
   471→                          notify(`Connected with ${search}! 🤝`, 'success', onToast);
   472→                        }}
   473→                        className="mt-3 px-3 py-1.5 rounded-lg text-[10px] font-bold bg-violet-600 hover:bg-violet-500 text-white transition inline-flex items-center gap-1"
   474→                      >
   475→                        <UserPlus className="w-3 h-3" /> Connect
   476→                      </button>
