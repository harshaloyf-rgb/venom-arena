   144→// ----------------------------------------------------------------------------
   145→// Spatial hash grid — slimmed-down client port of the server's grid.
   146→// Items are bucketed into square cells; queries return a deduplicated Map.
   147→// ----------------------------------------------------------------------------
   148→
   149→class SpatialHashGrid {
   150→  private readonly cellSize: number;
   151→  private readonly cells: Map<string, Map<string, GridItem>> = new Map();
   152→
   153→  constructor(cellSize = 120) {
   154→    this.cellSize = cellSize;
   155→  }
   156→
   157→  private key(cx: number, cy: number): string {
   158→    return cx + ':' + cy;
   159→  }
   160→
   161→  clear(): void {
   162→    this.cells.clear();
   163→  }
   164→
   165→  insert(item: GridItem): void {
   166→    const minCx = Math.floor((item.x - item.radius) / this.cellSize);
   167→    const maxCx = Math.floor((item.x + item.radius) / this.cellSize);
   168→    const minCy = Math.floor((item.y - item.radius) / this.cellSize);
   169→    const maxCy = Math.floor((item.y + item.radius) / this.cellSize);
   170→    for (let cx = minCx; cx <= maxCx; cx++) {
   171→      for (let cy = minCy; cy <= maxCy; cy++) {
   172→        const k = this.key(cx, cy);
   173→        let bucket = this.cells.get(k);
   174→        if (!bucket) {
   175→          bucket = new Map();
   176→          this.cells.set(k, bucket);
   177→        }
   178→        bucket.set(item.id, item);
   179→      }
   180→    }
   181→  }
   182→
   183→  queryRadius(x: number, y: number, r: number): Map<string, GridItem> {
   184→    const out = new Map<string, GridItem>();
   185→    const minCx = Math.floor((x - r) / this.cellSize);
   186→    const maxCx = Math.floor((x + r) / this.cellSize);
   187→    const minCy = Math.floor((y - r) / this.cellSize);
   188→    const maxCy = Math.floor((y + r) / this.cellSize);
   189→    for (let cx = minCx; cx <= maxCx; cx++) {
   190→      for (let cy = minCy; cy <= maxCy; cy++) {
   191→        const bucket = this.cells.get(this.key(cx, cy));
   192→        if (!bucket) continue;
   193→        for (const [id, item] of bucket) {
   194→          if (!out.has(id)) out.set(id, item);
   195→        }
   196→      }
   197→    }
   198→    return out;
   199→  }
   200→}

   256→function angularDelta(current: number, desired: number): number {
   257→  let diff = desired - current;
   258→  while (diff > Math.PI) diff -= 2 * Math.PI;
   259→  while (diff < -Math.PI) diff += 2 * Math.PI;
   260→  return diff;
   261→}
   262→
   263→function turnToward(current: number, desired: number, maxStep: number): number {
   264→  const diff = angularDelta(current, desired);
   265→  if (Math.abs(diff) <= maxStep) return desired;
   266→  return current + Math.sign(diff) * maxStep;
   267→}
   268→
   269→function dist(ax: number, ay: number, bx: number, by: number): number {
   270→  return Math.hypot(ax - bx, ay - by);
   271→}
   272→
   273→/** Random point within a circle of maxR around center. */
   274→function randomPointInCircle(cx: number, cy: number, maxR: number): Vec2 {
   275→  const r = Math.sqrt(Math.random()) * maxR;
   276→  const theta = Math.random() * Math.PI * 2;
   277→  return {
   278→    x: cx + Math.cos(theta) * r,
   279→    y: cy + Math.sin(theta) * r,
   280→  };
   281→}
   282→
   283→function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   284→  const pts: Vec2[] = [];
   285→  for (let i = 0; i < length; i++) {
   286→    pts.push({
   287→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   288→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   289→    });
   290→  }
   291→  return pts;
   292→}

   298→/** Pick a weighted random FoodOrbConfig. */
   299→function weightedRandomOrb(): FoodOrbConfig {
   300→  let r = Math.random();
   301→  for (let i = 0; i < ALL_FOOD_ORBS.length; i++) {
   302→    r -= FOOD_ORB_WEIGHTS[i];
   303→    if (r <= 0) return ALL_FOOD_ORBS[i];
   304→  }
   305→  return ALL_FOOD_ORBS[0]; // fallback to small
   306→}
   307→
   308→/** Create a Food object at a random position around the player. */
   309→function createFoodOrb(idPrefix: string, idCounter: { value: number }, cx: number, cy: number, farSpawn: boolean): Food {
   310→  const orb = weightedRandomOrb();
   311→  const radius = farSpawn ? FOOD_SPAWN_RADIUS_FAR : FOOD_SPAWN_RADIUS_NEAR;
   312→  const pos = randomPointInCircle(cx, cy, radius);
   313→  return {
   314→    id: `${idPrefix}-food-${idCounter.value++}`,
   315→    x: pos.x,
   316→    y: pos.y,
   317→    size: orb.radius,
   318→    value: orb.value,
   319→    orbSize: orb.size,
   320→    color: orb.color,
   321→    glowColor: orb.glowColor,
   322→  };
   323→}
   324→
   325→/** Create a specific Food object at an exact position (for death drops). */
   326→function createFoodOrbAt(idPrefix: string, idCounter: { value: number }, x: number, y: number, orb: FoodOrbConfig): Food {
   327→  return {
   328→    id: `${idPrefix}-drop-${idCounter.value++}`,
   329→    x,
   330→    y,
   331→    size: orb.radius,
   332→    value: orb.value,
   333→    orbSize: orb.size,
   334→    color: orb.color,

   336→  };

   339→// ----------------------------------------------------------------------------
   340→// Death food drop — greedily split score into Large(5)/Medium(3)/Small(1)
   341→// ----------------------------------------------------------------------------
   342→
   343→interface DeathDropResult {
   344→  foods: Food[];
   345→  /** IDs of dead snakes that should drop food. */
   346→}
   347→
   348→function computeDeathFoodDrop(
   349→  totalScore: number,
   350→  deathX: number,
   351→  deathY: number,
   352→  idPrefix: string,
   353→  idCounter: { value: number },
   354→): Food[] {
   355→  const result: Food[] = [];
   356→  let remaining = Math.max(0, totalScore);
   357→
   358→  // Large orbs (value 5)
   359→  const largeOrb = ALL_FOOD_ORBS[2]; // FOOD_ORB_LARGE
   360→  const largeCount = Math.floor(remaining / largeOrb.value);
   361→  for (let i = 0; i < largeCount; i++) {
   362→    const offset = randomPointInCircle(deathX, deathY, 30 + largeCount * 0.5);
   363→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, largeOrb));
   364→  }
   365→  remaining -= largeCount * largeOrb.value;
   366→
   367→  // Medium orbs (value 3)
   368→  const mediumOrb = ALL_FOOD_ORBS[1]; // FOOD_ORB_MEDIUM
   369→  const mediumCount = Math.floor(remaining / mediumOrb.value);
   370→  for (let i = 0; i < mediumCount; i++) {
   371→    const offset = randomPointInCircle(deathX, deathY, 30 + mediumCount * 0.5);
   372→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, mediumOrb));
   373→  }
   374→  remaining -= mediumCount * mediumOrb.value;
   375→
   376→  // Small orbs (value 1) for the remainder
   377→  const smallOrb = ALL_FOOD_ORBS[0]; // FOOD_ORB_SMALL
   378→  for (let i = 0; i < remaining; i++) {
   379→    const offset = randomPointInCircle(deathX, deathY, 30 + remaining * 0.5);
   380→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, smallOrb));
   381→  }
   382→
   383→  return result;
   384→}

   586→  private handleResize(): void {
   587→    const canvas = this.canvas;
   588→    const dpr = Math.min(window.devicePixelRatio || 1, 2);
   589→    const rect = canvas.getBoundingClientRect();
   590→    const w = Math.max(1, Math.floor(rect.width));
   591→    const h = Math.max(1, Math.floor(rect.height));
   592→    canvas.width = Math.floor(w * dpr);

   729→  private onTouchStart(e: TouchEvent): void {
   730→    if (this.joystick) return;
   731→    const t = this.findJoystickTouch(e.touches);
   732→    if (!t) return;
   733→    e.preventDefault();
   734→    const rect = this.canvas.getBoundingClientRect();
   735→    this.joystick = {
   736→      active: true,
   737→      pointerId: t.identifier,
   738→      originX: t.clientX - rect.left,
   739→      originY: t.clientY - rect.top,
   740→      curX: t.clientX - rect.left,
   741→      curY: t.clientY - rect.top,
   742→    };
   743→  }
   744→
   745→  private onTouchMove(e: TouchEvent): void {
   746→    const js = this.joystick;
   747→    if (!js) return;
   748→    let t: Touch | null = null;
   749→    for (let i = 0; i < e.touches.length; i++) {
   750→      if (e.touches[i].identifier === js.pointerId) {
   751→        t = e.touches[i];
   752→        break;
   753→      }
   754→    }
   755→    if (!t) return;
   756→    e.preventDefault();
   757→    const rect = this.canvas.getBoundingClientRect();
   758→    js.curX = t.clientX - rect.left;
   759→    js.curY = t.clientY - rect.top;
   760→    const dx = js.curX - js.originX;
   761→    const dy = js.curY - js.originY;
   762→    const d = Math.hypot(dx, dy);
   763→    const mag = Math.min(1, d / JOYSTICK_MAX_RADIUS_PX);
   764→    if (mag > JOYSTICK_DEADZONE) {
   765→      this.touchAngle = Math.atan2(dy, dx);
   766→      this.touchBoost = mag > JOYSTICK_BOOST_MAGNITUDE;
   767→    } else {
   768→      this.touchAngle = null;
   769→      this.touchBoost = false;
   770→    }
   771→  }
   772→
   773→  private onTouchEnd(e: TouchEvent): void {
   774→    const js = this.joystick;
   775→    if (!js) return;
   776→    let stillActive = false;
   777→    for (let i = 0; i < e.touches.length; i++) {
   778→      if (e.touches[i].identifier === js.pointerId) {
   779→        stillActive = true;
   780→        break;
   781→      }
   782→    }
   783→    if (!stillActive) {
   784→      this.joystick = null;
   785→      this.touchAngle = null;
   786→      this.touchBoost = false;
   787→    }
   788→  }
   789→
   790→  private findJoystickTouch(touches: TouchList): Touch | null {
   791→    const rect = this.canvas.getBoundingClientRect();
   792→    for (let i = 0; i < touches.length; i++) {
   793→      const t = touches[i];
   794→      const x = t.clientX - rect.left;
   795→      const y = t.clientY - rect.top;
   796→      if (x < rect.width / 2 && y > rect.height / 2) {
   797→        return t;
   798→      }
   799→    }
   800→    return null;
   801→  }

   904→  private frame = (now: number): void => {
   905→    if (this.stopped) return;
   906→    this.rafId = requestAnimationFrame(this.frame);
   907→    const dt = now - this.lastFrameTime;
   908→    this.lastFrameTime = now;
   909→
   910→    this.updateFps(now);
   911→    this.updateParticles(dt);
   912→
   913→    // Physics ticks at ~TICK_MS (30 Hz) cadence.
   914→    if (this.state === 'playing') {
   915→      this.accumulator += dt;
   916→      let safety = 4;
   917→      while (this.accumulator >= TICK_MS && safety > 0) {
   918→        this.tickPhysics(now);
   919→        this.accumulator -= TICK_MS;
   920→        safety--;
   921→      }
   922→      if (safety === 0) this.accumulator = 0;
   923→    }
   924→
   925→    this.render(now);
   926→    this.updateHUD();
   927→  };

   931→  // --------------------------------------------------------------------------
   932→  // Physics tick (one server-tick equivalent)
   933→  // --------------------------------------------------------------------------
   934→
   935→  private tickPhysics(now: number): void {
   936→    this.tick++;
   937→    const p = this.player;
   938→    if (!p || p.isDead) return;
   939→
   940→    // 1) Compute player desired angle + boost.
   941→    const { angle, boost } = this.computePlayerInput();
   942→    if (angle !== null) p.desiredAngle = angle;
   943→    p.wantsBoost = boost;
   944→    p.isBoosting = boost && p.points.length > BOOST_MIN_LENGTH + INITIAL_BODY_LENGTH - 20;
   945→
   946→    // Extraction progress (3-second hold).
   947→    if (p.isExtracting) {
   948→      p.extractionProgress += TICK_MS;
   949→      if (p.extractionProgress >= EXTRACT_DURATION_MS) {
   950→        this.finishExtract();
   951→        return;
   952→      }
   953→    }
   954→
   955→    // 2) Move player.
   956→    this.tickSnakeMovement(p, p.desiredAngle, p.wantsBoost);
   957→
   958→    // 3) Bots: think + move.
   959→    for (const bot of this.bots.values()) {
   960→      this.tickBot(bot, now);
   961→    }
   962→
   963→    // Process boost food drops (from tail shedding)
   964→    if (this.boostDropQueue.length > 0) {
   965→      for (const drop of this.boostDropQueue) {
   966→        this.foods.push({
   967→          id: `food-${this.arena.id}-${this.idCounterObj.value++}`,
   968→          x: drop.x,
   969→          y: drop.y,
   970→          size: 3,
   971→          value: 1,
   972→          isStarChip: false,
   973→          color: '#34d399',
   974→          glowColor: '#10b981',
   975→          orbSize: 'small',
   976→        });
   977→      }
   978→      this.boostDropQueue.length = 0;
   979→    }
   980→
   981→    // 4) Build spatial grid for collision + food queries.
   982→    this.grid.clear();
   983→    this.insertSnakeIntoGrid(p);
   984→    for (const bot of this.bots.values()) {
   985→      if (!bot.isDead) this.insertSnakeIntoGrid(bot);
   986→    }
   987→    for (const food of this.foods) {
   988→      this.grid.insert({
   989→        id: food.id,
   990→        kind: 'food',
   991→        x: food.x,
   992→        y: food.y,
   993→        radius: food.size,
   994→        value: food.value,
   995→        foodRef: food,
   996→      });
   997→    }
   998→
   999→    // 5) Eat food (player + bots).
  1000→    this.eatFood();
  1001→
  1002→    // 6) Collision detection (body collision + head-on collision). NO wall death.
  1003→    const deaths = this.detectCollisions(now);
  1004→
  1005→    // 7) Apply deaths + drop food.
  1006→    let playerDied = false;
  1007→    const newDropFoods: Food[] = [];
  1008→    for (const d of deaths) {
  1009→      if (d.deadId === p.id) {
  1010→        playerDied = true;
  1011→        // Drop player's food
  1012→        const playerTotal = INITIAL_SPAWN_SCORE + p.score;
  1013→        newDropFoods.push(...computeDeathFoodDrop(playerTotal, p.points[0].x, p.points[0].y, this.arena.id, this.idCounterObj));
  1014→        continue;
  1015→      }
  1016→      const bot = this.bots.get(d.deadId);
  1017→      if (bot) {
  1018→        // Drop bot's food
  1019→        const botTotal = INITIAL_SPAWN_SCORE + bot.score;
  1020→        newDropFoods.push(...computeDeathFoodDrop(botTotal, bot.points[0].x, bot.points[0].y, this.arena.id, this.idCounterObj));
  1021→        bot.isDead = true;
  1022→        // Credit kill to the killer if it's the player.
  1023→        if (d.killerId === p.id) {
  1024→          p.kills++;
  1025→        }
  1026→      }
  1027→    }
  1028→
  1029→    // Add dropped food to the world.
  1030→    if (newDropFoods.length > 0) {
  1031→      this.foods.push(...newDropFoods);
  1032→    }
  1033→
  1034→    // 8) Remove dead bots and respawn to maintain count.
  1035→    if (deaths.length > 0) {
  1036→      const toRemove: string[] = [];
  1037→      for (const [id, bot] of this.bots) {
  1038→        if (bot.isDead) toRemove.push(id);
  1039→      }
  1040→      for (const id of toRemove) {
  1041→        this.bots.delete(id);
  1042→      }
  1043→    }
  1044→    // Respawn bots near the player to maintain count.
  1045→    const pHead = p.points[0];
  1046→    while (this.bots.size < this.arena.botsCount) {
  1047→      const bot = this.spawnBot(pHead.x, pHead.y);
  1048→      // Ensure bots spawn in a safe area (far from player AND other bots)
  1049→      let safe = false;
  1050→      for (let attempt = 0; attempt < 20; attempt++) {
  1051→        const spawnPt = bot.points[0];
  1052→        const distFromPlayer = dist(spawnPt.x, spawnPt.y, pHead.x, pHead.y);
  1053→        if (distFromPlayer < 300) {
  1054→          // Re-spawn further from player
  1055→          const far = randomPointInCircle(pHead.x, pHead.y, BOT_SPAWN_RADIUS);
  1056→          bot.points = initialBody(far.x, far.y, bot.angle, INITIAL_BODY_LENGTH);
  1057→          continue;
  1058→        }
  1059→        // Also check distance from other bots
  1060→        let tooClose = false;
  1061→        for (const other of this.bots.values()) {
  1062→          if (other.isDead || other.points.length === 0) continue;
  1063→          if (dist(spawnPt.x, spawnPt.y, other.points[0].x, other.points[0].y) < 300) {
  1064→            tooClose = true;
  1065→            break;
  1066→          }
  1067→        }
  1068→        if (tooClose) {
  1069→          const far = randomPointInCircle(pHead.x, pHead.y, BOT_SPAWN_RADIUS);
  1070→          bot.points = initialBody(far.x, far.y, bot.angle, INITIAL_BODY_LENGTH);
  1071→          continue;
  1072→        }
  1073→        safe = true;
  1074→        break;
  1075→      }
  1076→      this.bots.set(bot.id, bot);
  1077→    }
  1078→
  1079→    // 9) Player death → death screen.
  1080→    if (playerDied) {
  1081→      p.isDead = true;
  1082→      this.handlePlayerDeath();
  1083→      return;
  1084→    }
  1085→
  1086→    // 10) Replenish food toward target (spawn around player).
  1087→    this.replenishFood();
  1088→
  1089→    // 11) Expire chat bubbles.
  1090→    this.expireChat(now);
  1091→
  1092→    // 12) Update camera target.
  1093→    this.updateCamera();
  1094→  }

  1096→  // --------------------------------------------------------------------------
  1097→  // Snake movement (server-authoritative formula)
  1098→  // --------------------------------------------------------------------------
  1099→
  1100→  private tickSnakeMovement(snake: SnakeBase, desiredAngle: number, wantsBoost: boolean): void {
  1101→    if (snake.points.length === 0 || snake.isDead) return;
  1102→
  1103→    // Turn rate: max(0.045, 0.15 - totalScore*0.0006).
  1104→    const totalScore = INITIAL_SPAWN_SCORE + snake.score;
  1105→    const turnRate = Math.max(TURN_MIN, TURN_BASE - totalScore * TURN_SCORE_FACTOR);
  1106→    snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
  1107→
  1108→    // Speed.
  1109→    let speed = BASE_SPEED;
  1110→    if (snake.isExtracting) {
  1111→      speed = EXTRACT_GLIDE_SPEED;
  1112→    } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
  1113→      speed = BOOST_SPEED;
  1114→      snake.isBoosting = true;
  1115→      snake.boostFrameCounter++;
  1116→      if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
  1117→        snake.boostFrameCounter = 0;
  1118→        if (snake.points.length > BOOST_MIN_LENGTH) {
  1119→          // Record tail position BEFORE popping (for food drop)
  1120→          const tail = snake.points[snake.points.length - 1];
  1121→          this.boostDropQueue.push({
  1122→            x: tail.x,
  1123→            y: tail.y,
  1124→          });
  1125→          snake.points.pop();
  1126→          // Reduce food-score (score can go below 0 but body is clamped)
  1127→          snake.score = Math.max(0, snake.score - 1);
  1128→        }
  1129→      }
  1130→    } else {
  1131→      snake.isBoosting = false;
  1132→    }
  1133→
  1134→    // Move head.
  1135→    const head = snake.points[0];
  1136→    const nx = head.x + Math.cos(snake.angle) * speed;
  1137→    const ny = head.y + Math.sin(snake.angle) * speed;
  1138→    snake.points.unshift({ x: nx, y: ny });
  1139→
  1140→    // Grow / shrink body to target length based on SCORE.
  1141→    // targetLen = INITIAL_BODY_LENGTH + foodScore, clamped to [BOOST_MIN_LENGTH+1, MAX_BODY_LENGTH]
  1142→    const targetLen = Math.max(
  1143→      BOOST_MIN_LENGTH + 1,
  1144→      Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + snake.score),
  1145→    );
  1146→    while (snake.points.length > targetLen) snake.points.pop();
  1147→
  1148→    // Size: 8 + sqrt(totalScore) * 0.4.
  1149→    snake.size = SIZE_BASE + Math.sqrt(INITIAL_SPAWN_SCORE + snake.score) * SIZE_SCORE_FACTOR;
  1150→  }

  1152→  // --------------------------------------------------------------------------
  1153→  // Bot AI — no self-destruct, seek food, predictive evasion
  1154→  // --------------------------------------------------------------------------
  1155→
  1156→  private tickBot(bot: BotSession, now: number): void {
  1157→    if (bot.points.length === 0 || bot.isDead) return;
  1158→
  1159→    if (now >= bot.nextThinkAt) {
  1160→      bot.nextThinkAt = now + BOT_THINK_INTERVAL_MS + Math.floor(Math.random() * BOT_THINK_JITTER_MS);
  1161→      const head = bot.points[0];
  1162→
  1163→      // --- Find nearest food ---
  1164→      let bestFood: GridItem | null = null;
  1165→      let bestFoodDist = Infinity;
  1166→      const foodQuery = this.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
  1167→      for (const item of foodQuery.values()) {
  1168→        if (item.kind !== 'food') continue;
  1169→        if ((item.value ?? 0) <= 0) continue;
  1170→        const d = dist(head.x, head.y, item.x, item.y);
  1171→        if (d < bestFoodDist) {
  1172→          bestFoodDist = d;
  1173→          bestFood = item;
  1174→        }
  1175→      }
  1176→
  1177→      // --- Find nearest threat (foreign body segment) ---
  1178→      let threatX = 0;
  1179→      let threatY = 0;
  1180→      let threatDist = Infinity;
  1181→      const threatQuery = this.grid.queryRadius(head.x, head.y, BOT_THREAT_SCAN_RADIUS);
  1182→      for (const item of threatQuery.values()) {
  1183→        if (item.kind !== 'segment') continue;
  1184→        if (item.snakeId === bot.id) continue;
  1185→        if (item.segIdx === 0) continue;
  1186→        const d = dist(head.x, head.y, item.x, item.y);
  1187→        if (d < threatDist) {
  1188→          threatDist = d;
  1189→          threatX = item.x;
  1190→          threatY = item.y;
  1191→        }
  1192→      }
  1193→
  1194→      let desired: number;
  1195→      const shouldFlee =
  1196→        threatDist < 140 &&
  1197→        (bot.personality === 'coward' ||
  1198→          bot.personality === 'extractor' ||
  1199→          (bot.personality === 'opportunist' && threatDist < 90));
  1200→
  1201→      // --- Predictive evasion against the human player ---
  1202→      const p = this.player;
  1203→      if (p && !p.isDead && p.points.length > 0 && now >= p.spawnProtectedUntil) {
  1204→        const playerHead = p.points[0];
  1205→        const playerDist = dist(head.x, head.y, playerHead.x, playerHead.y);
  1206→        if (playerDist < BOT_EVADE_RADIUS * 2) {
  1207→          // Predict where the player will be in N ticks.
  1208→          const predictedX = playerHead.x + Math.cos(p.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1209→          const predictedY = playerHead.y + Math.sin(p.angle) * BOT_PREDICT_AHEAD_TICKS;
  1210→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1211→
  1212→          // If the bot is on a collision course with the predicted position, evade.
  1213→          if (distToPredicted < (bot.size + p.size) * 3) {
  1214→            // Evade perpendicular to the player's heading.
  1215→            const evadeAngle = p.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1216→            desired = evadeAngle;
  1217→            bot.desiredAngle = desired;
  1218→            const wantsBoost = bot.personality === 'hunter' && bot.score > 10;
  1219→            this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1220→            return;
  1221→          }
  1222→        }
  1223→      }
  1224→
  1225→      if (shouldFlee) {
  1226→        desired = Math.atan2(head.y - threatY, head.x - threatX);
  1227→      } else if (bestFood) {
  1228→        desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
  1229→      } else {
  1230→        desired = bot.angle + (Math.random() - 0.5) * 0.4;
  1231→      }
  1232→
  1233→      // NO edge avoidance — infinite map, no boundaries.
  1234→
  1235→      bot.desiredAngle = desired;
  1236→    }
  1237→
  1238→    // Hunter personality occasionally boosts.
  1239→    const wantsBoost = bot.personality === 'hunter' && bot.score > 5 && Math.random() < 0.05;
  1240→    this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1241→  }

  1296→  // Collision detection (body + head-on) — NO wall death
  1297→  // --------------------------------------------------------------------------
  1298→
  1299→  private detectCollisions(now: number): { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] {
  1300→    const deaths: { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] = [];
  1301→    const seenDead = new Set<string>();
  1302→    const all: SnakeBase[] = [];
  1303→    if (this.player && !this.player.isDead) all.push(this.player);
  1304→    for (const bot of this.bots.values()) {
  1305→      if (!bot.isDead) all.push(bot);
  1306→    }
  1307→
  1308→    // --- Head-to-head collision (checked first, takes priority) ---
  1309→    const headOnChecked = new Set<string>();
  1310→    for (let i = 0; i < all.length; i++) {
  1311→      for (let j = i + 1; j < all.length; j++) {
  1312→        const a = all[i];
  1313→        const b = all[j];
  1314→        if (a.isDead || b.isDead) continue;
  1315→        if (now < a.spawnProtectedUntil || now < b.spawnProtectedUntil) continue;
  1316→        const pairKey = a.id < b.id ? `${a.id}:${b.id}` : `${b.id}:${a.id}`;
  1317→        if (headOnChecked.has(pairKey)) continue;
  1318→        headOnChecked.add(pairKey);
  1319→
  1320→        const ha = a.points[0];
  1321→        const hb = b.points[0];
  1322→        const d = dist(ha.x, ha.y, hb.x, hb.y);
  1323→        const hitDist = (a.size + b.size) * HEAD_ON_HIT_FACTOR;
  1324→        if (d >= hitDist) continue;
  1325→
  1326→        // Head-on collision! Determine winner based on rules.
  1327→        const aBoosting = a.isBoosting;
  1328→        const bBoosting = b.isBoosting;
  1329→        const aTotalScore = INITIAL_SPAWN_SCORE + a.score;
  1330→        const bTotalScore = INITIAL_SPAWN_SCORE + b.score;
  1331→
  1332→        let aDies = false;
  1333→        let bDies = false;
  1334→
  1335→        if (aTotalScore === bTotalScore) {
  1336→          // Tie — both die
  1337→          aDies = true;
  1338→          bDies = true;
  1339→        } else if (aBoosting && !bBoosting) {
  1340→          // A is boosting, B is not → A wins (boost advantage)
  1341→          bDies = true;
  1342→        } else if (bBoosting && !aBoosting) {
  1343→          // B is boosting, A is not → B wins (boost advantage)
  1344→          aDies = true;
  1345→        } else {
  1346→          // Both boosting or neither boosting → higher score wins
  1347→          if (aTotalScore > bTotalScore) {
  1348→            bDies = true;
  1349→          } else {
  1350→            aDies = true;
  1351→          }
  1352→        }
  1353→
  1354→        if (aDies && !seenDead.has(a.id)) {
  1355→          deaths.push({ deadId: a.id, killerId: b.id, cause: 'headon' });
  1356→          seenDead.add(a.id);
  1357→        }
  1358→        if (bDies && !seenDead.has(b.id)) {
  1359→          deaths.push({ deadId: b.id, killerId: a.id, cause: 'headon' });
  1360→          seenDead.add(b.id);
  1361→        }
  1362→      }
  1363→    }
  1364→
  1365→    // --- Body collision: head vs foreign non-head segment ---
  1366→    for (const snake of all) {
  1367→      if (snake.isDead) continue;
  1368→      if (snake.points.length === 0) continue;
  1369→      if (now < snake.spawnProtectedUntil) continue;
  1370→      if (seenDead.has(snake.id)) continue;
  1371→      const head = snake.points[0];
  1372→
  1373→      // NO wall collision — infinite map.
  1374→
  1375→      const queryR = snake.size + 30;
  1376→      const nearby = this.grid.queryRadius(head.x, head.y, queryR);
  1377→      for (const item of nearby.values()) {
  1378→        if (item.kind !== 'segment') continue;
  1379→        // Own-collision immunity — skip own segments.
  1380→        if (item.snakeId === snake.id) continue;
  1381→        // Skip head segments (handled by head-on collision above).
  1382→        if (item.segIdx === 0) continue;
  1383→        // Neck protection: skip first N segments behind the head (close-call safety)
  1384→        if (item.segIdx <= 5) continue;
  1385→        const d = dist(head.x, head.y, item.x, item.y);
  1386→        if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
  1387→          deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
  1388→          seenDead.add(snake.id);
  1389→          break;
  1390→        }
  1391→      }
  1392→    }
  1393→    return deaths;
  1394→  }

  1616→
  1617→  private drawFoodOrbs(ctx: CanvasRenderingContext2D, rc: FrameRenderCtx): void {
  1618→    const vis = computeVisibleRect(rc);
  1619→    const zoom = rc.zoom;
  1620→    const lowQ = rc.lowQuality;
  1621→
  1622→    // Batch by orb size for efficient rendering.
  1623→    // Small: tiny glowing dot, green
  1624→    // Medium: medium circle, blue
  1625→    // Large: larger circle, pink with brighter glow
  1626→    const smalls: Food[] = [];
  1627→    const mediums: Food[] = [];
  1628→    const larges: Food[] = [];
  1629→
  1630→    for (let i = 0; i < this.foods.length; i++) {
  1631→      const f = this.foods[i];
  1632→      if (f.value <= 0) continue;
  1633→      if (f.x < vis.left || f.x > vis.right || f.y < vis.top || f.y > vis.bottom) continue;
  1634→
  1635→      if (f.orbSize === 'small') smalls.push(f);
  1636→      else if (f.orbSize === 'medium') mediums.push(f);
  1637→      else larges.push(f);
  1638→    }
  1639→
  1640→    // --- Small orbs: batched simple circles ---
  1641→    if (smalls.length > 0) {
