   286→// Food orb rendering — three distinct sizes
   287→// ---------------------------------------------------------------------------
   288→
   289→/** Orb type configuration for the three visual tiers. */
   290→interface OrbConfig {
   291→  radius: number;
   292→  color: string;
   293→  glowColor: string;
   294→  shadowBlur: number;
   295→  label: 'small' | 'medium' | 'large';
   296→}
   297→
   298→const ORB_CONFIGS: Record<number, OrbConfig> = {
   299→  1: { radius: 3, color: '#34d399', glowColor: '#10b981', shadowBlur: 6, label: 'small' },
   300→  3: { radius: 5, color: '#38bdf8', glowColor: '#0ea5e9', shadowBlur: 10, label: 'medium' },
   301→  5: { radius: 8, color: '#f472b6', glowColor: '#ec4899', shadowBlur: 16, label: 'large' },
   302→};
   303→
   304→/** Default config used when orbSize doesn't match known values. */
   305→const DEFAULT_ORB: OrbConfig = ORB_CONFIGS[1];
   306→
   307→/**
   308→ * Draws a single food orb with glow effects. The orbSize determines which
   309→ * visual tier to use. Falls back to small orb for unknown sizes.
   310→ */
   311→export function drawFoodOrb(
   312→  ctx: CanvasRenderingContext2D,
   313→  x: number,
   314→  y: number,
   315→  orbSize: number,
   316→  value: number,
   317→  color: string,
   318→  glowColor: string,
   319→  now: number,
   320→  lowQuality: boolean,
   321→): void {
   322→  // Select config based on value; use provided color/glowColor if different
   323→  const config = ORB_CONFIGS[value] ?? DEFAULT_ORB;
   324→  const useColor = color || config.color;
   325→  const useGlow = glowColor || config.glowColor;
   326→  const r = Math.max(2, orbSize > 0 ? orbSize : config.radius);
   327→
   328→  ctx.save();
   329→
   330→  // Large orbs get a subtle pulse animation
   331→  let animR = r;
   332→  if (value === 5 && !lowQuality) {
   333→    const pulse = Math.sin(now * 0.004) * 1.5;
   334→    animR = r + pulse;
   335→  }
   336→
   337→  // Glow effect (skip in low quality)
   338→  if (!lowQuality) {
   339→    ctx.shadowColor = useGlow;
   340→    ctx.shadowBlur = config.shadowBlur;
   341→  }
   342→
   343→  // Radial gradient for a lit-from-within look
   344→  const grad = ctx.createRadialGradient(x, y, 0, x, y, animR);
   345→  grad.addColorStop(0, '#ffffff');
   346→  grad.addColorStop(0.3, useColor);
   347→  grad.addColorStop(1, useGlow);
   348→
   349→  ctx.beginPath();
   350→  ctx.arc(x, y, animR, 0, Math.PI * 2);
   351→  ctx.fillStyle = grad;
   352→  ctx.fill();
   353→
   354→  ctx.shadowBlur = 0;
   355→  ctx.restore();
   356→}

   358→// ---------------------------------------------------------------------------
   359→// Star collectible rendering — 5-pointed star shape
   360→// ---------------------------------------------------------------------------
   361→
   362→/**
   363→ * Draws a proper 5-pointed star path. Generic helper used by drawStarCollectible.
   364→ *
   365→ * @param ctx     Canvas context
   366→ * @param cx      Center X
   367→ * @param cy      Center Y
   368→ * @param outerR  Outer radius (tip of star)
   369→ * @param innerR  Inner radius (valley between points)
   370→ * @param points  Number of points (always 5 for stars)
   371→ * @param rotation Rotation in radians (for spin animation)
   372→ */
   373→export function drawStarShape(
   374→  ctx: CanvasRenderingContext2D,
   375→  cx: number,
   376→  cy: number,
   377→  outerR: number,
   378→  innerR: number,
   379→  points: number,
   380→  rotation: number,
   381→): void {
   382→  ctx.beginPath();
   383→  for (let i = 0; i < points * 2; i++) {
   384→    const r = i % 2 === 0 ? outerR : innerR;
   385→    const angle = (Math.PI / points) * i - Math.PI / 2 + rotation;
   386→    const px = cx + Math.cos(angle) * r;
   387→    const py = cy + Math.sin(angle) * r;
   388→    if (i === 0) ctx.moveTo(px, py);
   389→    else ctx.lineTo(px, py);
   390→  }
   391→  ctx.closePath();
   392→}
   393→
   394→/**
   395→ * Draws a 5-pointed star collectible with golden glow, subtle rotation,
   396→ * and pulsing glow effect. Stars are the rare high-value pickups.
   397→ */
   398→export function drawStarCollectible(
   399→  ctx: CanvasRenderingContext2D,
   400→  x: number,
   401→  y: number,
   402→  size: number,
   403→  tick: number,
   404→  lowQuality: boolean,
   405→): void {
   406→  const outerR = Math.max(4, size);
   407→  const innerR = Math.max(2, size * 0.4);
   408→
   409→  // Rotation: slow spin based on tick
   410→  const rotation = tick * 0.002;
   411→
   412→  // Pulsing glow: size oscillates ±1.5px over ~1.5s
   413→  const pulse = lowQuality ? 0 : Math.sin(tick * 0.004) * 1.5;
   414→  const animOuter = outerR + pulse;
   415→  const animInner = innerR + pulse * 0.4;
   416→
   417→  ctx.save();
   418→
   419→  // Glow
   420→  if (!lowQuality) {
   421→    ctx.shadowColor = '#f59e0b';
   422→    ctx.shadowBlur = 12 + Math.sin(tick * 0.003) * 4;
   423→  }
   424→
   425→  // Star shape
   426→  drawStarShape(ctx, x, y, animOuter, animInner, 5, rotation);
   427→
   428→  // Golden radial gradient fill
   429→  const grad = ctx.createRadialGradient(x, y, 0, x, y, animOuter);
   430→  grad.addColorStop(0, '#fef3c7'); // bright center
   431→  grad.addColorStop(0.4, '#fbbf24');
   432→  grad.addColorStop(1, '#f59e0b');
   433→  ctx.fillStyle = grad;
   434→  ctx.fill();
   435→
   436→  ctx.shadowBlur = 0;
   437→  ctx.restore();
   438→}

   440→// ---------------------------------------------------------------------------
   441→// Food (main entry point — batches regular orbs, draws stars individually)
   442→// ---------------------------------------------------------------------------
   443→
   444→/**
   445→ * Draws all food in two passes:
   446→ *  1. Regular food orbs — batched by orbSize tier into Path2D groups.
   447→ *     Each tier (small/medium/large) has its own glow gradient.
   448→ *  2. Star collectibles — drawn individually as 5-pointed gold stars with
   449→ *     rotation + pulsing glow.
   450→ */
   451→export function drawFood(rc: FrameRenderCtx, foods: FoodSnapshot[]): void {
   452→  const { ctx, lowQuality } = rc;
   453→  const vis = computeVisibleRect(rc);
   454→
   455→  // --- Regular food orbs: batch by value tier ---
   456→  const smallOrbs: FoodSnapshot[] = [];
   457→  const mediumOrbs: FoodSnapshot[] = [];
   458→  const largeOrbs: FoodSnapshot[] = [];
   459→  const starChips: FoodSnapshot[] = [];
   460→
   461→  for (let i = 0; i < foods.length; i++) {
   462→    const f = foods[i];
   463→    if (f.x < vis.left || f.x > vis.right || f.y < vis.top || f.y > vis.bottom) {
   464→      continue; // culled
   465→    }
   466→    if (f.isStarChip) {
   467→      starChips.push(f);
   468→      continue;
   469→    }
   470→    // Route to orb tier by value
   471→    if (f.value >= 5) largeOrbs.push(f);
   472→    else if (f.value >= 3) mediumOrbs.push(f);
   473→    else smallOrbs.push(f);
   474→  }
   475→
   476→  // Draw each orb tier
   477→  const drawOrbBatch = (orbs: FoodSnapshot[]) => {
   478→    for (let i = 0; i < orbs.length; i++) {
   479→      const f = orbs[i];
   480→      drawFoodOrb(ctx, f.x, f.y, f.orbSize ? f.size : f.size, f.value, f.color, f.glowColor ?? '', rc.now, lowQuality);
   481→    }
   482→  };
   483→
   484→  drawOrbBatch(smallOrbs);
   485→  drawOrbBatch(mediumOrbs);
   486→  drawOrbBatch(largeOrbs);
   487→
   488→  // --- Star collectibles: 5-pointed gold stars with rotation + glow ---
   489→  if (starChips.length > 0) {
   490→    ctx.save();
   491→    for (let i = 0; i < starChips.length; i++) {
   492→      const f = starChips[i];
   493→      drawStarCollectible(ctx, f.x, f.y, Math.max(6, f.size + 4), rc.now, lowQuality);
   494→    }
   495→    ctx.restore();
   496→  }
   497→}

   499→// ---------------------------------------------------------------------------
   500→// Snake rendering
   501→// ---------------------------------------------------------------------------
   502→
   503→/**
   504→ * Draws a single snake. The body is rendered as ONE thick stroked polyline
   505→ * (with an outline underlay). The head is drawn separately with eyes and
   506→ * (for the player's own snake, high-quality only) a glow halo.
   507→ *
   508→ * Player skin patterns (rainbow / neon / metallic / camo) are applied when
   509→ * `rc.playerSkin.pattern` is set and the snake is the local player.
   510→ *
   511→ * @param opacity  Optional global alpha override (0..1). Default = 1.
   512→ */
   513→export function drawSnake(rc: FrameRenderCtx, snake: SnakeSnapshot, opacity?: number): void {
   514→  const pts = snake.points;
   515→  if (!pts || pts.length === 0) return;
   516→
   517→  const vis = computeVisibleRect(rc);
   518→  if (!snakeIsVisible(pts, vis)) return;
   519→
   520→  const { ctx, zoom, lowQuality } = rc;
   521→  const isMe = snake.id === rc.myId;
   522→  const baseAlpha = opacity ?? 1;
   523→
   524→  // Downsample long snakes
   525→  const stride = pts.length > 60 ? 2 : 1;
   526→
   527→  // Body width in world units
   528→  const radius = Math.max(2, snake.size);
   529→  const width = radius * 2;
   530→
   531→  ctx.save();
   532→  ctx.globalAlpha = baseAlpha;
   533→
   534→  // --- Outline (underlay) ---
   535→  if (snake.secondaryColor) {
   536→    ctx.save();
   537→    ctx.lineCap = 'round';
   538→    ctx.lineJoin = 'round';
   539→    ctx.lineWidth = width + 4 / zoom;
   540→    ctx.strokeStyle = snake.secondaryColor;
   541→    ctx.globalAlpha = baseAlpha * 0.55;
   542→    ctx.beginPath();
   543→    ctx.moveTo(pts[0].x, pts[0].y);
   544→    for (let i = stride; i < pts.length; i += stride) {
   545→      ctx.lineTo(pts[i].x, pts[i].y);
   546→    }
   547→    ctx.stroke();
   548→    ctx.restore();
   549→  }
   550→
   551→  // --- Body ---
   552→  ctx.save();
   553→  ctx.globalAlpha = baseAlpha;
   554→  ctx.lineCap = 'round';
   555→  ctx.lineJoin = 'round';
   556→  ctx.lineWidth = width;
   557→
   558→  const pattern = isMe ? rc.playerSkin?.pattern : undefined;
   559→  if (pattern === 'metallic' && snake.secondaryColor) {
   560→    const g = getMetallicGradient(rc, radius, snake.color, snake.secondaryColor);
   561→    ctx.strokeStyle = g;
   562→    ctx.beginPath();
   563→    ctx.moveTo(pts[0].x, pts[0].y);
   564→    for (let i = stride; i < pts.length; i += stride) {
   565→      ctx.lineTo(pts[i].x, pts[i].y);
   566→    }
   567→    ctx.stroke();
   568→  } else if (pattern === 'rainbow' && !lowQuality) {
   569→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   570→    ctx.lineWidth = width;
   571→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   572→      const hue = (rc.now * 0.05 + i * 14) % 360;
   573→      ctx.strokeStyle = `hsl(${hue}, 90%, 55%)`;
   574→      ctx.beginPath();
   575→      ctx.moveTo(pts[i].x, pts[i].y);
   576→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   577→      for (let j = i + stride; j <= end; j += stride) {
   578→        ctx.lineTo(pts[j].x, pts[j].y);
   579→      }
   580→      ctx.stroke();
   581→    }
   582→  } else if (pattern === 'neon' && !lowQuality) {
   583→    const chunkSize = Math.max(4, Math.floor(pts.length / 10));
   584→    ctx.lineWidth = width;
   585→    let chunkIndex = 0;
   586→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   587→      const ratio = (Math.sin(rc.now * 0.009 - chunkIndex * 0.28) + 1) / 2;
   588→      ctx.strokeStyle = ratio > 0.5 ? '#06b6d4' : '#a855f7';
   589→      ctx.beginPath();
   590→      ctx.moveTo(pts[i].x, pts[i].y);
   591→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   592→      for (let j = i + stride; j <= end; j += stride) {
   593→        ctx.lineTo(pts[j].x, pts[j].y);
   594→      }
   595→      ctx.stroke();
   596→      chunkIndex += 1;
   597→    }
   598→  } else if (pattern === 'camo') {
   599→    const camoColors = ['#15803d', '#854d0e', '#78350f', '#166534'];
   600→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   601→    let chunkIndex = 0;
   602→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   603→      ctx.strokeStyle = camoColors[chunkIndex % camoColors.length];
   604→      ctx.beginPath();
   605→      ctx.moveTo(pts[i].x, pts[i].y);
   606→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   607→      for (let j = i + stride; j <= end; j += stride) {
   608→        ctx.lineTo(pts[j].x, pts[j].y);
   609→      }
   610→      ctx.stroke();
   611→      chunkIndex += 1;
   612→    }
   613→  } else {
   614→    // Default: solid snake.color
   615→    ctx.strokeStyle = snake.color;
   616→    ctx.beginPath();
   617→    ctx.moveTo(pts[0].x, pts[0].y);
   618→    for (let i = stride; i < pts.length; i += stride) {
   619→      ctx.lineTo(pts[i].x, pts[i].y);
   620→    }
   621→    ctx.stroke();
   622→  }
   623→  ctx.restore();
   624→
   625→  // --- Head ---
   626→  const head = pts[0];
   627→  ctx.save();
   628→  ctx.globalAlpha = baseAlpha;
   629→  if (isMe && !lowQuality) {
   630→    ctx.shadowColor = snake.color;
   631→    ctx.shadowBlur = 14;
   632→  }
   633→  ctx.beginPath();
   634→  ctx.arc(head.x, head.y, radius, 0, Math.PI * 2);
   635→  ctx.fillStyle = snake.color;
   636→  ctx.fill();
   637→  if (isMe && !lowQuality) ctx.shadowBlur = 0;
   638→
   639→  // Eyes
   640→  const eyeOffset = radius * 0.45;
   641→  const eyeR = Math.max(1.5, radius * 0.32);
   642→  const pupilR = Math.max(0.8, radius * 0.18);
   643→  const angle = snake.angle;
   644→  const perp = angle + Math.PI / 2;
   645→  const ex1 = head.x + Math.cos(angle) * eyeOffset * 0.4 + Math.cos(perp) * eyeOffset;
   646→  const ey1 = head.y + Math.sin(angle) * eyeOffset * 0.4 + Math.sin(perp) * eyeOffset;
   647→  const ex2 = head.x + Math.cos(angle) * eyeOffset * 0.4 - Math.cos(perp) * eyeOffset;
   648→  const ey2 = head.y + Math.sin(angle) * eyeOffset * 0.4 - Math.sin(perp) * eyeOffset;
   649→
   650→  ctx.fillStyle = '#ffffff';
   651→  ctx.beginPath();
   652→  ctx.arc(ex1, ey1, eyeR, 0, Math.PI * 2);
   653→  ctx.arc(ex2, ey2, eyeR, 0, Math.PI * 2);
   654→  ctx.fill();
   655→
   656→  ctx.fillStyle = '#0a0a0a';
   657→  ctx.beginPath();
   658→  ctx.arc(ex1 + Math.cos(angle) * pupilR, ey1 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   659→  ctx.arc(ex2 + Math.cos(angle) * pupilR, ey2 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   660→  ctx.fill();
   661→
   662→  // Spawn-protection ring
   663→  if (snake.spawnProtected) {
   664→    ctx.strokeStyle = 'rgba(255,255,255,0.6)';
   665→    ctx.lineWidth = 2 / zoom;
   666→    ctx.beginPath();
   667→    ctx.arc(head.x, head.y, radius + 4 / zoom, 0, Math.PI * 2);
   668→    ctx.stroke();
   669→  }
   670→
   671→  // Extraction ring
   672→  if (snake.isExtracting) {
   673→    ctx.strokeStyle = 'rgba(16, 185, 129, 0.95)';
   674→    ctx.lineWidth = 3 / zoom;
   675→    ctx.setLineDash([6 / zoom, 4 / zoom]);
   676→    ctx.beginPath();
   677→    ctx.arc(head.x, head.y, radius + 8 / zoom, 0, Math.PI * 2);
   678→    ctx.stroke();
   679→    ctx.setLineDash([]);
   680→  }
   681→  ctx.restore();
   682→
   683→  // --- Name label ---
   684→  if (snake.name) {
   685→    ctx.save();
   686→    ctx.globalAlpha = baseAlpha;
   687→    ctx.font = `${Math.max(10, 12 / zoom)}px monospace`;
   688→    ctx.textAlign = 'center';
   689→    ctx.textBaseline = 'bottom';
   690→    const labelY = head.y - radius - 6 / zoom;
   691→    ctx.fillStyle = isMe ? '#22c55e' : 'rgba(226, 232, 240, 0.85)';
   692→    ctx.fillText(snake.name, head.x, labelY);
   693→    if (snake.userTag) {
   694→      ctx.fillStyle = 'rgba(148, 163, 184, 0.7)';
   695→      ctx.font = `${Math.max(8, 9 / zoom)}px monospace`;
   696→      ctx.fillText(snake.userTag, head.x, labelY - 12 / zoom);
   697→    }
   698→    ctx.restore();
   699→  }
   700→
   701→  // --- Chat bubble ---
   702→  if (snake.chatMessage) {
   703→    ctx.save();
   704→    ctx.globalAlpha = baseAlpha;
   705→    drawChatBubble(rc, head.x, head.y - radius - 24 / zoom, snake.chatMessage);
   706→    ctx.restore();
   707→  }
   708→
   709→  ctx.restore(); // outer globalAlpha save
   710→}

   712→// ---------------------------------------------------------------------------
   713→// Snake opacity layering system
   714→// ---------------------------------------------------------------------------
   715→
   716→/** Proximity threshold for opacity fade (world-space pixels). */
   717→const LAYERING_PROXIMITY = 30;
   718→
   719→/**
   720→ * Draws a snake with opacity layering: when a smaller snake passes close to a
   721→ * larger snake, the larger snake fades to 75% opacity. This creates visual
   722→ * depth when snakes overlap spatially.
   723→ *
   724→ * Logic:
   725→ *  1. For each OTHER snake, check if its head is within ~30px of this snake's head
   726→ *  2. If a smaller snake (by size) is nearby, this snake fades to 0.75 opacity
   727→ *  3. Otherwise, render at full opacity
   728→ */
   729→export function drawSnakeWithLayering(
   730→  rc: FrameRenderCtx,
   731→  snake: SnakeSnapshot,
   732→  allSnakes: SnakeSnapshot[],
   733→): void {
   734→  const pts = snake.points;
   735→  if (!pts || pts.length === 0) return;
   736→
   737→  const myHead = pts[0];
   738→  let shouldFade = false;
   739→
   740→  for (let i = 0; i < allSnakes.length; i++) {
   741→    const other = allSnakes[i];
   742→    if (other.id === snake.id) continue;
   743→    const otherPts = other.points;
   744→    if (!otherPts || otherPts.length === 0) continue;
   745→
   746→    const otherHead = otherPts[0];
   747→    const dist = Math.hypot(otherHead.x - myHead.x, otherHead.y - myHead.y);
   748→
   749→    if (dist < LAYERING_PROXIMITY && snake.size > other.size) {
   750→      // A smaller snake is close by — fade this (larger) snake
   751→      shouldFade = true;
   752→      break;
   753→    }
   754→  }
   755→
   756→  drawSnake(rc, snake, shouldFade ? 0.75 : 1);
   757→}

   759→// ---------------------------------------------------------------------------
   760→// Chip label above snake head
   761→// ---------------------------------------------------------------------------
   762→
   763→/**
   764→ * Draws a chip count label above the snake's head. Uses Indian numbering
   765→ * (K, L, Cr). Only meaningful for real players (bots have 0 carriedChips).
   766→ *
   767→ * Rendered as a semi-transparent pill with white text, positioned centered
   768→ * above the head.
   769→ */
   770→export function drawChipLabel(
   771→  ctx: CanvasRenderingContext2D,
   772→  x: number,
   773→  y: number,
   774→  chips: number,
   775→  snakeSize: number,
   776→  zoom: number,
   777→): void {
   778→  if (chips <= 0) return; // bots have 0 chips, skip them
   779→
   780→  const label = formatChipDisplay(chips);
   781→  const fontSize = Math.max(9, 10 / zoom);
   782→  const offset = snakeSize + 12 / zoom;
   783→
   784→  ctx.save();
   785→  ctx.font = `bold ${fontSize}px monospace`;
   786→  ctx.textAlign = 'center';
   787→  ctx.textBaseline = 'bottom';
   788→
   789→  const metrics = ctx.measureText(label);
   790→  const padX = 5 / zoom;
   791→  const padY = 2 / zoom;
   792→  const tw = metrics.width + padX * 2;
   793→  const th = fontSize + padY * 2;
   794→  const rx = x - tw / 2;
   795→  const ry = y - offset - th;
   796→  const rr = 4 / zoom;
   797→
   798→  // Background pill
   799→  ctx.fillStyle = 'rgba(0, 0, 0, 0.65)';
   800→  ctx.beginPath();
   801→  ctx.moveTo(rx + rr, ry);
   802→  ctx.lineTo(rx + tw - rr, ry);
   803→  ctx.arcTo(rx + tw, ry, rx + tw, ry + rr, rr);
   804→  ctx.lineTo(rx + tw, ry + th - rr);
   805→  ctx.arcTo(rx + tw, ry + th, rx + tw - rr, ry + th, rr);
   806→  ctx.lineTo(rx + rr, ry + th);
   807→  ctx.arcTo(rx, ry + th, rx, ry + th - rr, rr);
   808→  ctx.lineTo(rx, ry + rr);
   809→  ctx.arcTo(rx, ry, rx + rr, ry, rr);
   810→  ctx.closePath();
   811→  ctx.fill();
   812→
   813→  // Gold accent border
   814→  ctx.strokeStyle = 'rgba(251, 191, 36, 0.7)';
   815→  ctx.lineWidth = 1 / zoom;
   816→  ctx.stroke();
   817→
   818→  // Text
   819→  ctx.fillStyle = '#ffffff';
   820→  ctx.textBaseline = 'middle';
   821→  ctx.fillText(label, x, ry + th / 2);
   822→
   823→  ctx.restore();
   824→}

   826→// ---------------------------------------------------------------------------
   827→// Chat bubble
   828→// ---------------------------------------------------------------------------
   829→
   830→/** Draws a small chat bubble above a snake head. */
   831→function drawChatBubble(
   832→  rc: FrameRenderCtx,
   833→  x: number,
   834→  y: number,
   835→  msg: string,
   836→): void {
   837→  const { ctx, zoom } = rc;
   838→  const text = msg.length > 80 ? msg.slice(0, 78) + '…' : msg;
   839→  ctx.save();
   840→  ctx.font = `${Math.max(10, 11 / zoom)}px monospace`;
   841→  const metrics = ctx.measureText(text);
   842→  const padX = 6 / zoom;
   843→  const padY = 3 / zoom;
   844→  const tw = metrics.width + padX * 2;
   845→  const th = 14 / zoom + padY * 2;
   846→  ctx.fillStyle = 'rgba(15, 23, 42, 0.92)';
   847→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.6)';
   848→  ctx.lineWidth = 1.5 / zoom;
   849→  const rx = x - tw / 2;
   850→  const ry = y - th;
   851→  const rr = 4 / zoom;
   852→  ctx.beginPath();
   853→  ctx.moveTo(rx + rr, ry);
   854→  ctx.lineTo(rx + tw - rr, ry);
   855→  ctx.arcTo(rx + tw, ry, rx + tw, ry + rr, rr);
   856→  ctx.lineTo(rx + tw, ry + th - rr);
   857→  ctx.arcTo(rx + tw, ry + th, rx + tw - rr, ry + th, rr);
   858→  ctx.lineTo(rx + rr, ry + th);
   859→  ctx.arcTo(rx, ry + th, rx, ry + th - rr, rr);
   860→  ctx.lineTo(rx, ry + rr);
   861→  ctx.arcTo(rx, ry, rx + rr, ry, rr);
   862→  ctx.closePath();
   863→  ctx.fill();
   864→  ctx.stroke();
   865→  ctx.fillStyle = '#f8fafc';
   866→  ctx.textAlign = 'center';
   867→  ctx.textBaseline = 'middle';
   868→  ctx.fillText(text, x, y - th / 2);
   869→  ctx.restore();
   870→}

   876→/** Draws the particle array (caller caps at MAX_PARTICLES = 200). */
   877→export function drawParticles(rc: FrameRenderCtx, particles: Particle[]): void {
   878→  if (particles.length === 0) return;
   879→  const { ctx, zoom } = rc;
   880→  ctx.save();
   881→  ctx.globalCompositeOperation = 'lighter';
   882→  for (let i = 0; i < particles.length; i++) {
   883→    const p = particles[i];
   884→    if (p.life <= 0) continue;
   885→    const alpha = Math.max(0, Math.min(1, p.life / p.maxLife));
   886→    ctx.globalAlpha = alpha;
   887→    ctx.fillStyle = p.color;
   888→    ctx.beginPath();
   889→    ctx.arc(p.x, p.y, p.size / zoom, 0, Math.PI * 2);
   890→    ctx.fill();
   891→  }
   892→  ctx.globalAlpha = 1;
   893→  ctx.globalCompositeOperation = 'source-over';
   894→  ctx.restore();
   895→}

   897→// ---------------------------------------------------------------------------
   898→// Minimap (drawn in screen-space, no camera transform)
   899→// ---------------------------------------------------------------------------
   900→
   901→export interface MinimapArgs {
   902→  ctx: CanvasRenderingContext2D;
   903→  /** Top-left of minimap in CSS pixels. */
   904→  x: number;
   905→  y: number;
   906→  /** Minimap size (square). */
   907→  size: number;
   908→  worldSize: number;
   909→  /** Current breathing arena radius (world units). */
   910→  arenaRadius: number;
   911→  snakes: SnakeSnapshot[];
   912→  myId: string;
   913→  /**
   914→   * World-space radius around the player the radar should cover. Snakes
   915→   * outside this range are NOT rendered on the small corner minimap.
   916→   * Defaults to WORLD_SIZE/2 (legacy full-radar behaviour) when omitted.
   917→   */
   918→  range?: number;
   919→}
   920→
   921→/**
   922→ * Draws the circular radar minimap. Player is rendered as an indigo dot at
   923→ * the centre (the radar is centered on the player's head). The arena
   924→ * boundary is a dashed rose circle. Bots render as rose dots, other real
   925→ * players as emerald dots — matches AUDIT-A radar visual language.
   926→ */
   927→export function drawMinimap(args: MinimapArgs): void {
   928→  const { ctx, x, y, size, worldSize, arenaRadius, snakes, myId } = args;
   929→  const cx = x + size / 2;
   930→  const cy = y + size / 2;
   931→  const r = size / 2;
   932→  const radarRange = args.range ?? WORLD_SIZE / 2;
   933→  const scale = r / radarRange;
   934→
   935→  ctx.save();
   936→  // Background circle (radar disc)
   937→  ctx.beginPath();
   938→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   939→  ctx.fillStyle = 'rgba(2, 6, 23, 0.85)';
   940→  ctx.fill();
   941→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.4)';
   942→  ctx.lineWidth = 1.5;
   943→  ctx.stroke();
   944→
   945→  // Clip to minimap circle.
   946→  ctx.beginPath();
   947→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
   948→  ctx.clip();
   949→
   950→  // Concentric rings.
   951→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
   952→  ctx.lineWidth = 1;
   953→  for (const inset of [2, 5, 8]) {
   954→    ctx.beginPath();
   955→    ctx.arc(cx, cy, r - inset, 0, Math.PI * 2);
   956→    ctx.stroke();
   957→  }
   958→
   959→  // Crosshairs.
   960→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.4)';
   961→  ctx.beginPath();
   962→  ctx.moveTo(cx, y);
   963→  ctx.lineTo(cx, y + size);
   964→  ctx.moveTo(x, cy);
   965→  ctx.lineTo(x + size, cy);
   966→  ctx.stroke();
   967→
   968→  // Arena boundary (dashed rose) — centered on world center.
   969→  const mySnake = snakes.find((s) => s.id === myId);
   970→  const myHead = mySnake?.points?.[0];
   971→  const px = myHead ? myHead.x : worldSize / 2;
   972→  const py = myHead ? myHead.y : worldSize / 2;
   973→  const worldCenterOffsetX = (worldSize / 2 - px) * scale;
   974→  const worldCenterOffsetY = (worldSize / 2 - py) * scale;
   975→  const arenaR = arenaRadius * scale;
   976→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.6)';
   977→  ctx.lineWidth = 1.2;
   978→  ctx.setLineDash([4, 4]);
   979→  ctx.beginPath();
   980→  ctx.arc(cx + worldCenterOffsetX, cy + worldCenterOffsetY, arenaR, 0, Math.PI * 2);
   981→  ctx.stroke();
   982→  ctx.setLineDash([]);
   983→
   984→  // Snakes (player centered → drawn at centre; others offset).
   985→  for (let i = 0; i < snakes.length; i++) {
   986→    const s = snakes[i];
   987→    if (!s.points || s.points.length === 0) continue;
   988→    const head = s.points[0];
   989→    const dx = (head.x - px) * scale;
   990→    const dy = (head.y - py) * scale;
   991→    const dist = Math.hypot(dx, dy);
   992→    if (dist > r) continue;
   993→    const mx = cx + dx;
   994→    const my = cy + dy;
   995→    ctx.beginPath();
   996→    if (s.id === myId) {
   997→      ctx.arc(mx, my, 3, 0, Math.PI * 2);
   998→      ctx.fillStyle = '#818cf8'; // indigo-400
   999→    } else if (s.isBot) {
  1000→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
  1001→      ctx.fillStyle = '#f43f5e'; // rose-500
  1002→    } else {
  1003→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
  1004→      ctx.fillStyle = '#34d399'; // emerald-400
  1005→    }
  1006→    ctx.fill();
  1007→  }
  1008→  ctx.restore();
  1009→}

  1011→// ---------------------------------------------------------------------------
  1012→// Full Map overlay — drawn in screen-space, no camera transform.
  1013→// Press `M` to toggle. Shows the ENTIRE arena with all snakes as dots.
  1014→// ---------------------------------------------------------------------------
  1015→
  1016→export interface FullMapArgs {
  1017→  ctx: CanvasRenderingContext2D;
  1018→  /** Canvas CSS-pixel width. */
  1019→  w: number;
  1020→  /** Canvas CSS-pixel height. */
  1021→  h: number;
  1022→  worldSize: number;
  1023→  /** Current breathing arena radius (world units). */
  1024→  arenaRadius: number;
  1025→  snakes: SnakeSnapshot[];
  1026→  myId: string;
  1027→}
  1028→
  1029→/**
  1030→ * Draws a full-screen overlay of the entire arena. The arena circle is
  1031→ * centered on screen and scaled to fit the smaller of (w, h) with padding.
  1032→ * All snakes are rendered as dots — player = indigo (larger), bots = rose,
  1033→ * other real humans = emerald. A "Press M to close" hint is drawn at the
  1034→ * bottom. The player's own dot pulses with a ring.
  1035→ */
  1036→export function drawFullMap(args: FullMapArgs): void {
  1037→  const { ctx, w, h, worldSize, arenaRadius, snakes, myId } = args;
  1038→  ctx.save();
  1039→  ctx.setTransform(1, 0, 0, 1, 0, 0);
  1040→
  1041→  // --- Background fill (dim slate) ---
  1042→  ctx.fillStyle = 'rgba(2, 6, 23, 0.94)';
  1043→  ctx.fillRect(0, 0, w, h);
  1044→
  1045→  // --- Layout: arena centred on screen ---
  1046→  const cx = w / 2;
  1047→  const cy = h / 2;
  1048→  const margin = 80;
  1049→  const fitDim = Math.min(w, h) - margin * 2;
  1050→  const arenaDiameter = arenaRadius * 2;
  1051→  const scale = fitDim / arenaDiameter;
  1052→  const screenR = arenaRadius * scale;
  1053→
  1054→  const wcx = worldSize / 2;
  1055→  const wcy = worldSize / 2;
  1056→  const toScreenX = (wx: number) => cx + (wx - wcx) * scale;
  1057→  const toScreenY = (wy: number) => cy + (wy - wcy) * scale;
  1058→
  1059→  // --- Concentric range rings (faint) ---
  1060→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
  1061→  ctx.lineWidth = 1;
  1062→  for (const frac of [0.25, 0.5, 0.75]) {
  1063→    ctx.beginPath();
  1064→    ctx.arc(cx, cy, screenR * frac, 0, Math.PI * 2);
  1065→    ctx.stroke();
  1066→  }
  1067→  // Crosshairs.
  1068→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.5)';
  1069→  ctx.beginPath();
  1070→  ctx.moveTo(cx, cy - screenR);
  1071→  ctx.lineTo(cx, cy + screenR);
  1072→  ctx.moveTo(cx - screenR, cy);
  1073→  ctx.lineTo(cx + screenR, cy);
  1074→  ctx.stroke();
  1075→
  1076→  // --- Arena boundary (dashed rose) ---
  1077→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.8)';
  1078→  ctx.lineWidth = 2;
  1079→  ctx.setLineDash([6, 6]);
  1080→  ctx.beginPath();
  1081→  ctx.arc(cx, cy, screenR, 0, Math.PI * 2);
  1082→  ctx.stroke();
  1083→  ctx.setLineDash([]);
  1084→
  1085→  // --- Snakes ---
  1086→  for (let i = 0; i < snakes.length; i++) {
  1087→    const s = snakes[i];
  1088→    if (!s.points || s.points.length === 0) continue;
  1089→    const head = s.points[0];
  1090→    const sx = toScreenX(head.x);
  1091→    const sy = toScreenY(head.y);
  1092→    ctx.beginPath();
  1093→    if (s.id === myId) {
  1094→      ctx.arc(sx, sy, 5, 0, Math.PI * 2);
  1095→      ctx.fillStyle = '#818cf8';
  1096→      ctx.fill();
  1097→      ctx.strokeStyle = 'rgba(129, 140, 248, 0.7)';
  1098→      ctx.lineWidth = 2;
  1099→      ctx.beginPath();
  1100→      ctx.arc(sx, sy, 9, 0, Math.PI * 2);
  1101→      ctx.stroke();
  1102→    } else if (s.isBot) {
  1103→      ctx.arc(sx, sy, 2.5, 0, Math.PI * 2);
  1104→      ctx.fillStyle = '#f43f5e';
  1105→      ctx.fill();
  1106→    } else {
  1107→      ctx.arc(sx, sy, 3, 0, Math.PI * 2);
  1108→      ctx.fillStyle = '#34d399';
  1109→      ctx.fill();
  1110→    }
  1111→  }
  1112→
  1113→  // --- Title + close hint ---
  1114→  ctx.fillStyle = 'rgba(226, 232, 240, 0.95)';
  1115→  ctx.font = 'bold 14px monospace';
  1116→  ctx.textAlign = 'center';
  1117→  ctx.textBaseline = 'top';
  1118→  ctx.fillText('ARENA OVERVIEW — ALL SNAKES', cx, 16);
  1119→  ctx.fillStyle = 'rgba(148, 163, 184, 0.85)';
  1120→  ctx.font = '11px monospace';
  1121→  ctx.fillText('Press M to close', cx, h - 24);
  1122→
  1123→  // Legend (top-left).
  1124→  ctx.textAlign = 'left';
  1125→  ctx.font = '11px monospace';
  1126→  const legendX = 20;
  1127→  let legendY = 20;
  1128→  const drawLegend = (color: string, label: string, dotR: number) => {
  1129→    ctx.beginPath();
  1130→    ctx.arc(legendX + 6, legendY + 6, dotR, 0, Math.PI * 2);
  1131→    ctx.fillStyle = color;
  1132→    ctx.fill();
  1133→    ctx.fillStyle = 'rgba(226, 232, 240, 0.85)';
  1134→    ctx.fillText(label, legendX + 18, legendY);
  1135→    legendY += 18;
  1136→  };
  1137→  drawLegend('#818cf8', 'You', 5);
  1138→  drawLegend('#34d399', 'Real Players', 3);
  1139→  drawLegend('#f43f5e', 'Bots', 2.5);
  1140→
  1141→  ctx.restore();
  1142→}
