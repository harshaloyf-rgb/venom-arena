   342→// ---------------------------------------------------------------------------
   343→// generateCustomSegments — exact replica of original helper.
   344→// ---------------------------------------------------------------------------
   345→function generateCustomSegments(
   346→  colors: string[],
   347→  shapeStyle: BodyStyle,
   348→  taperStyle: TaperStyle,
   349→  glowEnabled: boolean,
   350→): CustomSegment[] {
   351→  if (colors.length === 0) return [];
   352→  const result: CustomSegment[] = [];
   353→  const totalNodes = 16;
   354→
   355→  for (let i = 0; i < totalNodes; i++) {
   356→    const color = colors[i % colors.length];
   357→
   358→    let shape: SegShape = 'circle';
   359→    if (shapeStyle === 'smooth') {
   360→      shape = 'circle';
   361→    } else if (shapeStyle === 'dragon') {
   362→      shape = i === 0 ? 'circle' : i % 2 === 1 ? 'spike' : 'circle';
   363→    } else if (shapeStyle === 'armored') {
   364→      shape = i === 0 ? 'circle' : i % 2 === 1 ? 'square' : 'circle';
   365→    } else if (shapeStyle === 'crystal') {
   366→      shape = i === 0 ? 'circle' : i % 2 === 1 ? 'diamond' : 'circle';
   367→    } else if (shapeStyle === 'obsidian') {
   368→      shape = 'spike';
   369→    } else if (shapeStyle === 'basilisk') {
   370→      shape = 'diamond';
   371→    }
   372→
   373→    let sizeScale = 1.0;
   374→    if (taperStyle === 'uniform') {
   375→      sizeScale = i === 0 ? 1.3 : 1.0;
   376→    } else if (taperStyle === 'natural') {
   377→      sizeScale = i === 0 ? 1.35 : Math.max(0.65, 1.25 - (i / totalNodes) * 0.55);
   378→    } else if (taperStyle === 'wave') {
   379→      sizeScale = i === 0 ? 1.3 : 1.0 + Math.sin(i * 0.95) * 0.22;
   380→    } else if (taperStyle === 'heavy') {
   381→      sizeScale = i === 0 ? 1.6 : Math.max(0.55, 1.35 - (i / totalNodes) * 0.8);
   382→    }
   383→
   384→    result.push({
   385→      color,
   386→      shape,
   387→      glow: glowEnabled,
   388→      sizeScale: Number(sizeScale.toFixed(2)),
   389→    });
   390→  }
   391→
   392→  return result;
   393→}

   433→// ---------------------------------------------------------------------------
   434→// 1. REUSABLE REAL-TIME 60FPS SLITHERING SKIN PREVIEW
   435→// ---------------------------------------------------------------------------
   436→interface SkinsCanvasPreviewProps {
   437→  colors: string[];
   438→  pattern?: string;
   439→  shapeStyle?: BodyStyle;
   440→  glow?: boolean;
   441→}
   442→
   443→function SkinsCanvasPreview({
   444→  colors,
   445→  pattern,
   446→  shapeStyle = 'smooth',
   447→  glow = false,
   448→}: SkinsCanvasPreviewProps) {
   449→  const canvasRef = useRef<HTMLCanvasElement>(null);
   450→
   451→  useEffect(() => {
   452→    const canvas = canvasRef.current;
   453→    if (!canvas) return;
   454→    const ctx = canvas.getContext('2d');
   455→    if (!ctx) return;
   456→
   457→    let animId = 0;
   458→    let time = Math.random() * 100;
   459→
   460→    const numSegments = 10;
   461→    const points: Array<{ x: number; y: number }> = [];
   462→    for (let i = 0; i < numSegments; i++) points.push({ x: 0, y: 0 });
   463→
   464→    const render = () => {
   465→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   466→      time += 0.07;
   467→
   468→      // Grid background
   469→      ctx.strokeStyle = 'rgba(99, 102, 241, 0.05)';
   470→      ctx.lineWidth = 1;
   471→      for (let x = 15; x < canvas.width; x += 30) {
   472→        ctx.beginPath();
   473→        ctx.moveTo(x, 0);
   474→        ctx.lineTo(x, canvas.height);
   475→        ctx.stroke();
   476→      }
   477→
   478→      for (let i = 0; i < numSegments; i++) {
   479→        const x = canvas.width - 24 - (i * (canvas.width - 48)) / (numSegments - 1);
   480→        const wiggleVal = Math.sin(time - i * 0.42) * 9;
   481→        const y = canvas.height / 2 + wiggleVal;
   482→        points[i] = { x, y };
   483→      }
   484→
   485→      // Draw trailing segment shadows & nodes from tail to head
   486→      for (let i = points.length - 1; i >= 1; i--) {
   487→        const pt = points[i];
   488→        const nextPt = points[i - 1] || pt;
   489→        const segAngle = Math.atan2(nextPt.y - pt.y, nextPt.x - pt.x);
   490→
   491→        let fillColor = colors[i % colors.length] || '#ffffff';
   492→        let segmentGlow = glow;
   493→
   494→        if (pattern === 'rainbow') {
   495→          const hue = (Date.now() * 0.06 + i * 36) % 360;
   496→          fillColor = `hsl(${hue}, 85%, 55%)`;
   497→          segmentGlow = true;
   498→        } else if (pattern === 'neon') {
   499→          fillColor = i % 2 === 0 ? '#06b6d4' : '#a855f7';
   500→          segmentGlow = true;
   501→        } else if (pattern === 'metallic') {
   502→          fillColor = i % 2 === 0 ? '#cbd5e1' : '#475569';
   503→        } else if (pattern === 'camo') {
   504→          fillColor = i % 2 === 0 ? '#15803d' : '#854d0e';
   505→        }
   506→
   507→        const sizeRatio = 1 - (i / points.length) * 0.42;
   508→        const r = 6 * sizeRatio;
   509→
   510→        ctx.save();
   511→        if (segmentGlow) {
   512→          ctx.shadowBlur = 8;
   513→          ctx.shadowColor = fillColor;
   514→        }
   515→        ctx.fillStyle = fillColor;
   516→
   517→        let segmentShape: SegShape = 'circle';
   518→        if (shapeStyle === 'dragon') {
   519→          segmentShape = i % 2 === 1 ? 'spike' : 'circle';
   520→        } else if (shapeStyle === 'armored') {
   521→          segmentShape = i % 2 === 1 ? 'square' : 'circle';
   522→        } else if (shapeStyle === 'crystal') {
   523→          segmentShape = i % 2 === 1 ? 'diamond' : 'circle';
   524→        } else if (shapeStyle === 'obsidian') {
   525→          segmentShape = 'spike';
   526→        } else if (shapeStyle === 'basilisk') {
   527→          segmentShape = 'diamond';
   528→        }
   529→
   530→        ctx.beginPath();
   531→        if (segmentShape === 'circle') {
   532→          ctx.arc(pt.x, pt.y, r, 0, Math.PI * 2);
   533→          ctx.fill();
   534→        } else if (segmentShape === 'square') {
   535→          ctx.fillRect(pt.x - r, pt.y - r, r * 2, r * 2);
   536→        } else if (segmentShape === 'diamond') {
   537→          ctx.moveTo(pt.x, pt.y - r);
   538→          ctx.lineTo(pt.x + r, pt.y);
   539→          ctx.lineTo(pt.x, pt.y + r);
   540→          ctx.lineTo(pt.x - r, pt.y);
   541→          ctx.closePath();
   542→          ctx.fill();
   543→        } else if (segmentShape === 'spike') {
   544→          const perpAngle = segAngle + Math.PI / 2;
   545→          ctx.moveTo(
   546→            pt.x + Math.cos(segAngle) * r * 1.3,
   547→            pt.y + Math.sin(segAngle) * r * 1.3,
   548→          );
   549→          ctx.lineTo(
   550→            pt.x + Math.cos(perpAngle) * r * 0.8,
   551→            pt.y + Math.sin(perpAngle) * r * 0.8,
   552→          );
   553→          ctx.lineTo(
   554→            pt.x - Math.cos(segAngle) * r * 0.3,
   555→            pt.y - Math.sin(segAngle) * r * 0.3,
   556→          );
   557→          ctx.lineTo(
   558→            pt.x - Math.cos(perpAngle) * r * 0.8,
   559→            pt.y - Math.sin(perpAngle) * r * 0.8,
   560→          );
   561→          ctx.closePath();
   562→          ctx.fill();
   563→        }
   564→        ctx.restore();
   565→      }
   566→
   567→      // Head

   571→
   572→      let headColor = colors[0] || '#ffffff';
   573→      if (pattern === 'rainbow') {
   574→        const hue = (Date.now() * 0.06) % 360;
   575→        headColor = `hsl(${hue}, 85%, 55%)`;
   576→      } else if (pattern === 'neon') {
   577→        headColor = '#06b6d4';
   578→      }
   579→
   580→      ctx.save();
   581→      ctx.fillStyle = headColor;
   582→      if (glow) {
   583→        ctx.shadowBlur = 10;
   584→        ctx.shadowColor = headColor;
   585→      }
   586→      ctx.beginPath();
   587→      ctx.arc(head.x, head.y, 8.5, 0, Math.PI * 2);
   588→      ctx.fill();
   589→      ctx.restore();

   591→      // Eyes
   592→      const eyeL_Angle = headAngle + 0.52;
   593→      const eyeR_Angle = headAngle - 0.52;
   594→      const eyeL_Pos = {
   595→        x: head.x + Math.cos(eyeL_Angle) * 3.8,
   596→        y: head.y + Math.sin(eyeL_Angle) * 3.8,
   597→      };
   598→      const eyeR_Pos = {
   599→        x: head.x + Math.cos(eyeR_Angle) * 3.8,
   600→        y: head.y + Math.sin(eyeR_Angle) * 3.8,
   601→      };

   604→      ctx.beginPath();

   606→      ctx.arc(eyeR_Pos.x, eyeR_Pos.y, 2, 0, Math.PI * 2);
   607→      ctx.fill();

   609→      ctx.fillStyle = '#090d16';
   610→      ctx.beginPath();
   611→      ctx.arc(
   612→        eyeL_Pos.x + Math.cos(headAngle) * 0.6,
   613→        eyeL_Pos.y + Math.sin(headAngle) * 0.6,
   614→        1,
   615→        0,
   616→        Math.PI * 2,
   617→      );
   618→      ctx.arc(
   619→        eyeR_Pos.x + Math.cos(headAngle) * 0.6,
   620→        eyeR_Pos.y + Math.sin(headAngle) * 0.6,
   621→        1,
   622→        0,
   623→        Math.PI * 2,
   624→      );
   625→      ctx.fill();
   626→
   627→      animId = requestAnimationFrame(render);
   628→    };
   629→
   630→    render();
   631→    return () => cancelAnimationFrame(animId);

   647→// 2. INTERACTIVE TRY-ON PLAYGROUND (steer with mouse)

   649→interface TryOnPreviewProps {
   650→  colors: string[];
   651→  shapeStyle: BodyStyle;
   652→  taperStyle: TaperStyle;
   653→  glow: boolean;
   654→}

   656→function TryOnPreview({
   657→  colors,
   658→  shapeStyle,
   659→  taperStyle,
   660→  glow,
   661→}: TryOnPreviewProps) {
   662→  const canvasRef = useRef<HTMLCanvasElement>(null);
   663→  const mousePos = useRef({ x: 220, y: 100 });
   664→  const isHovered = useRef(false);
   665→
   666→  useEffect(() => {
   667→    const canvas = canvasRef.current;
   668→    if (!canvas) return;
   669→    const ctx = canvas.getContext('2d');
   670→    if (!ctx) return;
   671→
   672→    let animId = 0;
   673→    let time = 0;
   674→
   675→    const numPoints = 26;
   676→    const points: Array<{ x: number; y: number }> = [];
   677→    for (let i = 0; i < numPoints; i++) points.push({ x: 220, y: 100 });
   678→
   679→    const handleMouseMove = (e: MouseEvent) => {
   680→      const rect = canvas.getBoundingClientRect();
   681→      mousePos.current = {
   682→        x: e.clientX - rect.left,
   683→        y: e.clientY - rect.top,
   684→      };
   685→    };
   686→    const handleMouseEnter = () => {
   687→      isHovered.current = true;
   688→    };
   689→    const handleMouseLeave = () => {
   690→      isHovered.current = false;
   691→    };
   692→
   693→    canvas.addEventListener('mousemove', handleMouseMove);
   694→    canvas.addEventListener('mouseenter', handleMouseEnter);
   695→    canvas.addEventListener('mouseleave', handleMouseLeave);
   696→
   697→    let headX = 220;
   698→    let headY = 100;
   699→
   700→    const render = () => {
   701→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   702→      time += 0.045;
   703→
   704→      let targetX = canvas.width / 2 + Math.cos(time * 0.9) * 120;
   705→      let targetY = canvas.height / 2 + Math.sin(time * 1.6) * 45;
   706→
   707→      if (isHovered.current) {
   708→        targetX = mousePos.current.x;
   709→        targetY = mousePos.current.y;
   710→      }
   711→
   712→      const dx = targetX - headX;
   713→      const dy = targetY - headY;
   714→      const dist = Math.hypot(dx, dy);
   715→
   716→      if (dist > 3) {
   717→        const speed = isHovered.current ? 4.8 : 3.4;
   718→        const angle = Math.atan2(dy, dx);
   719→        headX += Math.cos(angle) * speed;
   720→        headY += Math.sin(angle) * speed;
   721→      }
   722→
   723→      points.unshift({ x: headX, y: headY });
   724→      if (points.length > numPoints) points.pop();
   725→
   726→      // Grid scanlines
   727→      ctx.strokeStyle = 'rgba(99, 102, 241, 0.04)';
   728→      ctx.lineWidth = 1;
   729→      for (let x = 0; x < canvas.width; x += 20) {
   730→        ctx.beginPath();
   731→        ctx.moveTo(x, 0);
   732→        ctx.lineTo(x, canvas.height);
   733→        ctx.stroke();
   734→      }
   735→      for (let y = 0; y < canvas.height; y += 20) {
   736→        ctx.beginPath();
   737→        ctx.moveTo(0, y);
   738→        ctx.lineTo(canvas.width, y);
   739→        ctx.stroke();
   740→      }
   741→
   742→      // Mouse radar ring
   743→      if (isHovered.current) {
   744→        ctx.strokeStyle = 'rgba(99, 102, 241, 0.2)';
   745→        ctx.beginPath();
   746→        ctx.arc(mousePos.current.x, mousePos.current.y, 12, 0, Math.PI * 2);
   747→        ctx.stroke();
   748→      }
   749→
   750→      // Body segments
   751→      for (let i = points.length - 1; i >= 1; i--) {
   752→        const pt = points[i];
   753→        const prevPt = points[i - 1] || pt;
   754→        const segAngle = Math.atan2(pt.y - prevPt.y, pt.x - prevPt.x);
   755→        const perpAngle = segAngle + Math.PI / 2;
   756→
   757→        const sizeRatio = 1 - (i / points.length) * 0.45;
   758→        const color = colors[i % colors.length] || '#ffffff';
   759→
   760→        let shape: SegShape = 'circle';
   761→        if (shapeStyle === 'smooth') {
   762→          shape = 'circle';
   763→        } else if (shapeStyle === 'dragon') {
   764→          shape = i % 2 === 1 ? 'spike' : 'circle';
   765→        } else if (shapeStyle === 'armored') {
   766→          shape = i % 2 === 1 ? 'square' : 'circle';
   767→        } else if (shapeStyle === 'crystal') {
   768→          shape = i % 2 === 1 ? 'diamond' : 'circle';
   769→        } else if (shapeStyle === 'obsidian') {
   770→          shape = 'spike';
   771→        } else if (shapeStyle === 'basilisk') {
   772→          shape = 'diamond';
   773→        }
   774→
   775→        let sizeScale = 1.0;
   776→        if (taperStyle === 'uniform') {
   777→          sizeScale = 1.0;
   778→        } else if (taperStyle === 'natural') {
   779→          sizeScale = Math.max(0.65, 1.25 - (i / points.length) * 0.55);
   780→        } else if (taperStyle === 'wave') {
   781→          sizeScale = 1.0 + Math.sin(i * 0.95) * 0.22;
   782→        } else if (taperStyle === 'heavy') {
   783→          sizeScale = Math.max(0.55, 1.35 - (i / points.length) * 0.8);
   784→        }
   785→
   786→        const r = 10 * sizeRatio * sizeScale;
   787→
   788→        ctx.save();
   789→        if (glow) {
   790→          ctx.shadowBlur = 12;
   791→          ctx.shadowColor = color;
   792→        }
   793→        ctx.fillStyle = color;
   794→
   795→        ctx.beginPath();
   796→        if (shape === 'circle') {
   797→          ctx.arc(pt.x, pt.y, r, 0, Math.PI * 2);
   798→          ctx.fill();
   799→        } else if (shape === 'square') {
   800→          ctx.fillRect(pt.x - r, pt.y - r, r * 2, r * 2);
   801→        } else if (shape === 'diamond') {
   802→          ctx.moveTo(pt.x, pt.y - r);
   803→          ctx.lineTo(pt.x + r, pt.y);
   804→          ctx.lineTo(pt.x, pt.y + r);
   805→          ctx.lineTo(pt.x - r, pt.y);
   806→          ctx.closePath();
   807→          ctx.fill();
   808→        } else if (shape === 'spike') {
   809→          const spikeAngle = segAngle + Math.PI;
   810→          ctx.moveTo(
   811→            pt.x + Math.cos(segAngle) * r * 1.35,
   812→            pt.y + Math.sin(segAngle) * r * 1.35,
   813→          );
   814→          ctx.lineTo(
   815→            pt.x + Math.cos(perpAngle) * r * 0.95,
   816→            pt.y + Math.sin(perpAngle) * r * 0.95,
   817→          );
   818→          ctx.lineTo(
   819→            pt.x + Math.cos(spikeAngle) * r * 0.4,
   820→            pt.y + Math.sin(spikeAngle) * r * 0.4,
   821→          );
   822→          ctx.lineTo(
   823→            pt.x + Math.cos(perpAngle - Math.PI) * r * 0.95,
   824→            pt.y + Math.sin(perpAngle - Math.PI) * r * 0.95,
   825→          );
   826→          ctx.closePath();
   827→          ctx.fill();
   828→        }
   829→        ctx.restore();
   830→      }
   831→
   832→      // Head
   833→      const head = points[0];
   834→      const headColor = colors[0] || '#ffffff';
   835→      ctx.save();
   836→      if (glow) {
   837→        ctx.shadowBlur = 15;
   838→        ctx.shadowColor = headColor;
   839→      }
   840→      ctx.fillStyle = headColor;
   841→      ctx.beginPath();
   842→      ctx.arc(head.x, head.y, 12, 0, Math.PI * 2);
   843→      ctx.fill();
   844→      ctx.restore();
   845→
   846→      // Forked tongue
   847→      if (Math.sin(Date.now() * 0.012) > 0.45) {
   848→        ctx.save();
   849→        ctx.strokeStyle = '#ef4444';
   850→        ctx.lineWidth = 2.5;
   851→        ctx.lineCap = 'round';
   852→        const nextPt = points[1] || head;
   853→        const headAngle = Math.atan2(head.y - nextPt.y, head.x - nextPt.x);
   854→        const startX = head.x + Math.cos(headAngle) * 12;
   855→        const startY = head.y + Math.sin(headAngle) * 12;
   856→        const endX = startX + Math.cos(headAngle) * 8;
   857→        const endY = startY + Math.sin(headAngle) * 8;
   858→
   859→        ctx.beginPath();
   860→        ctx.moveTo(startX, startY);
   861→        ctx.lineTo(endX, endY);
   862→        ctx.lineTo(
   863→          endX + Math.cos(headAngle + 0.45) * 5,
   864→          endY + Math.sin(headAngle + 0.45) * 5,
   865→        );
   866→        ctx.moveTo(endX, endY);
   867→        ctx.lineTo(
   868→          endX + Math.cos(headAngle - 0.45) * 5,
   869→          endY + Math.sin(headAngle - 0.45) * 5,
   870→        );
   871→        ctx.stroke();
   872→        ctx.restore();
   873→      }
   874→
   875→      // Eyes
   876→      const nextPt = points[1] || head;
   877→      const headAngle = Math.atan2(head.y - nextPt.y, head.x - nextPt.x);
   878→      const eyeL = {
   879→        x: head.x + Math.cos(headAngle + 0.45) * 6,
   880→        y: head.y + Math.sin(headAngle + 0.45) * 6,
   881→      };
   882→      const eyeR = {
   883→        x: head.x + Math.cos(headAngle - 0.45) * 6,
   884→        y: head.y + Math.sin(headAngle - 0.45) * 6,
   885→      };
   886→
   887→      ctx.fillStyle = '#ffffff';
   888→      ctx.beginPath();
   889→      ctx.arc(eyeL.x, eyeL.y, 2.8, 0, Math.PI * 2);
   890→      ctx.arc(eyeR.x, eyeR.y, 2.8, 0, Math.PI * 2);
   891→      ctx.fill();
   892→
   893→      ctx.fillStyle = '#090d16';
   894→      ctx.beginPath();
   895→      ctx.arc(
   896→        eyeL.x + Math.cos(headAngle) * 0.8,
   897→        eyeL.y + Math.sin(headAngle) * 0.8,
   898→        1.4,
   899→        0,
   900→        Math.PI * 2,
   901→      );
   902→      ctx.arc(
   903→        eyeR.x + Math.cos(headAngle) * 0.8,
   904→        eyeR.y + Math.sin(headAngle) * 0.8,
   905→        1.4,
   906→        0,
   907→        Math.PI * 2,
   908→      );
   909→      ctx.fill();
   910→
   911→      animId = requestAnimationFrame(render);

   914→    render();
   915→    return () => {

   917→      canvas.removeEventListener('mousemove', handleMouseMove);
   918→      canvas.removeEventListener('mouseenter', handleMouseEnter);
   919→      canvas.removeEventListener('mouseleave', handleMouseLeave);
   920→    };
   921→  }, [colors, shapeStyle, taperStyle, glow]);

   925→      <div className="absolute top-2 left-3 flex items-center gap-1.5 z-10 bg-slate-900/90 px-2 py-0.5 rounded border border-indigo-500/20 pointer-events-none select-none">
