    61→// Generate ranks 1..100 for the global board (mixing mock + auto-generated)
    62→function generateGlobalRanks(playerTag?: string): LeaderboardEntry[] {
    63→  const out: LeaderboardEntry[] = [];
    64→  const seenTags = new Set<string>();
    65→
    66→  // Hall-of-fame top achievers as the global top tier
    67→  const topAchievers: { name: string; userTag: string; country: string; chips: number; level: number }[] = [
    68→    { name: 'Hari', userTag: '#IND-001', country: 'IN', chips: 10_000_000, level: 50 },
    69→    { name: 'Apex_Viper', userTag: '#USA-882', country: 'US', chips: 9_400_000, level: 49 },
    70→    { name: 'K-Snake_Master', userTag: '#KOR-114', country: 'KR', chips: 8_900_000, level: 49 },
    71→    { name: 'Shadow_Ninja', userTag: '#JPN-309', country: 'JP', chips: 5_000_000, level: 48 },
    72→    { name: 'Viper_Zero', userTag: '#USA-402', country: 'US', chips: 2_500_000, level: 47 },
    73→    { name: 'Rookie_Striker', userTag: '#IND-104', country: 'IN', chips: 1_200_000, level: 32 },
    74→  ];
    75→
    76→  topAchievers.forEach((p, i) => {
    77→    out.push({
    78→      name: p.name,
    79→      userTag: p.userTag,
    80→      country: p.country,
    81→      bankedChips: p.chips,
    82→      level: p.level,
    83→      rank: i + 1,
    84→      isPlayer: p.userTag === playerTag,
    85→    });
    86→    seenTags.add(p.userTag);
    87→  });
    88→
    89→  // Mock leaderboard next
    90→  MOCK_LEADERBOARD.forEach((m, i) => {
    91→    if (seenTags.has(m.userTag)) return;
    92→    out.push({
    93→      name: m.name,
    94→      userTag: m.userTag,
    95→      country: m.country,
    96→      bankedChips: m.bankedChips,
    97→      level: m.level,
    98→      rank: out.length + 1,
    99→      isPlayer: m.userTag === playerTag,
   100→    });
   101→    seenTags.add(m.userTag);
   102→  });
   103→
   104→  // Auto-generate ranks 1-100 if sparse
   105→  while (out.length < 100) {
   106→    const i = out.length;
   107→    const chips = Math.max(50_000, 10_000_000 - i * 95_000 + Math.floor(Math.random() * 20_000));
   108→    const level = Math.max(5, 50 - Math.floor(i / 2.2));
   109→    out.push({
   110→      name: `Viper_Challenger_${i + 1}`,
   111→      userTag: `VRP-${1000 + i}`,
   112→      country: COUNTRIES[i % COUNTRIES.length].code,
   113→      bankedChips: chips,
   114→      level,
   115→      rank: i + 1,
   116→    });
   117→  }
   118→
   119→  return out.slice(0, 100);
   120→}

   122→// Country #1 summit table - for each country, show one champion
   123→function generateCountrySummit(): { country: string; name: string; userTag: string; chips: number; level: number }[] {
   124→  return COUNTRIES.map((c, idx) => {
   125→    const seed = COUNTRY_SEEDS[c.code]?.[0];
   126→    if (seed) {
   127→      return {
   128→        country: c.code,
   129→        name: seed.name,
   130→        userTag: seed.userTag,
   131→        chips: seed.chips,
   132→        level: seed.level,
   133→      };
   134→    }
   135→    return {
   136→      country: c.code,
   137→      name: `Apex_${c.code}_Leader`,
   138→      userTag: `#${c.code}-001`,
   139→      chips: 10_000_000 - idx * 450_000,
   140→      level: 50 - idx,
   141→    };
   142→  }).sort((a, b) => b.chips - a.chips);
   143→}

   145→// National board for a given country code - top 100
   146→function generateNationalBoard(countryCode: string): LeaderboardEntry[] {
   147→  const country = COUNTRIES.find((c) => c.code === countryCode);
   148→  const countryDisplayName = country?.name || countryCode;
   149→  const seeds = COUNTRY_SEEDS[countryCode] || [];
   150→  const out: LeaderboardEntry[] = [];
   151→  const seenTags = new Set<string>();
   152→
   153→  seeds.forEach((s, i) => {
   154→    out.push({
   155→      name: s.name,
   156→      userTag: s.userTag,
   157→      country: countryCode,
   158→      bankedChips: s.chips,
   159→      level: s.level,
   160→      rank: i + 1,
   161→    });
   162→    seenTags.add(s.userTag);
   163→  });
   164→
   165→  while (out.length < 100) {
   166→    const i = out.length;
   167→    const chips = Math.max(50_000, 5_000_000 - i * 47_000);
   168→    const level = Math.max(5, 45 - Math.floor(i / 2.5));
   169→    out.push({
   170→      name: `${countryDisplayName}_Challenger_${i + 1}`,
   171→      userTag: `#${countryCode}-${100 + i}`,
   172→      country: countryCode,
   173→      bankedChips: chips,
   174→      level,
   175→      rank: i + 1,
   176→    });
   177→    seenTags.add(`${countryCode}-${100 + i}`);
   178→  }
   179→
   180→  return out;
   181→}

   183→// Milestone-tier board - players at each milestone
   184→function generateMilestoneBoard(tierId: string): LeaderboardEntry[] {
   185→  if (tierId === 'all') {
   186→    return generateGlobalRanks();
   187→  }
   188→  const tier = MILESTONE_TIERS.find((t) => t.id === tierId);
   189→  if (!tier) return [];
   190→
   191→  const out: LeaderboardEntry[] = [];
   192→
   193→  // t-1crore tier special: only 3 achievers (Hari, Apex_Viper, K-Snake_Master)
   194→  if (tierId === 'omega' || tier.id === 'omega') {
   195→    out.push({ name: 'Hari', userTag: '#IND-001', country: 'IN', bankedChips: 10_000_000, level: 50, rank: 1 });
   196→    out.push({ name: 'Apex_Viper', userTag: '#USA-882', country: 'US', bankedChips: 10_000_000, level: 49, rank: 2 });
   197→    out.push({ name: 'K-Snake_Master', userTag: '#KOR-114', country: 'KR', bankedChips: 10_000_000, level: 49, rank: 3 });
   198→    return out;
   199→  }
   200→
   201→  // Rookie tier - all players below 100K chips
   202→  if (tierId === 'rookie') {
   203→    out.push({ name: 'Starter_Pawn', userTag: '#GEN-000', country: 'US', bankedChips: 45_000, level: 8, rank: 1 });
   204→    while (out.length < 100) {
   205→      const i = out.length;
   206→      out.push({
   207→        name: `${COUNTRIES[i % COUNTRIES.length].name.split(' ')[0]}_Rookie_${i + 1}`,
   208→        userTag: `#${COUNTRIES[i % COUNTRIES.length].code}-${200 + i}`,
   209→        country: COUNTRIES[i % COUNTRIES.length].code,
   210→        bankedChips: Math.max(1_000, 90_000 - i * 880),
   211→        level: Math.max(1, 10 - Math.floor(i / 12)),
   212→        rank: i + 1,
   213→      });
   214→    }
   215→    return out;
   216→  }
   217→
   218→  // First achiever for known tiers
   219→  const firstAchieverMap: Record<string, { name: string; userTag: string; country: string }> = {
   220→    bronze: { name: 'Rookie_Striker', userTag: '#IND-104', country: 'IN' },
   221→    silver: { name: 'Viper_Zero', userTag: '#USA-402', country: 'US' },
   222→    gold: { name: 'K-Snake_Master', userTag: '#KOR-114', country: 'KR' },
   223→    platinum: { name: 'Apex_Viper', userTag: '#USA-882', country: 'US' },
   224→    diamond: { name: 'Shadow_Ninja', userTag: '#JPN-309', country: 'JP' },
   225→  };
   226→  const achiever = firstAchieverMap[tierId];
   227→  if (achiever) {
   228→    out.push({
   229→      name: achiever.name,
   230→      userTag: achiever.userTag,
   231→      country: achiever.country,
   232→      bankedChips: tier.minChips,
   233→      level: 45 + Math.floor(Math.random() * 5),
   234→      rank: 1,
   235→    });
   236→  }
   237→
   238→  // Auto-generate ranks 2..100
   239→  while (out.length < 100) {
   240→    const i = out.length;
   241→    out.push({
   242→      name: `${COUNTRIES[i % COUNTRIES.length].name.split(' ')[0]}_Achiever_${i + 1}`,
   243→      userTag: `#${COUNTRIES[i % COUNTRIES.length].code}-${100 + i}`,
   244→      country: COUNTRIES[i % COUNTRIES.length].code,
   245→      bankedChips: tier.minChips,
   246→      level: Math.max(5, 45 - Math.floor(i / 2.5)),
   247→      rank: i + 1,
   248→    });
   249→  }
   250→  return out;
   251→}

   305→export function Leaderboards({ onToast, onInspectPlayer }: LeaderboardsProps) {

   310→  const [countrySearch, setCountrySearch] = useState('');
   311→  const [liveEntries, setLiveEntries] = useState<LeaderboardEntry[]>([]);
   312→  const [loading, setLoading] = useState(true);
   313→  const [lastUpdated, setLastUpdated] = useState<Date | null>(null);
   314→
   315→  const playerTag = player?.userTag;
   316→
   317→  const fetchLive = useCallback(async () => {
   318→    setLoading(true);
   319→    try {
   320→      const res = await fetch('/api/leaderboard?type=chips&limit=100', { cache: 'no-store' });
   321→      const data = (await res.json().catch(() => ({}))) as {
   322→        entries?: LeaderboardEntry[];
   323→        error?: string;
   324→      };
   325→      if (res.ok && data.entries && data.entries.length > 0) {
   326→        setLiveEntries(data.entries);
   327→      } else {
   328→        setLiveEntries(generateGlobalRanks(playerTag));
   329→      }
   330→      setLastUpdated(new Date());
   331→    } catch {
   332→      setLiveEntries(generateGlobalRanks(playerTag));
   333→      setLastUpdated(new Date());
   334→    } finally {
   335→      setLoading(false);
   336→    }
   337→  }, [playerTag]);
   338→
   339→  useEffect(() => {
   340→    void fetchLive();
   341→    const id = setInterval(() => void fetchLive(), 30 * 60 * 1000);
   342→    return () => clearInterval(id);
   343→  }, [fetchLive]);
   344→
   345→  const countrySummit = useMemo(() => generateCountrySummit(), []);
   346→  const nationalBoard = useMemo(
   347→    () => generateNationalBoard(selectedCountry),
   348→    [selectedCountry],
   349→  );
   350→  const milestoneBoard = useMemo(
   351→    () => generateMilestoneBoard(selectedTierId),
   352→    [selectedTierId],
   353→  );
   354→
   355→  // Merge live + generated global ranks for the Global view
   356→  const globalRanks = useMemo<LeaderboardEntry[]>(() => {
   357→    const generated = generateGlobalRanks(playerTag);
   358→    if (liveEntries.length === 0) return generated;
   359→    const liveSet = new Set(liveEntries.map((e) => e.userTag));
   360→    const merged = [...liveEntries];
   361→    let rankCounter = merged.length;
   362→    for (const g of generated) {
   363→      if (!liveSet.has(g.userTag)) {
   364→        merged.push({ ...g, rank: ++rankCounter });
   365→      }
   366→    }
   367→    return merged.slice(0, 100);
   368→  }, [liveEntries, playerTag]);
   369→
   370→  // Compute player rank info for the Your Rank card
   371→  const playerRankInfo = useMemo(() => {
   372→    if (!player) return null;
   373→    const globalEntry = globalRanks.find((e) => e.userTag === player.userTag);
   374→    const national = generateNationalBoard(player.country || 'US');
   375→    const nationalEntry = national.find((e) => e.userTag === player.userTag);
   376→    const tier = milestoneTierForChips(player.bankedChips);
   377→    return {
   378→      globalRank: globalEntry?.rank ?? null,
   379→      nationalRank: nationalEntry?.rank ?? null,
   380→      tierBadge: tier.badge,
   381→      tierName: tier.name,
   382→      bankedChips: player.bankedChips,
   383→      level: player.level,
   384→    };
   385→  }, [player, globalRanks]);
   386→
   387→  if (!player) return <NotSignedIn />;
   388→

   394→      userTag: e.userTag,
   395→      country: e.country,
   396→      flag: countryFlag(e.country),
   397→      bankedChips: e.chips ?? e.bankedChips ?? 0,
   398→      level: e.level,
   399→      clanTag: 'APEX',
   400→      clanName: 'Viper Apex Syndicate',
   401→      achievedAt: '26 Jul 2026, 05:42 PM UTC',
   402→      globalRank: e.rank,
   403→      countryRank: Math.floor(e.rank / 1.4) || 1,
   404→      regionalRank: Math.floor(e.rank / 2) || 1,
   405→    });
   406→    void tier;
   407→  }
   408→
   409→  const filteredNational = nationalBoard.filter((e) => {
   410→    if (!countrySearch.trim()) return true;
   411→    const q = countrySearch.toLowerCase();
   412→    return e.name.toLowerCase().includes(q) || e.userTag.toLowerCase().includes(q);
   413→  });
   414→
   415→  const tabs: { id: TopTab; icon: typeof Crown; label: string; color: string }[] = [
   416→    { id: 'summit', icon: Crown, label: 'Summit', color: '#f59e0b' },
   417→    { id: 'global', icon: Globe, label: 'Global', color: '#06b6d4' },
   418→    { id: 'national', icon: MapPin, label: 'National', color: '#8b5cf6' },
   419→    { id: 'tiers', icon: Medal, label: 'Tiers', color: '#eab308' },
   420→  ];
   421→
   422→  return (
   423→    <div className="relative rounded-2xl border border-slate-800/80 bg-slate-900/60 shadow-md p-5 sm:p-6 overflow-hidden">
   424→      <GlowBlob color="bg-amber-500/10" className="-top-12 -right-12 w-56 h-56" />
   425→
   426→      {/* Header */}
   427→      <div className="relative flex flex-col sm:flex-row sm:items-center justify-between gap-3 mb-5 pb-5 border-b border-slate-800">
   428→        <div>
   429→          <div className="flex items-center gap-2 flex-wrap">
   430→            <span className="bg-amber-500/10 border border-amber-500/30 text-amber-300 text-[9px] font-mono font-bold px-2 py-0.5 rounded uppercase tracking-widest">
   431→              CURRENT YEAR (2026) CONCURRENT TOURNAMENT

   438→            <Trophy className="w-5.5 h-5.5 text-amber-400" />
   439→            Official World Tournament Leaderboards
