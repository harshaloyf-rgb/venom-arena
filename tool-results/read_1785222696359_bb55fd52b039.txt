    15→//  * **Bot AI** — no self-destruct, predictive evasion, food-seeking.
    16→//  * **Opacity layering** — larger snakes fade when a smaller snake is near.
    17→//  * **Score model** — INITIAL_SPAWN_SCORE + food eaten, body from score.

   176→// ----------------------------------------------------------------------------
   177→// Spatial hash grid — slimmed-down client port of the server's grid.
   178→// Items are bucketed into square cells; queries return a deduplicated Map.
   179→// ----------------------------------------------------------------------------
   180→
   181→class SpatialHashGrid {
   182→  private readonly cellSize: number;
   183→  private readonly cells: Map<string, Map<string, GridItem>> = new Map();
   184→
   185→  constructor(cellSize = 120) {
   186→    this.cellSize = cellSize;
   187→  }
   188→
   189→  private key(cx: number, cy: number): string {
   190→    return cx + ':' + cy;
   191→  }
   192→
   193→  clear(): void {
   194→    this.cells.clear();
   195→  }
   196→
   197→  insert(item: GridItem): void {
   198→    const minCx = Math.floor((item.x - item.radius) / this.cellSize);
   199→    const maxCx = Math.floor((item.x + item.radius) / this.cellSize);
   200→    const minCy = Math.floor((item.y - item.radius) / this.cellSize);
   201→    const maxCy = Math.floor((item.y + item.radius) / this.cellSize);
   202→    for (let cx = minCx; cx <= maxCx; cx++) {
   203→      for (let cy = minCy; cy <= maxCy; cy++) {
   204→        const k = this.key(cx, cy);
   205→        let bucket = this.cells.get(k);
   206→        if (!bucket) {
   207→          bucket = new Map();
   208→          this.cells.set(k, bucket);
   209→        }
   210→        bucket.set(item.id, item);
   211→      }
   212→    }
   213→  }
   214→
   215→  queryRadius(x: number, y: number, r: number): Map<string, GridItem> {
   216→    const out = new Map<string, GridItem>();
   217→    const minCx = Math.floor((x - r) / this.cellSize);
   218→    const maxCx = Math.floor((x + r) / this.cellSize);
   219→    const minCy = Math.floor((y - r) / this.cellSize);
   220→    const maxCy = Math.floor((y + r) / this.cellSize);
   221→    for (let cx = minCx; cx <= maxCx; cx++) {
   222→      for (let cy = minCy; cy <= maxCy; cy++) {
   223→        const bucket = this.cells.get(this.key(cx, cy));
   224→        if (!bucket) continue;
   225→        for (const [id, item] of bucket) {
   226→          if (!out.has(id)) out.set(id, item);
   227→        }
   228→      }
   229→    }
   230→    return out;
   231→  }
   232→}

   294→function angularDelta(current: number, desired: number): number {
   295→  let diff = desired - current;
   296→  while (diff > Math.PI) diff -= 2 * Math.PI;
   297→  while (diff < -Math.PI) diff += 2 * Math.PI;
   298→  return diff;
   299→}
   300→
   301→function turnToward(current: number, desired: number, maxStep: number): number {
   302→  const diff = angularDelta(current, desired);
   303→  if (Math.abs(diff) <= maxStep) return desired;
   304→  return current + Math.sign(diff) * maxStep;
   305→}

   307→function dist(ax: number, ay: number, bx: number, by: number): number {
   308→  return Math.hypot(ax - bx, ay - by);
   309→}

   311→/** Random point within a circle of maxR around center. */
   312→function randomPointInCircle(cx: number, cy: number, maxR: number): Vec2 {
   313→  const r = Math.sqrt(Math.random()) * maxR;
   314→  const theta = Math.random() * Math.PI * 2;
   315→  return {
   316→    x: cx + Math.cos(theta) * r,
   317→    y: cy + Math.sin(theta) * r,
   318→  };
   319→}

   321→function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   322→  const pts: Vec2[] = [];
   323→  for (let i = 0; i < length; i++) {
   324→    pts.push({
   325→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   326→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   327→    });
   328→  }
   329→  return pts;
   330→}

   336→/** Pick a weighted random FoodOrbConfig. */
   337→function weightedRandomOrb(): FoodOrbConfig {
   338→  let r = Math.random();
   339→  for (let i = 0; i < ALL_FOOD_ORBS.length; i++) {
   340→    r -= FOOD_ORB_WEIGHTS[i];
   341→    if (r <= 0) return ALL_FOOD_ORBS[i];
   342→  }
   343→  return ALL_FOOD_ORBS[0]; // fallback to small
   344→}

   346→/** Create a Food object at a random position around the player. */
   347→function createFoodOrb(idPrefix: string, idCounter: { value: number }, cx: number, cy: number, farSpawn: boolean): Food {
   348→  const orb = weightedRandomOrb();
   349→  const radius = farSpawn ? FOOD_SPAWN_RADIUS_FAR : FOOD_SPAWN_RADIUS_NEAR;

   352→    id: `${idPrefix}-food-${idCounter.value++}`,
   353→    x: pos.x,
   354→    y: pos.y,
   355→    size: orb.radius,
   356→    value: orb.value,

   377→// ----------------------------------------------------------------------------
   378→// Death food drop — greedily split score into Large(5)/Medium(3)/Small(1)
   379→// ----------------------------------------------------------------------------
   380→
   381→interface DeathDropResult {
   382→  foods: Food[];
   383→  /** IDs of dead snakes that should drop food. */
   384→}
   385→
   386→function computeDeathOrbs(totalScore: number): { small: number; medium: number; large: number } {
   387→  let remaining = totalScore;
   388→  const large = Math.floor(remaining / 5);
   389→  remaining -= large * 5;
   390→  const medium = Math.floor(remaining / 3);
   391→  remaining -= medium * 3;
   392→  const small = remaining;
   393→  return { small, medium, large };
   394→}
   395→
   396→function computeDeathFoodDrop(
   397→  totalScore: number,
   398→  bodyPoints: Vec2[],
   399→  idPrefix: string,
   400→  idCounter: { value: number },
   401→): Food[] {
   402→  const result: Food[] = [];
   403→  if (!bodyPoints || bodyPoints.length === 0 || totalScore <= 0) return result;
   404→
   405→  const { small, medium, large } = computeDeathOrbs(totalScore);
   406→  let orbIdx = 0;
   407→  const totalOrbs = small + medium + large;
   408→  if (totalOrbs === 0) return result;
   409→
   410→  // Build orb sequence
   411→  const orbSequence: Array<{ value: number; size: number; color: string; glowColor: string; orbSize: string }> = [];
   412→  for (let i = 0; i < large; i++) orbSequence.push({ value: 5, size: 8, color: '#f472b6', glowColor: '#ec4899', orbSize: 'large' });
   413→  for (let i = 0; i < medium; i++) orbSequence.push({ value: 3, size: 5, color: '#38bdf8', glowColor: '#0ea5e9', orbSize: 'medium' });
   414→  for (let i = 0; i < small; i++) orbSequence.push({ value: 1, size: 3, color: '#34d399', glowColor: '#10b981', orbSize: 'small' });
   415→
   416→  // Shuffle
   417→  for (let i = orbSequence.length - 1; i > 0; i--) {
   418→    const j = Math.floor(Math.random() * (i + 1));
   419→    [orbSequence[i], orbSequence[j]] = [orbSequence[j], orbSequence[i]];
   420→  }
   421→
   422→  // Distribute orbs evenly along the body
   423→  const scatter = 15;
   424→  for (const orb of orbSequence) {
   425→    const segIdx = Math.min(bodyPoints.length - 1, Math.floor((orbIdx / totalOrbs) * bodyPoints.length));
   426→    const pt = bodyPoints[segIdx];
   427→    result.push({
   428→      id: `${idPrefix}-death-${idCounter.value++}`,
   429→      x: pt.x + (Math.random() - 0.5) * scatter,
   430→      y: pt.y + (Math.random() - 0.5) * scatter,
   431→      size: orb.size,
   432→      value: orb.value,
   433→      orbSize: orb.orbSize as 'small' | 'medium' | 'large',
   434→      color: orb.color,
   435→      glowColor: orb.glowColor,
   436→    });
   437→    orbIdx++;
   438→  }
   439→
   440→  return result;
   441→}

   810→  private onTouchStart(e: TouchEvent): void {
   811→    if (this.joystick) return;
   812→    const t = this.findJoystickTouch(e.touches);
   813→    if (!t) return;
   814→    e.preventDefault();
   815→    const rect = this.canvas.getBoundingClientRect();
   816→    this.joystick = {
   817→      active: true,
   818→      pointerId: t.identifier,
   819→      originX: t.clientX - rect.left,
   820→      originY: t.clientY - rect.top,
   821→      curX: t.clientX - rect.left,
   822→      curY: t.clientY - rect.top,
   823→    };
   824→  }
   825→
   826→  private onTouchMove(e: TouchEvent): void {
   827→    const js = this.joystick;
   828→    if (!js) return;
   829→    let t: Touch | null = null;
   830→    for (let i = 0; i < e.touches.length; i++) {
   831→      if (e.touches[i].identifier === js.pointerId) {
   832→        t = e.touches[i];
   833→        break;
   834→      }
   835→    }
   836→    if (!t) return;
   837→    e.preventDefault();
   838→    const rect = this.canvas.getBoundingClientRect();
   839→    js.curX = t.clientX - rect.left;
   840→    js.curY = t.clientY - rect.top;
   841→    const dx = js.curX - js.originX;
   842→    const dy = js.curY - js.originY;
   843→    const d = Math.hypot(dx, dy);
   844→    const mag = Math.min(1, d / JOYSTICK_MAX_RADIUS_PX);
   845→    if (mag > JOYSTICK_DEADZONE) {
   846→      this.touchAngle = Math.atan2(dy, dx);
   847→      this.touchBoost = mag > JOYSTICK_BOOST_MAGNITUDE;
   848→    } else {
   849→      this.touchAngle = null;
   850→      this.touchBoost = false;
   851→    }
   852→  }
   853→
   854→  private onTouchEnd(e: TouchEvent): void {
   855→    const js = this.joystick;
   856→    if (!js) return;
   857→    let stillActive = false;
   858→    for (let i = 0; i < e.touches.length; i++) {
   859→      if (e.touches[i].identifier === js.pointerId) {
   860→        stillActive = true;
   861→        break;
   862→      }
   863→    }
   864→    if (!stillActive) {
   865→      this.joystick = null;
   866→      this.touchAngle = null;
   867→      this.touchBoost = false;
   868→    }
   869→  }
   870→
   871→  private findJoystickTouch(touches: TouchList): Touch | null {
   872→    const rect = this.canvas.getBoundingClientRect();
   873→    for (let i = 0; i < touches.length; i++) {
   874→      const t = touches[i];
   875→      const x = t.clientX - rect.left;
   876→      const y = t.clientY - rect.top;
   877→      if (x < rect.width / 2 && y > rect.height / 2) {
   878→        return t;
   879→      }
   880→    }
   881→    return null;
   882→  }

   884→  private onKeyDown(e: KeyboardEvent): void {
   885→    const k = e.key.toLowerCase();
   886→    if (
   887→      k === 'arrowup' ||
   888→      k === 'arrowdown' ||
   889→      k === 'arrowleft' ||
   890→      k === 'arrowright' ||
   891→      k === ' ' ||
   892→      k === 'spacebar'

   894→      e.preventDefault();

   896→    if (k === 'escape' && this.state !== 'playing') {
   897→      this.handleExitToLobby();

   900→    if (k === 'e' && this.state === 'playing' && !this.extractHold) {
   901→      this.extractHold = true;
   902→      this.beginExtract();

   904→    if (this.state === 'playing' && ['1', '2', '3', '4', '5'].includes(k)) {
   905→      const idx = parseInt(k, 10) - 1;
   906→      if (idx >= 0 && idx < QUICK_EMOTES.length) {
   907→        this.setPlayerChat(QUICK_EMOTES[idx]);
   908→      }
   909→    }
   910→    const normalized = k === 'spacebar' ? ' ' : k;
   911→    this.keys.add(normalized);
   912→  }

   994→  private frame = (now: number): void => {
   995→    if (this.stopped) return;
   996→    this.rafId = requestAnimationFrame(this.frame);
   997→    const dt = now - this.lastFrameTime;
   998→    this.lastFrameTime = now;
   999→
  1000→    this.updateFps(now);
  1001→    this.updateParticles(dt);
  1002→
  1003→    // Physics ticks at ~TICK_MS (30 Hz) cadence.
  1004→    if (this.state === 'playing' || this.isPostDeathRecording) {
  1005→      this.accumulator += dt;
  1006→      let safety = 4;
  1007→      while (this.accumulator >= TICK_MS && safety > 0) {
  1008→        this.tickPhysics(now);
  1009→        this.accumulator -= TICK_MS;
  1010→        safety--;
  1011→      }
  1012→      if (safety === 0) this.accumulator = 0;
  1013→    }
  1014→
  1015→    // Post-death recording: count down ticks
  1016→    if (this.isPostDeathRecording) {
  1017→      this.postDeathTicksRemaining--;
  1018→      if (this.postDeathTicksRemaining <= 0) {
  1019→        this.finishPostDeathRecording();
  1020→      }
  1021→    }
  1022→
  1023→    this.render(now);
  1024→    this.updateHUD();
  1025→  };
  1026→
  1027→  private accumulator: number = 0;
  1028→
  1029→  // --------------------------------------------------------------------------
  1030→  // Physics tick (one server-tick equivalent)
  1031→  // --------------------------------------------------------------------------
  1032→
  1033→  private tickPhysics(now: number): void {
  1034→    this.tick++;
  1035→    const p = this.player;
  1036→    if (!p) return;
  1037→
  1038→    // During post-death recording, run simplified bot-only physics
  1039→    if (this.isPostDeathRecording) {
  1040→      this.tickPostDeathPhysics(now);
  1041→      this.captureReplaySnapshot();
  1042→      return;
  1043→    }
  1044→
  1045→    if (p.isDead) return;
  1046→
  1047→    // 1) Compute player desired angle + boost.
  1048→    const { angle, boost } = this.computePlayerInput();
  1049→    if (angle !== null) p.desiredAngle = angle;
  1050→    p.wantsBoost = boost;
  1051→    p.isBoosting = boost && p.points.length > BOOST_MIN_LENGTH + INITIAL_BODY_LENGTH - 20;
  1052→
  1053→    // Extraction progress (3-second hold).
  1054→    if (p.isExtracting) {
  1055→      p.extractionProgress += TICK_MS;
  1056→      if (p.extractionProgress >= EXTRACT_DURATION_MS) {
  1057→        this.finishExtract();
  1058→        return;
  1059→      }
  1060→    }
  1061→
  1062→    // 2) Move player.
  1063→    this.tickSnakeMovement(p, p.desiredAngle, p.wantsBoost);
  1064→
  1065→    // 3) Bots: think + move.
  1066→    for (const bot of this.bots.values()) {
  1067→      this.tickBot(bot, now);
  1068→    }
  1069→
  1070→    // Process boost food drops (from tail shedding)
  1071→    if (this.boostDropQueue.length > 0) {
  1072→      for (const drop of this.boostDropQueue) {
  1073→        this.foods.push({
  1074→          id: `food-${this.arena.id}-${this.idCounterObj.value++}`,
  1075→          x: drop.x,
  1076→          y: drop.y,
  1077→          size: 3,
  1078→          value: 1,
  1079→          isStarChip: false,
  1080→          color: '#34d399',
  1081→          glowColor: '#10b981',
  1082→          orbSize: 'small',
  1083→        });
  1084→      }
  1085→      this.boostDropQueue.length = 0;
  1086→    }
  1087→
  1088→    // 4) Build spatial grid for collision + food queries.
  1089→    this.grid.clear();
  1090→    this.insertSnakeIntoGrid(p);
  1091→    for (const bot of this.bots.values()) {
  1092→      if (!bot.isDead) this.insertSnakeIntoGrid(bot);
  1093→    }
  1094→    for (const food of this.foods) {
  1095→      this.grid.insert({
  1096→        id: food.id,
  1097→        kind: 'food',
  1098→        x: food.x,
  1099→        y: food.y,
  1100→        radius: food.size,
  1101→        value: food.value,
  1102→        foodRef: food,
  1103→      });
  1104→    }
  1105→
  1106→    // 5) Eat food (player + bots).
  1107→    this.eatFood();
  1108→
  1109→    // 6) Collision detection (body collision + head-on collision). NO wall death.
  1110→    const deaths = this.detectCollisions(now);
  1111→
  1112→    // 7) Apply deaths + drop food.
  1113→    let playerDied = false;
  1114→    const newDropFoods: Food[] = [];
  1115→    for (const d of deaths) {
  1116→      if (d.deadId === p.id) {
  1117→        playerDied = true;
  1118→        // Drop player's food
  1119→        const playerTotal = INITIAL_SPAWN_SCORE + p.score;
  1120→        newDropFoods.push(...computeDeathFoodDrop(playerTotal, p.points, this.arena.id, this.idCounterObj));
  1121→        continue;
  1122→      }
  1123→      const bot = this.bots.get(d.deadId);
  1124→      if (bot) {
  1125→        // Drop bot's food
  1126→        const botTotal = INITIAL_SPAWN_SCORE + bot.score;
  1127→        newDropFoods.push(...computeDeathFoodDrop(botTotal, bot.points, this.arena.id, this.idCounterObj));
  1128→        bot.isDead = true;
  1129→        // Credit kill to the killer if it's the player.
  1130→        if (d.killerId === p.id) {
  1131→          p.kills++;
  1132→        }
  1133→      }
  1134→    }
  1135→
  1136→    // Add dropped food to the world.
  1137→    if (newDropFoods.length > 0) {
  1138→      this.foods.push(...newDropFoods);
  1139→    }
  1140→
  1141→    // 8) Remove dead bots and respawn to maintain count.
  1142→    if (deaths.length > 0) {
  1143→      const toRemove: string[] = [];
  1144→      for (const [id, bot] of this.bots) {
  1145→        if (bot.isDead) toRemove.push(id);
  1146→      }
  1147→      for (const id of toRemove) {
  1148→        this.bots.delete(id);
  1149→      }
  1150→    }
  1151→    // Respawn bots near the player to maintain count.
  1152→    const pHead = p.points[0];
  1153→    while (this.bots.size < this.arena.botsCount) {
  1154→      const bot = this.spawnBot(pHead.x, pHead.y);
  1155→      // Ensure bots spawn in a safe area (far from player AND other bots)
  1156→      let safe = false;
  1157→      for (let attempt = 0; attempt < 20; attempt++) {
  1158→        const spawnPt = bot.points[0];
  1159→        const distFromPlayer = dist(spawnPt.x, spawnPt.y, pHead.x, pHead.y);
  1160→        if (distFromPlayer < 500) {
  1161→          // Re-spawn further from player
  1162→          const far = randomPointInCircle(pHead.x, pHead.y, BOT_SPAWN_RADIUS);
  1163→          bot.points = initialBody(far.x, far.y, bot.angle, INITIAL_BODY_LENGTH);
  1164→          continue;
  1165→        }
  1166→        // Also check distance from other bots
  1167→        let tooClose = false;
  1168→        for (const other of this.bots.values()) {
  1169→          if (other.isDead || other.points.length === 0) continue;
  1170→          if (dist(spawnPt.x, spawnPt.y, other.points[0].x, other.points[0].y) < 500) {
  1171→            tooClose = true;
  1172→            break;
  1173→          }
  1174→        }
  1175→        if (tooClose) {
  1176→          const far = randomPointInCircle(pHead.x, pHead.y, BOT_SPAWN_RADIUS);
  1177→          bot.points = initialBody(far.x, far.y, bot.angle, INITIAL_BODY_LENGTH);
  1178→          continue;
  1179→        }
  1180→        safe = true;
  1181→        break;
  1182→      }
  1183→      this.bots.set(bot.id, bot);
  1184→    }
  1185→
  1186→    // 9) Player death → enter post-death recording (15s of continued simulation).
  1187→    if (playerDied && !this.isPostDeathRecording) {
  1188→      p.isDead = true;
  1189→      this.enterPostDeathRecording();
  1190→      // Don't return — continue to replenish food, expire chat, update camera
  1191→    }
  1192→
  1193→    // 10) Replenish food toward target (spawn around player).
  1194→    this.replenishFood();
  1195→
  1196→    // 11) Expire chat bubbles.
  1197→    this.expireChat(now);
  1198→
  1199→    // 12) Update camera target.
  1200→    this.updateCamera();
  1201→
  1202→    // 13) Capture replay snapshot.
  1203→    this.captureReplaySnapshot();
  1204→  }

  1207→  // Snake movement (server-authoritative formula)
  1208→  // --------------------------------------------------------------------------
  1209→
  1210→  private tickSnakeMovement(snake: SnakeBase, desiredAngle: number, wantsBoost: boolean): void {
  1211→    if (snake.points.length === 0 || snake.isDead) return;
  1212→
  1213→    // Turn rate: max(0.045, 0.15 - totalScore*0.0006).
  1214→    const totalScore = INITIAL_SPAWN_SCORE + snake.score;
  1215→    const turnRate = Math.max(TURN_MIN, TURN_BASE - totalScore * TURN_SCORE_FACTOR);
  1216→    snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
  1217→
  1218→    // Speed.
  1219→    let speed = BASE_SPEED;
  1220→    if (snake.isExtracting) {
  1221→      speed = EXTRACT_GLIDE_SPEED;
  1222→    } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
  1223→      speed = BOOST_SPEED;
  1224→      snake.isBoosting = true;
  1225→      snake.boostFrameCounter++;
  1226→      if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
  1227→        snake.boostFrameCounter = 0;
  1228→        if (snake.points.length > BOOST_MIN_LENGTH) {
  1229→          // Record tail position BEFORE popping (for food drop)
  1230→          const tail = snake.points[snake.points.length - 1];
  1231→          this.boostDropQueue.push({
  1232→            x: tail.x,
  1233→            y: tail.y,
  1234→          });
  1235→          snake.points.pop();
  1236→          // Reduce food-score (score can go below 0 but body is clamped)
  1237→          snake.score = Math.max(0, snake.score - 1);
  1238→        }
  1239→      }
  1240→    } else {
  1241→      snake.isBoosting = false;
  1242→    }
  1243→
  1244→    // Move head.
  1245→    const head = snake.points[0];
  1246→    const nx = head.x + Math.cos(snake.angle) * speed;
  1247→    const ny = head.y + Math.sin(snake.angle) * speed;
  1248→    snake.points.unshift({ x: nx, y: ny });
  1249→
  1250→    // Grow / shrink body to target length based on SCORE.
  1251→    // targetLen = INITIAL_BODY_LENGTH + foodScore, clamped to [BOOST_MIN_LENGTH+1, MAX_BODY_LENGTH]
  1252→    const targetLen = Math.max(
  1253→      BOOST_MIN_LENGTH + 1,
  1254→      Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + snake.score),
  1255→    );
  1256→    while (snake.points.length > targetLen) snake.points.pop();
  1257→
  1258→    // Size: 8 + sqrt(totalScore) * 0.4.
  1259→    snake.size = SIZE_BASE + Math.sqrt(INITIAL_SPAWN_SCORE + snake.score) * SIZE_SCORE_FACTOR;
  1260→  }

  1263→  // Bot AI — no self-destruct, seek food, predictive evasion
  1264→  // --------------------------------------------------------------------------
  1265→
  1266→  private tickBot(bot: BotSession, now: number): void {
  1267→    if (bot.points.length === 0 || bot.isDead) return;
  1268→
  1269→    if (now >= bot.nextThinkAt) {
  1270→      bot.nextThinkAt = now + BOT_THINK_INTERVAL_MS + Math.floor(Math.random() * BOT_THINK_JITTER_MS);
  1271→      const head = bot.points[0];
  1272→
  1273→      // --- Find nearest food ---
  1274→      let bestFood: GridItem | null = null;
  1275→      let bestFoodDist = Infinity;
  1276→      const foodQuery = this.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
  1277→      for (const item of foodQuery.values()) {
  1278→        if (item.kind !== 'food') continue;
  1279→        if ((item.value ?? 0) <= 0) continue;
  1280→        const d = dist(head.x, head.y, item.x, item.y);
  1281→        if (d < bestFoodDist) {
  1282→          bestFoodDist = d;
  1283→          bestFood = item;
  1284→        }
  1285→      }
  1286→
  1287→      // --- Find nearest threat (foreign body segment) ---
  1288→      let threatX = 0;
  1289→      let threatY = 0;
  1290→      let threatDist = Infinity;
  1291→      const threatQuery = this.grid.queryRadius(head.x, head.y, BOT_THREAT_SCAN_RADIUS);
  1292→      for (const item of threatQuery.values()) {
  1293→        if (item.kind !== 'segment') continue;
  1294→        if (item.snakeId === bot.id) continue;
  1295→        if (item.segIdx === 0) continue;
  1296→        const d = dist(head.x, head.y, item.x, item.y);
  1297→        if (d < threatDist) {
  1298→          threatDist = d;
  1299→          threatX = item.x;
  1300→          threatY = item.y;
  1301→        }
  1302→      }
  1303→
  1304→      let desired: number;
  1305→      const shouldFlee = threatDist < 150; // ALL bots dodge body segments when close
  1306→
  1307→      // --- Predictive evasion against the human player ---
  1308→      const p = this.player;
  1309→      if (p && !p.isDead && p.points.length > 0 && now >= p.spawnProtectedUntil) {
  1310→        const playerHead = p.points[0];
  1311→        const playerDist = dist(head.x, head.y, playerHead.x, playerHead.y);
  1312→        if (playerDist < BOT_EVADE_RADIUS * 2) {
  1313→          // Predict where the player will be in N ticks.
  1314→          const predictedX = playerHead.x + Math.cos(p.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1315→          const predictedY = playerHead.y + Math.sin(p.angle) * BOT_PREDICT_AHEAD_TICKS;
  1316→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1317→
  1318→          // If the bot is on a collision course with the predicted position, evade.
  1319→          if (distToPredicted < (bot.size + p.size) * 3) {
  1320→            // Evade perpendicular to the player's heading.
  1321→            const evadeAngle = p.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1322→            desired = evadeAngle;
  1323→            bot.desiredAngle = desired;
  1324→            const wantsBoost = bot.personality === 'hunter' && bot.score > 10;
  1325→            this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1326→            return;
  1327→          }
  1328→        }
  1329→      }
  1330→
  1331→      // Also check nearby bot heads for predictive evasion
  1332→      for (const other of this.bots.values()) {
  1333→        if (other.id === bot.id || other.isDead || other.points.length === 0) continue;
  1334→        const otherHead = other.points[0];
  1335→        const otherDist = dist(head.x, head.y, otherHead.x, otherHead.y);
  1336→        if (otherDist < 200) {
  1337→          const predictedX = otherHead.x + Math.cos(other.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1338→          const predictedY = otherHead.y + Math.sin(other.angle) * BOT_PREDICT_AHEAD_TICKS;
  1339→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1340→          if (distToPredicted < (bot.size + other.size) * 3) {
  1341→            const evadeAngle = other.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1342→            desired = evadeAngle;
  1343→            bot.desiredAngle = desired;
  1344→            this.tickSnakeMovement(bot, bot.desiredAngle, false);
  1345→            return;
  1346→          }
  1347→        }
  1348→      }
  1349→
  1350→      if (shouldFlee) {
  1351→        desired = Math.atan2(head.y - threatY, head.x - threatX);
  1352→      } else if (bestFood) {
  1353→        desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
  1354→      } else {
  1355→        desired = bot.angle + (Math.random() - 0.5) * 0.4;
  1356→      }
  1357→
  1358→      // NO edge avoidance — infinite map, no boundaries.
  1359→
  1360→      bot.desiredAngle = desired;
  1361→    }
  1362→
  1363→    // Hunter personality occasionally boosts.
  1364→    const wantsBoost = bot.personality === 'hunter' && bot.score > 5 && Math.random() < 0.05;
  1365→    this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1366→  }

  1399→  private eatFoodForSnake(snake: SnakeBase): void {
  1400→    if (snake.points.length === 0) return;
  1401→    const head = snake.points[0];
  1402→    const nearby = this.grid.queryRadius(head.x, head.y, snake.size + 20);
  1403→    for (const item of nearby.values()) {
  1404→      if (item.kind !== 'food') continue;
  1405→      if ((item.value ?? 0) <= 0) continue;

  1407→      if (d < snake.size + (item.radius || 6) + 6) {
  1408→        // Eat food — add its value to the snake's food-score.
  1409→        snake.score += item.value ?? 0;
  1410→        item.value = 0;
  1411→        if (item.foodRef) item.foodRef.value = 0;
  1412→        // Spawn small particle burst for player only.
  1413→        if (snake.isPlayer) {
  1414→          this.spawnEatParticles(item.x, item.y, item.radius || 6, item.foodRef?.color ?? '#fbbf24');
  1415→        }
  1416→      }
  1417→    }
  1418→  }

  1421→  // Collision detection (body + head-on) — NO wall death
  1422→  // --------------------------------------------------------------------------

  1424→  private detectCollisions(now: number): { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] {
  1425→    const deaths: { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] = [];
  1426→    const seenDead = new Set<string>();
  1427→    const all: SnakeBase[] = [];
  1428→    if (this.player && !this.player.isDead) all.push(this.player);
  1429→    for (const bot of this.bots.values()) {
  1430→      if (!bot.isDead) all.push(bot);
  1431→    }
  1432→
  1433→    // --- Head-to-head collision (checked first, takes priority) ---
  1434→    const headOnChecked = new Set<string>();
  1435→    for (let i = 0; i < all.length; i++) {
  1436→      for (let j = i + 1; j < all.length; j++) {
  1437→        const a = all[i];
  1438→        const b = all[j];
  1439→        if (a.isDead || b.isDead) continue;
  1440→        if (now < a.spawnProtectedUntil || now < b.spawnProtectedUntil) continue;
  1441→        const pairKey = a.id < b.id ? `${a.id}:${b.id}` : `${b.id}:${a.id}`;
  1442→        if (headOnChecked.has(pairKey)) continue;
  1443→        headOnChecked.add(pairKey);
  1444→
  1445→        const ha = a.points[0];
  1446→        const hb = b.points[0];
  1447→        const d = dist(ha.x, ha.y, hb.x, hb.y);
  1448→        const hitDist = (a.size + b.size) * HEAD_ON_HIT_FACTOR;
  1449→        if (d >= hitDist) continue;
  1450→
  1451→        // Head-on collision! Determine winner based on rules.
  1452→        const aBoosting = a.isBoosting;
  1453→        const bBoosting = b.isBoosting;
  1454→        const aTotalScore = INITIAL_SPAWN_SCORE + a.score;
  1455→        const bTotalScore = INITIAL_SPAWN_SCORE + b.score;
  1456→
  1457→        let aDies = false;
  1458→        let bDies = false;
  1459→
  1460→        if (aTotalScore === bTotalScore) {
  1461→          // Tie — both die
  1462→          aDies = true;
  1463→          bDies = true;
  1464→        } else if (aBoosting && !bBoosting) {
  1465→          // A is boosting, B is not → A wins (boost advantage)
  1466→          bDies = true;
  1467→        } else if (bBoosting && !aBoosting) {
  1468→          // B is boosting, A is not → B wins (boost advantage)
  1469→          aDies = true;
  1470→        } else {
  1471→          // Both boosting or neither boosting → higher score wins
  1472→          if (aTotalScore > bTotalScore) {
  1473→            bDies = true;
  1474→          } else {
  1475→            aDies = true;
  1476→          }
  1477→        }
  1478→
  1479→        if (aDies && !seenDead.has(a.id)) {
  1480→          deaths.push({ deadId: a.id, killerId: b.id, cause: 'headon' });
  1481→          seenDead.add(a.id);
  1482→        }
  1483→        if (bDies && !seenDead.has(b.id)) {
  1484→          deaths.push({ deadId: b.id, killerId: a.id, cause: 'headon' });
  1485→          seenDead.add(b.id);
  1486→        }
  1487→      }
  1488→    }
  1489→
  1490→    // --- Body collision: head vs foreign non-head segment ---
  1491→    for (const snake of all) {
  1492→      if (snake.isDead) continue;
  1493→      if (snake.points.length === 0) continue;
  1494→      if (now < snake.spawnProtectedUntil) continue;
  1495→      if (seenDead.has(snake.id)) continue;
  1496→      const head = snake.points[0];
  1497→
  1498→      // NO wall collision — infinite map.
  1499→
  1500→      const queryR = snake.size + 30;
  1501→      const nearby = this.grid.queryRadius(head.x, head.y, queryR);
  1502→      for (const item of nearby.values()) {
  1503→        if (item.kind !== 'segment') continue;
  1504→        // Own-collision immunity — skip own segments.
  1505→        if (item.snakeId === snake.id) continue;
  1506→        // Skip head segments (handled by head-on collision above).
  1507→        if (item.segIdx === 0) continue;
  1508→        // Neck protection: skip first N segments behind the head (close-call safety)
  1509→        if (item.segIdx <= 5) continue;
  1510→        const d = dist(head.x, head.y, item.x, item.y);
  1511→        if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
  1512→          deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
  1513→          seenDead.add(snake.id);
  1514→          break;
  1515→        }
  1516→      }
  1517→    }
  1518→    return deaths;
  1519→  }

  1596→    if (newDropFoods.length > 0) {
  1597→      this.foods.push(...newDropFoods);

  1599→
  1600→    // Remove dead bots and respawn to maintain count
  1601→    if (deaths.length > 0) {
  1602→      const toRemove: string[] = [];
  1603→      for (const [id, bot] of this.bots) {
  1604→        if (bot.isDead) toRemove.push(id);
  1605→      }

  1615→    }
  1616→
  1617→    // Replenish food
  1618→    this.replenishFood();
  1619→  }

  1621→  private captureReplaySnapshot(): void {
  1622→    const camX = this.isPostDeathRecording ? this.deathCamX : this.cam.x;
  1623→    const camY = this.isPostDeathRecording ? this.deathCamY : this.cam.y;
  1624→    const camZoom = this.cam.zoom;
  1625→
  1626→    const snakes: ReplaySnakeData[] = [];
  1627→    const allSnakes: SnakeBase[] = [];
  1628→    for (const bot of this.bots.values()) {
  1629→      if (!bot.isDead) allSnakes.push(bot);
  1630→    }
  1631→    if (this.player && !this.player.isDead) allSnakes.push(this.player);
  1632→    if (this.player && this.player.isDead && this.postDeathTicksRemaining > REPLAY_POST_MAX - 30) {
  1633→      allSnakes.push(this.player);
  1634→    }
  1635→
  1636→    for (const snake of allSnakes) {
  1637→      if (snake.points.length === 0) continue;
  1638→      const head = snake.points[0];
  1639→      const d = dist(head.x, head.y, camX, camY);
