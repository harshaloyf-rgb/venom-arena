    82→/** Returns the world-space rectangle visible in the viewport, plus margin. */
    83→export function computeVisibleRect(rc: FrameRenderCtx, marginPx = 100): VisibleRect {
    84→  const halfW = rc.w / 2 / rc.zoom;
    85→  const halfH = rc.h / 2 / rc.zoom;
    86→  const m = marginPx / rc.zoom;
    87→  return {
    88→    left: rc.camX - halfW - m,
    89→    right: rc.camX + halfW + m,
    90→    top: rc.camY - halfH - m,
    91→    bottom: rc.camY + halfH + m,
    92→  };
    93→}
    94→
    95→/** Quick AABB-vs-rect test. */
    96→function rectContainsPoint(rect: VisibleRect, x: number, y: number): boolean {
    97→  return x >= rect.left && x <= rect.right && y >= rect.top && y <= rect.bottom;
    98→}
    99→
   100→/** Returns true if any point in the list is inside the visible rect. */
   101→function snakeIsVisible(pts: { x: number; y: number }[], rect: VisibleRect): boolean {
   102→  if (pts.length === 0) return false;
   103→  if (rectContainsPoint(rect, pts[0].x, pts[0].y)) return true;
   104→  const last = pts[pts.length - 1];
   105→  if (rectContainsPoint(rect, last.x, last.y)) return true;
   106→  const stride = pts.length > 60 ? 4 : 2;
   107→  for (let i = 0; i < pts.length; i += stride) {
   108→    if (rectContainsPoint(rect, pts[i].x, pts[i].y)) return true;
   109→  }
   110→  return false;
   111→}

   113→/**
   114→ * Cache a metallic radial gradient. Bucketed by 2-pixel radius so we don't
   115→ * cache a gradient for every floating-point size value.
   116→ */
   117→function getMetallicGradient(
   118→  rc: FrameRenderCtx,
   119→  radius: number,
   120→  color: string,
   121→  secondary: string | undefined,
   122→): CanvasGradient | string {
   123→  const bucket = Math.max(4, Math.round(radius));
   124→  const key = `${color}:${secondary ?? ''}:${bucket}`;
   125→  const cached = rc.metallicCache.get(key);
   126→  if (cached) return cached;
   127→  const { ctx } = rc;
   128→  const g = ctx.createRadialGradient(-bucket * 0.35, -bucket * 0.35, bucket * 0.1, 0, 0, bucket);
   129→  g.addColorStop(0, '#f8fafc');
   130→  g.addColorStop(0.35, secondary || color);
   131→  g.addColorStop(1, color);
   132→  rc.metallicCache.set(key, g);
   133→  return g;
   134→}

   136→/**
   137→ * Breathing arena radius at a given time. Matches the server's
   138→ * `getMapRadius(elapsedMs)` formula exactly:
   139→ *   baseRadius + sin(cycleProgress * 2π) * amplitude
   140→ */
   141→export function getArenaRadius(now: number): number {
   142→  const cycleTime = (now % MAP_BREATH_CYCLE_MS) / MAP_BREATH_CYCLE_MS;
   143→  const sinVal = Math.sin(cycleTime * Math.PI * 2);
   144→  return MAP_BASE_RADIUS + sinVal * MAP_BREATH_AMPLITUDE;
   145→}

   147→// ---------------------------------------------------------------------------
   148→// Indian number formatting for chip display
   149→// ---------------------------------------------------------------------------
   150→
   151→/**
   152→ * Formats a chip count using Indian numbering system.
   153→ *  - Under 1000: plain number (e.g. "500")
   154→ *  - 1K to 99K: "1.5K", "25K"
   155→ *  - 1L (1,00,000) to 99L: "1L", "50L"
   156→ *  - 1Cr (1,00,00,000)+: "1.2Cr", "15Cr"
   157→ */
   158→export function formatChipDisplay(chips: number): string {
   159→  if (chips < 1000) return String(chips);
   160→  if (chips < 100_000) {
   161→    // K range: 1K to 99.9K
   162→    const k = chips / 1000;
   163→    return k % 1 === 0 ? `${k}K` : `${k.toFixed(1).replace(/\.0$/, '')}K`;
   164→  }
   165→  if (chips < 10_000_000) {
   166→    // L (lakh) range: 1L to 99.9L
   167→    const l = chips / 100_000;
   168→    return l % 1 === 0 ? `${l}L` : `${l.toFixed(1).replace(/\.0$/, '')}L`;
   169→  }
   170→  // Cr (crore) range
   171→  const cr = chips / 10_000_000;
   172→  return cr % 1 === 0 ? `${cr}Cr` : `${cr.toFixed(1).replace(/\.0$/, '')}Cr`;
   173→}

   179→/**
   180→ * Draws the breathing circular arena: background fill (deep slate `#020617`)
   181→ * clipped to the circle, a subtle grid (`#1e293b`, gridSize 60) inside the
   182→ * circle, and the neon-rose boundary (`#f43f5e`, lineWidth 10, shadowBlur 16).
   183→ *
   184→ * The radius breathes using the same sinusoidal formula as the server.
   185→ */
   186→export function drawGrid(rc: FrameRenderCtx): void {
   187→  const { ctx, worldSize, camX, camY, zoom, w, h } = rc;
   188→  const cx = worldSize / 2;
   189→  const cy = worldSize / 2;
   190→  const radius = getArenaRadius(rc.now);
   191→
   192→  // --- Background fill (entire arena disc) ---
   193→  ctx.save();
   194→  ctx.beginPath();
   195→  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   196→  ctx.fillStyle = '#020617';
   197→  ctx.fill();
   198→  ctx.clip();
   199→
   200→  // --- Grid lines (subtle slate-800, gridSize 60) ---
   201→  const grid = 60;
   202→  const vis = computeVisibleRect(rc);
   203→  const startX = Math.max(0, Math.floor(vis.left / grid) * grid);
   204→  const endX = Math.min(worldSize, Math.ceil(vis.right / grid) * grid);
   205→  const startY = Math.max(0, Math.floor(vis.top / grid) * grid);
   206→  const endY = Math.min(worldSize, Math.ceil(vis.bottom / grid) * grid);
   207→
   208→  ctx.strokeStyle = '#1e293b';
   209→  ctx.lineWidth = 1 / zoom;
   210→  ctx.beginPath();
   211→  for (let x = startX; x <= endX; x += grid) {
   212→    ctx.moveTo(x, startY);
   213→    ctx.lineTo(x, endY);
   214→  }
   215→  for (let y = startY; y <= endY; y += grid) {
   216→    ctx.moveTo(startX, y);
   217→    ctx.lineTo(endX, y);
   218→  }
   219→  ctx.stroke();
   220→  ctx.restore(); // remove clip
   221→
   222→  // --- Outer boundary (neon rose, breathing) ---
   223→  ctx.save();
   224→  ctx.lineWidth = 10;
   225→  ctx.strokeStyle = '#f43f5e';
   226→  if (!rc.lowQuality) {
   227→    ctx.shadowColor = '#f43f5e';
   228→    ctx.shadowBlur = 16;
   229→  }
   230→  ctx.beginPath();
   231→  ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   232→  ctx.stroke();
   233→  ctx.shadowBlur = 0;
   234→  ctx.restore();
   235→
   236→  // Silence unused-param lint for camX/camY/w/h.
   237→  void camX;
   238→  void camY;
   239→  void w;
   240→  void h;
   241→}

   244→// Dynamic map boundary (for online games with dynamic radius)

   247→/**
   248→ * Draws a circular map boundary with neon-rose glow and a subtle breathing
   249→ * pulse. Used for online arenas where `mapRadius` comes from the GameSnapshot
   250→ * rather than the fixed breathing formula.
   251→ */
   252→export function drawMapBoundary(
   253→  ctx: CanvasRenderingContext2D,
   254→  centerX: number,
   255→  centerY: number,
   256→  radius: number,
   257→  tick: number,
   258→): void {
   259→  // Subtle breathing pulse: ±3px oscillation over a 4-second cycle
   260→  const breathe = Math.sin(tick * 0.0015) * 3;
   261→  const r = Math.max(100, radius + breathe);
   262→
   263→  ctx.save();
   264→  // Outer glow
   265→  ctx.lineWidth = 10;
   266→  ctx.strokeStyle = '#f43f5e';
   267→  ctx.shadowColor = '#f43f5e';
   268→  ctx.shadowBlur = 20;
   269→  ctx.beginPath();
   270→  ctx.arc(centerX, centerY, r, 0, Math.PI * 2);
   271→  ctx.stroke();
   272→
   273→  // Inner subtle glow ring
   274→  ctx.shadowBlur = 8;
   275→  ctx.lineWidth = 3;
   276→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.35)';
   277→  ctx.beginPath();
   278→  ctx.arc(centerX, centerY, r - 6, 0, Math.PI * 2);
   279→  ctx.stroke();
   280→
   281→  ctx.shadowBlur = 0;
   282→  ctx.restore();
   283→}

   289→/** Orb type configuration for the three visual tiers. */
   290→interface OrbConfig {
   291→  radius: number;
   292→  color: string;
   293→  glowColor: string;
   294→  shadowBlur: number;
   295→  label: 'small' | 'medium' | 'large';
   296→}
   297→
   298→const ORB_CONFIGS: Record<number, OrbConfig> = {
   299→  1: { radius: 3, color: '#34d399', glowColor: '#10b981', shadowBlur: 6, label: 'small' },
   300→  3: { radius: 5, color: '#38bdf8', glowColor: '#0ea5e9', shadowBlur: 10, label: 'medium' },
   301→  5: { radius: 8, color: '#f472b6', glowColor: '#ec4899', shadowBlur: 16, label: 'large' },
   302→};
   303→
   304→/** Default config used when orbSize doesn't match known values. */
   305→const DEFAULT_ORB: OrbConfig = ORB_CONFIGS[1];
   306→
   307→/**
   308→ * Draws a single food orb with glow effects. The orbSize determines which
   309→ * visual tier to use. Falls back to small orb for unknown sizes.
   310→ */
   311→export function drawFoodOrb(
   312→  ctx: CanvasRenderingContext2D,
   313→  x: number,
   314→  y: number,
   315→  orbSize: number,
   316→  value: number,
   317→  color: string,
   318→  glowColor: string,
   319→  now: number,
   320→  lowQuality: boolean,
   321→): void {
   322→  // Select config based on value; use provided color/glowColor if different
   323→  const config = ORB_CONFIGS[value] ?? DEFAULT_ORB;
   324→  const useColor = color || config.color;
   325→  const useGlow = glowColor || config.glowColor;
   326→  const r = Math.max(2, orbSize > 0 ? orbSize : config.radius);
   327→
   328→  ctx.save();
   329→
   330→  // Large orbs get a subtle pulse animation
   331→  let animR = r;
   332→  if (value === 5 && !lowQuality) {
   333→    const pulse = Math.sin(now * 0.004) * 1.5;
   334→    animR = r + pulse;
   335→  }
   336→
   337→  // Glow effect (skip in low quality)
   338→  if (!lowQuality) {
   339→    ctx.shadowColor = useGlow;
   340→    ctx.shadowBlur = config.shadowBlur;
   341→  }
   342→
   343→  // Radial gradient for a lit-from-within look
   344→  const grad = ctx.createRadialGradient(x, y, 0, x, y, animR);
   345→  grad.addColorStop(0, '#ffffff');
   346→  grad.addColorStop(0.3, useColor);
   347→  grad.addColorStop(1, useGlow);
   348→
   349→  ctx.beginPath();
   350→  ctx.arc(x, y, animR, 0, Math.PI * 2);
   351→  ctx.fillStyle = grad;
   352→  ctx.fill();
   353→
   354→  ctx.shadowBlur = 0;
   355→  ctx.restore();
   356→}

   358→// ---------------------------------------------------------------------------
   359→// Star collectible rendering — 5-pointed star shape
   360→// ---------------------------------------------------------------------------
   361→
   362→/**
   363→ * Draws a proper 5-pointed star path. Generic helper used by drawStarCollectible.
   364→ *
   365→ * @param ctx     Canvas context
   366→ * @param cx      Center X
   367→ * @param cy      Center Y
   368→ * @param outerR  Outer radius (tip of star)
   369→ * @param innerR  Inner radius (valley between points)
   370→ * @param points  Number of points (always 5 for stars)
   371→ * @param rotation Rotation in radians (for spin animation)
   372→ */
   373→export function drawStarShape(
   374→  ctx: CanvasRenderingContext2D,
   375→  cx: number,
   376→  cy: number,
   377→  outerR: number,
   378→  innerR: number,
   379→  points: number,
   380→  rotation: number,
   381→): void {
   382→  ctx.beginPath();
   383→  for (let i = 0; i < points * 2; i++) {
   384→    const r = i % 2 === 0 ? outerR : innerR;
   385→    const angle = (Math.PI / points) * i - Math.PI / 2 + rotation;
   386→    const px = cx + Math.cos(angle) * r;
   387→    const py = cy + Math.sin(angle) * r;
   388→    if (i === 0) ctx.moveTo(px, py);
   389→    else ctx.lineTo(px, py);
   390→  }
   391→  ctx.closePath();
   392→}

   394→/**
   395→ * Draws a 5-pointed star collectible with golden glow, subtle rotation,
   396→ * and pulsing glow effect. Stars are the rare high-value pickups.
   397→ */
   398→export function drawStarCollectible(
   399→  ctx: CanvasRenderingContext2D,
   400→  x: number,
   401→  y: number,
   402→  size: number,
   403→  tick: number,
   404→  lowQuality: boolean,
   405→): void {
   406→  const outerR = Math.max(4, size);
   407→  const innerR = Math.max(2, size * 0.4);
   408→
   409→  // Rotation: slow spin based on tick
   410→  const rotation = tick * 0.002;
   411→
   412→  // Pulsing glow: size oscillates ±1.5px over ~1.5s
   413→  const pulse = lowQuality ? 0 : Math.sin(tick * 0.004) * 1.5;
   414→  const animOuter = outerR + pulse;
   415→  const animInner = innerR + pulse * 0.4;
   416→
   417→  ctx.save();
   418→
   419→  // Glow
   420→  if (!lowQuality) {
   421→    ctx.shadowColor = '#f59e0b';
   422→    ctx.shadowBlur = 12 + Math.sin(tick * 0.003) * 4;
   423→  }
   424→
   425→  // Star shape
   426→  drawStarShape(ctx, x, y, animOuter, animInner, 5, rotation);
   427→
   428→  // Golden radial gradient fill
   429→  const grad = ctx.createRadialGradient(x, y, 0, x, y, animOuter);
   430→  grad.addColorStop(0, '#fef3c7'); // bright center
   431→  grad.addColorStop(0.4, '#fbbf24');
   432→  grad.addColorStop(1, '#f59e0b');
   433→  ctx.fillStyle = grad;
   434→  ctx.fill();
   435→
   436→  ctx.shadowBlur = 0;
   437→  ctx.restore();
   438→}

   440→// ---------------------------------------------------------------------------
   441→// Extraction ring — white-to-green circle near an extracting snake's head
   442→// ---------------------------------------------------------------------------
   443→
   444→/**
   445→ * Draws the extraction progress ring around a snake's head.
   446→ * Visible to ALL players as a warning that someone is banking chips.
   447→ */
   448→export function drawExtractionRing(
   449→  ctx: CanvasRenderingContext2D,
   450→  x: number,
   451→  y: number,
   452→  snakeSize: number,
   453→  progress: number, // 0..1
   454→  zoom: number,
   455→): void {
   456→  if (progress <= 0 || progress > 1) return;
   457→
   458→  const ringRadius = Math.max(8, snakeSize + 10 / zoom);
   459→  const lineWidth = Math.max(2, 3 / zoom);
   460→
   461→  ctx.save();
   462→
   463→  // Background ring (track)
   464→  ctx.beginPath();
   465→  ctx.arc(x, y, ringRadius, 0, Math.PI * 2);
   466→  ctx.strokeStyle = 'rgba(255, 255, 255, 0.15)';
   467→  ctx.lineWidth = lineWidth;
   468→  ctx.stroke();
   469→
   470→  // Progress arc (white → green gradient)
   471→  const startAngle = -Math.PI / 2;
   472→  const endAngle = startAngle + progress * Math.PI * 2;
   473→  ctx.beginPath();
   474→  ctx.arc(x, y, ringRadius, startAngle, endAngle);
   475→
   476→  // Color: white at start → emerald green at end
   477→  const r = Math.round(255 - progress * 227); // 255 → 28
   478→  const g = Math.round(255 - progress * 55);  // 255 → 200
   479→  const b = Math.round(255 - progress * 215); // 255 → 40
   480→  ctx.strokeStyle = `rgba(${r}, ${g}, ${b}, 0.9)`;
   481→  ctx.lineWidth = lineWidth + 1 / zoom;
   482→  ctx.lineCap = 'round';
   483→  ctx.stroke();
   484→
   485→  // Glow
   486→  if (!lowQualityCheck(ctx)) {
   487→    ctx.shadowColor = `rgba(${r}, ${g}, ${b}, 0.5)`;
   488→    ctx.shadowBlur = 8 / zoom;
   489→    ctx.beginPath();
   490→    ctx.arc(x, y, ringRadius, startAngle, endAngle);
   491→    ctx.strokeStyle = `rgba(${r}, ${g}, ${b}, 0.4)`;
   492→    ctx.lineWidth = lineWidth;
   493→    ctx.stroke();
   494→    ctx.shadowBlur = 0;
   495→  }
   496→
   497→  ctx.restore();
   498→}

   505→// ---------------------------------------------------------------------------
   506→// Food (main entry point — batches regular orbs, draws stars individually)
   507→// ---------------------------------------------------------------------------
   508→
   509→/**
   510→ * Draws all food in two passes:
   511→ *  1. Regular food orbs — batched by orbSize tier into Path2D groups.
   512→ *     Each tier (small/medium/large) has its own glow gradient.
   513→ *  2. Star collectibles — drawn individually as 5-pointed gold stars with
   514→ *     rotation + pulsing glow.
   515→ */
   516→export function drawFood(rc: FrameRenderCtx, foods: FoodSnapshot[]): void {
   517→  const { ctx, lowQuality } = rc;
   518→  const vis = computeVisibleRect(rc);
   519→
   520→  // --- Regular food orbs: batch by value tier ---
   521→  const smallOrbs: FoodSnapshot[] = [];
   522→  const mediumOrbs: FoodSnapshot[] = [];
   523→  const largeOrbs: FoodSnapshot[] = [];
   524→  const starChips: FoodSnapshot[] = [];
   525→
   526→  for (let i = 0; i < foods.length; i++) {
   527→    const f = foods[i];
   528→    if (f.x < vis.left || f.x > vis.right || f.y < vis.top || f.y > vis.bottom) {
   529→      continue; // culled
   530→    }
   531→    if (f.isStarChip) {
   532→      starChips.push(f);
   533→      continue;
   534→    }
   535→    // Route to orb tier by value
   536→    if (f.value >= 5) largeOrbs.push(f);
   537→    else if (f.value >= 3) mediumOrbs.push(f);
   538→    else smallOrbs.push(f);
   539→  }
   540→
   541→  // Draw each orb tier
   542→  const drawOrbBatch = (orbs: FoodSnapshot[]) => {
   543→    for (let i = 0; i < orbs.length; i++) {
   544→      const f = orbs[i];
   545→      drawFoodOrb(ctx, f.x, f.y, f.orbSize ? f.size : f.size, f.value, f.color, f.glowColor ?? '', rc.now, lowQuality);
   546→    }
   547→  };
   548→
   549→  drawOrbBatch(smallOrbs);
   550→  drawOrbBatch(mediumOrbs);
   551→  drawOrbBatch(largeOrbs);
   552→
   553→  // --- Star collectibles: 5-pointed gold stars with rotation + glow + value label ---
   554→  if (starChips.length > 0) {
   555→    ctx.save();
   556→    for (let i = 0; i < starChips.length; i++) {
   557→      const f = starChips[i];
   558→      drawStarCollectible(ctx, f.x, f.y, Math.max(6, f.size + 4), rc.now, lowQuality);
   559→      // Draw chip value label inside the star
   560→      if (f.value > 0) {
   561→        const starRadius = Math.max(4, f.size + 4);
   562→        const labelSize = Math.max(7, Math.min(11, starRadius * 0.55));
   563→        ctx.font = `bold ${labelSize}px monospace`;
   564→        ctx.textAlign = 'center';
   565→        ctx.textBaseline = 'middle';
   566→        ctx.fillStyle = '#7c2d12'; // dark brown, readable inside golden star
   567→        const label = f.value >= 1000 ? `${Math.round(f.value / 1000)}k` : `${Math.round(f.value)}`;
   568→        ctx.fillText(label, f.x, f.y + 0.5);
   569→      }
   570→    }
   571→    ctx.restore();
   572→  }
   573→}
   574→
   575→// ---------------------------------------------------------------------------
   576→// Snake rendering
   577→// ---------------------------------------------------------------------------
   578→
   579→/**
   580→ * Draws a single snake. The body is rendered as ONE thick stroked polyline
   581→ * (with an outline underlay). The head is drawn separately with eyes and
   582→ * (for the player's own snake, high-quality only) a glow halo.
   583→ *
   584→ * Player skin patterns (rainbow / neon / metallic / camo) are applied when
   585→ * `rc.playerSkin.pattern` is set and the snake is the local player.
   586→ *
   587→ * @param opacity  Optional global alpha override (0..1). Default = 1.
   588→ */
   589→export function drawSnake(rc: FrameRenderCtx, snake: SnakeSnapshot, opacity?: number): void {
   590→  const pts = snake.points;
   591→  if (!pts || pts.length === 0) return;
   592→
   593→  const vis = computeVisibleRect(rc);
   594→  if (!snakeIsVisible(pts, vis)) return;
   595→
   596→  const { ctx, zoom, lowQuality } = rc;
   597→  const isMe = snake.id === rc.myId;
   598→  const baseAlpha = opacity ?? 1;
   599→
   600→  // Downsample long snakes
   601→  const stride = pts.length > 60 ? 2 : 1;
   602→
   603→  // Body width in world units
   604→  const radius = Math.max(2, snake.size);
   605→  const width = radius * 2;
   606→
   607→  ctx.save();
   608→  ctx.globalAlpha = baseAlpha;
   609→
   610→  // --- Outline (underlay) ---
   611→  if (snake.secondaryColor) {
   612→    ctx.save();
   613→    ctx.lineCap = 'round';
   614→    ctx.lineJoin = 'round';
   615→    ctx.lineWidth = width + 4 / zoom;
   616→    ctx.strokeStyle = snake.secondaryColor;
   617→    ctx.globalAlpha = baseAlpha * 0.55;
   618→    ctx.beginPath();
   619→    ctx.moveTo(pts[0].x, pts[0].y);
   620→    for (let i = stride; i < pts.length; i += stride) {
   621→      ctx.lineTo(pts[i].x, pts[i].y);
   622→    }
   623→    ctx.stroke();
   624→    ctx.restore();
   625→  }
   626→
   627→  // --- Body ---
   628→  ctx.save();
   629→  ctx.globalAlpha = baseAlpha;
   630→  ctx.lineCap = 'round';
   631→  ctx.lineJoin = 'round';
   632→  ctx.lineWidth = width;
   633→
   634→  const pattern = isMe ? rc.playerSkin?.pattern : undefined;
   635→  if (pattern === 'metallic' && snake.secondaryColor) {
   636→    const g = getMetallicGradient(rc, radius, snake.color, snake.secondaryColor);
   637→    ctx.strokeStyle = g;
   638→    ctx.beginPath();
   639→    ctx.moveTo(pts[0].x, pts[0].y);
   640→    for (let i = stride; i < pts.length; i += stride) {
   641→      ctx.lineTo(pts[i].x, pts[i].y);
   642→    }
   643→    ctx.stroke();
   644→  } else if (pattern === 'rainbow' && !lowQuality) {
   645→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   646→    ctx.lineWidth = width;
   647→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   648→      const hue = (rc.now * 0.05 + i * 14) % 360;
   649→      ctx.strokeStyle = `hsl(${hue}, 90%, 55%)`;
   650→      ctx.beginPath();
   651→      ctx.moveTo(pts[i].x, pts[i].y);
   652→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   653→      for (let j = i + stride; j <= end; j += stride) {
   654→        ctx.lineTo(pts[j].x, pts[j].y);
   655→      }
   656→      ctx.stroke();
   657→    }
   658→  } else if (pattern === 'neon' && !lowQuality) {
   659→    const chunkSize = Math.max(4, Math.floor(pts.length / 10));
   660→    ctx.lineWidth = width;
   661→    let chunkIndex = 0;
   662→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   663→      const ratio = (Math.sin(rc.now * 0.009 - chunkIndex * 0.28) + 1) / 2;
   664→      ctx.strokeStyle = ratio > 0.5 ? '#06b6d4' : '#a855f7';
   665→      ctx.beginPath();
   666→      ctx.moveTo(pts[i].x, pts[i].y);
   667→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   668→      for (let j = i + stride; j <= end; j += stride) {
   669→        ctx.lineTo(pts[j].x, pts[j].y);
   670→      }
   671→      ctx.stroke();
   672→      chunkIndex += 1;
   673→    }
   674→  } else if (pattern === 'camo') {
   675→    const camoColors = ['#15803d', '#854d0e', '#78350f', '#166534'];
   676→    const chunkSize = Math.max(4, Math.floor(pts.length / 12));
   677→    let chunkIndex = 0;
   678→    for (let i = 0; i < pts.length - 1; i += chunkSize) {
   679→      ctx.strokeStyle = camoColors[chunkIndex % camoColors.length];
   680→      ctx.beginPath();
   681→      ctx.moveTo(pts[i].x, pts[i].y);
   682→      const end = Math.min(pts.length - 1, i + chunkSize + 1);
   683→      for (let j = i + stride; j <= end; j += stride) {
   684→        ctx.lineTo(pts[j].x, pts[j].y);
   685→      }
   686→      ctx.stroke();
   687→      chunkIndex += 1;
   688→    }
   689→  } else {
   690→    // Default: solid snake.color
   691→    ctx.strokeStyle = snake.color;
   692→    ctx.beginPath();
   693→    ctx.moveTo(pts[0].x, pts[0].y);
   694→    for (let i = stride; i < pts.length; i += stride) {
   695→      ctx.lineTo(pts[i].x, pts[i].y);
   696→    }
   697→    ctx.stroke();
   698→  }
   699→  ctx.restore();
   700→
   701→  // --- Head ---
   702→  const head = pts[0];
   703→  ctx.save();
   704→  ctx.globalAlpha = baseAlpha;
   705→  if (isMe && !lowQuality) {
   706→    ctx.shadowColor = snake.color;
   707→    ctx.shadowBlur = 14;
   708→  }
   709→  ctx.beginPath();
   710→  ctx.arc(head.x, head.y, radius, 0, Math.PI * 2);
   711→  ctx.fillStyle = snake.color;
   712→  ctx.fill();
   713→  if (isMe && !lowQuality) ctx.shadowBlur = 0;
   714→
   715→  // Eyes
   716→  const eyeOffset = radius * 0.45;
   717→  const eyeR = Math.max(1.5, radius * 0.32);
   718→  const pupilR = Math.max(0.8, radius * 0.18);
   719→  const angle = snake.angle;
   720→  const perp = angle + Math.PI / 2;
   721→  const ex1 = head.x + Math.cos(angle) * eyeOffset * 0.4 + Math.cos(perp) * eyeOffset;
   722→  const ey1 = head.y + Math.sin(angle) * eyeOffset * 0.4 + Math.sin(perp) * eyeOffset;
   723→  const ex2 = head.x + Math.cos(angle) * eyeOffset * 0.4 - Math.cos(perp) * eyeOffset;
   724→  const ey2 = head.y + Math.sin(angle) * eyeOffset * 0.4 - Math.sin(perp) * eyeOffset;
   725→
   726→  ctx.fillStyle = '#ffffff';
   727→  ctx.beginPath();
   728→  ctx.arc(ex1, ey1, eyeR, 0, Math.PI * 2);
   729→  ctx.arc(ex2, ey2, eyeR, 0, Math.PI * 2);
   730→  ctx.fill();
   731→
   732→  ctx.fillStyle = '#0a0a0a';
   733→  ctx.beginPath();
   734→  ctx.arc(ex1 + Math.cos(angle) * pupilR, ey1 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   735→  ctx.arc(ex2 + Math.cos(angle) * pupilR, ey2 + Math.sin(angle) * pupilR, pupilR, 0, Math.PI * 2);
   736→  ctx.fill();
   737→
   738→  // Spawn-protection ring
   739→  if (snake.spawnProtected) {
   740→    ctx.strokeStyle = 'rgba(255,255,255,0.6)';
   741→    ctx.lineWidth = 2 / zoom;
   742→    ctx.beginPath();
   743→    ctx.arc(head.x, head.y, radius + 4 / zoom, 0, Math.PI * 2);
   744→    ctx.stroke();
   745→  }
   746→
   747→  // Extraction ring
   748→  if (snake.isExtracting) {
   749→    ctx.strokeStyle = 'rgba(16, 185, 129, 0.95)';
   750→    ctx.lineWidth = 3 / zoom;
   751→    ctx.setLineDash([6 / zoom, 4 / zoom]);
   752→    ctx.beginPath();
   753→    ctx.arc(head.x, head.y, radius + 8 / zoom, 0, Math.PI * 2);
   754→    ctx.stroke();
   755→    ctx.setLineDash([]);
   756→  }
   757→  ctx.restore();
   758→
   759→  // --- Name label ---
   760→  if (snake.name) {
   761→    ctx.save();
   762→    ctx.globalAlpha = baseAlpha;
   763→    ctx.font = `${Math.max(10, 12 / zoom)}px monospace`;
   764→    ctx.textAlign = 'center';
   765→    ctx.textBaseline = 'bottom';
   766→    const labelY = head.y - radius - 6 / zoom;
   767→    ctx.fillStyle = isMe ? '#22c55e' : 'rgba(226, 232, 240, 0.85)';
   768→    ctx.fillText(snake.name, head.x, labelY);
   769→    if (snake.userTag) {
   770→      ctx.fillStyle = 'rgba(148, 163, 184, 0.7)';
   771→      ctx.font = `${Math.max(8, 9 / zoom)}px monospace`;
   772→      ctx.fillText(snake.userTag, head.x, labelY - 12 / zoom);
   773→    }
   774→    ctx.restore();
   775→  }
   776→
   777→  // --- Chat bubble ---
   778→  if (snake.chatMessage) {
   779→    ctx.save();
   780→    ctx.globalAlpha = baseAlpha;
   781→    drawChatBubble(rc, head.x, head.y - radius - 24 / zoom, snake.chatMessage);
   782→    ctx.restore();
   783→  }
   784→
   785→  ctx.restore(); // outer globalAlpha save
   786→}
   787→
   788→// ---------------------------------------------------------------------------
   789→// Snake opacity layering system
   790→// ---------------------------------------------------------------------------
   791→
   792→/** Proximity threshold for opacity fade (world-space pixels). */
   793→const LAYERING_PROXIMITY = 30;
   794→
   795→/**
   796→ * Draws a snake with opacity layering: when a smaller snake passes close to a
   797→ * larger snake, the larger snake fades to 75% opacity. This creates visual
   798→ * depth when snakes overlap spatially.
   799→ *
   800→ * Logic:
   801→ *  1. For each OTHER snake, check if its head is within ~30px of this snake's head
   802→ *  2. If a smaller snake (by size) is nearby, this snake fades to 0.75 opacity
   803→ *  3. Otherwise, render at full opacity
   804→ */
   805→export function drawSnakeWithLayering(
   806→  rc: FrameRenderCtx,
   807→  snake: SnakeSnapshot,
   808→  allSnakes: SnakeSnapshot[],
   809→): void {
   810→  const pts = snake.points;
   811→  if (!pts || pts.length === 0) return;
   812→
   813→  const myHead = pts[0];
   814→  let shouldFade = false;
   815→
   816→  for (let i = 0; i < allSnakes.length; i++) {
   817→    const other = allSnakes[i];
   818→    if (other.id === snake.id) continue;
   819→    const otherPts = other.points;
   820→    if (!otherPts || otherPts.length === 0) continue;
   821→
   822→    const otherHead = otherPts[0];
   823→    const dist = Math.hypot(otherHead.x - myHead.x, otherHead.y - myHead.y);
   824→
   825→    if (dist < LAYERING_PROXIMITY && snake.size > other.size) {
   826→      // A smaller snake is close by — fade this (larger) snake
   827→      shouldFade = true;
   828→      break;
   829→    }
   830→  }
   831→
   832→  drawSnake(rc, snake, shouldFade ? 0.75 : 1);
   833→}

   835→// ---------------------------------------------------------------------------
   836→// Chip label above snake head
   837→// ---------------------------------------------------------------------------
   838→
   839→/**
   840→ * Draws a chip count label above the snake's head. Uses Indian numbering
   841→ * (K, L, Cr). Only meaningful for real players (bots have 0 carriedChips).
   842→ *
   843→ * Rendered as a semi-transparent pill with white text, positioned centered
   844→ * above the head.
   845→ */
   846→export function drawChipLabel(
   847→  ctx: CanvasRenderingContext2D,
   848→  x: number,
   849→  y: number,
   850→  chips: number,
   851→  snakeSize: number,
   852→  zoom: number,
   853→): void {
   854→  if (chips <= 0) return; // bots have 0 chips, skip them
   855→
   856→  const label = formatChipDisplay(chips);
   857→  const fontSize = Math.max(9, 10 / zoom);
   858→  const offset = snakeSize + 12 / zoom;
   859→
   860→  ctx.save();
   861→  ctx.font = `bold ${fontSize}px monospace`;
   862→  ctx.textAlign = 'center';
   863→  ctx.textBaseline = 'bottom';
   864→
   865→  const metrics = ctx.measureText(label);
   866→  const padX = 5 / zoom;
   867→  const padY = 2 / zoom;
   868→  const tw = metrics.width + padX * 2;
   869→  const th = fontSize + padY * 2;
   870→  const rx = x - tw / 2;
   871→  const ry = y - offset - th;
   872→  const rr = 4 / zoom;
   873→
   874→  // Background pill
   875→  ctx.fillStyle = 'rgba(0, 0, 0, 0.65)';
   876→  ctx.beginPath();
   877→  ctx.moveTo(rx + rr, ry);
   878→  ctx.lineTo(rx + tw - rr, ry);
   879→  ctx.arcTo(rx + tw, ry, rx + tw, ry + rr, rr);
   880→  ctx.lineTo(rx + tw, ry + th - rr);
   881→  ctx.arcTo(rx + tw, ry + th, rx + tw - rr, ry + th, rr);
   882→  ctx.lineTo(rx + rr, ry + th);
   883→  ctx.arcTo(rx, ry + th, rx, ry + th - rr, rr);
   884→  ctx.lineTo(rx, ry + rr);
   885→  ctx.arcTo(rx, ry, rx + rr, ry, rr);
   886→  ctx.closePath();
   887→  ctx.fill();
   888→
   889→  // Gold accent border
   890→  ctx.strokeStyle = 'rgba(251, 191, 36, 0.7)';
   891→  ctx.lineWidth = 1 / zoom;
   892→  ctx.stroke();
   893→
   894→  // Text
   895→  ctx.fillStyle = '#ffffff';
   896→  ctx.textBaseline = 'middle';
   897→  ctx.fillText(label, x, ry + th / 2);
   898→
   899→  ctx.restore();
   900→}

   907→function drawChatBubble(
   908→  rc: FrameRenderCtx,

   910→  y: number,
   911→  msg: string,

   919→  const padY = 3 / zoom;
   920→  const tw = metrics.width + padX * 2;
   921→  const th = 14 / zoom + padY * 2;
   922→  ctx.fillStyle = 'rgba(15, 23, 42, 0.92)';
   923→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.6)';
   924→  ctx.lineWidth = 1.5 / zoom;
   925→  const rx = x - tw / 2;

   927→  const rr = 4 / zoom;

   932→  ctx.lineTo(rx + tw, ry + th - rr);
   933→  ctx.arcTo(rx + tw, ry + th, rx + tw - rr, ry + th, rr);
   934→  ctx.lineTo(rx + rr, ry + th);

   943→  ctx.textBaseline = 'middle';
   944→  ctx.fillText(text, x, y - th / 2);
   945→  ctx.restore();
   946→}

   952→/** Draws the particle array (caller caps at MAX_PARTICLES = 200). */
   953→export function drawParticles(rc: FrameRenderCtx, particles: Particle[]): void {
   954→  if (particles.length === 0) return;
   955→  const { ctx, zoom } = rc;
   956→  ctx.save();
   957→  ctx.globalCompositeOperation = 'lighter';
   958→  for (let i = 0; i < particles.length; i++) {
   959→    const p = particles[i];
   960→    if (p.life <= 0) continue;
   961→    const alpha = Math.max(0, Math.min(1, p.life / p.maxLife));
   962→    ctx.globalAlpha = alpha;
   963→    ctx.fillStyle = p.color;
   964→    ctx.beginPath();
   965→    ctx.arc(p.x, p.y, p.size / zoom, 0, Math.PI * 2);
   966→    ctx.fill();
   967→  }
   968→  ctx.globalAlpha = 1;
   969→  ctx.globalCompositeOperation = 'source-over';
   970→  ctx.restore();
   971→}

   973→// ---------------------------------------------------------------------------
   974→// Minimap (drawn in screen-space, no camera transform)
   975→// ---------------------------------------------------------------------------
   976→
   977→export interface MinimapArgs {
   978→  ctx: CanvasRenderingContext2D;
   979→  /** Top-left of minimap in CSS pixels. */
   980→  x: number;
   981→  y: number;
   982→  /** Minimap size (square). */
   983→  size: number;
   984→  worldSize: number;
   985→  /** Current breathing arena radius (world units). */
   986→  arenaRadius: number;
   987→  snakes: SnakeSnapshot[];
   988→  myId: string;
   989→  /**
   990→   * World-space radius around the player the radar should cover. Snakes
   991→   * outside this range are NOT rendered on the small corner minimap.
   992→   * Defaults to WORLD_SIZE/2 (legacy full-radar behaviour) when omitted.
   993→   */
   994→  range?: number;
   995→}
   996→
   997→/**
   998→ * Draws the circular radar minimap. Player is rendered as an indigo dot at
   999→ * the centre (the radar is centered on the player's head). The arena
  1000→ * boundary is a dashed rose circle. Bots render as rose dots, other real
  1001→ * players as emerald dots — matches AUDIT-A radar visual language.
  1002→ */
  1003→export function drawMinimap(args: MinimapArgs): void {
  1004→  const { ctx, x, y, size, worldSize, arenaRadius, snakes, myId } = args;
  1005→  const cx = x + size / 2;
  1006→  const cy = y + size / 2;
  1007→  const r = size / 2;
  1008→  const radarRange = args.range ?? WORLD_SIZE / 2;
  1009→  const scale = r / radarRange;
  1010→
  1011→  ctx.save();
  1012→  // Background circle (radar disc)
  1013→  ctx.beginPath();
  1014→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  1015→  ctx.fillStyle = 'rgba(2, 6, 23, 0.85)';
  1016→  ctx.fill();
  1017→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.4)';
  1018→  ctx.lineWidth = 1.5;
  1019→  ctx.stroke();
  1020→
  1021→  // Clip to minimap circle.
  1022→  ctx.beginPath();
  1023→  ctx.arc(cx, cy, r, 0, Math.PI * 2);
  1024→  ctx.clip();
  1025→
  1026→  // Concentric rings.
  1027→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
  1028→  ctx.lineWidth = 1;
  1029→  for (const inset of [2, 5, 8]) {
  1030→    ctx.beginPath();
  1031→    ctx.arc(cx, cy, r - inset, 0, Math.PI * 2);
  1032→    ctx.stroke();
  1033→  }
  1034→
  1035→  // Crosshairs.
  1036→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.4)';
  1037→  ctx.beginPath();
  1038→  ctx.moveTo(cx, y);
  1039→  ctx.lineTo(cx, y + size);
  1040→  ctx.moveTo(x, cy);
  1041→  ctx.lineTo(x + size, cy);
  1042→  ctx.stroke();
  1043→
  1044→  // Arena boundary (dashed rose) — centered on world center.
  1045→  const mySnake = snakes.find((s) => s.id === myId);
  1046→  const myHead = mySnake?.points?.[0];
  1047→  const px = myHead ? myHead.x : worldSize / 2;
  1048→  const py = myHead ? myHead.y : worldSize / 2;
  1049→  const worldCenterOffsetX = (worldSize / 2 - px) * scale;
  1050→  const worldCenterOffsetY = (worldSize / 2 - py) * scale;
  1051→  const arenaR = arenaRadius * scale;
  1052→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.6)';
  1053→  ctx.lineWidth = 1.2;
  1054→  ctx.setLineDash([4, 4]);
  1055→  ctx.beginPath();
  1056→  ctx.arc(cx + worldCenterOffsetX, cy + worldCenterOffsetY, arenaR, 0, Math.PI * 2);
  1057→  ctx.stroke();
  1058→  ctx.setLineDash([]);
  1059→
  1060→  // Snakes (player centered → drawn at centre; others offset).
  1061→  for (let i = 0; i < snakes.length; i++) {
  1062→    const s = snakes[i];
  1063→    if (!s.points || s.points.length === 0) continue;
  1064→    const head = s.points[0];
  1065→    const dx = (head.x - px) * scale;
  1066→    const dy = (head.y - py) * scale;
  1067→    const dist = Math.hypot(dx, dy);
  1068→    if (dist > r) continue;
  1069→    const mx = cx + dx;
  1070→    const my = cy + dy;
  1071→    ctx.beginPath();
  1072→    if (s.id === myId) {
  1073→      ctx.arc(mx, my, 3, 0, Math.PI * 2);
  1074→      ctx.fillStyle = '#818cf8'; // indigo-400
  1075→    } else if (s.isBot) {
  1076→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
  1077→      ctx.fillStyle = '#f43f5e'; // rose-500
  1078→    } else {
  1079→      ctx.arc(mx, my, 2, 0, Math.PI * 2);
  1080→      ctx.fillStyle = '#34d399'; // emerald-400
  1081→    }
  1082→    ctx.fill();
  1083→  }
  1084→  ctx.restore();
  1085→}

  1087→// ---------------------------------------------------------------------------
  1088→// Full Map overlay — drawn in screen-space, no camera transform.
  1089→// Press `M` to toggle. Shows the ENTIRE arena with all snakes as dots.
  1090→// ---------------------------------------------------------------------------
  1091→
  1092→export interface FullMapArgs {
  1093→  ctx: CanvasRenderingContext2D;
  1094→  /** Canvas CSS-pixel width. */
  1095→  w: number;
  1096→  /** Canvas CSS-pixel height. */
  1097→  h: number;
  1098→  worldSize: number;
  1099→  /** Current breathing arena radius (world units). */
  1100→  arenaRadius: number;
  1101→  snakes: SnakeSnapshot[];
  1102→  myId: string;
  1103→}
  1104→
  1105→/**
  1106→ * Draws a full-screen overlay of the entire arena. The arena circle is
  1107→ * centered on screen and scaled to fit the smaller of (w, h) with padding.
  1108→ * All snakes are rendered as dots — player = indigo (larger), bots = rose,
  1109→ * other real humans = emerald. A "Press M to close" hint is drawn at the
  1110→ * bottom. The player's own dot pulses with a ring.
  1111→ */
  1112→export function drawFullMap(args: FullMapArgs): void {
  1113→  const { ctx, w, h, worldSize, arenaRadius, snakes, myId } = args;
  1114→  ctx.save();
  1115→  ctx.setTransform(1, 0, 0, 1, 0, 0);
  1116→
  1117→  // --- Background fill (dim slate) ---
  1118→  ctx.fillStyle = 'rgba(2, 6, 23, 0.94)';
  1119→  ctx.fillRect(0, 0, w, h);
  1120→
  1121→  // --- Layout: arena centred on screen ---
  1122→  const cx = w / 2;
  1123→  const cy = h / 2;
  1124→  const margin = 80;
  1125→  const fitDim = Math.min(w, h) - margin * 2;
  1126→  const arenaDiameter = arenaRadius * 2;
  1127→  const scale = fitDim / arenaDiameter;
  1128→  const screenR = arenaRadius * scale;
  1129→
  1130→  const wcx = worldSize / 2;
  1131→  const wcy = worldSize / 2;
  1132→  const toScreenX = (wx: number) => cx + (wx - wcx) * scale;
  1133→  const toScreenY = (wy: number) => cy + (wy - wcy) * scale;
  1134→
  1135→  // --- Concentric range rings (faint) ---
  1136→  ctx.strokeStyle = 'rgba(99, 102, 241, 0.10)';
  1137→  ctx.lineWidth = 1;
  1138→  for (const frac of [0.25, 0.5, 0.75]) {
  1139→    ctx.beginPath();
  1140→    ctx.arc(cx, cy, screenR * frac, 0, Math.PI * 2);
  1141→    ctx.stroke();
  1142→  }
  1143→  // Crosshairs.
  1144→  ctx.strokeStyle = 'rgba(15, 23, 42, 0.5)';
  1145→  ctx.beginPath();
  1146→  ctx.moveTo(cx, cy - screenR);
  1147→  ctx.lineTo(cx, cy + screenR);
  1148→  ctx.moveTo(cx - screenR, cy);
  1149→  ctx.lineTo(cx + screenR, cy);
  1150→  ctx.stroke();
  1151→
  1152→  // --- Arena boundary (dashed rose) ---
  1153→  ctx.strokeStyle = 'rgba(244, 63, 94, 0.8)';
  1154→  ctx.lineWidth = 2;
  1155→  ctx.setLineDash([6, 6]);
  1156→  ctx.beginPath();
  1157→  ctx.arc(cx, cy, screenR, 0, Math.PI * 2);
  1158→  ctx.stroke();
  1159→  ctx.setLineDash([]);
  1160→
  1161→  // --- Snakes ---
  1162→  for (let i = 0; i < snakes.length; i++) {
  1163→    const s = snakes[i];
  1164→    if (!s.points || s.points.length === 0) continue;
  1165→    const head = s.points[0];
  1166→    const sx = toScreenX(head.x);
  1167→    const sy = toScreenY(head.y);
  1168→    ctx.beginPath();
  1169→    if (s.id === myId) {
  1170→      ctx.arc(sx, sy, 5, 0, Math.PI * 2);
  1171→      ctx.fillStyle = '#818cf8';
  1172→      ctx.fill();
  1173→      ctx.strokeStyle = 'rgba(129, 140, 248, 0.7)';
  1174→      ctx.lineWidth = 2;
  1175→      ctx.beginPath();
  1176→      ctx.arc(sx, sy, 9, 0, Math.PI * 2);
  1177→      ctx.stroke();
  1178→    } else if (s.isBot) {
  1179→      ctx.arc(sx, sy, 2.5, 0, Math.PI * 2);
  1180→      ctx.fillStyle = '#f43f5e';
  1181→      ctx.fill();
  1182→    } else {
  1183→      ctx.arc(sx, sy, 3, 0, Math.PI * 2);
  1184→      ctx.fillStyle = '#34d399';
  1185→      ctx.fill();
  1186→    }
  1187→  }
  1188→
  1189→  // --- Title + close hint ---
  1190→  ctx.fillStyle = 'rgba(226, 232, 240, 0.95)';
  1191→  ctx.font = 'bold 14px monospace';
  1192→  ctx.textAlign = 'center';
  1193→  ctx.textBaseline = 'top';
  1194→  ctx.fillText('ARENA OVERVIEW — ALL SNAKES', cx, 16);
  1195→  ctx.fillStyle = 'rgba(148, 163, 184, 0.85)';
  1196→  ctx.font = '11px monospace';
  1197→  ctx.fillText('Press M to close', cx, h - 24);
  1198→
  1199→  // Legend (top-left).
  1200→  ctx.textAlign = 'left';
  1201→  ctx.font = '11px monospace';
  1202→  const legendX = 20;
  1203→  let legendY = 20;
  1204→  const drawLegend = (color: string, label: string, dotR: number) => {
  1205→    ctx.beginPath();
  1206→    ctx.arc(legendX + 6, legendY + 6, dotR, 0, Math.PI * 2);
  1207→    ctx.fillStyle = color;
  1208→    ctx.fill();
  1209→    ctx.fillStyle = 'rgba(226, 232, 240, 0.85)';
  1210→    ctx.fillText(label, legendX + 18, legendY);
  1211→    legendY += 18;
  1212→  };
  1213→  drawLegend('#818cf8', 'You', 5);
  1214→  drawLegend('#34d399', 'Real Players', 3);
  1215→  drawLegend('#f43f5e', 'Bots', 2.5);
  1216→
  1217→  ctx.restore();
  1218→}
