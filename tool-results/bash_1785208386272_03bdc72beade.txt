     8	     8	// Key design decisions that FIX the old server's bugs:
     9	     9	//  * NO client-supplied position — the server is the sole authority on
    10	    10	//    snake coordinates. Client only sends a desired `angle`.
    11	    11	//  * Bounded body length — a snake's `points` array is capped at
    12	    12	//    `MAX_BODY_POINTS` so a runaway chip count can't OOM the server.
    13	    13	//  * All `points[0]` accesses are guarded; an empty `points` array
    14	    14	//    short-circuits the snake's tick (no TypeError crash).
    15	    15	//  * Map iteration + late-removal: dead snakes are collected into an
    16	    16	//    array and removed AFTER iteration completes (no Map mutation mid-iter).
    17	    17	//  * Bot AI reuses the same grid queries as players and never allocates
    18	    18	//    new arrays per tick (it walks the query Map directly).

   194	   194	/** Smallest angular difference from `current` to `desired`, in [-π, π]. */
   195	   195	function angularDelta(current: number, desired: number): number {
   196	   196	  let diff = desired - current;
   197	   197	  while (diff > Math.PI) diff -= 2 * Math.PI;
   198	   198	  while (diff < -Math.PI) diff += 2 * Math.PI;
   199	   199	  return diff;
   200	   200	}
   201	   201	
   202	   202	/** Step `current` toward `desired` by at most `maxStep` radians. */
   203	   203	export function turnToward(current: number, desired: number, maxStep: number): number {
   204	   204	  const diff = angularDelta(current, desired);
   205	   205	  if (Math.abs(diff) <= maxStep) return desired;
   206	   206	  return current + Math.sign(diff) * maxStep;
   207	   207	}

   209	   209	/** Distance between two points. */
   210	   210	function dist(ax: number, ay: number, bx: number, by: number): number {
   211	   211	  return Math.hypot(ax - bx, ay - by);
   212	   212	}

   218	   218	/** Random point inside a disc of radius `maxR` centered on the WORLD center. */
   219	   219	export function randomSpawnPoint(maxR: number): Vec2 {
   220	   220	  // sqrt for uniform area distribution (avoids clustering at center)
   221	   221	  const r = Math.sqrt(Math.random()) * maxR;
   222	   222	  const theta = Math.random() * Math.PI * 2;
   223	   223	  return { x: WORLD_RADIUS + Math.cos(theta) * r, y: WORLD_RADIUS + Math.sin(theta) * r };
   224	   224	}
   225	   225	
   226	   226	/** Breathing map radius — oscillates +/- MAP_BREATH_AMPLITUDE every MAP_BREATH_CYCLE_MS. */
   227	   227	export function getMapRadius(elapsedMs: number): number {
   228	   228	  const cycle = (elapsedMs % MAP_BREATH_CYCLE_MS) / MAP_BREATH_CYCLE_MS;
   229	   229	  return MAP_BASE_RADIUS + Math.sin(cycle * Math.PI * 2) * MAP_BREATH_AMPLITUDE;
   230	   230	}

   232	   232	/**
   233	   233	 * Build an initial body: `length` points strung out behind the head at
   234	   234	 * SEGMENT_SPACING px apart, all along the angle's reverse direction.
   235	   235	 */
   236	   236	export function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   237	   237	  const pts: Vec2[] = [];
   238	   238	  for (let i = 0; i < length; i++) {
   239	   239	    pts.push({
   240	   240	      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   241	   241	      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   242	   242	    });
   243	   243	  }
   244	   244	  return pts;
   245	   245	}

   372	   372	// Snake movement
   373	   373	// ----------------------------------------------------------------------------
   374	   374	
   375	   375	/**
   376	   376	 * Move a snake one tick. The server is the sole authority on coordinates:
   377	   377	 * `desiredAngle` is the only client influence (via rate-limited turn speed).
   378	   378	 *
   379	   379	 * ORIGINAL MECHANICS (matching server.ts):
   380	   380	 *  * Regular food = +1 score (grows body), NO chips.
   381	   381	 *  * Star chips = +chips to carriedChips, +3 score.
   382	   382	 *  * Boost costs BODY LENGTH: every 40 frames drops 1 tail segment (score -1).
   383	   383	 *  * Boost requires body.length > 8.
   384	   384	 *  * Turn rate: max(0.045, 0.15 - score*0.0006) — bigger snakes turn slower.
   385	   385	 *  * Size: 8 + sqrt(score) * 0.4.
   386	   386	 *  * Extraction glide speed = 3.2 (can still steer while extracting).
   387	   387	 *  * Body length = INITIAL_BODY_LENGTH + score, capped at MAX_BODY_LENGTH (120).
   388	   388	 */
   389	   389	export function tickSnakeMovement(
   390	   390	  snake: SnakeBase,
   391	   391	  desiredAngle: number,
   392	   392	  wantsBoost: boolean,
   393	   393	): void {
   394	   394	  if (snake.points.length === 0) return;
   395	   395	  if (snake.isDead) return;
   396	   396	
   397	   397	  // 1) Turn — rate-limited, bigger snakes turn slower (original formula).
   398	   398	  const turnRate = Math.max(TURN_MIN, TURN_BASE - snake.score * TURN_SCORE_FACTOR);
   399	   399	  snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
   400	   400	
   401	   401	  // 2) Speed: extracting glide < normal < boost.
   402	   402	  let speed = BASE_SPEED;
   403	   403	  if (snake.isExtracting) {
   404	   404	    speed = EXTRACT_GLIDE_SPEED;
   405	   405	  } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
   406	   406	    speed = BOOST_SPEED;
   407	   407	    // Boost cost: drop 1 tail segment every BOOST_DROP_INTERVAL frames.
   408	   408	    snake.boostFrameCounter++;
   409	   409	    if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
   410	   410	      snake.boostFrameCounter = 0;
   411	   411	      if (snake.points.length > BOOST_MIN_LENGTH) {
   412	   412	        snake.points.pop();
   413	   413	        snake.score = Math.max(BOOST_MIN_LENGTH, snake.score - 1);
   414	   414	      }
   415	   415	    }
   416	   416	  }
   417	   417	
   418	   418	  // 3) Move head.
   419	   419	  const head = snake.points[0];
   420	   420	  const nx = head.x + Math.cos(snake.angle) * speed;
   421	   421	  const ny = head.y + Math.sin(snake.angle) * speed;
   422	   422	
   423	   423	  // 4) Unshift new head — body trails naturally. NO world clamp (wall = death).
   424	   424	  snake.points.unshift({ x: nx, y: ny });
   425	   425	
   426	   426	  // 5) Grow/shrink body to target length based on SCORE (not chips).
   427	   427	  const targetLen = Math.min(
   428	   428	    MAX_BODY_LENGTH,
   429	   429	    INITIAL_BODY_LENGTH + snake.score,
   430	   430	  );
   431	   431	  while (snake.points.length > targetLen) snake.points.pop();
   432	   432	
   433	   433	  // 6) Size formula: 8 + sqrt(score) * 0.4 (original).
   434	   434	  snake.size = SIZE_BASE + Math.sqrt(snake.score) * SIZE_SCORE_FACTOR;
   435	   435	}

   437	   437	// ----------------------------------------------------------------------------
   438	   438	// Bot AI
   439	   439	// ----------------------------------------------------------------------------
   440	   440	
   441	   441	/**
   442	   442	 * Cheap per-bot AI. Re-evaluates a target angle every ~150ms; between
   443	   443	 * re-thinks it just moves toward the cached angle. The bot seeks the
   444	   444	 * nearest food pellet and flees from any foreign body segment within a
   445	   445	 * threat radius. No allocations on the hot path — we walk the grid query
   446	   446	 * Map directly.
   447	   447	 */
   448	   448	export function tickBot(bot: BotSession, room: ArenaRoom, now: number): void {
   449	   449	  if (bot.points.length === 0 || bot.isDead) return;
   450	   450	
   451	   451	  if (now >= bot.nextThinkAt) {
   452	   452	    bot.nextThinkAt = now + 120 + Math.floor(Math.random() * 80);
   453	   453	    const head = bot.points[0];
   454	   454	
   455	   455	    // Find nearest food (scan ~250px radius).
   456	   456	    let bestFood: GridItem | null = null;
   457	   457	    let bestFoodDist = Infinity;
   458	   458	    const foodQuery = room.grid.queryRadius(head.x, head.y, 260);
   459	   459	    for (const item of foodQuery.values()) {
   460	   460	      if (item.kind !== 'food') continue;
   461	   461	      if ((item.value ?? 0) <= 0) continue;
   462	   462	      const d = dist(head.x, head.y, item.x, item.y);
   463	   463	      if (d < bestFoodDist) {
   464	   464	        bestFoodDist = d;
   465	   465	        bestFood = item;
   466	   466	      }
   467	   467	    }
   468	   468	
   469	   469	    // Find nearest threat (foreign body segment within ~140px).
   470	   470	    let threatX = 0;
   471	   471	    let threatY = 0;
   472	   472	    let threatDist = Infinity;
   473	   473	    const threatQuery = room.grid.queryRadius(head.x, head.y, 150);
   474	   474	    for (const item of threatQuery.values()) {
   475	   475	      if (item.kind !== 'segment') continue;
   476	   476	      if (item.snakeId === bot.id) continue;
   477	   477	      if (item.segIdx === 0) continue; // ignore head-on collisions
   478	   478	      const d = dist(head.x, head.y, item.x, item.y);
   479	   479	      if (d < threatDist) {
   480	   480	        threatDist = d;
   481	   481	        threatX = item.x;
   482	   482	        threatY = item.y;
   483	   483	      }
   484	   484	    }
   485	   485	
   486	   486	    let desired: number;
   487	   487	    const cowardFleeDist = 140;
   488	   488	    const opportunistFleeDist = 90;
   489	   489	    const shouldFlee =
   490	   490	      threatDist < cowardFleeDist &&
   491	   491	      (bot.personality === 'coward' ||
   492	   492	        bot.personality === 'extractor' ||
   493	   493	        (bot.personality === 'opportunist' && threatDist < opportunistFleeDist));
   494	   494	
   495	   495	    if (shouldFlee) {
   496	   496	      // Flee directly away from the threat segment.
   497	   497	      desired = Math.atan2(head.y - threatY, head.x - threatX);
   498	   498	    } else if (bestFood) {
   499	   499	      desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
   500	   500	    } else {
   501	   501	      // Wander: nudge current angle slightly. Avoids getting stuck on walls.
   502	   502	      desired = bot.angle + (Math.random() - 0.5) * 0.4;
   503	   503	    }
   504	   504	
   505	   505	    // Edge avoidance: if near world border, turn back toward center.
   506	   506	    const headR = Math.hypot(head.x, head.y);
   507	   507	    if (headR > WORLD_RADIUS - 250) {
   508	   508	      const toCenter = Math.atan2(-head.y, -head.x);
   509	   509	      desired = turnToward(desired, toCenter, MAX_TURN_PER_TICK * 2);
   510	   510	    }
   511	   511	
   512	   512	    bot.desiredAngle = desired;
   513	   513	  }
   514	   514	
   515	   515	  // Hunter personality occasionally boosts.
   516	   516	  const wantsBoost = bot.personality === 'hunter' && bot.carriedChips > 5 && Math.random() < 0.05;
   517	   517	  tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
   518	   518	}

   525	   525	export interface PendingDeath {
   526	   526	  deadId: string;
   527	   527	  killerId?: string;
   528	   528	  cause: 'body' | 'wall';
   529	   529	}

   531	   531	/**
   532	   532	 * Detect head-to-body collisions via the spatial grid. For each living,
   533	   533	 * non-spawn-protected snake's head, query nearby segments; if any
   534	   534	 * foreign non-head segment is within `head.size + seg.radius`, the head's
   535	   535	 * owner dies (and the segment's owner is the killer).
   536	   536	 *
   537	   537	 * Returns the list of deaths. Caller must apply them AFTER iteration.
   538	   538	 */
   539	   539	export function detectCollisions(room: ArenaRoom, now: number): PendingDeath[] {
   540	   540	  const deaths: PendingDeath[] = [];
   541	   541	  const seenDead = new Set<string>();
   542	   542	
   543	   543	  const allSnakes: SnakeBase[] = collectAllSnakes(room);
   544	   544	  for (const snake of allSnakes) {
   545	   545	    if (snake.isDead) continue;
   546	   546	    if (snake.points.length === 0) continue;
   547	   547	    if (now < snake.spawnProtectedUntil) continue;
   548	   548	    if (seenDead.has(snake.id)) continue;
   549	   549	
   550	   550	    const head = snake.points[0];
   551	   551	
   552	   552	    // Wall collision: death if outside the breathing map radius.
   553	   553	    const mapRadius = getMapRadius(now);
   554	   554	    const distFromCenter = Math.hypot(head.x - WORLD_RADIUS, head.y - WORLD_RADIUS);
   555	   555	    if (distFromCenter > mapRadius) {
   556	   556	      deaths.push({ deadId: snake.id, killerId: 'wall', cause: 'wall' });
   557	   557	      seenDead.add(snake.id);
   558	   558	      continue;
   559	   559	    }
   560	   560	
   561	   561	    const queryR = snake.size + 30;
   562	   562	    const nearby = room.grid.queryRadius(head.x, head.y, queryR);
   563	   563	
   564	   564	    for (const item of nearby.values()) {
   565	   565	      if (item.kind !== 'segment') continue;
   566	   566	      if (item.snakeId === snake.id) continue;
   567	   567	      if (item.segIdx === 0) continue; // skip head-to-head (head-to-body only)
   568	   568	      const d = dist(head.x, head.y, item.x, item.y);
   569	   569	      // Original hit factor: (radiusA + radiusB) * 0.75
   570	   570	      if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
   571	   571	        deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
   572	   572	        seenDead.add(snake.id);
   573	   573	        break;
   574	   574	      }
   575	   575	    }
   576	   576	  }
   577	   577	
   578	   578	  return deaths;
   579	   579	}

   581	   581	/**
   582	   582	 * Eat food via the spatial grid. RULES:
   583	   583	 *  * Regular food (isStarChip=false): +1 score (grows body), NO chips. ALL snakes eat this.
   584	   584	 *  * Star chip (isStarChip=true): +value chips to carriedChips, +3 score. ONLY real players eat this.
   585	   585	 *  * Bots NEVER collect star chips (they have no economic role).
   586	   586	 * Eaten food is zeroed on both the grid item AND the real Food object.
   587	   587	 */
   588	   588	export function eatFood(room: ArenaRoom): void {
   589	   589	  const allSnakes: SnakeBase[] = collectAllSnakes(room);
   590	   590	  for (const snake of allSnakes) {
   591	   591	    if (snake.isDead) continue;
   592	   592	    if (snake.points.length === 0) continue;
   593	   593	    const head = snake.points[0];
   594	   594	    const nearby = room.grid.queryRadius(head.x, head.y, snake.size + 20);
   595	   595	    for (const item of nearby.values()) {
   596	   596	      if (item.kind !== 'food') continue;
   597	   597	      if ((item.value ?? 0) <= 0) continue;
   598	   598	      const d = dist(head.x, head.y, item.x, item.y);
   599	   599	      if (d < snake.size + (item.radius || 6) + 6) {
   600	   600	        if (item.isStarChip) {
   601	   601	          // Star chip: ONLY real players collect (bots skip).
   602	   602	          if (!snake.isPlayer) continue;
   603	   603	          snake.carriedChips += item.value ?? 0;
   604	   604	          snake.score += STAR_CHIP_GROW;
   605	   605	        } else {
   606	   606	          // Regular food: all snakes eat (+1 score, no chips).
   607	   607	          snake.score += REGULAR_FOOD_GROW;
   608	   608	        }
   609	   609	        // Zero BOTH the grid item (intra-tick sentinel) AND the real food object.
   610	   610	        item.value = 0;
   611	   611	        if (item.foodRef) item.foodRef.value = 0;
   612	   612	      }
   613	   613	    }
   614	   614	  }
   615	   615	}

   621	   621	/** Iterator helper: yield every living-or-dead snake (player + bot) in the room. */
   622	   622	export function collectAllSnakes(room: ArenaRoom): SnakeBase[] {
   623	   623	  const out: SnakeBase[] = [];
   624	   624	  for (const p of room.players.values()) out.push(p);
   625	   625	  for (const b of room.bots.values()) out.push(b);
   626	   626	  return out;
   627	   627	}

   629	   629	/**
   630	   630	 * Build the broadcast GameSnapshot for an arena.
   631	   631	 * @param viewerId — the socket ID of the player receiving this snapshot (for yourRank computation).
   632	   632	 */
   633	   633	export function buildSnapshot(room: ArenaRoom, viewerId?: string): GameSnapshot {
   634	   634	  const snakes: SnakeSnapshot[] = [];
   635	   635	  const now = Date.now();
   636	   636	  for (const snake of collectAllSnakes(room)) {
   637	   637	    let points = snake.points;
   638	   638	    if (points.length > MAX_SNAPSHOT_POINTS) {
   639	   639	      const step = points.length / MAX_SNAPSHOT_POINTS;
   640	   640	      const downsampled: Vec2[] = [];
   641	   641	      for (let i = 0; i < MAX_SNAPSHOT_POINTS; i++) {
   642	   642	        downsampled.push(points[Math.floor(i * step)]);
   643	   643	      }
   644	   644	      points = downsampled;
   645	   645	    }
   646	   646	    snakes.push({
   647	   647	      id: snake.id,
   648	   648	      name: snake.name,
   649	   649	      userTag: snake.userTag,
   650	   650	      points,
   651	   651	      angle: snake.angle,
   652	   652	      size: snake.size,
   653	   653	      color: snake.color,
   654	   654	      secondaryColor: snake.secondaryColor,
   655	   655	      isPlayer: snake.isPlayer,
   656	   656	      isBot: snake.isBot,
   657	   657	      carriedChips: Math.floor(snake.carriedChips),
   658	   658	      score: snake.score,
   659	   659	      isExtracting: snake.isExtracting,
   660	   660	      extractionProgress: Math.min(1, snake.extractionProgress / EXTRACT_DURATION_MS),
   661	   661	      isDead: snake.isDead,
   662	   662	      spawnProtected: now < snake.spawnProtectedUntil,
   663	   663	      chatMessage: snake.chatMessage,
   664	   664	      country: snake.country,
   665	   665	    });
   666	   666	  }
   667	   667	
   668	   668	  const foods: FoodSnapshot[] = room.foods.map((f) => ({
   669	   669	    id: f.id,
   670	   670	    x: f.x,
   671	   671	    y: f.y,
   672	   672	    size: f.size,
   673	   673	    value: f.value,
   674	   674	    isStarChip: f.isStarChip,
   675	   675	    color: f.color,
   676	   676	  }));
   677	   677	
   678	   678	  // Build arena leaderboard from REAL players only, sorted by carriedChips desc.
   679	   679	  const realPlayers = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling);
   680	   680	  const sorted = realPlayers.sort((a, b) => b.carriedChips - a.carriedChips);
   681	   681	  const arenaLeaderboard = sorted.slice(0, 10).map(p => ({
   682	   682	    id: p.id,
   683	   683	    name: p.name,
   684	   684	    userTag: p.userTag,
   685	   685	    carriedChips: Math.floor(p.carriedChips),
   686	   686	    score: p.score,
   687	   687	    kills: p.kills,
   688	   688	    isPlayer: p.id === viewerId,
   689	   689	    country: p.country,
   690	   690	  }));
   691	   691	
   692	   692	  // Your rank: position in sorted real players by carriedChips.
   693	   693	  const yourRank = viewerId ? sorted.findIndex(p => p.id === viewerId) + 1 : 0;
   694	   694	
   695	   695	  // Dynamic commission: 0% if <=3 real players, 35% if >=4.
   696	   696	  const realPlayerCount = realPlayers.length;
   697	   697	  const commissionRate = realPlayerCount <= 3 ? 0 : 0.35;
   698	   698	
   699	   699	  return {
   700	   700	    arenaId: room.arena.id,
   701	   701	    tick: room.tick,
   702	   702	    snakes,
   703	   703	    foods,
   704	   704	    worldSize: WORLD_SIZE,
   705	   705	    leaderId: room.leaderId,
   706	   706	    leaderChips: room.leaderChips,
   707	   707	    realPlayerCount,
   708	   708	    yourRank: yourRank > 0 ? yourRank : 0,
   709	   709	    arenaLeaderboard,
   710	   710	    commissionRate,
   711	   711	  };
   712	   712	}

   715	   715	 * Recompute the arena leader (highest carriedChips). Called once per tick
   716	   716	 * after collisions + food have been resolved.
   717	   717	 */
   718	   718	export function recomputeLeader(room: ArenaRoom): void {
   719	   719	  let topId: string | null = null;
   720	   720	  let topChips = 0;
   721	   721	  for (const snake of collectAllSnakes(room)) {
   722	   722	    if (snake.carriedChips > topChips) {
   723	   723	      topChips = snake.carriedChips;
   724	   724	      topId = snake.id;
   725	   725	    }
   726	   726	  }
   727	   727	  room.leaderId = topId;
   728	   728	  room.leaderChips = Math.floor(topChips);
   729	   729	}

   731	   731	/** Re-populate food up to FOOD_COUNT_TARGET. Called once per tick. */
   732	   732	export function replenishFood(room: ArenaRoom): void {
   733	   733	  // Filter out eaten food first (sentinel: value === 0).
   734	   734	  if (room.foods.some((f) => f.value <= 0)) {
   735	   735	    room.foods = room.foods.filter((f) => f.value > 0);
   736	   736	  }
   737	   737	  let guard = 0;
   738	   738	  while (room.foods.length < FOOD_COUNT_TARGET && guard < 50) {
   739	   739	    room.foods.push(spawnRandomFood(room));
   740	   740	    guard++;
   741	   741	  }
   742	   742	}
