   332→      const buf = replayPreBufferRef.current;
   333→      if (buf.length < REPLAY_PRE_MAX) {
   334→        buf.push(snap);
   335→      } else {
   336→        buf[replayWriteIdxRef.current % REPLAY_PRE_MAX] = snap;
   337→      }

   351→      preFrames.push(...preBuf);

   354→      const start = replayWriteIdxRef.current % REPLAY_PRE_MAX;

   356→        preFrames.push(preBuf[(start + i) % REPLAY_PRE_MAX]);
   357→      }
   358→    }
