   103→   322→export const MAP_MAX_RADIUS = 5000;  // radius when 1000 players

   108→   327→/** Compute dynamic map radius based on real player count. */
   109→   328→export function getDynamicMapRadius(realPlayerCount: number, elapsedMs?: number): number {
   110→   329→  const minP = 1;
   111→   330→  const maxP = MAX_ARENA_PLAYERS;
   112→   331→  const count = Math.max(minP, Math.min(maxP, realPlayerCount));
   113→   332→  // sqrt scaling: 1 player -> 1500, ~31 players -> ~3000, 1000 players -> 5000
   114→   333→  const baseRadius = MAP_MIN_RADIUS + (MAP_MAX_RADIUS - MAP_MIN_RADIUS) * Math.sqrt((count - 1) / (maxP - 1));
   115→   334→  // Add breathing
   116→   335→  if (elapsedMs !== undefined) {
   117→   336→    const cycle = (elapsedMs % MAP_BREATH_CYCLE_MS) / MAP_BREATH_CYCLE_MS;
   118→   337→    return baseRadius + Math.sin(cycle * Math.PI * 2) * MAP_BREATH_AMPLITUDE;
   119→   338→  }
   120→   339→  return baseRadius;
   121→   340→}
