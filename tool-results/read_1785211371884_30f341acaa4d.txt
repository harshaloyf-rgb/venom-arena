   143→   143→// ----------------------------------------------------------------------------
   144→   144→// Spatial hash grid — slimmed-down client port of the server's grid.
   145→   145→// Items are bucketed into square cells; queries return a deduplicated Map.
   146→   146→// ----------------------------------------------------------------------------
   147→   147→
   148→   148→class SpatialHashGrid {
   149→   149→  private readonly cellSize: number;
   150→   150→  private readonly cells: Map<string, Map<string, GridItem>> = new Map();
   151→   151→
   152→   152→  constructor(cellSize = 120) {
   153→   153→    this.cellSize = cellSize;
   154→   154→  }
   155→   155→
   156→   156→  private key(cx: number, cy: number): string {
   157→   157→    return cx + ':' + cy;
   158→   158→  }
   159→   159→
   160→   160→  clear(): void {
   161→   161→    this.cells.clear();
   162→   162→  }
   163→   163→
   164→   164→  insert(item: GridItem): void {
   165→   165→    const minCx = Math.floor((item.x - item.radius) / this.cellSize);
   166→   166→    const maxCx = Math.floor((item.x + item.radius) / this.cellSize);
   167→   167→    const minCy = Math.floor((item.y - item.radius) / this.cellSize);
   168→   168→    const maxCy = Math.floor((item.y + item.radius) / this.cellSize);
   169→   169→    for (let cx = minCx; cx <= maxCx; cx++) {
   170→   170→      for (let cy = minCy; cy <= maxCy; cy++) {
   171→   171→        const k = this.key(cx, cy);
   172→   172→        let bucket = this.cells.get(k);
   173→   173→        if (!bucket) {
   174→   174→          bucket = new Map();
   175→   175→          this.cells.set(k, bucket);
   176→   176→        }
   177→   177→        bucket.set(item.id, item);
   178→   178→      }
   179→   179→    }
   180→   180→  }
   181→   181→
   182→   182→  queryRadius(x: number, y: number, r: number): Map<string, GridItem> {
   183→   183→    const out = new Map<string, GridItem>();
   184→   184→    const minCx = Math.floor((x - r) / this.cellSize);
   185→   185→    const maxCx = Math.floor((x + r) / this.cellSize);
   186→   186→    const minCy = Math.floor((y - r) / this.cellSize);
   187→   187→    const maxCy = Math.floor((y + r) / this.cellSize);
   188→   188→    for (let cx = minCx; cx <= maxCx; cx++) {
   189→   189→      for (let cy = minCy; cy <= maxCy; cy++) {
   190→   190→        const bucket = this.cells.get(this.key(cx, cy));
   191→   191→        if (!bucket) continue;
   192→   192→        for (const [id, item] of bucket) {
   193→   193→          if (!out.has(id)) out.set(id, item);
   194→   194→        }
   195→   195→      }
   196→   196→    }
   197→   197→    return out;
   198→   198→  }
   199→   199→}

   255→   255→function angularDelta(current: number, desired: number): number {
   256→   256→  let diff = desired - current;
   257→   257→  while (diff > Math.PI) diff -= 2 * Math.PI;
   258→   258→  while (diff < -Math.PI) diff += 2 * Math.PI;
   259→   259→  return diff;
   260→   260→}
   261→   261→
   262→   262→function turnToward(current: number, desired: number, maxStep: number): number {
   263→   263→  const diff = angularDelta(current, desired);
   264→   264→  if (Math.abs(diff) <= maxStep) return desired;
   265→   265→  return current + Math.sign(diff) * maxStep;
   266→   266→}
   267→   267→
   268→   268→function dist(ax: number, ay: number, bx: number, by: number): number {
   269→   269→  return Math.hypot(ax - bx, ay - by);
   270→   270→}
   271→   271→
   272→   272→/** Random point within a circle of maxR around center. */
   273→   273→function randomPointInCircle(cx: number, cy: number, maxR: number): Vec2 {
   274→   274→  const r = Math.sqrt(Math.random()) * maxR;
   275→   275→  const theta = Math.random() * Math.PI * 2;
   276→   276→  return {
   277→   277→    x: cx + Math.cos(theta) * r,
   278→   278→    y: cy + Math.sin(theta) * r,
   279→   279→  };
   280→   280→}

   282→   282→function initialBody(headX: number, headY: number, angle: number, length: number): Vec2[] {
   283→   283→  const pts: Vec2[] = [];
   284→   284→  for (let i = 0; i < length; i++) {
   285→   285→    pts.push({
   286→   286→      x: headX - Math.cos(angle) * i * SEGMENT_SPACING,
   287→   287→      y: headY - Math.sin(angle) * i * SEGMENT_SPACING,
   288→   288→    });
   289→   289→  }
   290→   290→  return pts;
   291→   291→}

   297→   297→/** Pick a weighted random FoodOrbConfig. */
   298→   298→function weightedRandomOrb(): FoodOrbConfig {
   299→   299→  let r = Math.random();
   300→   300→  for (let i = 0; i < ALL_FOOD_ORBS.length; i++) {
   301→   301→    r -= FOOD_ORB_WEIGHTS[i];
   302→   302→    if (r <= 0) return ALL_FOOD_ORBS[i];
   303→   303→  }
   304→   304→  return ALL_FOOD_ORBS[0]; // fallback to small
   305→   305→}

   307→   307→/** Create a Food object at a random position around the player. */
   308→   308→function createFoodOrb(idPrefix: string, idCounter: { value: number }, cx: number, cy: number, farSpawn: boolean): Food {
   309→   309→  const orb = weightedRandomOrb();
   310→   310→  const radius = farSpawn ? FOOD_SPAWN_RADIUS_FAR : FOOD_SPAWN_RADIUS_NEAR;
   311→   311→  const pos = randomPointInCircle(cx, cy, radius);
   312→   312→  return {
   313→   313→    id: `${idPrefix}-food-${idCounter.value++}`,
   314→   314→    x: pos.x,
   315→   315→    y: pos.y,
   316→   316→    size: orb.radius,
   317→   317→    value: orb.value,

   346→   346→
   347→   347→function computeDeathFoodDrop(
   348→   348→  totalScore: number,
   349→   349→  deathX: number,
   350→   350→  deathY: number,
   351→   351→  idPrefix: string,
   352→   352→  idCounter: { value: number },
   353→   353→): Food[] {
   354→   354→  const result: Food[] = [];
   355→   355→  let remaining = Math.max(0, totalScore);
   356→   356→
   357→   357→  // Large orbs (value 5)
   358→   358→  const largeOrb = ALL_FOOD_ORBS[2]; // FOOD_ORB_LARGE
   359→   359→  const largeCount = Math.floor(remaining / largeOrb.value);
   360→   360→  for (let i = 0; i < largeCount; i++) {
   361→   361→    const offset = randomPointInCircle(deathX, deathY, 30 + largeCount * 0.5);
   362→   362→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, largeOrb));
   363→   363→  }
   364→   364→  remaining -= largeCount * largeOrb.value;
   365→   365→
   366→   366→  // Medium orbs (value 3)
   367→   367→  const mediumOrb = ALL_FOOD_ORBS[1]; // FOOD_ORB_MEDIUM
   368→   368→  const mediumCount = Math.floor(remaining / mediumOrb.value);
   369→   369→  for (let i = 0; i < mediumCount; i++) {
   370→   370→    const offset = randomPointInCircle(deathX, deathY, 30 + mediumCount * 0.5);
   371→   371→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, mediumOrb));
   372→   372→  }
   373→   373→  remaining -= mediumCount * mediumOrb.value;
   374→   374→
   375→   375→  // Small orbs (value 1) for the remainder
   376→   376→  const smallOrb = ALL_FOOD_ORBS[0]; // FOOD_ORB_SMALL
   377→   377→  for (let i = 0; i < remaining; i++) {
   378→   378→    const offset = randomPointInCircle(deathX, deathY, 30 + remaining * 0.5);
   379→   379→    result.push(createFoodOrbAt(idPrefix, idCounter, offset.x, offset.y, smallOrb));
   380→   380→  }
   381→   381→
   382→   382→  return result;
   383→   383→}

   727→   727→  private onTouchStart(e: TouchEvent): void {
   728→   728→    if (this.joystick) return;
   729→   729→    const t = this.findJoystickTouch(e.touches);
   730→   730→    if (!t) return;
   731→   731→    e.preventDefault();
   732→   732→    const rect = this.canvas.getBoundingClientRect();
   733→   733→    this.joystick = {
   734→   734→      active: true,
   735→   735→      pointerId: t.identifier,
   736→   736→      originX: t.clientX - rect.left,
   737→   737→      originY: t.clientY - rect.top,
   738→   738→      curX: t.clientX - rect.left,
   739→   739→      curY: t.clientY - rect.top,
   740→   740→    };
   741→   741→  }
   742→   742→
   743→   743→  private onTouchMove(e: TouchEvent): void {
   744→   744→    const js = this.joystick;
   745→   745→    if (!js) return;
   746→   746→    let t: Touch | null = null;
   747→   747→    for (let i = 0; i < e.touches.length; i++) {
   748→   748→      if (e.touches[i].identifier === js.pointerId) {
   749→   749→        t = e.touches[i];
   750→   750→        break;
   751→   751→      }
   752→   752→    }
   753→   753→    if (!t) return;
   754→   754→    e.preventDefault();
   755→   755→    const rect = this.canvas.getBoundingClientRect();
   756→   756→    js.curX = t.clientX - rect.left;
   757→   757→    js.curY = t.clientY - rect.top;
   758→   758→    const dx = js.curX - js.originX;
   759→   759→    const dy = js.curY - js.originY;
   760→   760→    const d = Math.hypot(dx, dy);
   761→   761→    const mag = Math.min(1, d / JOYSTICK_MAX_RADIUS_PX);
   762→   762→    if (mag > JOYSTICK_DEADZONE) {
   763→   763→      this.touchAngle = Math.atan2(dy, dx);
   764→   764→      this.touchBoost = mag > JOYSTICK_BOOST_MAGNITUDE;
   765→   765→    } else {
   766→   766→      this.touchAngle = null;
   767→   767→      this.touchBoost = false;
   768→   768→    }
   769→   769→  }
   770→   770→
   771→   771→  private onTouchEnd(e: TouchEvent): void {
   772→   772→    const js = this.joystick;
   773→   773→    if (!js) return;
   774→   774→    let stillActive = false;
   775→   775→    for (let i = 0; i < e.touches.length; i++) {
   776→   776→      if (e.touches[i].identifier === js.pointerId) {
   777→   777→        stillActive = true;
   778→   778→        break;
   779→   779→      }
   780→   780→    }
   781→   781→    if (!stillActive) {
   782→   782→      this.joystick = null;
   783→   783→      this.touchAngle = null;
   784→   784→      this.touchBoost = false;
   785→   785→    }
   786→   786→  }
   787→   787→
   788→   788→  private findJoystickTouch(touches: TouchList): Touch | null {
   789→   789→    const rect = this.canvas.getBoundingClientRect();
   790→   790→    for (let i = 0; i < touches.length; i++) {
   791→   791→      const t = touches[i];
   792→   792→      const x = t.clientX - rect.left;
   793→   793→      const y = t.clientY - rect.top;
   794→   794→      if (x < rect.width / 2 && y > rect.height / 2) {
   795→   795→        return t;
   796→   796→      }
   797→   797→    }
   798→   798→    return null;
   799→   799→  }

   801→   801→  private onKeyDown(e: KeyboardEvent): void {
   802→   802→    const k = e.key.toLowerCase();
   803→   803→    if (
   804→   804→      k === 'arrowup' ||
   805→   805→      k === 'arrowdown' ||
   806→   806→      k === 'arrowleft' ||
   807→   807→      k === 'arrowright' ||
   808→   808→      k === ' ' ||
   809→   809→      k === 'spacebar'
   810→   810→    ) {
   811→   811→      e.preventDefault();
   812→   812→    }
   813→   813→    if (k === 'escape' && this.state !== 'playing') {
   814→   814→      this.handleExitToLobby();
   815→   815→      return;
   816→   816→    }
   817→   817→    if (k === 'e' && this.state === 'playing' && !this.extractHold) {
   818→   818→      this.extractHold = true;
   819→   819→      this.beginExtract();
   820→   820→    }
   821→   821→    if (this.state === 'playing' && ['1', '2', '3', '4', '5'].includes(k)) {
   822→   822→      const idx = parseInt(k, 10) - 1;
   823→   823→      if (idx >= 0 && idx < QUICK_EMOTES.length) {
   824→   824→        this.setPlayerChat(QUICK_EMOTES[idx]);
   825→   825→      }
   826→   826→    }
   827→   827→    const normalized = k === 'spacebar' ? ' ' : k;
   828→   828→    this.keys.add(normalized);
   829→   829→  }

   831→   831→  private onKeyUp(e: KeyboardEvent): void {
   832→   832→    const k = e.key.toLowerCase();

   834→   834→      this.extractHold = false;
   835→   835→      this.cancelExtract();
   836→   836→    }
   837→   837→    const normalized = k === 'spacebar' ? ' ' : k;
   838→   838→    this.keys.delete(normalized);
   839→   839→  }

   899→   899→  // Main rAF loop
   900→   900→  // --------------------------------------------------------------------------
   901→   901→
   902→   902→  private frame = (now: number): void => {
   903→   903→    if (this.stopped) return;
   904→   904→    this.rafId = requestAnimationFrame(this.frame);
   905→   905→    const dt = now - this.lastFrameTime;
   906→   906→    this.lastFrameTime = now;
   907→   907→
   908→   908→    this.updateFps(now);
   909→   909→    this.updateParticles(dt);
   910→   910→
   911→   911→    // Physics ticks at ~TICK_MS (30 Hz) cadence.
   912→   912→    if (this.state === 'playing') {
   913→   913→      this.accumulator += dt;
   914→   914→      let safety = 4;
   915→   915→      while (this.accumulator >= TICK_MS && safety > 0) {
   916→   916→        this.tickPhysics(now);
   917→   917→        this.accumulator -= TICK_MS;
   918→   918→        safety--;
   919→   919→      }
   920→   920→      if (safety === 0) this.accumulator = 0;
   921→   921→    }
   922→   922→
   923→   923→    this.render(now);
   924→   924→    this.updateHUD();
   925→   925→  };
   926→   926→
   927→   927→  private accumulator: number = 0;
   928→   928→
   929→   929→  // --------------------------------------------------------------------------
   930→   930→  // Physics tick (one server-tick equivalent)
   931→   931→  // --------------------------------------------------------------------------
   932→   932→
   933→   933→  private tickPhysics(now: number): void {
   934→   934→    this.tick++;
   935→   935→    const p = this.player;
   936→   936→    if (!p || p.isDead) return;
   937→   937→
   938→   938→    // 1) Compute player desired angle + boost.
   939→   939→    const { angle, boost } = this.computePlayerInput();
   940→   940→    if (angle !== null) p.desiredAngle = angle;
   941→   941→    p.wantsBoost = boost;
   942→   942→    p.isBoosting = boost && p.points.length > BOOST_MIN_LENGTH + INITIAL_BODY_LENGTH - 20;
   943→   943→
   944→   944→    // Extraction progress (3-second hold).
   945→   945→    if (p.isExtracting) {
   946→   946→      p.extractionProgress += TICK_MS;
   947→   947→      if (p.extractionProgress >= EXTRACT_DURATION_MS) {
   948→   948→        this.finishExtract();
   949→   949→        return;
   950→   950→      }
   951→   951→    }
   952→   952→
   953→   953→    // 2) Move player.
   954→   954→    this.tickSnakeMovement(p, p.desiredAngle, p.wantsBoost);
   955→   955→
   956→   956→    // 3) Bots: think + move.
   957→   957→    for (const bot of this.bots.values()) {
   958→   958→      this.tickBot(bot, now);
   959→   959→    }
   960→   960→
   961→   961→    // 4) Build spatial grid for collision + food queries.
   962→   962→    this.grid.clear();
   963→   963→    this.insertSnakeIntoGrid(p);
   964→   964→    for (const bot of this.bots.values()) {
   965→   965→      if (!bot.isDead) this.insertSnakeIntoGrid(bot);
   966→   966→    }
   967→   967→    for (const food of this.foods) {
   968→   968→      this.grid.insert({
   969→   969→        id: food.id,
   970→   970→        kind: 'food',
   971→   971→        x: food.x,
   972→   972→        y: food.y,
   973→   973→        radius: food.size,
   974→   974→        value: food.value,
   975→   975→        foodRef: food,
   976→   976→      });
   977→   977→    }
   978→   978→
   979→   979→    // 5) Eat food (player + bots).
   980→   980→    this.eatFood();
   981→   981→
   982→   982→    // 6) Collision detection (body collision + head-on collision). NO wall death.
   983→   983→    const deaths = this.detectCollisions(now);
   984→   984→
   985→   985→    // 7) Apply deaths + drop food.
   986→   986→    let playerDied = false;
   987→   987→    const newDropFoods: Food[] = [];
   988→   988→    for (const d of deaths) {
   989→   989→      if (d.deadId === p.id) {
   990→   990→        playerDied = true;
   991→   991→        // Drop player's food
   992→   992→        const playerTotal = INITIAL_SPAWN_SCORE + p.score;
   993→   993→        newDropFoods.push(...computeDeathFoodDrop(playerTotal, p.points[0].x, p.points[0].y, this.arena.id, this.idCounterObj));
   994→   994→        continue;
   995→   995→      }
   996→   996→      const bot = this.bots.get(d.deadId);
   997→   997→      if (bot) {
   998→   998→        // Drop bot's food
   999→   999→        const botTotal = INITIAL_SPAWN_SCORE + bot.score;
  1000→  1000→        newDropFoods.push(...computeDeathFoodDrop(botTotal, bot.points[0].x, bot.points[0].y, this.arena.id, this.idCounterObj));
  1001→  1001→        bot.isDead = true;
  1002→  1002→        // Credit kill to the killer if it's the player.
  1003→  1003→        if (d.killerId === p.id) {
  1004→  1004→          p.kills++;
  1005→  1005→        }
  1006→  1006→      }
  1007→  1007→    }
  1008→  1008→
  1009→  1009→    // Add dropped food to the world.
  1010→  1010→    if (newDropFoods.length > 0) {
  1011→  1011→      this.foods.push(...newDropFoods);
  1012→  1012→    }
  1013→  1013→
  1014→  1014→    // 8) Remove dead bots and respawn to maintain count.
  1015→  1015→    if (deaths.length > 0) {
  1016→  1016→      const toRemove: string[] = [];
  1017→  1017→      for (const [id, bot] of this.bots) {
  1018→  1018→        if (bot.isDead) toRemove.push(id);
  1019→  1019→      }
  1020→  1020→      for (const id of toRemove) {
  1021→  1021→        this.bots.delete(id);
  1022→  1022→      }
  1023→  1023→    }
  1024→  1024→    // Respawn bots near the player to maintain count.
  1025→  1025→    const pHead = p.points[0];
  1026→  1026→    while (this.bots.size < this.arena.botsCount) {
  1027→  1027→      const bot = this.spawnBot(pHead.x, pHead.y);
  1028→  1028→      // Ensure bots spawn far enough from the player.
  1029→  1029→      const spawnDist = dist(bot.points[0].x, bot.points[0].y, pHead.x, pHead.y);
  1030→  1030→      if (spawnDist < 400) {
  1031→  1031→        // Re-spawn further away
  1032→  1032→        const far = randomPointInCircle(pHead.x, pHead.y, BOT_SPAWN_RADIUS);
  1033→  1033→        bot.points = initialBody(far.x, far.y, bot.angle, INITIAL_BODY_LENGTH);
  1034→  1034→      }
  1035→  1035→      this.bots.set(bot.id, bot);
  1036→  1036→    }
  1037→  1037→
  1038→  1038→    // 9) Player death → death screen.
  1039→  1039→    if (playerDied) {
  1040→  1040→      p.isDead = true;
  1041→  1041→      this.handlePlayerDeath();
  1042→  1042→      return;
  1043→  1043→    }
  1044→  1044→
  1045→  1045→    // 10) Replenish food toward target (spawn around player).
  1046→  1046→    this.replenishFood();
  1047→  1047→
  1048→  1048→    // 11) Expire chat bubbles.
  1049→  1049→    this.expireChat(now);
  1050→  1050→
  1051→  1051→    // 12) Update camera target.
  1052→  1052→    this.updateCamera();
  1053→  1053→  }

  1055→  1055→  // --------------------------------------------------------------------------
  1056→  1056→  // Snake movement (server-authoritative formula)
  1057→  1057→  // --------------------------------------------------------------------------
  1058→  1058→
  1059→  1059→  private tickSnakeMovement(snake: SnakeBase, desiredAngle: number, wantsBoost: boolean): void {
  1060→  1060→    if (snake.points.length === 0 || snake.isDead) return;
  1061→  1061→
  1062→  1062→    // Turn rate: max(0.045, 0.15 - totalScore*0.0006).
  1063→  1063→    const totalScore = INITIAL_SPAWN_SCORE + snake.score;
  1064→  1064→    const turnRate = Math.max(TURN_MIN, TURN_BASE - totalScore * TURN_SCORE_FACTOR);
  1065→  1065→    snake.angle = turnToward(snake.angle, desiredAngle, turnRate);
  1066→  1066→
  1067→  1067→    // Speed.
  1068→  1068→    let speed = BASE_SPEED;
  1069→  1069→    if (snake.isExtracting) {
  1070→  1070→      speed = EXTRACT_GLIDE_SPEED;
  1071→  1071→    } else if (wantsBoost && snake.points.length > BOOST_MIN_LENGTH) {
  1072→  1072→      speed = BOOST_SPEED;
  1073→  1073→      snake.isBoosting = true;
  1074→  1074→      snake.boostFrameCounter++;
  1075→  1075→      if (snake.boostFrameCounter >= BOOST_DROP_INTERVAL) {
  1076→  1076→        snake.boostFrameCounter = 0;
  1077→  1077→        if (snake.points.length > BOOST_MIN_LENGTH) {
  1078→  1078→          snake.points.pop();
  1079→  1079→          // Reduce food-score (score can go below 0 but body is clamped)
  1080→  1080→          snake.score = Math.max(0, snake.score - 1);
  1081→  1081→        }
  1082→  1082→      }
  1083→  1083→    } else {
  1084→  1084→      snake.isBoosting = false;
  1085→  1085→    }
  1086→  1086→
  1087→  1087→    // Move head.
  1088→  1088→    const head = snake.points[0];
  1089→  1089→    const nx = head.x + Math.cos(snake.angle) * speed;
  1090→  1090→    const ny = head.y + Math.sin(snake.angle) * speed;
  1091→  1091→    snake.points.unshift({ x: nx, y: ny });
  1092→  1092→
  1093→  1093→    // Grow / shrink body to target length based on SCORE.
  1094→  1094→    // targetLen = INITIAL_BODY_LENGTH + foodScore, clamped to [BOOST_MIN_LENGTH+1, MAX_BODY_LENGTH]
  1095→  1095→    const targetLen = Math.max(
  1096→  1096→      BOOST_MIN_LENGTH + 1,
  1097→  1097→      Math.min(MAX_BODY_LENGTH, INITIAL_BODY_LENGTH + snake.score),
  1098→  1098→    );
  1099→  1099→    while (snake.points.length > targetLen) snake.points.pop();
  1100→  1100→
  1101→  1101→    // Size: 8 + sqrt(totalScore) * 0.4.
  1102→  1102→    snake.size = SIZE_BASE + Math.sqrt(INITIAL_SPAWN_SCORE + snake.score) * SIZE_SCORE_FACTOR;
  1103→  1103→  }

  1106→  1106→  // Bot AI — no self-destruct, seek food, predictive evasion
  1107→  1107→  // --------------------------------------------------------------------------
  1108→  1108→
  1109→  1109→  private tickBot(bot: BotSession, now: number): void {
  1110→  1110→    if (bot.points.length === 0 || bot.isDead) return;
  1111→  1111→
  1112→  1112→    if (now >= bot.nextThinkAt) {
  1113→  1113→      bot.nextThinkAt = now + BOT_THINK_INTERVAL_MS + Math.floor(Math.random() * BOT_THINK_JITTER_MS);
  1114→  1114→      const head = bot.points[0];
  1115→  1115→
  1116→  1116→      // --- Find nearest food ---
  1117→  1117→      let bestFood: GridItem | null = null;
  1118→  1118→      let bestFoodDist = Infinity;
  1119→  1119→      const foodQuery = this.grid.queryRadius(head.x, head.y, BOT_FOOD_SCAN_RADIUS);
  1120→  1120→      for (const item of foodQuery.values()) {
  1121→  1121→        if (item.kind !== 'food') continue;
  1122→  1122→        if ((item.value ?? 0) <= 0) continue;
  1123→  1123→        const d = dist(head.x, head.y, item.x, item.y);
  1124→  1124→        if (d < bestFoodDist) {
  1125→  1125→          bestFoodDist = d;
  1126→  1126→          bestFood = item;
  1127→  1127→        }
  1128→  1128→      }
  1129→  1129→
  1130→  1130→      // --- Find nearest threat (foreign body segment) ---
  1131→  1131→      let threatX = 0;
  1132→  1132→      let threatY = 0;
  1133→  1133→      let threatDist = Infinity;
  1134→  1134→      const threatQuery = this.grid.queryRadius(head.x, head.y, BOT_THREAT_SCAN_RADIUS);
  1135→  1135→      for (const item of threatQuery.values()) {
  1136→  1136→        if (item.kind !== 'segment') continue;
  1137→  1137→        if (item.snakeId === bot.id) continue;
  1138→  1138→        if (item.segIdx === 0) continue;
  1139→  1139→        const d = dist(head.x, head.y, item.x, item.y);
  1140→  1140→        if (d < threatDist) {
  1141→  1141→          threatDist = d;
  1142→  1142→          threatX = item.x;
  1143→  1143→          threatY = item.y;
  1144→  1144→        }
  1145→  1145→      }
  1146→  1146→
  1147→  1147→      let desired: number;
  1148→  1148→      const shouldFlee =
  1149→  1149→        threatDist < 140 &&
  1150→  1150→        (bot.personality === 'coward' ||
  1151→  1151→          bot.personality === 'extractor' ||
  1152→  1152→          (bot.personality === 'opportunist' && threatDist < 90));
  1153→  1153→
  1154→  1154→      // --- Predictive evasion against the human player ---
  1155→  1155→      const p = this.player;
  1156→  1156→      if (p && !p.isDead && p.points.length > 0 && now >= p.spawnProtectedUntil) {
  1157→  1157→        const playerHead = p.points[0];
  1158→  1158→        const playerDist = dist(head.x, head.y, playerHead.x, playerHead.y);
  1159→  1159→        if (playerDist < BOT_EVADE_RADIUS * 2) {
  1160→  1160→          // Predict where the player will be in N ticks.
  1161→  1161→          const predictedX = playerHead.x + Math.cos(p.angle) * BOT_PREDICT_SPEED * BOT_PREDICT_AHEAD_TICKS;
  1162→  1162→          const predictedY = playerHead.y + Math.sin(p.angle) * BOT_PREDICT_AHEAD_TICKS;
  1163→  1163→          const distToPredicted = dist(head.x, head.y, predictedX, predictedY);
  1164→  1164→
  1165→  1165→          // If the bot is on a collision course with the predicted position, evade.
  1166→  1166→          if (distToPredicted < (bot.size + p.size) * 3) {
  1167→  1167→            // Evade perpendicular to the player's heading.
  1168→  1168→            const evadeAngle = p.angle + (Math.random() > 0.5 ? Math.PI / 2 : -Math.PI / 2);
  1169→  1169→            desired = evadeAngle;
  1170→  1170→            bot.desiredAngle = desired;
  1171→  1171→            const wantsBoost = bot.personality === 'hunter' && bot.score > 10;
  1172→  1172→            this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1173→  1173→            return;
  1174→  1174→          }
  1175→  1175→        }
  1176→  1176→      }
  1177→  1177→
  1178→  1178→      if (shouldFlee) {
  1179→  1179→        desired = Math.atan2(head.y - threatY, head.x - threatX);
  1180→  1180→      } else if (bestFood) {
  1181→  1181→        desired = Math.atan2(bestFood.y - head.y, bestFood.x - head.x);
  1182→  1182→      } else {
  1183→  1183→        desired = bot.angle + (Math.random() - 0.5) * 0.4;
  1184→  1184→      }
  1185→  1185→
  1186→  1186→      // NO edge avoidance — infinite map, no boundaries.
  1187→  1187→
  1188→  1188→      bot.desiredAngle = desired;
  1189→  1189→    }
  1190→  1190→
  1191→  1191→    // Hunter personality occasionally boosts.
  1192→  1192→    const wantsBoost = bot.personality === 'hunter' && bot.score > 5 && Math.random() < 0.05;
  1193→  1193→    this.tickSnakeMovement(bot, bot.desiredAngle, wantsBoost);
  1194→  1194→  }

  1205→  1205→        kind: 'segment',
  1206→  1206→        x: pts[i].x,
  1207→  1207→        y: pts[i].y,
  1208→  1208→        radius: snake.size,
  1209→  1209→        snakeId: snake.id,

  1211→  1211→      });

  1219→  1219→  private eatFood(): void {
  1220→  1220→    const p = this.player;
  1221→  1221→    if (p && !p.isDead) this.eatFoodForSnake(p);
  1222→  1222→    for (const bot of this.bots.values()) {
  1223→  1223→      if (!bot.isDead) this.eatFoodForSnake(bot);
  1224→  1224→    }
  1225→  1225→  }
  1226→  1226→
  1227→  1227→  private eatFoodForSnake(snake: SnakeBase): void {
  1228→  1228→    if (snake.points.length === 0) return;
  1229→  1229→    const head = snake.points[0];
  1230→  1230→    const nearby = this.grid.queryRadius(head.x, head.y, snake.size + 20);
  1231→  1231→    for (const item of nearby.values()) {
  1232→  1232→      if (item.kind !== 'food') continue;
  1233→  1233→      if ((item.value ?? 0) <= 0) continue;
  1234→  1234→      const d = dist(head.x, head.y, item.x, item.y);
  1235→  1235→      if (d < snake.size + (item.radius || 6) + 6) {
  1236→  1236→        // Eat food — add its value to the snake's food-score.
  1237→  1237→        snake.score += item.value ?? 0;
  1238→  1238→        item.value = 0;
  1239→  1239→        if (item.foodRef) item.foodRef.value = 0;
  1240→  1240→        // Spawn small particle burst for player only.
  1241→  1241→        if (snake.isPlayer) {
  1242→  1242→          this.spawnEatParticles(item.x, item.y, item.radius || 6, item.foodRef?.color ?? '#fbbf24');
  1243→  1243→        }
  1244→  1244→      }
  1245→  1245→    }
  1246→  1246→  }

  1249→  1249→  // Collision detection (body + head-on) — NO wall death
  1250→  1250→  // --------------------------------------------------------------------------
  1251→  1251→
  1252→  1252→  private detectCollisions(now: number): { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] {
  1253→  1253→    const deaths: { deadId: string; killerId?: string; cause: 'body' | 'headon' }[] = [];
  1254→  1254→    const seenDead = new Set<string>();
  1255→  1255→    const all: SnakeBase[] = [];
  1256→  1256→    if (this.player && !this.player.isDead) all.push(this.player);
  1257→  1257→    for (const bot of this.bots.values()) {
  1258→  1258→      if (!bot.isDead) all.push(bot);
  1259→  1259→    }
  1260→  1260→
  1261→  1261→    // --- Head-to-head collision (checked first, takes priority) ---
  1262→  1262→    const headOnChecked = new Set<string>();
  1263→  1263→    for (let i = 0; i < all.length; i++) {
  1264→  1264→      for (let j = i + 1; j < all.length; j++) {
  1265→  1265→        const a = all[i];
  1266→  1266→        const b = all[j];
  1267→  1267→        if (a.isDead || b.isDead) continue;
  1268→  1268→        if (now < a.spawnProtectedUntil || now < b.spawnProtectedUntil) continue;
  1269→  1269→        const pairKey = a.id < b.id ? `${a.id}:${b.id}` : `${b.id}:${a.id}`;
  1270→  1270→        if (headOnChecked.has(pairKey)) continue;
  1271→  1271→        headOnChecked.add(pairKey);
  1272→  1272→
  1273→  1273→        const ha = a.points[0];
  1274→  1274→        const hb = b.points[0];
  1275→  1275→        const d = dist(ha.x, ha.y, hb.x, hb.y);
  1276→  1276→        const hitDist = (a.size + b.size) * HEAD_ON_HIT_FACTOR;
  1277→  1277→        if (d >= hitDist) continue;
  1278→  1278→
  1279→  1279→        // Head-on collision! Determine winner based on rules.
  1280→  1280→        const aBoosting = a.isBoosting;
  1281→  1281→        const bBoosting = b.isBoosting;
  1282→  1282→        const aTotalScore = INITIAL_SPAWN_SCORE + a.score;
  1283→  1283→        const bTotalScore = INITIAL_SPAWN_SCORE + b.score;
  1284→  1284→
  1285→  1285→        let aDies = false;
  1286→  1286→        let bDies = false;
  1287→  1287→
  1288→  1288→        if (aTotalScore === bTotalScore) {
  1289→  1289→          // Tie — both die
  1290→  1290→          aDies = true;
  1291→  1291→          bDies = true;
  1292→  1292→        } else if (aBoosting && !bBoosting) {
  1293→  1293→          // A is boosting, B is not → A wins (boost advantage)
  1294→  1294→          bDies = true;
  1295→  1295→        } else if (bBoosting && !aBoosting) {
  1296→  1296→          // B is boosting, A is not → B wins (boost advantage)
  1297→  1297→          aDies = true;
  1298→  1298→        } else {
  1299→  1299→          // Both boosting or neither boosting → higher score wins
  1300→  1300→          if (aTotalScore > bTotalScore) {
  1301→  1301→            bDies = true;
  1302→  1302→          } else {
  1303→  1303→            aDies = true;
  1304→  1304→          }
  1305→  1305→        }
  1306→  1306→
  1307→  1307→        if (aDies && !seenDead.has(a.id)) {
  1308→  1308→          deaths.push({ deadId: a.id, killerId: b.id, cause: 'headon' });
  1309→  1309→          seenDead.add(a.id);
  1310→  1310→        }
  1311→  1311→        if (bDies && !seenDead.has(b.id)) {
  1312→  1312→          deaths.push({ deadId: b.id, killerId: a.id, cause: 'headon' });
  1313→  1313→          seenDead.add(b.id);
  1314→  1314→        }
  1315→  1315→      }
  1316→  1316→    }
  1317→  1317→
  1318→  1318→    // --- Body collision: head vs foreign non-head segment ---
  1319→  1319→    for (const snake of all) {
  1320→  1320→      if (snake.isDead) continue;
  1321→  1321→      if (snake.points.length === 0) continue;
  1322→  1322→      if (now < snake.spawnProtectedUntil) continue;
  1323→  1323→      if (seenDead.has(snake.id)) continue;
  1324→  1324→      const head = snake.points[0];
  1325→  1325→
  1326→  1326→      // NO wall collision — infinite map.
  1327→  1327→
  1328→  1328→      const queryR = snake.size + 30;
  1329→  1329→      const nearby = this.grid.queryRadius(head.x, head.y, queryR);
  1330→  1330→      for (const item of nearby.values()) {
  1331→  1331→        if (item.kind !== 'segment') continue;
  1332→  1332→        // Own-collision immunity — skip own segments.
  1333→  1333→        if (item.snakeId === snake.id) continue;
  1334→  1334→        // Skip head segments (handled by head-on collision above).
  1335→  1335→        if (item.segIdx === 0) continue;
  1336→  1336→        const d = dist(head.x, head.y, item.x, item.y);
  1337→  1337→        if (d < (snake.size + item.radius) * COLLISION_HIT_FACTOR) {
  1338→  1338→          deaths.push({ deadId: snake.id, killerId: item.snakeId, cause: 'body' });
  1339→  1339→          seenDead.add(snake.id);
  1340→  1340→          break;
  1341→  1341→        }
  1342→  1342→      }
  1343→  1343→    }
  1344→  1344→    return deaths;
  1345→  1345→  }

  1347→  1347→  // --------------------------------------------------------------------------
  1348→  1348→  // Player death
  1349→  1349→  // --------------------------------------------------------------------------
  1350→  1350→
  1351→  1351→  private handlePlayerDeath(): void {
  1352→  1352→    const p = this.player;
  1353→  1353→    if (!p) return;

  1690→  1690→
  1691→  1691→      for (let j = 0; j < allSnakes.length; j++) {
  1692→  1692→        if (i === j) continue;
  1693→  1693→        const b = allSnakes[j];

  1695→  1695→        const bTotalScore = INITIAL_SPAWN_SCORE + b.score;

  1698→  1698→        if (aTotalScore < bTotalScore) {
  1699→  1699→          const d = dist(a.points[0].x, a.points[0].y, b.points[0].x, b.points[0].y);

  1702→  1702→            // The larger snake (b) fades.
  1703→  1703→            const current = opacityMap.get(b.id) ?? 1.0;
  1704→  1704→            opacityMap.set(b.id, Math.min(current, OPACITY_FADE_TO));
  1705→  1705→          }
  1706→  1706→        }
  1707→  1707→      }
  1708→  1708→    }
  1709→  1709→
  1710→  1710→    // Render bots first, player last (on top).
  1711→  1711→    for (const bot of this.bots.values()) {

  1714→  1714→      ctx.save();

  1716→  1716→      this.drawSnakeSnapshot(rc, bot);
  1717→  1717→      ctx.restore();

  1719→  1719→    if (!p.isDead) {
  1720→  1720→      const opacity = opacityMap.get(p.id) ?? 1.0;
