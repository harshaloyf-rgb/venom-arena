     1→     1→// ============================================================================
     2→     2→// index.ts — Venom Arena Socket.IO game server (mini-service, port 3001).
     3→     3→// ----------------------------------------------------------------------------
     4→     4→// Server-authoritative multiplayer snake. The Next.js app (port 3000) talks
     5→     5→// to this server via the Caddy gateway using io("/?XTransformPort=3001").
     6→     6→//
     7→     7→// Architectural fixes vs the OLD broken server (see worklog ANALYSIS-1):
     8→     8→//  * Auth is mandatory — sockets without a valid JWT (verified via
     9→     9→//    POST /api/match/verify) are disconnected at the middleware. No
    10→    10→//    "auth optional, admit anyway" path.
    11→    11→//  * Identity is the verify response only. We never trust any subsequent
    12→    12→//    client-supplied userTag/name/color/skin.
    13→    13→//  * One socket per userTag — a new connection kicks the prior socket.
    14→    14→//  * Movement is server-authoritative — clients send only `angle`; the
    15→    15→//    server computes position. Teleport is impossible.
    16→    16→//  * Buy-in is deducted atomically by /api/match/join (Next.js + Prisma
    17→    17→//    transaction). We never touch the DB directly.
    18→    18→//  * Match results are reported exactly once via /api/match/result, guarded
    19→    19→//    by a `matchSettling` flag to prevent double-credit on disconnect races.
    20→    20→//  * Recursive setTimeout (not setInterval) for both tick and broadcast
    21→    21→//    loops — a slow tick can't overlap the next.
    22→    22→//  * Every tick is wrapped in try/catch; one bad snake never kills the loop.
    23→    23→//  * Spatial hash grid (see spatial-grid.ts) keeps collision detection
    24→    24→//    near-linear instead of O(n²).
    25→    25→//  * Broadcasts happen at 20 Hz, not every tick, and snapshot points are
    26→    26→//    downsampled to 60 to keep payloads small.
    27→    27→//  * uncaughtException / unhandledRejection are logged, never fatal.
    28→    28→//  * Graceful shutdown on SIGINT/SIGTERM broadcasts `server_shutdown` then exits.
    29→    29→// ============================================================================

   137→   137→  try {
   138→   138→    const res = await fetch(`${NEXT_APP_URL}/api/match/verify`, {
   139→   139→      method: 'POST',
   140→   140→      headers: { 'Content-Type': 'application/json', 'x-internal-secret': INTERNAL_SECRET },
   141→   141→      body: JSON.stringify({ token }),
   142→   142→      signal: controller.signal,
   143→   143→    });
   144→   144→    return (await res.json()) as VerifyResponse;
   145→   145→  } finally {
   146→   146→    clearTimeout(timer);

   230→   230→/** Max real players per arena shard before auto-creating a new instance. */
   231→   231→const MAX_PLAYERS_PER_SHARD = 1000;
   232→   232→
   233→   233→/**
   234→   234→ * Get-or-create an ArenaRoom for `arenaId`. If the primary shard is full
   235→   235→ * (>= MAX_PLAYERS_PER_SHARD real players), find or create the next shard.
   236→   236→ * Shards are keyed as `{arenaId}` (shard 0), `{arenaId}#2` (shard 1), etc.
   237→   237→ */
   238→   238→function getOrCreateRoom(arenaId: string): ArenaRoom | null {
   239→   239→  const arena = getArenaById(arenaId);
   240→   240→  if (!arena) return null;
   241→   241→
   242→   242→  // Try shard 0, then 1, 2, ... until we find one with capacity.
   243→   243→  let shardIdx = 0;
   244→   244→  while (true) {
   245→   245→    const roomKey = shardIdx === 0 ? arenaId : `${arenaId}#${shardIdx + 1}`;
   246→   246→    let room = rooms.get(roomKey);
   247→   247→    if (room && room.players.size < MAX_PLAYERS_PER_SHARD) {
   248→   248→      return room;
   249→   249→    }
   250→   250→    if (!room) {
   251→   251→      // Create a new shard.
   252→   252→      room = createArenaRoom(arena);
   253→   253→      rooms.set(roomKey, room);
   254→   254→      ensureBots(room);
   255→   255→      log('info', `Created arena shard ${roomKey} with ${room.bots.size} bots (shard ${shardIdx})`);
   256→   256→      return room;
   257→   257→    }
   258→   258→    shardIdx++;
   259→   259→    // Safety: don't infinite-loop if somehow all shards are full (shouldn't happen).
   260→   260→    if (shardIdx > 200) {
   261→   261→      log('error', `Arena ${arenaId} has >200 shards, rejecting join`);
   262→   262→      return null;
   263→   263→    }
   264→   264→  }
   265→   265→}

   279→   279→interface MatchResultClientPayload {
   280→   280→  outcome: 'extract' | 'death';
   281→   281→  arenaId: string;
   282→   282→  arenaName: string;
   283→   283→  chipsExtracted: number;

   293→   293→
   294→   294→/**
   295→   295→ * Settle a player's match: call /api/match/result, emit `match_result` to
   296→   296→ * the player, and remove them from the room. Idempotent via the
   297→   297→ * `matchSettling` flag — concurrent calls short-circuit.
   298→   298→ */
   299→   299→async function settleMatch(
   300→   300→  room: ArenaRoom,
   301→   301→  session: PlayerSession,
   302→   302→  outcome: 'extract' | 'death',
   303→   303→  killer?: SnakeBase,
   304→   304→): Promise<void> {
   305→   305→  if (session.matchSettling) return;
   306→   306→  session.matchSettling = true;
   307→   307→  session.isExtracting = false;
   308→   308→  session.extractionProgress = 0;
   309→   309→
   310→   310→  const now = Date.now();
   311→   311→  const durationSeconds = Math.max(0, Math.floor((now - session.joinedAt) / 1000));
   312→   312→  const carriedChips = Math.max(0, Math.floor(session.carriedChips));
   313→   313→  const kills = session.kills;
   314→   314→  const killerTag = killer?.userTag;
   315→   315→  const score = session.score;
   316→   316→
   317→   317→  // Dynamic commission: 0% if <=3 real players in arena, 35% if >=4.
   318→   318→  const realPlayerCount = room.players.size;
   319→   319→  const commissionRate = realPlayerCount <= 3 ? 0 : 0.35;
   320→   320→  const commission = outcome === 'extract' ? Math.floor(carriedChips * commissionRate) : 0;
   321→   321→  const bankedAmount = outcome === 'extract' ? (carriedChips - commission) : 0;
   322→   322→
   323→   323→  const result = await reportMatchResult({
   324→   324→    userTag: session.identity.userTag,
   325→   325→    arenaId: room.arena.id,
   326→   326→    outcome,
   327→   327→    carriedChips,
   328→   328→    kills,
   329→   329→    durationSeconds,
   330→   330→    killerTag,
   331→   331→    score,
   332→   332→    bankedAmount,
   333→   333→  });
   334→   334→
   335→   335→  const payload: MatchResultClientPayload = {
   336→   336→    outcome,
   337→   337→    arenaId: room.arena.id,
   338→   338→    arenaName: room.arena.name,
   339→   339→    chipsExtracted: outcome === 'extract' ? carriedChips : 0,
   340→   340→    commission,
   341→   341→    bankedAmount,
   342→   342→    kills,
   343→   343→    score,
   344→   344→    xpGained: result?.xpGained ?? 0,
   345→   345→    newLevel: result?.newLevel ?? session.identity.level,
   346→   346→    newBankedChips: result?.newBankedChips ?? 0,
   347→   347→    durationSeconds,
   348→   348→  };
   349→   349→
   350→   350→  const socket = io.sockets.sockets.get(session.id);
   351→   351→  if (socket) {
   352→   352→    // Emit death BEFORE match_result so the client's onDeath handler
   353→   353→    // runs first (sets up post-death replay recording). If match_result
   354→   354→    // arrived first it would set matchEndedRef=true and cause onDeath
   355→   355→    // to bail via its idempotency guard.
   356→   356→    if (outcome === 'death') {
   357→   357→      socket.emit('death', {
   358→   358→        killerId: killer?.id,
   359→   359→        killerName: killer?.name,
   360→   360→        killerTag: killer?.userTag,
   361→   361→        killerColor: killer?.color,
   362→   362→        killerIsBot: killer?.isBot ?? true,
   363→   363→      });
   364→   364→    }
   365→   365→    socket.emit('match_result', payload);
   366→   366→  }
   367→   367→
   368→   368→  // For death: keep the player in the room for 16 s so the broadcast
   369→   369→  // loop continues sending snapshots.  The client records 15 s of
   370→   370→  // post-death frames (300 frames at 20 Hz) for the replay.  After the
   371→   371→  // window expires we clean up.  The session is already marked
   372→   372→  // isDead + matchSettling, so the tick loop skips movement/collisions.
   373→   373→  if (outcome === 'death') {
   374→   374→    const sid = session.id;
   375→   375→    const roomId = room.arena.id;
   376→   376→    setTimeout(() => {
   377→   377→      const r = rooms.get(roomId);
   378→   378→      if (r) r.players.delete(sid);
   379→   379→      const s = io.sockets.sockets.get(sid);
   380→   380→      if (s) {
   381→   381→        const d = getSocketData(s);
   382→   382→        d.playerSession = undefined;
   383→   383→      }
   384→   384→      log('info', `Post-death replay window expired — removed ${session.identity.userTag} from ${roomId}`);
   385→   385→    }, 16_000);
   386→   386→  } else {
   387→   387→    // Extract: remove immediately (no post-death replay needed).
   388→   388→    room.players.delete(session.id);
   389→   389→    if (socket) {
   390→   390→      const data = getSocketData(socket);
   391→   391→      data.playerSession = undefined;
   392→   392→    }
   393→   393→  }
   394→   394→  log('info', `${session.identity.userTag} ${outcome} in ${room.arena.id}: chips=${carriedChips} kills=${kills} dur=${durationSeconds}s`);
   395→   395→}

   397→   397→// ----------------------------------------------------------------------------
   398→   398→// Tick loop (recursive setTimeout — a slow tick can't overlap the next)
   399→   399→// ----------------------------------------------------------------------------
   400→   400→
   401→   401→function tickRoom(room: ArenaRoom, now: number): void {
   402→   402→  room.tick++;
   403→   403→
   404→   404→  // 1) Rebuild spatial grid from scratch each tick.
   405→   405→  room.grid.clear();
   406→   406→
   407→   407→  // 2) Insert all snake body segments (skip dead / empty).
   408→   408→  for (const snake of collectAllSnakes(room)) {
   409→   409→    if (snake.isDead) continue;
   410→   410→    if (snake.points.length === 0) continue;
   411→   411→    for (let i = 0; i < snake.points.length; i++) {
   412→   412→      const p = snake.points[i];
   413→   413→      room.grid.insert({
   414→   414→        id: `${snake.id}:${i}`,
   415→   415→        kind: 'segment',
   416→   416→        x: p.x,
   417→   417→        y: p.y,
   418→   418→        radius: snake.size,
   419→   419→        snakeId: snake.id,
   420→   420→        segIdx: i,
   421→   421→      });
   422→   422→    }
   423→   423→  }
   424→   424→
   425→   425→  // 3) Insert all food (skip already-eaten with value=0).
   426→   426→  for (const food of room.foods) {
   427→   427→    if (food.value <= 0) continue;
   428→   428→    room.grid.insert({
   429→   429→      id: food.id,
   430→   430→      kind: 'food',
   431→   431→      x: food.x,
   432→   432→      y: food.y,
   433→   433→      radius: food.size,
   434→   434→      value: food.value,
   435→   435→      isStarChip: food.isStarChip,
   436→   436→      color: food.color,
   437→   437→      foodRef: food, // <-- reference to the REAL food object so eatFood can zero it
   438→   438→    });
   439→   439→  }
   440→   440→
   441→   441→  // 4) Bot AI tick.
   442→   442→  for (const bot of room.bots.values()) {
   443→   443→    if (bot.isDead) continue;
   444→   444→    try {
   445→   445→      tickBot(bot, room, now);
   446→   446→    } catch (err) {
   447→   447→      log('error', `Bot tick error in ${room.arena.id}: ${(err as Error).message}`);
   448→   448→    }
   449→   449→  }
   450→   450→
   451→   451→  // 5) Player movement (server-authoritative — apply desired angle + speed).
   452→   452→  for (const session of room.players.values()) {
   453→   453→    if (session.isDead || session.matchSettling) continue;
   454→   454→    try {
   455→   455→      const dropped = tickSnakeMovement(session, session.desiredAngle, session.wantsBoost);
   456→   456→      // Add boost-dropped food orbs to the room.
   457→   457→      for (const pt of dropped) {
   458→   458→        room.foods.push({
   459→   459→          id: `food-${room.arena.id}-${room.foodIdCounter++}`,
   460→   460→          x: pt.x,
   461→   461→          y: pt.y,
   462→   462→          size: FOOD_ORB_SMALL.radius,
   463→   463→          value: 1,
   464→   464→          isStarChip: false,
   465→   465→          color: FOOD_ORB_SMALL.color,
   466→   466→          glowColor: FOOD_ORB_SMALL.glowColor,
   467→   467→          orbSize: 'small',
   468→   468→        });
   469→   469→      }
   470→   470→    } catch (err) {
   471→   471→      log('error', `Player movement error: ${(err as Error).message}`);
   472→   472→    }
   473→   473→  }
   474→   474→
   475→   475→  // 6) Collision detection (body + head-on).
   476→   476→  let deaths: PendingDeath[] = [];
   477→   477→  try {
   478→   478→    deaths = detectCollisions(room, now);
   479→   479→    // Head-on collisions processed AFTER body collisions (already-dead snakes skipped).
   480→   480→    const headOnDeaths = detectHeadOnCollisions(room, now);
   481→   481→    deaths = deaths.concat(headOnDeaths);
   482→   482→  } catch (err) {
   483→   483→    log('error', `Collision detection error in ${room.arena.id}: ${(err as Error).message}`);
   484→   484→  }
   485→   485→
   486→   486→  // 7) Apply deaths with CORRECT drop rules per spec:
   487→   487→  //    Body/headOn collision: drop score orbs (ALL snakes, including selfDestruct bots) + 10 stars (real players only).
   488→   488→  //    Wall death: drop 0 food (score destroyed) + 10 stars (real players only).
   489→   489→  //    Bot selfDestruct WALL death: 0 food, 0 stars (vanish cleanly).
   490→   490→  //    Bot selfDestruct COLLISION death: STILL drops food (only wall death vanishes cleanly).
   491→   491→  for (const death of deaths) {
   492→   492→    const dead = room.players.get(death.deadId) || (room.bots.get(death.deadId) as SnakeBase | undefined);
   493→   493→    if (!dead || dead.isDead) continue;
   494→   494→    dead.isDead = true;
   495→   495→
   496→   496→    const headX = dead.points[0]?.x ?? 0;
   497→   497→    const headY = dead.points[0]?.y ?? 0;
   498→   498→
   499→   499→    if (death.cause === 'wall') {
   500→   500→      // Wall death: NO food orbs (score destroyed).
   501→   501→      // Stars: real players always get 10 stars; selfDestruct bots get 0.
   502→   502→      if (dead.isPlayer && dead.carriedChips > 0) {
   503→   503→        dropStarsAtDeath(room, headX, headY, dead.carriedChips);
   504→   504→      }
   505→   505→      // Bot wall death (selfDestruct): 0 food, 0 stars — vanish cleanly.
   506→   506→    } else {
   507→   507→      // Body or headOn collision: ALL snakes drop score orbs (sum = snake.score).
   508→   508→      // selfDestruct bots that die by COLLISION still drop food (only WALL death vanishes cleanly).
   509→   509→      dropScoreOrbsAtBody(room, dead.points, dead.score, dead.color);
   510→   510→      // Stars: real players only.
   511→   511→      if (dead.isPlayer && dead.carriedChips > 0) {
   512→   512→        dropStarsAtDeath(room, headX, headY, dead.carriedChips);
   513→   513→      }
   514→   514→    }
   515→   515→
   516→   516→    // Credit killer's kill counter (only if killer is alive). Bots don't track kills.
   517→   517→    const killer = death.killerId && death.killerId !== 'wall'
   518→   518→      ? (room.players.get(death.killerId) ?? room.bots.get(death.killerId))
   519→   519→      : undefined;
   520→   520→    if (killer && !killer.isDead && killer.isPlayer) {
   521→   521→      (killer as PlayerSession).kills += 1;
   522→   522→    }
   523→   523→
   524→   524→    if (dead.isPlayer) {
   525→   525→      const session = dead as PlayerSession;
   526→   526→      log('info', `${session.identity.userTag} died in ${room.arena.id} (cause=${death.cause}, killer=${death.killerId ?? 'none'})`);
   527→   527→      // Settle asynchronously — never throw out of the tick loop.
   528→   528→      settleMatch(room, session, 'death', death.cause === 'wall' ? undefined : killer).catch((err) => {
   529→   529→        log('error', `settleMatch(death) failed: ${(err as Error).message}`);
   530→   530→      });
   531→   531→    } else {
   532→   532→      // Bot dies: remove + respawn a fresh one to keep the bot count stable.
   533→   533→      room.bots.delete(dead.id);
   534→   534→      try {
   535→   535→        const fresh = spawnNewBot(room);
   536→   536→        if (fresh) room.bots.set(fresh.id, fresh);
   537→   537→      } catch (err) {
   538→   538→        log('error', `Bot respawn error: ${(err as Error).message}`);
   539→   539→      }
   540→   540→    }
   541→   541→  }
   542→   542→
   543→   543→  // 8) Food collision (credits carriedChips, marks food as eaten via sentinel).
   544→   544→  try {
   545→   545→    eatFood(room);
   546→   546→  } catch (err) {
   547→   547→    log('error', `eatFood error in ${room.arena.id}: ${(err as Error).message}`);
   548→   548→  }
   549→   549→
   550→   550→  // 9) Replenish food up to target.
   551→   551→  try {
   552→   552→    replenishFood(room);
   553→   553→  } catch (err) {
   554→   554→    log('error', `replenishFood error in ${room.arena.id}: ${(err as Error).message}`);
   555→   555→  }
   556→   556→
   557→   557→  // 10) Extraction progress for players. NO zone check — extract anywhere.
   558→   558→  for (const session of room.players.values()) {
   559→   559→    if (session.isDead || session.matchSettling) continue;
   560→   560→    if (!session.isExtracting) continue;
   561→   561→
   562→   562→    session.extractionProgress += TICK_MS;
   563→   563→    const progress = Math.min(1, session.extractionProgress / EXTRACT_DURATION_MS);
   564→   564→    const sock = io.sockets.sockets.get(session.id);
   565→   565→    if (sock) sock.emit('extract_progress', { progress });
   566→   566→
   567→   567→    if (session.extractionProgress >= EXTRACT_DURATION_MS) {
   568→   568→      // Extraction success — settle asynchronously.
   569→   569→      settleMatch(room, session, 'extract').catch((err) => {
   570→   570→        log('error', `settleMatch(extract) failed: ${(err as Error).message}`);
   571→   571→      });
   572→   572→    }
   573→   573→  }
   574→   574→
   575→   575→  // 11) Recompute leader + expire chat bubbles.
   576→   576→  try {
   577→   577→    recomputeLeader(room);
   578→   578→  } catch (err) {
   579→   579→    log('error', `recomputeLeader error: ${(err as Error).message}`);
   580→   580→  }
   581→   581→  try {
   582→   582→    expireChat(room, now);
   583→   583→  } catch (err) {
   584→   584→    log('error', `expireChat error: ${(err as Error).message}`);
   585→   585→  }
   586→   586→}

   588→   588→function tickOnce(): void {
   589→   589→  const now = Date.now();
   590→   590→  try {
   591→   591→    for (const room of rooms.values()) {
   592→   592→      // Skip empty rooms (no players) for CPU savings — bots stay idle.
   593→   593→      if (room.players.size === 0) continue;
   594→   594→      try {
   595→   595→        tickRoom(room, now);
   596→   596→      } catch (err) {
   597→   597→        log('error', `Tick error in arena ${room.arena.id}: ${(err as Error).message}`);
   598→   598→      }
   599→   599→    }
   600→   600→  } catch (err) {
   601→   601→    log('error', `Tick loop fatal: ${(err as Error).message}`);
   602→   602→  } finally {
   603→   603→    setTimeout(tickOnce, TICK_MS);
   604→   604→  }
   605→   605→}

   637→   637→    log('error', `Broadcast loop fatal: ${(err as Error).message}`);
   638→   638→  } finally {
   639→   639→    setTimeout(broadcastOnce, BROADCAST_MS);
   640→   640→  }

   714→   714→

   719→   719→
   720→   720→  // ----- disconnect -----
   721→   721→  socket.on('disconnect', (reason: string) => {

   724→   724→    const ident = getSocketData(socket).identity;
   725→   725→    if (ident && userTagToSocket.get(ident.userTag) === socket.id) {
   726→   726→      userTagToSocket.delete(ident.userTag);
   727→   727→    }
   728→   728→    log('info', `Socket disconnected: ${ident?.userTag ?? 'unknown'} (${reason})`);
   729→   729→  });

   731→   731→  socket.on('error', (err: Error) => {
   732→   732→    log('error', `Socket error (${socket.id}): ${err.message}`);
   733→   733→  });
   734→   734→});

   740→   740→async function handleJoinArena(socket: Socket, payload: unknown): Promise<void> {
   741→   741→  const data = getSocketData(socket);
   742→   742→  const identity = data.identity;
   743→   743→  if (!identity) return;
   744→   744→
   745→   745→  if (!payload || typeof payload !== 'object') {
   746→   746→    socket.emit('join_error', { reason: 'invalid_arena' });
   747→   747→    return;
   748→   748→  }
   749→   749→  const arenaId = String((payload as { arenaId?: unknown }).arenaId || '');
   750→   750→  const arena = getArenaById(arenaId);
   751→   751→  if (!arena) {
   752→   752→    socket.emit('join_error', { reason: 'invalid_arena' });
   753→   753→    return;
   754→   754→  }
   755→   755→
   756→   756→  // Already in a match? Reject.
   757→   757→  if (data.playerSession) {
   758→   758→    socket.emit('join_error', { reason: 'already_in_match' });
   759→   759→    return;
   760→   760→  }
   761→   761→
   762→   762→  // Server-to-server: atomically deduct buyIn on the Next.js side.
   763→   763→  let joinResult;
   764→   764→  try {
   765→   765→    joinResult = await joinMatch(identity.userTag, arenaId);
   766→   766→  } catch (err) {
   767→   767→    log('error', `joinMatch fetch failed: ${(err as Error).message}`);
   768→   768→    socket.emit('join_error', { reason: 'invalid_arena' });
   769→   769→    return;
   770→   770→  }
   771→   771→
   772→   772→  if (!joinResult.ok) {
   773→   773→    const reason = joinResult.reason;
   774→   774→    if (reason === 'insufficient_chips') {
   775→   775→      socket.emit('join_error', { reason: 'insufficient_chips' });
   776→   776→      return;
   777→   777→    }
   778→   778→    if (reason === 'banned') {
   779→   779→      socket.emit('join_error', { reason: 'banned' });
   780→   780→      socket.emit('kicked', { reason: 'Account banned' });
   781→   781→      socket.disconnect(true);
   782→   782→      return;
   783→   783→    }
   784→   784→    socket.emit('join_error', { reason: 'invalid_arena' });
   785→   785→    return;
   786→   786→  }
   787→   787→
   788→   788→  // Spawn the player.
   789→   789→  const room = getOrCreateRoom(arenaId);
   790→   790→  if (!room) {
   791→   791→    socket.emit('join_error', { reason: 'invalid_arena' });
   792→   792→    return;
   793→   793→  }
   794→   794→
   795→   795→  const realPlayerCount = [...room.players.values()].filter(p => !p.isDead && !p.matchSettling).length;
   796→   796→  const mapRadius = getDynamicMapRadius(Math.max(1, realPlayerCount));
   797→   797→  const spawn = findSafeSpawnPoint(room, mapRadius - 200, room.mapCenterX, room.mapCenterY);
   798→   798→  const angle = Math.random() * Math.PI * 2;
   799→   799→  const session: PlayerSession = {
   800→   800→    id: socket.id,
   801→   801→    name: identity.name,
   802→   802→    userTag: identity.userTag,
   803→   803→    country: identity.country,
   804→   804→    points: initialBody(spawn.x, spawn.y, angle, INITIAL_BODY_LENGTH),
   805→   805→    angle,
   806→   806→    size: SIZE_BASE + Math.sqrt(INITIAL_SPAWN_SCORE) * SIZE_SCORE_FACTOR,
   807→   807→    color: identity.color,
   808→   808→    secondaryColor: identity.secondaryColor,
   809→   809→    isPlayer: true,
   810→   810→    isBot: false,
   811→   811→    carriedChips: 0,
   812→   812→    score: INITIAL_SPAWN_SCORE,
   813→   813→    boostFrameCounter: 0,
   814→   814→    isExtracting: false,
   815→   815→    extractionProgress: 0,
   816→   816→    isDead: false,
   817→   817→    spawnProtectedUntil: Date.now() + RESPAWN_INVULN_MS,
   818→   818→    identity,
   819→   819→    desiredAngle: angle,
   820→   820→    wantsBoost: false,
   821→   821→    kills: 0,
   822→   822→    joinedAt: Date.now(),
   823→   823→    lastInputAt: 0,
   824→   824→    inputDropCount: 0,
   825→   825→    lastChatAt: 0,
   826→   826→    arenaId,
   827→   827→    matchSettling: false,
   828→   828→  };
   829→   829→
   830→   830→  room.players.set(socket.id, session);
   831→   831→  data.playerSession = session;
   832→   832→
   833→   833→  socket.emit('joined', {
   834→   834→    arenaId,
   835→   835→    worldSize: WORLD_SIZE,
   836→   836→    yourId: socket.id,
   837→   837→  });
   838→   838→
   839→   839→  log('info', `${identity.userTag} joined ${arenaId} (buyIn=${arena.buyIn})`);
   840→   840→}

   842→   842→/** Validate + rate-limit a client input packet. Server applies position; client only sends angle. */
   843→   843→function handleInput(socket: Socket, payload: unknown): void {
   844→   844→  const session = getSocketData(socket).playerSession;
   845→   845→  if (!session || session.isDead || session.matchSettling) return;
   846→   846→
   847→   847→  if (!payload || typeof payload !== 'object') return;
   848→   848→  const obj = payload as { angle?: unknown; wantsBoost?: unknown };
   849→   849→  const angleRaw = obj.angle;
   850→   850→  const wantsBoostRaw = obj.wantsBoost;
   851→   851→
   852→   852→  if (typeof angleRaw !== 'number' || !Number.isFinite(angleRaw)) return;
   853→   853→  if (typeof wantsBoostRaw !== 'boolean') return;
   854→   854→
   855→   855→  // Normalize angle into [0, 2π).
   856→   856→  let angle = angleRaw;
   857→   857→  while (angle < 0) angle += 2 * Math.PI;
   858→   858→  while (angle >= 2 * Math.PI) angle -= 2 * Math.PI;
   859→   859→
   860→   860→  // Rate-limit: max MAX_SNAPSHOTS_PER_SECOND per second.
   861→   861→  const now = Date.now();
   862→   862→  const delta = now - session.lastInputAt;
   863→   863→  if (delta < INPUT_MIN_INTERVAL_MS) {
   864→   864→    session.inputDropCount++;
   865→   865→    if (session.inputDropCount % 20 === 0) {
   866→   866→      log('warn', `${session.identity.userTag} input flood: ${session.inputDropCount} packets dropped`);
   867→   867→    }
   868→   868→    return;
   869→   869→  }
   870→   870→  session.lastInputAt = now;
   871→   871→  session.inputDropCount = 0;
   872→   872→
   873→   873→  session.desiredAngle = angle;
   874→   874→  session.wantsBoost = wantsBoostRaw;
   875→   875→}

   897→   897→/** Validate + rate-limit a chat message; broadcast to the room. */
   898→   898→function handleChat(socket: Socket, payload: unknown): void {
   899→   899→  const session = getSocketData(socket).playerSession;
   900→   900→  const identity = getSocketData(socket).identity;
   901→   901→  if (!session || !identity) return;
   902→   902→  if (!payload || typeof payload !== 'object') return;
   903→   903→  const raw = (payload as { message?: unknown }).message;
   904→   904→  if (typeof raw !== 'string') return;
   905→   905→  const message = raw.trim().slice(0, CHAT_MAX_LEN);
   906→   906→  if (message.length === 0) return;
   907→   907→
   908→   908→  const now = Date.now();
   909→   909→  if (now - session.lastChatAt < CHAT_MIN_INTERVAL_MS) return;
   910→   910→  session.lastChatAt = now;
   911→   911→
   912→   912→  session.chatMessage = message;
   913→   913→  session.chatExpiry = now + 4000;
   914→   914→
   915→   915→  const room = session.arenaId ? rooms.get(session.arenaId) : null;
   916→   916→  if (!room) return;
   917→   917→  const chatPayload = {
   918→   918→    senderId: session.id,
   919→   919→    senderName: session.name,
   920→   920→    senderTag: session.userTag,
   921→   921→    message,
   922→   922→  };
   923→   923→  for (const socketId of room.players.keys()) {
   924→   924→    io.to(socketId).emit('chat', chatPayload);
   925→   925→  }
   926→   926→}

   928→   928→/** Remove a player from their arena; scatter carried chips as star-chips (no DB write). */
   929→   929→function handleLeave(socket: Socket, reason: string): void {
   930→   930→  const data = getSocketData(socket);
   931→   931→  const session = data.playerSession;
   932→   932→  if (!session) return;
   933→   933→
   934→   934→  const room = session.arenaId ? rooms.get(session.arenaId) : null;
   935→   935→  if (room) {
   936→   936→    // Forfeit carried chips as 10 star collectibles at death position.
   937→   937→    if (!session.isDead && !session.matchSettling && session.carriedChips > 0) {
   938→   938→      try {
   939→   939→        dropStarsAtDeath(room, session.points[0]?.x ?? 0, session.points[0]?.y ?? 0, session.carriedChips);
   940→   940→      } catch (err) {
   941→   941→        log('error', `dropStarsAtDeath error on leave: ${(err as Error).message}`);
   942→   942→      }
   943→   943→    }
   944→   944→    room.players.delete(socket.id);
   945→   945→  }
   946→   946→
   947→   947→  data.playerSession = undefined;
   948→   948→  log('info', `${session.identity.userTag} left arena ${session.arenaId ?? '?'} (${reason})`);
   949→   949→}

   964→   964→function gracefulShutdown(signal: string): void {
   965→   965→  log('info', `${signal} received — broadcasting server_shutdown and exiting`);
   966→   966→  try {
   967→   967→    io.emit('server_shutdown', {});
   968→   968→  } catch (err) {
   969→   969→    log('error', `shutdown broadcast failed: ${(err as Error).message}`);
   970→   970→  }
   971→   971→  // Give sockets a moment to flush, then exit.
   972→   972→  setTimeout(() => process.exit(0), 400);
   973→   973→  // Hard exit fallback if flush hangs.
   974→   974→  setTimeout(() => process.exit(1), 2000).unref();
   975→   975→}
   976→   976→
   977→   977→process.on('SIGINT', () => gracefulShutdown('SIGINT'));
   978→   978→process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
   979→   979→// Log SIGHUP so we can diagnose if the sandbox is hanging up the process.
   980→   980→process.on('SIGHUP', () => {
   981→   981→  log('warn', 'SIGHUP received — ignoring (nohup-style)');
